    class UniformSampler(SamplingMethod):
    """Uniform-grid samples mapped onto a [`Shape`][pinnjax.geometry.base.Shape].

    - `mode` determines how `sampling_value` is interpreted:
    - `"grid_resolution"` is based on [`estimate_nsamples()`][pinnjax.geometry.base.Shape.estimate_nsamples].
        - This value is then reduced based on the dimensionality of the sample space.
    - `"density"` is based on the measured length, area, or volume of the associated [`Shape`][pinnjax.geometry.base.Shape].

    """

    def __init__(  # noqa: D107
        self,
        sampling_value: float = 0.1,
        *,
        mode: SpatialSampleMode = "grid_resolution",
        multiplier: float = 1.0,
    ) -> None:
        self.sampling_value = sampling_value
        self.mode = mode
        self.multiplier = multiplier

    def __call__(
        self,
        shape: Shape,
        sampling_value: float | None = None,
    ) -> Array:
        """Draw a uniform grid and map it onto the shape.

        Args:
            shape: [`Shape`][pinnjax.Shape] to sample.
            sampling_value: Value interpreted by ``mode``.

        Returns:
            Points shaped ``(n_samples, dims)``.

        Raises:
            ValueError: If `multiplier` or `sampling_value` is less than or equal to 0.
            TypeError: If ``mode`` is not one of the two spatial sampling modes.

        """
        if sampling_value is None:
            sampling_value = self.sampling_value

        if sampling_value <= 0:
            msg = f"`sampling_value` must be positive. Got {sampling_value}."
            config.LOGGER.error(msg)
            raise ValueError(msg)

        if self.multiplier <= 0:
            msg = f"`multiplier` must be positive. Got {self.multiplier}."
            config.LOGGER.error(msg)
            raise ValueError(msg)

        mode = self.mode.lower()
        if mode not in ("grid_resolution", "density"):
            msg = f"Invalid 'mode': {self.mode}."
            config.LOGGER.error(msg)
            raise TypeError(msg)

        grid_dims = len(shape.grid_shape(1.0))
        if isinstance(shape.times, Time):
            grid_dims -= 1

        if mode == "grid_resolution":
            resolution = sampling_value
            if grid_dims > 0:
                resolution /= self.multiplier ** (1 / grid_dims)
        else:
            if grid_dims > 0:
                resolution = (sampling_value * self.multiplier) ** (-1 / grid_dims)
            else:
                resolution = 1.0

        if isinstance(shape, Triangle):
            return self._triangle_grid(shape, resolution)

        grid_shape = tuple(max(n, 1) for n in shape.grid_shape(resolution))

        if grid_shape:
            axes = [(jnp.arange(n, dtype=config.get_dtype()) + 0.5) / n for n in grid_shape]
            mesh = jnp.meshgrid(*axes, indexing="ij")
            unit_samples = jnp.stack([axis.reshape(-1) for axis in mesh], axis=-1)
        else:
            unit_samples = jnp.empty((1, 0), dtype=config.get_dtype())

        samples = shape.transform(unit_samples)

        return jnp.asarray(samples, dtype=config.get_dtype())

    def _triangle_grid(self, shape: Triangle, resolution: float) -> Array:
        """Create a uniform triangular lattice inside a triangle."""
        area = float(shape.measure())
        target_samples = max(1, round(area / resolution**2))

        n = max(1, round((float(jnp.sqrt(1 + 8 * target_samples)) - 1) / 2))

        points = []

        for i in range(n):
            for j in range(n - i):
                u = (i + 1 / 3) / n
                v = (j + 1 / 3) / n

                point = shape.a + u * (shape.b - shape.a) + v * (shape.c - shape.a)
                points.append(point)

        spatial = jnp.stack(points)

        if shape.single_time is not None:
            time = jnp.full(
                (spatial.shape[0], 1),
                shape.single_time,
                dtype=config.get_dtype(),
            )
            return jnp.concatenate([spatial, time], axis=1)

        if shape.times is not None:
            n_time = max(1, round(float((shape.tmax - shape.tmin) / shape.tstep)))
            unit_time = (jnp.arange(n_time, dtype=config.get_dtype()) + 0.5) / n_time
            time = shape.tmin + unit_time * (shape.tmax - shape.tmin)

            n_space = spatial.shape[0]
            spatial = jnp.repeat(spatial, n_time, axis=0)
            time = jnp.tile(time, n_space)[:, None]

            return jnp.concatenate([spatial, time], axis=1)

        return jnp.asarray(spatial, dtype=config.get_dtype())

    class SobolSampler(SamplingMethod):
    """Sobol samples mapped onto a shape.

    - `mode` determines how `sampling_value` is interpreted:
        - `"grid_resolution"` is based on [`estimate_nsamples()`][pinnjax.geometry.base.Shape.estimate_nsamples].
            - This value is then reduced based on the dimensionality of the sample space.
        - `"point_count"` uses the set scalar value.
        - `"density"` is based on the measured length, area, or volume of the associated [`Shape`][pinnjax.geometry.base.Shape].

    Points are drawn in the unit hypercube
    and mapped with [`Shape.transform`][pinnjax.geometry.base.Shape.transform].

    Attributes:
        sampling_value: Value interpreted by `mode`.
        mode: `"grid_resolution"`, `"point_count"`, or `"density"`.
        multiplier: Factor applied to the sample count.
        scramble: Scramble the Sobol sequence.
        seed: Seed from [`get_seed`][pinnjax.get_seed].
        key: JAX random key. A call replaces it with the next key. Initialized from [`pinnjax.get_seed`][pinnjax.get_seed].

    """

    # TODO(@mjryande): Add z-warping
    def __init__(
        self,
        sampling_value: float = 1000,
        *,
        mode: SampleCountMode = "point_count",
        multiplier: float = 1.0,
        scramble: bool = True,
    ) -> None:
        """Store the sampling settings.

        The seed and the first JAX key come from [`get_seed`][pinnjax.get_seed].

        Note:
            `"point_count"` requires `sampling_value` to be an integer.

        Args:
            sampling_value: Value interpreted by ``mode``.
            mode: `"grid_resolution"`, `"point_count"`, or `"density"`.
            multiplier: Factor applied to the sample count.
            scramble: Scramble the Sobol sequence.

        Raises:
            TypeError: If ``mode`` is `"point_count"` and ``sampling_value``
                is not an integer.

        """
        self.scramble = scramble
        self.mode = mode
        if self.mode == "point_count" and not isinstance(sampling_value, int):
            msg = (
                "When using `point_count`, `sampling_value` must be an integer. "
                f"Got {type(sampling_value)}."
            )
            config.LOGGER.error(msg)
            raise TypeError(msg)
        self.sampling_value = sampling_value  # default value -- can be overwritten
        self.multiplier = multiplier
        self.seed = config.get_seed()
        key, _ = jax.random.split(jax.random.key(self.seed))
        self.key = key

    def __call__(
        self,
        shape: Shape,
        sampling_value: float | None = None,
    ) -> Array:
        """Draw Sobol points and map them onto the [`Shape`][pinnjax.geometry.base.Shape] using [`transform()`][pinnjax.geometry.base.Shape.transform].

        Args:
            shape: [`Shape`][pinnjax.Shape] to sample.
            sampling_value: Value interpreted by ``mode``.

        Returns:
            Points shaped ``(n_samples, dims)``.

        Raises:
            TypeError: If ``mode`` is `"point_count"` and ``sampling_value``
                is not an integer.
            TypeError: If ``mode`` is not one of the three sampling modes.

        """
        if sampling_value is None:
            sampling_value = self.sampling_value
        elif self.mode == "point_count" and not isinstance(sampling_value, int):
            msg = (
                "When using 'point_count', 'sampling_value' must be "
                f"an integer. Got {type(sampling_value)}."
            )
            config.LOGGER.error(msg)
            raise TypeError(msg)

        match self.mode.lower():
            case "grid_resolution":
                estimated_uniform_samples = shape.estimate_nsamples(sampling_value)

                n_samples = int(
                    jnp.round(
                        max(
                            self._sobol_equivalent_points(
                                estimated_uniform_samples,
                                shape.dims,
                            ),
                            1,
                        )
                        * self.multiplier,
                    ),
                )

            case "point_count":
                n_samples = int(sampling_value * self.multiplier)

            case "density":
                volume = shape.measure()
                times = shape.times
                if isinstance(times, Time):
                    n_samples = max(
                        int(
                            jnp.round(
                                (volume * sampling_value) * times.steps() * self.multiplier,
                            ),
                        ),
                        1,
                    )
                else:
                    n_samples = max(
                        int(
                            jnp.round(volume * sampling_value * self.multiplier),
                        ),
                        1,
                    )
            case _:
                msg = f"Invalid 'mode': {self.mode}."
                config.LOGGER.error(msg)
                raise TypeError(msg)

        lower, _ = shape.bounds()
        _next_power_of_2 = max(next_power_of_2(int(n_samples)), 2)
        key, subkey = jax.random.split(self.key, 2)
        self.key = key
        seed = np.random.default_rng(np.asarray(jax.random.key_data(subkey)))
        sampler = qmc.Sobol(d=lower.shape[0], scramble=self.scramble, rng=seed)
        sobol_samples = jnp.array(sampler.random(_next_power_of_2)[:n_samples])

        samples = shape.transform(sobol_samples)

        return jnp.asarray(samples, dtype=config.get_dtype())

    @staticmethod
    def _sobol_equivalent_points(
        n_uniform: int,
        dim: int,
        smoothness_factor: float = 1.0,
    ) -> int:
        """Estimate how many Sobol points match a uniform grid.

        The uniform count is divided by 2, 3, or 4 when ``dim`` is at most 3,
        at most 4, or larger, then divided by ``smoothness_factor``.

        Args:
            n_uniform: Number of points in the uniform grid.
            dim: Number of axes, including time.
            smoothness_factor: Extra divisor. Values below 1 keep more points.

        Returns:
            The rounded Sobol count.

        """
        THREE_DIMENSIONS = 3
        FOUR_DIMENSIONS = 4
        if dim <= THREE_DIMENSIONS:
            k = 2
        elif dim <= FOUR_DIMENSIONS:
            k = 3
        else:
            k = 4
        return round(n_uniform / (k * smoothness_factor))

    self.multiplier = multiplier
    self.seed = config.get_seed()
    key, _ = jax.random.split(jax.random.key(self.seed))
    self.key = key
