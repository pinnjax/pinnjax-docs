    def line_waypoint_generator(
    speed: float,
    direction: (tuple[float | int, float | int] | tuple[float | int, float | int, float | int]),
    initial: (tuple[float | int, float | int] | tuple[float | int, float | int, float | int]),
    time: Time | float,
    *,
    segments: int = 50,
    active: bool | None = True,
) -> Array:
    """Generate waypoints along a straight line.

    Times run from ``0`` to the duration. A [`Time`][pinnjax.Time] contributes
    its range, not its minimum. ``segments`` intervals produce ``segments + 1``
    points. The last column is time.

    Args:
        speed: Linear speed [m/s].
        direction: Direction of travel, ``(x, y)`` or ``(x, y, z)``.
        initial: Position at ``t = 0``, ``(x, y)`` or ``(x, y, z)``.
        time: A [`Time`][pinnjax.Time], or a duration in seconds.
        segments: Number of intervals along the line. Default: ``50``.
        active: Active flag stored on the
            [`LinearPathConfig`][pinnjax.LinearPathConfig]. It does not
            change the returned waypoints. Default: ``True``.

    Returns:
        Array of shape ``(segments + 1, 3)`` or ``(segments + 1, 4)`` with
        ``(x, y, t)`` or ``(x, y, z, t)``.

    Raises:
        TypeError: If ``time`` is not a [`Time`][pinnjax.Time], ``int``, or
            ``float``.
        InputError: If ``direction`` and ``initial`` differ in length, or if
            that length is not 2 or 3.

    """
    if isinstance(time, Time):
        duration = float(jnp.asarray(time.range()))
    elif isinstance(time, (int, float)):
        duration = float(time)
    else:
        msg = f"Invalid time type: {type(time)}. Time must either be pinnjax.Time, int, or float."
        config.LOGGER.error(msg)
        raise TypeError(msg)

    _config = LinearPathConfig(
        initial=initial,
        direction=direction,
        speed=speed,
        active=active,
    )
    path = create_path(_config)

    times = jnp.linspace(0, duration, segments + 1)
    points = path.eval(times)

    return jnp.hstack([points, times[:, None]])

    def circle_waypoint_generator(
    center: tuple[float | int, float | int] | tuple[float | int, float | int, float | int],
    speed: float | None = None,
    *,
    radius: float | None = None,
    diameter: float | None = None,
    starting_angle: float = 0.0,
    ending_angle: float | None = None,
    rotation_angle: float | None = None,
    time: Time | float | None = None,
    segments: int = 50,
    angle_units: _AngleUnits = "degrees",
) -> Array:
    """Generate waypoints along a circular arc for use with ComplexLinearPathConfig.

    Args:
        center: Center of the circle, (x, y) or (x, y, z).
        speed: Linear speed along the path [m/s]. Optional when time and
            a sweep are both provided, in which case speed is computed.
            When speed and a sweep are both set, those two set the duration.
            Default: ``None``.
        radius: Radius of the circle [m]. Mutually exclusive with diameter.
        diameter: Diameter of the circle [m]. Mutually exclusive with radius.
        starting_angle: Angle at first waypoint (0 = positive x-axis).
            Default: ``0``.
        ending_angle: Angle at last waypoint. Mutually exclusive with
            rotation_angle. When provided with time, enables speed to be computed.
        rotation_angle: Total angular sweep (e.g. 360 or -360 for full circle).
            Positive = counterclockwise, negative = clockwise. Mutually exclusive
            with ending_angle.
        time: A [`Time`][pinnjax.Time], or a duration in seconds. A
            [`Time`][pinnjax.Time] starts at its minimum and lasts for its
            range. A number starts at ``0``. Ignored as a duration when
            speed and a sweep are both set. Default: ``None``.
        segments: Number of waypoint intervals (produces segments + 1 points).
            Default: ``50``.
        angle_units: Unit for starting_angle, ending_angle, and rotation_angle
            (``"degrees"`` or ``"radians"``). Converted to radians internally.
            Default: ``"degrees"``.

    Returns:
        Array of shape (segments + 1, 3) or (segments + 1, 4) with (x, y, t) or (x, y, z, t)
        for use with ComplexLinearPathConfig(waypoints=..., is_temporal=True).

    Raises:
        InputError: If both radius and diameter are set, or if neither is set.
        InputError: If both ending_angle and rotation_angle are set.
        InputError: If speed and the duration cannot be determined from
            speed, time, and the sweep.

    """
    if radius is not None and diameter is not None:
        msg = "Provide either radius or diameter, not both."
        config.LOGGER.error(msg)
        raise config.InputError(msg)
    if radius is not None:
        r = float(radius)
    elif diameter is not None:
        r = float(diameter) / 2
    else:
        msg = "Either radius or diameter must be provided."
        config.LOGGER.error(msg)
        raise config.InputError(msg)

    if angle_units == "degrees":
        starting_angle = math.radians(float(starting_angle))
        if ending_angle is not None:
            ending_angle = math.radians(float(ending_angle))
        if rotation_angle is not None:
            rotation_angle = math.radians(float(rotation_angle))
    else:
        starting_angle = float(starting_angle)
        if ending_angle is not None:
            ending_angle = float(ending_angle)
        if rotation_angle is not None:
            rotation_angle = float(rotation_angle)

    if ending_angle is not None and rotation_angle is not None:
        msg = "Provide either ending_angle or rotation_angle, not both."
        config.LOGGER.error(msg)
        raise config.InputError(msg)
    if ending_angle is not None:
        rot = float(ending_angle) - starting_angle
    elif rotation_angle is not None:
        rot = float(rotation_angle)
    else:
        rot = None

    t_start = 0.0
    if isinstance(time, Time):
        duration = float(jnp.asarray(time.range()))
        t_start = float(jnp.asarray(time.min))
    elif isinstance(time, (int, float)):
        duration = float(time)
    else:
        duration = None

    if speed is not None and rot is not None:
        duration_s = (abs(rot) * r) / speed
        speed_s = speed
        rot_s = rot
    elif speed is not None and duration is not None:
        duration_s = duration
        speed_s = speed
        rot_s = speed * duration / r
    elif duration is not None and rot is not None:
        duration_s = duration
        rot_s = rot
        speed_s = (abs(rot) * r) / duration
    else:
        msg = (
            "Provide either: (speed + rotation_angle), (speed + time), "
            "or (time + rotation_angle/ending_angle)."
        )
        config.LOGGER.error(msg)
        raise config.InputError(msg)

    times = jnp.linspace(
        t_start,
        duration_s + t_start,
        segments + 1,
        dtype=config.get_dtype(),
    )
    # Positive rotation_angle = counterclockwise, negative = clockwise
    direction = 1.0 if rot_s >= 0 else -1.0
    theta = starting_angle + direction * (speed_s / r) * (times - t_start)

    center_arr = jnp.asarray(center, dtype=config.get_dtype())
    x = center_arr[0] + r * jnp.cos(theta)
    y = center_arr[1] + r * jnp.sin(theta)

    if center_arr.shape[0] == 2:
        points = jnp.stack([x, y], axis=-1)
    else:
        z = jnp.full_like(x, center_arr[2])
        points = jnp.stack([x, y, z], axis=-1)

    return jnp.hstack([points, times[:, None]])
