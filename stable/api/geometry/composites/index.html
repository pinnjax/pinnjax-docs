    Composite geometry made of several shapes, with shared residuals and losses.

    ComplexShape #

    class ComplexShape(Shape, ABC):
    """A group of shapes sampled, residual-bound, and drawn together.

    Child shapes stay in ``shapes``. Residuals and losses can be one object for
    every child, one object per child, or one object per residual term.
    ``assign_residuals()`` records those terms and fills the class and type maps
    used by the loss history.

    Attributes:
        shapes: Child shapes, in assignment order.
        times: Shared time range, or ``None`` when the children carry their own time.
        path: Optional path associated with this composite. Default: ``None``.
        class_map: Residual class to integer id, filled by ``assign_residuals()``.
        type_map: Residual category to integer id. Categories are boundary
            condition, partial differential equation, initial condition, data, and
            evaluation.
        kinds_class: Class id of each flattened residual term.
        kinds_type: Category id of each flattened residual term.
        labels_class: Class name for each id in ``class_map``.
        labels_type: Category name for each id in ``type_map``.
        loss_dictionary: Per-epoch loss histories, keyed by ``"class"`` and ``"type"``.
        evaluation_dictionary: Metric histories used by ``plot_evaluation()``.
        final_loss_dictionary: Last value of each series in ``loss_dictionary``.
        best_epoch: Epoch marked on the loss and evaluation plots.
        last_epoch: Latest recorded epoch.
        loss_weights: Weights aligned with the active training groups.
        n_class: Number of residual classes.
        n_type: Number of residual categories.

    """

    times: Time | None

    def __init__(
        self,
        shapes: list[Shape],
        *,
        times: Time | float | None = None,
    ) -> None:
        """Store the child shapes and the optional time range.

        A non-list ``shapes`` value is copied into a list. Loss maps and histories
        start empty. Call ``assign_residuals()`` and ``assign_loss()`` before
        evaluating a loss.

        Args:
            shapes: Child shapes.
            times: Time range shared by this composite. Default: ``None``.

        """
        super().__init__()
        if not isinstance(shapes, list):
            shapes = list(shapes)
        self.shapes: list[Shape] = shapes
        self.times: Time | float | None = times
        self.path: Path | None = None
        self.class_map: dict = {}
        self.type_map: dict = {}
        self.kinds_class: tuple[int, ...] = ()
        self.kinds_type: tuple[int, ...] = ()
        self.labels_class: tuple[str, ...] = ()
        self.labels_type: tuple[str, ...] = ()
        self.loss_dictionary = {}
        self.evaluation_dictionary = {}
        self.final_loss_dictionary = {}
        self.best_epoch: int = 0
        self.last_epoch: int = 0
        self.loss_weights: Iterable = []

        self._residual_terms_by_shape: tuple[tuple[Residual, ...], ...] = ()
        self._loss_terms_by_shape: tuple[tuple[Loss, ...], ...] = ()
        self._term_shape_indices: tuple[int, ...] = ()
        self._term_residual_functions: tuple[Residual, ...] = ()
        self._term_loss_functions: tuple[Loss, ...] = ()
        self._loss_spec: Loss | Sequence[Loss | Sequence[Loss]] | None = None

        self.n_class: int = 0
        self.n_type: int = 0

    def assign_residuals(
        self,
        residuals: Residual | Sequence[Residual | Sequence[Residual]],
        *,
        class_map: dict | None = None,
        type_map: dict | None = None,
        labels_class: list | None = None,
        labels_type: list | None = None,
        kinds_class: list | None = None,
        kinds_type: list | None = None,
    ) -> None:
        """Bind residuals to each child and record one term per bound residual.

        One [`Residual`][pinnjax.Residual] is copied onto every child. A flat
        sequence is one residual per child, unless this composite has a single child,
        in which case the sequence is that child's term list. A sequence of sequences
        is one term list per child and must have one entry per child.

        A residual with ``bind`` is replaced by ``bind(shape)``. Each bound residual
        is stored on the child as ``residual_functions``, and the first term is also
        ``residual_function``. Nested composites are rejected. Call ``unpack()`` first.

        Passing ``None`` for a map or label list starts a new one. Entries already
        present are kept, and new classes are appended. When a loss specification was
        stored before the residuals, ``assign_loss()`` runs again at the end.

        Args:
            residuals: One residual, one residual per child, or one term list per child.
            class_map: Residual class to integer id. Default: a new map.
            type_map: Residual category to integer id. Default: a new map.
            labels_class: Display names parallel to ``class_map``. Default: a new list.
            labels_type: Display names parallel to ``type_map``. Default: a new list.
            kinds_class: Class id per term. Built here and stored on this composite.
            kinds_type: Category id per term. Built here and stored on this composite.

        Raises:
            pinnjax.InputError: If the residual layout does not match ``shapes``, a
                term group is empty, or a child is itself a ``ComplexShape``.
            TypeError: If a bound residual is not a supported residual category.

        """
        if class_map is None or type_map is None:
            class_map, type_map = {}, {}
        if labels_class is None or labels_type is None:
            labels_class, labels_type = [], []
        if kinds_class is None or kinds_type is None:
            kinds_class, kinds_type = [], []

        residual_terms_by_shape = _normalize_residual_terms(residuals, len(self.shapes))

        term_shape_indices: list[int] = []
        term_residuals: list[Residual] = []

        for shape_i, (shape, residual_group) in enumerate(
            zip(self.shapes, residual_terms_by_shape, strict=True),
        ):
            if isinstance(shape, ComplexShape):
                msg = "Multiple residual terms on nested ComplexShape are not supported directly. Call unpack() first."
                config.LOGGER.error(msg)
                raise config.InputError(msg)

            bound_group: list[Residual] = []
            for residual in residual_group:
                bind = getattr(residual, "bind", None)
                if bind is None:
                    bound = residual
                else:
                    bound = cast("Callable[[Shape], Residual]", bind)(shape)
                bound_group.append(bound)

                term_shape_indices.append(shape_i)
                term_residuals.append(bound)

                cls = bound.__class__
                if cls not in class_map:
                    class_map[cls] = len(class_map)
                    labels_class.append(cls.__name__)
                kinds_class.append(class_map[cls])

                type_cls = _residual_type_class(bound)
                if type_cls not in type_map:
                    type_map[type_cls] = len(type_map)
                    labels_type.append(type_cls.__name__)
                kinds_type.append(type_map[type_cls])

            shape.residual_functions = tuple(bound_group)
            shape.residual_function = bound_group[0]

        self._residual_terms_by_shape = tuple(tuple(g) for g in residual_terms_by_shape)
        self._term_shape_indices = tuple(term_shape_indices)
        self._term_residual_functions = tuple(term_residuals)

        self.class_map = class_map
        self.type_map = type_map
        self.kinds_class = tuple(kinds_class)
        self.kinds_type = tuple(kinds_type)
        self.labels_class = tuple(labels_class)
        self.labels_type = tuple(labels_type)

        self.n_class = int(max(self.kinds_class) + 1) if self.kinds_class else 0
        self.n_type = int(max(self.kinds_type) + 1) if self.kinds_type else 0

        self.loss_dictionary = {
            "class": {lab: [] for lab in self.labels_class},
            "type": {lab: [] for lab in self.labels_type},
        }
        self.final_loss_dictionary = {}

        if self._loss_spec is not None:
            self.assign_loss(self._loss_spec)

    def print_samples(self) -> None:
        """Log the collocation-point count of each child.

        The message is the one-based index, the child class name, and ``n_points``.
        """
        for i, s in enumerate(self.shapes):
            config.LOGGER.info(
                "Shape %d (%s): %d collocation points",
                i + 1,
                s.__class__.__name__,
                s.n_points,
            )

    def final_loss(self, *, overwrite: bool = False) -> dict:
        """Return the last value of every series in the loss history.

        The result keeps the ``"class"`` and ``"type"`` keys. Each label maps to the
        last recorded value instead of the full series. A later call returns the
        cached dictionary until ``overwrite`` is true.

        Args:
            overwrite: Recompute the snapshot when one is already stored.
                Default: ``False``.

        Returns:
            The last value of each class and type series.

        Raises:
            ValueError: If ``loss_dictionary`` has not been filled.

        """
        if not self.loss_dictionary:
            msg = "The Loss Dictionary not yet populated. Train the model first."
            config.LOGGER.error(msg)
            raise ValueError(msg)

        if self.final_loss_dictionary and overwrite is False:
            return self.final_loss_dictionary

        self.final_loss_dictionary = _final_loss(self.loss_dictionary)

        return self.final_loss_dictionary

    def assign_loss(
        self,
        loss: Loss | Sequence[Loss | Sequence[Loss]],
    ) -> None:
        """Attach a loss to every child, or one loss to every residual term.

        The specification is stored and reused when ``assign_residuals()`` runs later.

        Before residuals are assigned, pass one [`Loss`][pinnjax.Loss] or one loss
        per child. A nested composite receives ``assign_loss()`` with its own entry.
        A leaf receives ``set_loss()``. A sequence of losses for one leaf is rejected
        until residuals exist.

        After residuals are assigned, one loss is repeated for every term. A flat
        sequence is one loss per child, or, for a single child, one loss per term.
        A sequence of sequences must match the term counts from ``assign_residuals()``.
        Each child then stores ``loss_functions`` and keeps the first loss as
        ``loss_function``.

        Args:
            loss: One loss, one loss per child, or one loss per residual term.

        Raises:
            ValueError: If a sequence length does not match the children or the
                residual terms.
            TypeError: If several losses are given for one leaf before residuals
                are assigned.

        """
        self._loss_spec = loss

        if not self._residual_terms_by_shape:
            if isinstance(loss, Loss):
                losses = [loss] * len(self.shapes)
            else:
                losses = list(loss)
            if len(losses) != len(self.shapes):
                msg = (
                    f"Length of losses must equal length of shapes: "
                    f"got {len(losses)} for {len(self.shapes)}."
                )
                config.LOGGER.error(msg)
                raise ValueError(msg)
            for shape, lf in zip(self.shapes, losses, strict=True):
                if isinstance(shape, ComplexShape):
                    shape.assign_loss(lf)
                else:
                    if not isinstance(lf, Loss):
                        msg = "Assign residuals before assigning multiple losses per shape."
                        config.LOGGER.error(msg)
                        raise TypeError(msg)
                    shape.set_loss(lf)
            return

        term_counts = [len(group) for group in self._residual_terms_by_shape]
        loss_terms_by_shape = _normalize_loss_terms(loss, len(self.shapes), term_counts)

        term_losses: list[Loss] = []
        for shape, loss_group in zip(self.shapes, loss_terms_by_shape, strict=True):
            shape.loss_functions = tuple(loss_group)
            shape.loss_function = loss_group[0]
            term_losses.extend(loss_group)

        self._loss_terms_by_shape = tuple(tuple(g) for g in loss_terms_by_shape)

    def sample(
        self,
        method: SamplingMethod,
        sampling_values: float | Sequence[float | int] | None = None,
        *,
        overwrite: bool = False,
        descriptor_count: int = 32,
    ) -> tuple[Array, int]:
        """Sample every child and store the total point count on this composite.

        ``sampling_values`` is one number per child. ``None`` leaves each child to
        its sampler default. One number is repeated for every child. A nested
        ``ComplexShape`` is sampled through this method. A ``Shape`` is sampled
        only when it has no points or ``overwrite`` is true. A ``Shape`` that
        already has points logs a warning and keeps them.

        The points stay on each child. This composite does not store one combined
        array on ``samples``. The returned array is empty. The returned count is
        the total ``n_points``.

        ``descriptor_count`` is used by
        [`OperatorGeometry.sample`][pinnjax.geometry.operator.OperatorGeometry.sample] and ignored
        here.

        Args:
            method: Sampler used for every child.
            sampling_values: One sampling size, or one size per child. ``None``
                leaves each child to its sampler default.
            overwrite: Replace points that were already generated.
            descriptor_count: Descriptor sample count for an operator geometry.

        Returns:
            samples: An empty array. The points are stored on the children.
            n_points: Total number of collocation points.

        Raises:
            ValueError: If a sequence of sampling sizes does not have one entry
                per child.

        """
        del descriptor_count
        n_shapes = len(self.shapes)
        if sampling_values is None:
            values: list[float | int | None] = [None] * n_shapes
        elif isinstance(sampling_values, (float, int)):
            values = [sampling_values] * n_shapes
        else:
            values = list(sampling_values)

        if n_shapes != len(values):
            msg = f"Number of Shapes: {n_shapes} does not equal number of sampling_values: {values}"
            config.LOGGER.error(msg)
            raise ValueError(msg)

        self.n_points = 0
        for shape, value in zip(self.shapes, values, strict=True):
            if isinstance(shape, ComplexShape):
                ComplexShape.sample(shape, method, value, overwrite=overwrite)
                self.n_points += shape.n_points
            elif shape.n_points == 0 or overwrite is True:
                _, count = shape.sample(method, value, overwrite=overwrite)
                self.n_points += count
            else:
                config.LOGGER.warning("Shape already sampled.")

        return jnp.empty((0,), dtype=config.get_dtype()), self.n_points

    def residual_and_losses(
        self,
        model: NeuralNetwork,
        ops: DifferentialOperators,
        *,
        samples_list_override: Sequence[Array] | None = None,
        sample_indices_override: Sequence[Array | None] | None = None,
        sample_targets_override: Sequence[Array | None] | None = None,
        microbatch_size: int | None = None,
        gradient_checkpoint: bool = False,
    ) -> tuple[Array, Array, Array]:
        """Compute the total loss and the class and type vectors.

        Samples come from each child's ``samples`` attribute unless
        ``samples_list_override`` is set. Index and target overrides follow the same
        child order and are used by data residuals. ``microbatch_size`` splits each
        leaf evaluation into sequential chunks. It does not add optimizer steps.

        Args:
            model: Network to evaluate.
            ops: Differential operators for ``model``.
            samples_list_override: Per-child point arrays. ``None`` uses each child's
                ``samples``. Default: ``None``.
            sample_indices_override: Per-child row indices into that child's training
                rows. An entry may be ``None``. Default: ``None``.
            sample_targets_override: Per-child targets for a data residual. An entry
                may be ``None``. Default: ``None``.
            microbatch_size: Rows per chunk, or ``None`` to evaluate each term at once.
                Default: ``None``.
            gradient_checkpoint: Checkpoint the forward pass inside a gradient.
                Default: ``False``.

        Returns:
            total: Scalar loss summed over terms.
            by_class: Loss summed into one entry per residual class.
            by_type: Loss summed into one entry per residual category.

        Raises:
            ValueError: If sampling has not been run, or an override length does not
                match the number of children.

        """
        if samples_list_override is None:
            collected: list[Array] = []
            for shape in self.shapes:
                collected.append(_points_for_loss(shape))
            samples_list = tuple(collected)
        else:
            if len(samples_list_override) != len(self.shapes):
                msg = (
                    f"samples_list_override must have length {len(self.shapes)}, "
                    f"got {len(samples_list_override)}."
                )
                config.LOGGER.error(msg)
                raise ValueError(msg)
            samples_list = tuple(samples_list_override)

        if sample_indices_override is not None and len(sample_indices_override) != len(self.shapes):
            msg = (
                f"sample_indices_override must have length {len(self.shapes)}, "
                f"got {len(sample_indices_override)}."
            )
            config.LOGGER.error(msg)
            raise ValueError(msg)

        if sample_targets_override is not None and len(sample_targets_override) != len(self.shapes):
            msg = (
                f"sample_targets_override must have length {len(self.shapes)}, "
                f"got {len(sample_targets_override)}."
            )
            config.LOGGER.error(msg)
            raise ValueError(msg)

        residual_functions = self._term_residual_functions
        if not residual_functions:
            residual_functions = tuple(s.residual_function for s in self.shapes)
            term_shape_indices = tuple(range(len(self.shapes)))
        else:
            term_shape_indices = self._term_shape_indices

        loss_functions = self._term_loss_functions
        if not loss_functions:
            loss_functions = tuple(ResidualMSError() for _ in residual_functions)

        kinds_class = jnp.asarray(self.kinds_class, dtype=jnp.int32)
        n_class = self.n_class
        kinds_type = jnp.asarray(self.kinds_type, dtype=jnp.int32)
        n_type = self.n_type

        total, by_class, by_type = _losses_kernel(
            samples_list=samples_list,
            residual_functions=residual_functions,
            loss_functions=loss_functions,
            term_shape_indices=term_shape_indices,
            kinds_class=kinds_class,
            kinds_type=kinds_type,
            n_class=n_class,
            n_type=n_type,
            ops=ops,
            sample_indices_list=sample_indices_override,
            sample_targets_list=sample_targets_override,
            microbatch_size=microbatch_size,
            checkpoint=gradient_checkpoint,
        )

        return total, by_class, by_type

        sample_targets_override: Sequence[Array | None] | None = None,
        microbatch_size: int | None = None,
        gradient_checkpoint: bool = True,
        train_mode: str = "type",
        term_weights: Array,
        loss_scale: float = 1.0,
    ) -> tuple[Array, Array, Array, Array, object]:
        """Compute one scalar loss and its gradient in a single reverse-mode pass.

        The scalar is the dot product of ``term_weights`` with either the category
        vector or the class vector, multiplied by ``loss_scale``. ``train_mode``
        ``"type"`` uses the category vector. Any other mode uses the class vector.

                ``None``. Default: ``None``.
            microbatch_size: Rows per chunk, or ``None`` to evaluate each term at once.
                Default: ``None``.

                Default: ``True``.

                residual classes. Default: ``"type"``.
            term_weights: One weight per class, or per category when ``train_mode`` is

            by_type: Loss summed into one entry per residual category.

        if sample_indices_override is not None and len(sample_indices_override) != len(self.shapes):
            msg = (
                f"sample_indices_override must have length {len(self.shapes)}, "
                f"got {len(sample_indices_override)}."
            )
            config.LOGGER.error(msg)
            raise ValueError(msg)

        if sample_targets_override is not None and len(sample_targets_override) != len(self.shapes):
            msg = (
                f"sample_targets_override must have length {len(self.shapes)}, "
                f"got {len(sample_targets_override)}."
            )
            config.LOGGER.error(msg)
            raise ValueError(msg)

        residual_functions = self._term_residual_functions
        if not residual_functions:
            residual_functions = tuple(s.residual_function for s in self.shapes)
            term_shape_indices = tuple(range(len(self.shapes)))

        n_class = self.n_class
        kinds_type = jnp.asarray(self.kinds_type, dtype=jnp.int32)

    def unpack(self, *, flatten: bool = True) -> None:
        """Replace nested composites with simpler children.

        The walk uses the children present at the start of the call. Shapes appended
        during the call are not visited again.

        ``flatten`` false replaces each nested composite with its immediate children.
        ``flatten`` true keeps a nested composite only when ``verify_repetition()``
        accepts it. The shared child type is then constructed from the oriented
        bounding box of ``encompassing_domain()``, with that domain's time. The shared
        residual and loss are copied onto the replacement, and existing child samples
        are stacked. A nested composite that fails the repetition check is removed and
        a warning is logged. Its children are not inserted.

        Args:
            flatten: Merge repeated children into one shape. Default: ``True``.

        Raises:
            TypeError: If the repetition check does not return its sets.

        """
        for _i, shape in enumerate(list(self.shapes)):
            if isinstance(shape, ComplexShape):
                self.shapes.remove(shape)
                if flatten:
                    result = shape.verify_repetition(set_return=True)
                    if isinstance(result, bool):
                        raise TypeError("verify_repetition did not return the repetition sets.")
                    valid, _shapes, _residual, _loss = result
                    if valid:
                        _shape = next(iter(_shapes), None)
                        if _shape is not None:
                            domain = shape.encompassing_domain()
                            obb = domain.to_obb()
                            flattened_shape = _shape(*obb, times=domain.times)
                            if _residual:
                                flattened_shape.set_residual(_residual)
                            if _loss:
                                flattened_shape.set_loss(_loss)
                            samples = []
                            n_points = 0
                            for s in shape.shapes:
                                if s.samples is not None:
                                    samples.append(s.samples)
                                    n_points += round(s.n_points)

                            if len(samples) > 0:
                                flattened_shape.samples = jnp.vstack(samples)
                            else:
                                flattened_shape.samples = None

                            flattened_shape.n_points = round(n_points)
                            self.shapes.append(flattened_shape)
                    else:
                        config.LOGGER.warning(
                            "Complex Shape Identified, but repetition not verified.",
                        )
                else:
                    for s in shape.shapes:
                        self.shapes.append(s)

    def encompassing_domain(self) -> SpatialDomain:
        """Return a domain that contains every child.

        The spatial bounds are the lowest and highest coordinates reported by
        ``bounds()``. When ``times`` is set, the last bound coordinate is treated as
        time and ``times`` is reused. Otherwise, the last bound coordinate becomes a
        [`Time`][pinnjax.Time] range with step ``0.1``.

        Returns:
            A domain covering every child.

        """
        mins = []
        maxs = []

        if isinstance(self.times, float):
            raise TypeError("Cannot create a SpatialDomain from a ComplexShape at a single time.")

        for shape in self.shapes:
            bmin, bmax = shape.bounds()
            mins.append(bmin)
            maxs.append(bmax)

        mins = jnp.min(jnp.stack(mins), axis=0)
        maxs = jnp.max(jnp.stack(maxs), axis=0)

        if self.times is not None:
            return SpatialDomain(
                min=tuple(mins[:-1]),
                max=tuple(maxs[:-1]),
                times=self.times,
            )

        return SpatialDomain(
            min=tuple(mins),
            max=tuple(maxs),
        )

    def verify_repetition(
        self,
        *,
        set_return: bool = False,
    ) -> bool | tuple[bool, set, set, set]:
        """Check whether every child has the same type, residual, and loss.

        A child with no residual, or no loss, does not add an entry. The check still
        passes when the children that do have a residual all share one object, and the
        same is true for the loss. Two different residual objects fail the check.

        Args:
            set_return: Return the three sets along with the boolean.
                Default: ``False``.

        Returns:
            True when the type, residual, and loss each have at most one distinct
            value. If ``set_return`` is true, also return the set of child types, the
            set of residual objects, and the set of loss objects.

        """
        shapes = set()
        residual = set()
        loss = set()
        for shape in self.shapes:
            shapes.add(type(shape))
            if shape.has_residual():
                residual.add(shape.residual_function)
            if shape.has_loss():
                loss.add(shape.loss_function)

        valid = all(len(s) <= 1 for s in (shapes, residual, loss))
        if set_return:
            return valid, shapes, residual, loss
        return valid

    def draw(
        self,
        *,
        show_normals: bool = False,
        normal_scale: float = 1.0,
        fig: Figure | None = None,
        ax: Axes | None = None,
        fig_size: tuple[int, int] | None = None,
        show: bool = True,
    ) -> tuple[Figure, Axes]:
        """Draw every child on one figure.

        A new figure and axes are created when either ``fig`` or ``ax`` is omitted.
        The axes are three-dimensional when any child has at least three spatial
        dimensions. Each child's ``draw`` is called with ``show`` false. A child that
        cannot be drawn logs a warning and is skipped. The title reports how many
        children were plotted.

        Args:
            show_normals: Forwarded to each child. Default: ``False``.
            normal_scale: Scale forwarded with ``show_normals``. Default: ``1``.
            fig: Figure to draw on. Default: a new figure.
            ax: Axes to draw on. Default: axes matching the spatial dimension.
            fig_size: Width and height, in inches, of a figure created here.
            show: Display the figure before returning it. Default: ``True``.

        Returns:
            The Matplotlib figure and axes.

        Raises:
            TypeError: If the plot is three-dimensional and ``ax`` is not a 3D axes.

        """
        # Determine if we need 2D or 3D plotting by checking the shapes
        needs_3d = any(
            (hasattr(shape, "spatial_dims") and shape.spatial_dims >= THREE_DIMENSIONAL)
            for shape in self.shapes
        )

        # Create appropriate axes if not provided
        if fig is None or ax is None:
            fig = plt.figure(figsize=fig_size)
            if needs_3d:
                ax = fig.add_subplot(111, projection="3d")
            else:
                ax = fig.add_subplot(111)

        # Draw each shape
        for shape in self.shapes:
            if hasattr(shape, "draw"):
                try:
                    fig, ax = shape.draw(
                        show_normals=show_normals,
                        normal_scale=normal_scale,
                        fig=fig,
                        ax=ax,
                        show=False,
                    )
                except (ValueError, TypeError, AttributeError) as e:
                    config.LOGGER.warning(
                        "Could not draw shape %s: %s",
                        type(shape).__name__,
                        e,
                    )

        mins = []
        maxs = []

        for shape in self.shapes:
            if isinstance(shape, ComplexShape):
                domain = shape.encompassing_domain()
                mins.append(jnp.asarray(domain.min))
                maxs.append(jnp.asarray(domain.max))
            else:
                lower, upper = shape.bounds()
                mins.append(lower[: shape.spatial_dims])
                maxs.append(upper[: shape.spatial_dims])

        domain_min = jnp.min(jnp.stack(mins), axis=0)
        domain_max = jnp.max(jnp.stack(maxs), axis=0)

        # Set appropriate properties based on 2D vs 3D
        if needs_3d:
            if not isinstance(ax, Axes3D):
                raise TypeError("A 3D plot requires a 3D axes.")

            ax.set_xlim(float(domain_min[0]), float(domain_max[0]))
            ax.set_ylim(float(domain_min[1]), float(domain_max[1]))
            ax.set_zlim(float(domain_min[2]), float(domain_max[2]))

            dimensions = domain_max - domain_min
            max_dim = float(jnp.max(dimensions))
            aspect_ratios = [max(0.2, float(d) / max_dim) for d in dimensions]

            ax.set_box_aspect(aspect_ratios)
            ax.set_xlabel("X")
            ax.set_ylabel("Y")
            ax.set_zlabel("Z")

        else:
            ax.set_xlim(float(domain_min[0]), float(domain_max[0]))
            ax.set_ylim(float(domain_min[1]), float(domain_max[1]))

            ax.set_xlabel("X")
            ax.set_ylabel("Y")
            ax.grid(True)
            ax.set_aspect("equal")

        ax.set_title(f"Complex Shape: {len(self.shapes)} components")

        if show:
            plt.show()

        return fig, ax

    def assign_loss(
    self,
    loss: Loss | Sequence[Loss | Sequence[Loss]],

    """Attach a loss to every child, or one loss to every residual term.

    The specification is stored and reused when ``assign_residuals()`` runs later.

    Before residuals are assigned, pass one [`Loss`][pinnjax.Loss] or one loss
    per child. A nested composite receives ``assign_loss()`` with its own entry.
    A leaf receives ``set_loss()``. A sequence of losses for one leaf is rejected
    until residuals exist.

    A sequence of sequences must match the term counts from ``assign_residuals()``.
    Each child then stores ``loss_functions`` and keeps the first loss as
    ``loss_function``.

    Args:
        loss: One loss, one loss per child, or one loss per residual term.

    Raises:
        ValueError: If a sequence length does not match the children or the
            residual terms.
        TypeError: If several losses are given for one leaf before residuals
            are assigned.

    """
    self._loss_spec = loss

    if not self._residual_terms_by_shape:
        if isinstance(loss, Loss):
            losses = [loss] * len(self.shapes)
        else:
            losses = list(loss)
        if len(losses) != len(self.shapes):
            msg = (
                f"Length of losses must equal length of shapes: "
                f"got {len(losses)} for {len(self.shapes)}."
            )
            config.LOGGER.error(msg)
            raise ValueError(msg)
        for shape, lf in zip(self.shapes, losses, strict=True):
            if isinstance(shape, ComplexShape):
                shape.assign_loss(lf)
            else:
                if not isinstance(lf, Loss):
                    msg = "Assign residuals before assigning multiple losses per shape."
                    config.LOGGER.error(msg)
                    raise TypeError(msg)
                shape.set_loss(lf)
        return

    term_counts = [len(group) for group in self._residual_terms_by_shape]
    loss_terms_by_shape = _normalize_loss_terms(loss, len(self.shapes), term_counts)

    term_losses: list[Loss] = []
    for shape, loss_group in zip(self.shapes, loss_terms_by_shape, strict=True):
        shape.loss_functions = tuple(loss_group)
        shape.loss_function = loss_group[0]
        term_losses.extend(loss_group)

    self._loss_terms_by_shape = tuple(tuple(g) for g in loss_terms_by_shape)
    self._term_loss_functions = tuple(term_losses)

    def assign_residuals(
    self,
    residuals: Residual | Sequence[Residual | Sequence[Residual]],
    *,
    class_map: dict | None = None,
    type_map: dict | None = None,
    labels_class: list | None = None,
    labels_type: list | None = None,
    kinds_class: list | None = None,
    kinds_type: list | None = None,
) -> None:
    """Bind residuals to each child and record one term per bound residual.

    One [`Residual`][pinnjax.Residual] is copied onto every child. A flat
    sequence is one residual per child, unless this composite has a single child,
    in which case the sequence is that child's term list. A sequence of sequences
    is one term list per child and must have one entry per child.

    A residual with ``bind`` is replaced by ``bind(shape)``. Each bound residual
    is stored on the child as ``residual_functions``, and the first term is also
    ``residual_function``. Nested composites are rejected. Call ``unpack()`` first.

    Passing ``None`` for a map or label list starts a new one. Entries already
    present are kept, and new classes are appended. When a loss specification was
    stored before the residuals, ``assign_loss()`` runs again at the end.

    Args:
        residuals: One residual, one residual per child, or one term list per child.
        class_map: Residual class to integer id. Default: a new map.
        type_map: Residual category to integer id. Default: a new map.
        labels_class: Display names parallel to ``class_map``. Default: a new list.
        labels_type: Display names parallel to ``type_map``. Default: a new list.
        kinds_class: Class id per term. Built here and stored on this composite.
        kinds_type: Category id per term. Built here and stored on this composite.

    Raises:
        pinnjax.InputError: If the residual layout does not match ``shapes``, a
            term group is empty, or a child is itself a ``ComplexShape``.
        TypeError: If a bound residual is not a supported residual category.

    """
    if class_map is None or type_map is None:
        class_map, type_map = {}, {}
    if labels_class is None or labels_type is None:
        labels_class, labels_type = [], []
    if kinds_class is None or kinds_type is None:
        kinds_class, kinds_type = [], []

    residual_terms_by_shape = _normalize_residual_terms(residuals, len(self.shapes))

    term_shape_indices: list[int] = []
    term_residuals: list[Residual] = []

    for shape_i, (shape, residual_group) in enumerate(
        zip(self.shapes, residual_terms_by_shape, strict=True),
    ):
        if isinstance(shape, ComplexShape):
            msg = "Multiple residual terms on nested ComplexShape are not supported directly. Call unpack() first."
            config.LOGGER.error(msg)
            raise config.InputError(msg)

        bound_group: list[Residual] = []
        for residual in residual_group:
            bind = getattr(residual, "bind", None)
            if bind is None:
                bound = residual
            else:
                bound = cast("Callable[[Shape], Residual]", bind)(shape)
            bound_group.append(bound)

            term_shape_indices.append(shape_i)
            term_residuals.append(bound)

            cls = bound.__class__
            if cls not in class_map:
                class_map[cls] = len(class_map)
                labels_class.append(cls.__name__)
            kinds_class.append(class_map[cls])

            type_cls = _residual_type_class(bound)
            if type_cls not in type_map:
                type_map[type_cls] = len(type_map)
                labels_type.append(type_cls.__name__)
            kinds_type.append(type_map[type_cls])

        shape.residual_functions = tuple(bound_group)
        shape.residual_function = bound_group[0]

    self._residual_terms_by_shape = tuple(tuple(g) for g in residual_terms_by_shape)
    self._term_shape_indices = tuple(term_shape_indices)
    self._term_residual_functions = tuple(term_residuals)

    self.class_map = class_map
    self.type_map = type_map
    self.kinds_class = tuple(kinds_class)
    self.kinds_type = tuple(kinds_type)
    self.labels_class = tuple(labels_class)
    self.labels_type = tuple(labels_type)

    self.n_class = int(max(self.kinds_class) + 1) if self.kinds_class else 0
    self.n_type = int(max(self.kinds_type) + 1) if self.kinds_type else 0

    self.loss_dictionary = {
        "class": {lab: [] for lab in self.labels_class},
        "type": {lab: [] for lab in self.labels_type},
    }
    self.final_loss_dictionary = {}

    if self._loss_spec is not None:
        self.assign_loss(self._loss_spec)

    """Draw every child on one figure.

    A new figure and axes are created when either ``fig`` or ``ax`` is omitted.
    The axes are three-dimensional when any child has at least three spatial
    dimensions. Each child's ``draw`` is called with ``show`` false. A child that
    cannot be drawn logs a warning and is skipped. The title reports how many
    children were plotted.

    Args:
        show_normals: Forwarded to each child. Default: ``False``.
        normal_scale: Scale forwarded with ``show_normals``. Default: ``1``.
        fig: Figure to draw on. Default: a new figure.

    Raises:
        TypeError: If the plot is three-dimensional and ``ax`` is not a 3D axes.

    needs_3d = any(
        (hasattr(shape, "spatial_dims") and shape.spatial_dims >= THREE_DIMENSIONAL)
        for shape in self.shapes
    )

    # Create appropriate axes if not provided
    if fig is None or ax is None:
        fig = plt.figure(figsize=fig_size)
        if needs_3d:
            ax = fig.add_subplot(111, projection="3d")
        else:
            ax = fig.add_subplot(111)

    # Draw each shape
    for shape in self.shapes:
        if hasattr(shape, "draw"):
            try:
                fig, ax = shape.draw(
                    show_normals=show_normals,
                    normal_scale=normal_scale,
                    fig=fig,
                    ax=ax,
                    show=False,
                )
            except (ValueError, TypeError, AttributeError) as e:
                config.LOGGER.warning(
                    "Could not draw shape %s: %s",
                    type(shape).__name__,
                    e,
                )

    mins = []
    maxs = []

    for shape in self.shapes:
        if isinstance(shape, ComplexShape):
            domain = shape.encompassing_domain()
            mins.append(jnp.asarray(domain.min))
            maxs.append(jnp.asarray(domain.max))
        else:
            lower, upper = shape.bounds()
            mins.append(lower[: shape.spatial_dims])
            maxs.append(upper[: shape.spatial_dims])

    domain_min = jnp.min(jnp.stack(mins), axis=0)
    domain_max = jnp.max(jnp.stack(maxs), axis=0)

    # Set appropriate properties based on 2D vs 3D
    if needs_3d:
        if not isinstance(ax, Axes3D):
            raise TypeError("A 3D plot requires a 3D axes.")

        ax.set_xlim(float(domain_min[0]), float(domain_max[0]))
        ax.set_ylim(float(domain_min[1]), float(domain_max[1]))
        ax.set_zlim(float(domain_min[2]), float(domain_max[2]))

        dimensions = domain_max - domain_min
        max_dim = float(jnp.max(dimensions))
        aspect_ratios = [max(0.2, float(d) / max_dim) for d in dimensions]

        ax.set_ylabel("Y")
        ax.set_zlabel("Z")

        ax.set_ylim(float(domain_min[1]), float(domain_max[1]))

        ax.set_ylabel("Y")
        ax.grid(True)
        ax.set_aspect("equal")

    ax.set_title(f"Complex Shape: {len(self.shapes)} components")

    if show:
        plt.show()

    return fig, ax

    Samples come from each child's ``samples`` attribute unless
    ``samples_list_override`` is set. Index and target overrides follow the same
    child order and are used by data residuals. ``microbatch_size`` splits each

            match the number of children.

            msg = (
                f"samples_list_override must have length {len(self.shapes)}, "

            raise ValueError(msg)

    if sample_indices_override is not None and len(sample_indices_override) != len(self.shapes):
        msg = (
            f"sample_indices_override must have length {len(self.shapes)}, "
            f"got {len(sample_indices_override)}."
        )
        config.LOGGER.error(msg)
        raise ValueError(msg)

        msg = (
            f"sample_targets_override must have length {len(self.shapes)}, "
            f"got {len(sample_targets_override)}."
        )
        config.LOGGER.error(msg)
        raise ValueError(msg)

        residual_functions = tuple(s.residual_function for s in self.shapes)

        sample_targets_list=sample_targets_override,
        microbatch_size=microbatch_size,

    )

    ``ComplexShape`` is sampled through this method. A ``Shape`` is sampled

    already has points logs a warning and keeps them.

    the total ``n_points``.

    [`OperatorGeometry.sample`][pinnjax.geometry.operator.OperatorGeometry.sample] and ignored
    here.

        method: Sampler used for every child.
        sampling_values: One sampling size, or one size per child. ``None``
            leaves each child to its sampler default.
        overwrite: Replace points that were already generated.
        descriptor_count: Descriptor sample count for an operator geometry.

        samples: An empty array. The points are stored on the children.
        n_points: Total number of collocation points.

    Raises:
        ValueError: If a sequence of sampling sizes does not have one entry
            per child.

    """
    del descriptor_count
    n_shapes = len(self.shapes)
    if sampling_values is None:
        values: list[float | int | None] = [None] * n_shapes
    elif isinstance(sampling_values, (float, int)):
        values = [sampling_values] * n_shapes
    else:
        values = list(sampling_values)

    if n_shapes != len(values):
        msg = f"Number of Shapes: {n_shapes} does not equal number of sampling_values: {values}"
        config.LOGGER.error(msg)
        raise ValueError(msg)

    self.n_points = 0
    for shape, value in zip(self.shapes, values, strict=True):
        if isinstance(shape, ComplexShape):
            ComplexShape.sample(shape, method, value, overwrite=overwrite)
            self.n_points += shape.n_points
        elif shape.n_points == 0 or overwrite is True:
            _, count = shape.sample(method, value, overwrite=overwrite)
            self.n_points += count
        else:
            config.LOGGER.warning("Shape already sampled.")

    return jnp.empty((0,), dtype=config.get_dtype()), self.n_points

    def unpack(self, *, flatten: bool = True) -> None:
    """Replace nested composites with simpler children.

    The walk uses the children present at the start of the call. Shapes appended
    during the call are not visited again.

    ``flatten`` false replaces each nested composite with its immediate children.
    ``flatten`` true keeps a nested composite only when ``verify_repetition()``
    accepts it. The shared child type is then constructed from the oriented
    bounding box of ``encompassing_domain()``, with that domain's time. The shared
    residual and loss are copied onto the replacement, and existing child samples
    are stacked. A nested composite that fails the repetition check is removed and
    a warning is logged. Its children are not inserted.

    Args:
        flatten: Merge repeated children into one shape. Default: ``True``.

    Raises:
        TypeError: If the repetition check does not return its sets.

    """
    for _i, shape in enumerate(list(self.shapes)):
        if isinstance(shape, ComplexShape):
            self.shapes.remove(shape)
            if flatten:
                result = shape.verify_repetition(set_return=True)
                if isinstance(result, bool):
                    raise TypeError("verify_repetition did not return the repetition sets.")
                valid, _shapes, _residual, _loss = result
                if valid:
                    _shape = next(iter(_shapes), None)
                    if _shape is not None:
                        domain = shape.encompassing_domain()
                        obb = domain.to_obb()
                        flattened_shape = _shape(*obb, times=domain.times)
                        if _residual:
                            flattened_shape.set_residual(_residual)
                        if _loss:
                            flattened_shape.set_loss(_loss)
                        samples = []
                        n_points = 0
                        for s in shape.shapes:
                            if s.samples is not None:
                                samples.append(s.samples)
                                n_points += round(s.n_points)

                        if len(samples) > 0:
                            flattened_shape.samples = jnp.vstack(samples)
                        else:
                            flattened_shape.samples = None

                        flattened_shape.n_points = round(n_points)
                        self.shapes.append(flattened_shape)
                else:
                    config.LOGGER.warning(
                        "Complex Shape Identified, but repetition not verified.",
                    )
            else:
                for s in shape.shapes:
                    self.shapes.append(s)
