    class DataResidual(Residual):
    """[`Residual`][pinnjax.Residual] for a fixed set of inputs and targets.

    The residual at each sample is ``truth - prediction``.
    `prediction_transformation`, when set, maps `(inputs, prediction)` onto the
    targets before that difference. Non-finite residuals are replaced with zeros.

    `bind` copies `train_outputs` when that array is set. Otherwise, it copies `outputs`.

    Attributes:
        shape: [`DataShape`][pinnjax.DataShape] this residual was copied onto.
            `None` until `bind`.
        prediction_transformation: Optional map from `(inputs, prediction)` to the
            values compared with the targets.
        ground_truth: Targets copied by `bind`. `None` until then.

    Args:
        prediction_transformation: Optional callable applied to the prediction.
            Default: None.

    """

    shape: Shape | None = eqx.field(
        default=None,
        static=True,
    )

    prediction_transformation: (
        Callable[
            [Array, Array],
            Array,
        ]
        | None
    ) = eqx.field(
        default=None,
        static=True,
    )

    ground_truth: Array | None = None

    def __init__(
        self,
        prediction_transformation: Callable[
            [Array, Array],
            Array,
        ]
        | None = None,
    ) -> None:
        """Store ``prediction_transformation`` and leave the shape and targets unset."""
        self.shape = None
        self.ground_truth = None
        self.prediction_transformation = prediction_transformation

    def bind(self, shape: DataShape) -> DataResidual:
        """Copy this residual onto a [`DataShape`][pinnjax.DataShape] and store its targets.

        Uses `train_outputs` when that array is set. Otherwise, uses `outputs`.
        The original residual is left unchanged.

        Args:
            shape: A [`DataShape`][pinnjax.DataShape] object.

        Returns:
            The bound copy.

        Raises:
            ValueError: If the shape has neither `train_outputs` nor `outputs`.

        """
        dr = copy.copy(self)
        dr.shape = shape

        if hasattr(shape, "train_outputs") and shape.train_outputs is not None:
            dr.ground_truth = jnp.asarray(shape.train_outputs, dtype=config.get_dtype())
        elif hasattr(shape, "outputs") and shape.outputs is not None:
            dr.ground_truth = jnp.asarray(shape.outputs, dtype=config.get_dtype())
        else:
            msg = (
                "DataShape must have 'train_outputs' or 'outputs' attribute. "
                "Call test_train_split() first if using train/test split."
            )
            config.LOGGER.error(msg)
            raise ValueError(msg)

        return dr

    @eqx.filter_jit
    def __call__(
        self,
        inputs: Array,
        ops: DifferentialOperators,
        *,
        batch_indices: Array | None = None,
        batch_targets: Array | None = None,
    ) -> Array:
        """Calculate the data residual for a batch of inputs.

        Utilizes `jax.jit` and `jax.vmap` for computational efficiency.
        `batch_targets` is used when it is passed. Otherwise, `batch_indices` selects
        rows of the stored targets. With neither, the stored targets are used when the
        batch has the same length.

        Args:
            inputs: Model inputs. One sample per row.
            ops: A [`DifferentialOperators`][pinnjax.DifferentialOperators] object.
                The prediction is `ops.model` applied to `inputs`.
            batch_indices: Rows of the stored targets for this batch. Default: None.
            batch_targets: Targets for this batch. Default: None.

        Returns:
            ``truth - prediction`` at each input. Non-finite values are replaced with zeros.

        Raises:
            ValueError: If `bind` has not stored targets, the batch does not match the
                stored targets, or the prediction cannot be reshaped to the target shape.

        """
        if self.ground_truth is None:
            msg = "DataResidual must be bound to a DataShape before use. Call bind() first."
            config.LOGGER.error(msg)
            raise ValueError(msg)

        return self._residuals(
            inputs,
            ops,
            batch_indices=batch_indices,
            batch_targets=batch_targets,
        )

    def _residuals(
        self,
        inputs: Array,
        ops: DifferentialOperators,
        *,
        batch_indices: Array | None = None,
        batch_targets: Array | None = None,
    ) -> Array:
        predictions = jax.vmap(ops.model)(inputs)

        if self.prediction_transformation is not None:
            predictions = jax.vmap(self.prediction_transformation)(
                inputs,
                predictions,
            )

        truth = self._targets_for_batch(
            inputs,
            batch_indices,
            batch_targets,
        )

        if predictions.shape != truth.shape:
            if predictions.size == truth.size:
                predictions = predictions.reshape(truth.shape)

            else:
                msg = (
                    f"Prediction shape {predictions.shape} "
                    f"does not match target shape "
                    f"{truth.shape} and cannot be reshaped."
                )

                config.LOGGER.error(msg)
                raise ValueError(msg)

        return jnp.nan_to_num(truth - predictions)

    def _targets_for_batch(
        self,
        inputs: Array,
        batch_indices: Array | None,
        batch_targets: Array | None,
    ) -> Array:
        assert self.ground_truth is not None
        n_truth = int(self.ground_truth.shape[0])
        n_batch = int(inputs.shape[0])

        if batch_targets is not None:
            return batch_targets

        if batch_indices is not None:
            return self.ground_truth[batch_indices]

        if n_batch == n_truth:
            return self.ground_truth

        msg = (
            f"Batch size ({n_batch}) does not match stored targets ({n_truth}). "
            "For mini-batch training, pass batch_targets "
            "or batch_indices (e.g. via Training.batch_size)."
        )
        config.LOGGER.error(msg)
        raise ValueError(msg)

    class EvaluationResidual(Residual):
    """[`Residual`][pinnjax.Residual] whose value is one evaluation metric.

    Built by [`Evaluation.to_residual`][pinnjax.loss.evaluation.Evaluation.to_residual].
    `bind` does not attach a shape.

    Attributes:
        evaluation: The [`Evaluation`][pinnjax.Evaluation] that computes the metrics.
        selection_metric: Key returned from that evaluation.

    Args:
        evaluation: A [`Evaluation`][pinnjax.Evaluation] object.
        selection_metric: Metric name to return.

    """

    def __init__(
        self,
        evaluation: Evaluation,
        selection_metric: str,
    ) -> None:
        """Store the evaluation and the metric name."""
        self.evaluation = evaluation
        self.selection_metric = selection_metric

    def bind(self, shape):
        """Return this residual unchanged.

        Args:
            shape: Ignored. An evaluation residual is not attached to a shape.

        Returns:
            This residual.

        """
        return self

    def __call__(
        self,
        points: Array,
        ops: DifferentialOperators,
    ) -> Array:
        """Return one evaluation metric as a residual value.

        The metric comes from ``evaluation.evaluate`` on ``ops.model``.
        ``points`` are not used. The selected entry is returned with at least
        one dimension.

        Args:
            points: Collocation points. This residual does not use them.
            ops: Operators whose network is evaluated.

        Returns:
            Array: The selected metric, with shape ``(1,)`` when it is a
                scalar.

        Raises:
            ValueError: If ``evaluation.evaluate`` is missing.

        """
        evaluate = self.evaluation.evaluate
        if evaluate is None:
            msg = "EvaluationResidual requires a callable evaluate."
            config.LOGGER.error(msg)
            raise ValueError(msg)
        metrics = evaluate(ops.model)

        return jnp.atleast_1d(metrics[self.selection_metric])

    class BoundaryCondition(Residual):
    """Class container for all Boundary Condition residual as a child class of [`Residual`][pinnjax.Residual].

    Children of BoundaryCondition are expected to have a `_residuals` function with a
        `points` Array and the
        `ops` [`DifferentialOperators`][pinnjax.DifferentialOperators] object.

    This is then called by a `_call_()` function which is JIT compiled using `jax.jit`.
        Therefore, all nested function in its call chain must be JIT compliant.
        It must remain side effect free and only use JAX-traceable operations.

    This class, as with all [`Residual`][pinnjax.Residual] child classes are
        functionally identical, they are only distinguished primarily for
        distinction in results and plotting.

    """

    shape: Shape | None = eqx.field(default=None, static=True)
    normal: Array | None = None

    def __init__(self) -> None:
        """Set ``shape`` and ``normal`` to ``None``."""
        self.shape = None
        self.normal = None

    class RobinHeatFlux(BoundaryCondition, ABC):
    """[`BoundaryCondition`][pinnjax.BoundaryCondition] for conduction, convection, and radiation.

    The residual at each point is

    ``-k dT/dn - h (T - T0) - 5.670374419e-8 * eps * (T**4 - T0**4)``.

    ``T`` is the prediction from [`DifferentialOperators`][pinnjax.DifferentialOperators].
    ``dT/dn`` is the derivative along the outward normal stored by `bind()`.
    ``k``, ``h``, ``eps``, and ``T0`` are taken from ``material``.
    The constant is the Stefan-Boltzmann constant [W/m^2/K^4].
    Non-finite residuals are replaced with zeros.

    Attributes:
        material: A [`ThermalMaterial`][pinnjax.ThermalMaterial] object.
        STEFAN_BOLTZMANN: Stefan-Boltzmann constant [W/m^2/K^4].
        k: Thermal conductivity [W/m-K].
        h: Convection coefficient [W/m^2-K].
        eps: Radiation coefficient.
        T0: Ambient temperature [K].
        T0_4: Fourth power of the ambient temperature.

    Args:
        material: A [`ThermalMaterial`][pinnjax.ThermalMaterial] object containing
            `thermal_conductivity`, `convection_coefficient`, `radiation_coefficient`,
            and `ambient_temperature`.

    Raises:
        ValueError: If `bind()` has not been called before evaluation.

    """

    material: ThermalMaterial = eqx.field(static=True)

    def __init__(self, material: ThermalMaterial) -> None:
        """Store ``material`` and the coefficients used by the residual."""
        super().__init__()
        self.STEFAN_BOLTZMANN = jnp.asarray(
            5.670374419e-8,
            dtype=config.get_dtype(),
        )  # W/m^2/K^4
        self.material = material

        self.k = jnp.asarray(material.thermal_conductivity, dtype=config.get_dtype())
        self.h = jnp.asarray(material.convection_coefficient, dtype=config.get_dtype())
        self.eps = jnp.asarray(material.radiation_coefficient, dtype=config.get_dtype())
        self.T0 = jnp.asarray(material.ambient_temperature, dtype=config.get_dtype())

        self.T0_4 = self.T0**4

    def _residuals(self, points: Array, ops: DifferentialOperators) -> Array:
        n = self.normal
        if n is None:
            msg = "Call bind() before evaluating RobinHeatFlux."
            config.LOGGER.error(msg)
            raise ValueError(msg)

        def residual(u: Array) -> Array:
            normal = n.astype(u.dtype)
            tangent = jnp.zeros_like(u).at[: normal.shape[0]].set(normal)
            T, dTdn = jax.jvp(ops.T, (u,), (tangent,))

            cond = -self.k * dTdn
            conv = self.h * (T - self.T0)
            rad = self.STEFAN_BOLTZMANN * self.eps * (T**4 - self.T0_4)

            return cond - conv - rad

        return jnp.nan_to_num(jax.vmap(residual)(points))

    class GoldakHeatFlux(PartialDifferentialEquation):
    """[`PartialDifferentialEquation`][pinnjax.PartialDifferentialEquation] particularly for the Goldak heat flux residual.

    Since the
        [`goldak_heat_flux_rotate()`][pinnjax.loss.pde.GoldakHeatFlux.goldak_heat_flux_rotate]
        method expects 3-dimensional coordinates, this PartialDifferentialEquation only
        works for 3-dimensional domains. Naturally, given the traversal of the heat
        source, it also only works for time-dependent domains.


    Args:
        material: A [`ThermalMaterial`][pinnjax.ThermalMaterial] object containing `density`,
            `heat_capacity`, and `thermal_conductivity`.
        goldak: A [`Goldak`][pinnjax.Goldak] object.
        path: A [`Path`][pinnjax.Path] object containing `.offset_in_local_frame()`
            and `.get_active()`.

    """

    material: ThermalMaterial = eqx.field(static=True)
    goldak: Goldak = eqx.field(static=True)
    path: Path = eqx.field(static=True)

    def __init__(
        self,
        material: ThermalMaterial,
        goldak: Goldak,
        path: Path,
    ) -> None:
        """Store the material, Goldak source, and path, and precompute the flux coefficients."""
        self.material = material
        self.goldak = goldak
        self.path = path

        g = goldak
        # precompute as much as possible.
        a_f = jnp.asarray(g.front_ellipsoid_length, dtype=config.get_dtype())
        a_r = jnp.asarray(g.rear_ellipsoid_length, dtype=config.get_dtype())
        b = jnp.asarray(g.width / 2, dtype=config.get_dtype())
        c = jnp.asarray(g.depth, dtype=config.get_dtype())

        f_f = jnp.asarray(
            g.front_ellipsoid_distribution_factor,
            dtype=config.get_dtype(),
        )
        r_f = jnp.asarray(
            g.rear_ellipsoid_distribution_factor,
            dtype=config.get_dtype(),
        )

        self.front_coefficient = jnp.asarray(
            (6.0 * jnp.sqrt(3.0) * f_f * g.power * g.efficiency) / (a_f * b * c * (jnp.pi**1.5)),
            dtype=config.get_dtype(),
        )

        self.rear_coefficient = jnp.asarray(
            (6.0 * jnp.sqrt(3.0) * r_f * g.power * g.efficiency) / (a_r * b * c * (jnp.pi**1.5)),
            dtype=config.get_dtype(),
        )

        self.rhoCp = jnp.asarray(
            material.density * material.heat_capacity,
            dtype=config.get_dtype(),
        )
        self.thermal_conductivity = jnp.asarray(
            material.thermal_conductivity,
            dtype=config.get_dtype(),
        )

        self.inv_af2 = 1.0 / (a_f * a_f)
        self.inv_ar2 = 1.0 / (a_r * a_r)
        self.inv_b2 = 1.0 / (b * b)
        self.inv_c2 = 1.0 / (c * c)

    def goldak_heat_flux_rotate(self, u: Array) -> Array:
        r"""Goldak heat flux \( Q_{\text{goldak}}(\mathbf{x}, t)\) of point `u`.

        Accounts for the rotation of the heat source along the path.
        Used internally in the [`__call__`][pinnjax.loss.pde.GoldakHeatFlux.__call__] method.

        The point `[dx, dy, dz]` is first transformed into the rotated coordinates
        `[xi, eta, zeta]` using the inverse of the local rotation at `u`. The flux is
        then computed as:

        \[
            Q_{\text{goldak}} =
            \begin{cases}
            \displaystyle
            \frac{6 \sqrt{3} f_f P \eta}{a_f b c \pi^{3/2}}
            \exp\!\left(
                -3 \left(
                    \frac{\xi^2}{a_f^2} + \frac{\eta^2}{b^2} + \frac{\zeta^2}{c^2}
                \right)
            \right), & \xi \geq 0 \\[2ex]
            \displaystyle
            \frac{6 \sqrt{3} f_r P \eta}{a_r b c \pi^{3/2}}
            \exp\!\left(
                -3 \left(
                    \frac{\xi^2}{a_r^2} + \frac{\eta^2}{b^2} + \frac{\zeta^2}{c^2}
                \right)
            \right), & \xi < 0
            \end{cases}
        \]

        Args:
            u: single point collocation sample in the form [x, y, z, t]

        Returns:
            A JAX Array of the Goldak heat flux at point `u`.

        """
        xi, eta, zeta = self.path.offset_in_local_frame(u)
        mask = jnp.asarray(self.path.get_active(u), dtype=xi.dtype)

        exp_front = jnp.exp(
            -3.0
            * (
                (xi * xi) * self.inv_af2
                + (eta * eta) * self.inv_b2
                + (zeta * zeta) * self.inv_c2
            ),
        )
        exp_rear = jnp.exp(
            -3.0
            * (
                (xi * xi) * self.inv_ar2
                + (eta * eta) * self.inv_b2
                + (zeta * zeta) * self.inv_c2
            ),
        )
        q_front = self.front_coefficient * exp_front * mask
        q_rear = self.rear_coefficient * exp_rear * mask
        return jnp.where(xi >= 0.0, q_front, q_rear)

    def goldak_heat_flux(self, u: Array) -> Array:
        r"""Goldak heat flux \( Q_{\text{goldak}}(\mathbf{x}, t)\) of point \(u\).

        Deprecated in favor of
            [`goldak_heat_flux_rotate`][pinnjax.loss.pde.GoldakHeatFlux.goldak_heat_flux_rotate].

        \[
            Q_{\text{goldak}} =
            \begin{cases}
            \displaystyle
            \frac{6 \sqrt{3} f_f P \eta}{a_f b c \pi^{3/2}}
            \exp\!\left(
                -3 \left(
                    \frac{x^2}{a_f^2} + \frac{y^2}{b^2} + \frac{z^2}{c^2}
                \right)
            \right), & x \geq 0 \\[2ex]
            \displaystyle
            \frac{6 \sqrt{3} f_r P \eta}{a_r b c \pi^{3/2}}
            \exp\!\left(
                -3 \left(
                    \frac{x^2}{a_r^2} + \frac{y^2}{b^2} + \frac{z^2}{c^2}
                \right)
            \right), & x < 0
            \end{cases}
        \]


        Args:
            u: single point collocation sample in the form [x, y, z, t]

        Returns:
            A JAX Array of the Goldak heat flux at point `u`.

        """
        dx, dy, dz = self.path.offset(u)
        t = u[-1]
        idx = cast("Any", self.path).get_segment_index(t)

        def _front(_: None) -> Array:
            exp_front = jnp.exp(
                -3.0
                * ((dx * dx) * self.inv_af2 + (dy * dy) * self.inv_b2 + (dz * dz) * self.inv_c2),
            )

            mask = cast("Any", self.path).active[idx].astype(exp_front.dtype)
            return self.front_coefficient * exp_front * mask

        def _rear(_: None) -> Array:
            exp_rear = jnp.exp(
                -3.0
                * ((dx * dx) * self.inv_ar2 + (dy * dy) * self.inv_b2 + (dz * dz) * self.inv_c2),
            )

            mask = cast("Any", self.path).active[idx].astype(exp_rear.dtype)
            return self.rear_coefficient * exp_rear * mask

        # jax.jit compilation doesn't like conditionals (if), thus this format
        # is necessary to maintain jit-compatibility.
        return lax.cond(dx >= 0.0, _front, _rear, operand=None)

    def _residual(self, u: Array, ops: DifferentialOperators) -> Array:
        r"""Calculate the transient heat conduction residual at one point, with a Goldak heat source.

        \[
            R_{\text{PDE}} = \rho C_p \frac{\partial T}{\partial t}
            - k \nabla^2 T
            - Q_{\text{goldak}}(\mathbf{x}, t)
        \]

        Args:
            u: A JAX Array of one collocation point sample.
            ops: A [`DifferentialOperators`][pinnjax.DifferentialOperators] object.

        Returns:
            JAX Array of transient heat conduction residual with Goldak heat source
                for a single point `u`.

        """
        dTdt, lap = ops.evolution_and_laplacian(u)
        return (
                self.rhoCp * dTdt
                - self.thermal_conductivity * lap
                - self.goldak_heat_flux_rotate(u)
        )

    # noinspection PyArgumentList
    def _residuals(self, points: Array, ops: DifferentialOperators) -> Array:
        """`jax.vmap` automatic vectorization for all collocation points.

        Args:
            points: A JAX Array of all collocation point samples.
            ops: A [`DifferentialOperators`][pinnjax.DifferentialOperators] object.

        Returns:
            JAX Array of transient heat conduction residual with Goldak heat source
                for all collocation point samples.

        """
        vm = jax.vmap(lambda u: self._residual(u, ops), in_axes=0)
        return jnp.nan_to_num(vm(points))

    @eqx.filter_jit
    def __call__(self, points: Array, ops: DifferentialOperators) -> Array:
        """`jax.jit` compliant call to full residual calculation.

        Args:
            points: A JAX Array of all collocation point samples.
            ops: A [`DifferentialOperators`][pinnjax.DifferentialOperators] object.

        Returns:
            JAX Array of the transient heat conduction residual with Goldak heat source
                for all collocation points.

        """
        return self._residuals(points, ops)
