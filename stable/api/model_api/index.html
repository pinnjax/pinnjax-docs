    class NeuralNetwork(eqx.Module):
    input_dim: int | None = eqx.field(static=True)

    def __init__(self):
        pass

    def __call__(self, inputs):
        raise NotImplementedError

    def single_point(self, u: Array) -> Array:
        # ensure shape (d,)
        u = jnp.atleast_1d(u)
        if u.ndim > 1:
            u = jnp.squeeze(u, axis=0)
        return self(u)

    def to_meta(self) -> dict:
        raise NotImplementedError

    class FNN(NeuralNetwork, ABC):
    layers: list[eqx.nn.Linear]
    domain: pinnjax.Domain = eqx.field(static=True)
    topology: tuple[int, ...] = eqx.field(static=True)

    activation: pinnjax.Activation = eqx.field(static=True)
    activation_function: Callable = eqx.field(static=True)
    initialization: pinnjax.Initialization = eqx.field(static=True)

    input_transformation: Callable | None = eqx.field(static=True)
    output_transformation: Callable | None = eqx.field(static=True)
    hard_constraints: tuple[Callable, ...] = eqx.field(static=True)

    spatial_dims: int | None = eqx.field(static=True)
    time_index: int | None = eqx.field(static=True)

    def __init__(
            self,
            domain: pinnjax.Domain | None,
            topology: tuple[int, ...],
            activation: pinnjax.Activation,
            *,
            initialization: pinnjax.Initialization = None,
            optimization: pinnjax.Optimization = None,
    ):
        super().__init__()
        keys = jax.random.split(jax.random.key(_CONFIG.SEED), len(topology) - 1)
        self.topology = topology
        self.input_dim = self.topology[0]
        self.activation = activation
        self.activation_function = functools.partial(
            activation.act_,
            *activation.args,
            **activation.kwargs,
        )
        self.domain = domain

        if optimization is None:
            optimization = pinnjax.Optimization()

        self.input_transformation = (
                optimization.input_transformation or IdentityInputTransform()
        )
        self.output_transformation = (
                optimization.output_transformation or IdentityOutputTransformation()
        )
        hc = getattr(optimization, "hard_constraint", None)
        if hc is None:
            self.hard_constraints = (IdentityHardConstraint(),)
        elif isinstance(hc, tuple):
            self.hard_constraints = tuple(hc)
        else:
            self.hard_constraints = (hc,)

        self.layers = [
            eqx.nn.Linear(self.topology[i], self.topology[i + 1], key=keys[i])
            for i in range(len(self.topology) - 1)
        ]

        self.initialization = initialization or pinnjax.Initialization()

        for i, (layer, key) in enumerate(zip(self.layers, keys, strict=False)):
            weight_key, bias_key = jax.random.split(key, 2)
            if self.initialization.weights is not None:
                weight = self.initialization.weights(
                    weight_key,
                    layer.weight.shape,
                    layer.weight.dtype,
                )
                layer = eqx.tree_at(lambda L: L.weight, layer, weight)
            if self.initialization.bias is not None:
                bias = self.initialization.bias(
                    bias_key,
                    layer.bias.shape,
                    layer.bias.dtype,
                )
                layer = eqx.tree_at(lambda L: L.bias, layer, bias)
            self.layers[i] = layer

        spatial_dims = len(self.domain.min) or None
        temporal_dims = 1 if hasattr(self.domain, "times") is not None else 0

        if self.domain is not None:
            if (spatial_dims + temporal_dims) != self.topology[0]:
                msg = (
                    f"Mismatch in Input Size between Domain ({spatial_dims}) "
                    f"and Topology ({temporal_dims})."
                )
                _CONFIG.logger.error(msg)
                raise InputError(msg)

            self.spatial_dims = spatial_dims
            if temporal_dims == 1:
                self.time_index = -1
            else:
                self.time_index = None

    @eqx.filter_jit
    def __call__(self, inputs: Array) -> Array:
        x = self.input_transformation(inputs)
        for layer in self.layers[:-1]:
            x = self.activation_function(layer(x))
        x = self.layers[-1](x)
        x = self.output_transformation(x)
        for hc in self.hard_constraints:
            x = hc(inputs, x)
        return x

    def to_meta(self) -> dict:
        t = getattr(self.domain, "times", None)
        times_meta = (
            None
            if t is None
            else {
                "min" : float(t.min),
                "max" : float(t.max),
                "step": float(t.step),
            }
        )
        return {
            "model_class": "FNN",
            "topology"   : list(self.topology),
            "activation" : {
                "func"  : self.activation.act_.__name__,
                "args"  : list(self.activation.args),
                "kwargs": dict(self.activation.kwargs),
            },
            "domain"     : {
                "min"  : list(self.domain.min) or None,
                "max"  : list(self.domain.max) or None,
                "times": times_meta,
            },
        }

    class NormalizationBetweenNeg1and1(InputTransformation):
    def __init__(self, domain: pinnjax.Domain):
        mins = jnp.asarray(domain.min, dtype=_CONFIG.DTYPE)
        maxs = jnp.asarray(domain.max, dtype=_CONFIG.DTYPE)

        # append time if provided
        if domain.times is not None:
            mins = jnp.concatenate(
                [mins, jnp.asarray([domain.times.min], dtype=_CONFIG.DTYPE)],
            )
            maxs = jnp.concatenate(
                [maxs, jnp.asarray([domain.times.max], dtype=_CONFIG.DTYPE)],
            )

        self.min = mins
        self.max = maxs
        self.range = self.max - self.min  # shape (d,)

    def __call__(self, inputs: Array) -> Array:
        mins = self.min.astype(inputs.dtype)
        rng = self.range.astype(inputs.dtype)
        rng = jnp.where(rng == 0, _CONFIG.EPS.astype(rng.dtype), rng)
        return (2 * ((inputs - mins) / rng)) - 1

    class HardScalarInitialConstraint(HardConstraint):
    def __init__(
            self,
            times: pinnjax.Time,
            initial_condition: float | tuple,
    ):
        self.times = times
        self.initial_condition = jnp.asarray(initial_condition, dtype=_CONFIG.DTYPE)
        phi95 = jnp.asarray(0.95, dtype=_CONFIG.DTYPE)
        rng = jnp.asarray(self.times.range(), dtype=_CONFIG.DTYPE)
        rng = jnp.where(jnp.abs(rng) < _CONFIG.EPS, _CONFIG.EPS, rng)
        self.tau = jnp.asarray(-rng / jnp.log1p(-phi95), dtype=_CONFIG.DTYPE)

    def __call__(self, inputs, output):
        if inputs.ndim == 1:
            val = self.point(inputs, jnp.squeeze(output))
            return jnp.atleast_1d(val)
        # multiple inputs
        fn = jax.vmap(self.point, in_axes=(0, 0))
        batched = fn(inputs, output)

        return batched[:, None] if batched.ndim == 1 else batched

    def point(self, inputs: jnp.ndarray, output: jnp.ndarray):
        t = inputs[-1]  # scalar
        tau = self.tau.astype(inputs.dtype)
        phi = jnp.asarray(1.0, dtype=inputs.dtype) - jnp.exp(-t / tau)  # scalar

        ic = self.initial_condition

        # --- validate initial_condition vs output ---
        if ic.ndim > 1:
            msg = f"initial_condition must be scalar or 1D, got shape {ic.shape}"
            _CONFIG.logger.error(msg)
            raise ValueError(msg)
        if output.ndim == 0:
            if ic.ndim == 1 and ic.size != 1:
                msg = (
                    f"Scalar output but initial_condition has size {ic.size} - "
                    f"Consider using 'HardArrayInitialConstraint instead."
                )
                _CONFIG.logger.error(msg)
                raise ValueError(msg)
        else:
            C = output.shape[-1]
            if ic.ndim == 1 and ic.size not in (1, C):
                msg = f"initial_condition length {ic.size} does not match outputs {C}."
                _CONFIG.logger.error(msg)
                raise ValueError(msg)
        # ---------------------------------------------

        ic = jnp.asarray(ic, dtype=output.dtype)
        return ic + phi * (output - ic)

    class HardArrayInitialConstraint(HardConstraint):
    def __init__(
            self,
            times: pinnjax.Time,
            initial_condition: jnp.ndarray | Callable[[jnp.ndarray], jnp.ndarray],
    ):
        self.times = times
        self.initial_condition = initial_condition
        phi95 = jnp.asarray(0.95, dtype=_CONFIG.DTYPE)
        rng = jnp.asarray(self.times.range(), dtype=_CONFIG.DTYPE)
        rng = jnp.where(jnp.abs(rng) < _CONFIG.EPS, _CONFIG.EPS, rng)
        self.tau = jnp.asarray(-rng / jnp.log1p(-phi95), dtype=_CONFIG.DTYPE)

    def __call__(self, inputs, output):
        if inputs.ndim == 1:
            return jnp.atleast_1d(self.point(inputs, jnp.squeeze(output)))
        fn = jax.vmap(self.point, in_axes=(0, 0))
        return fn(inputs, output)

    def point(self, inputs: jnp.ndarray, output: jnp.ndarray):
        t = inputs[-1]
        tau = self.tau.astype(inputs.dtype)
        phi = jnp.asarray(1.0, dtype=inputs.dtype) - jnp.exp(-t / tau)  # scalar

        # --- get initial condition ---
        if callable(self.initial_condition):
            ic = self.initial_condition(inputs)
        else:
            ic = self.initial_condition  # fixed array
        ic = jnp.asarray(ic, dtype=output.dtype)
        # --------------------------------

        if ic.shape != output.shape:
            msg = f"IC shape {ic.shape} != output shape {output.shape}"
            _CONFIG.logger.error(msg)
            raise ValueError(msg)

        return ic + phi * (output - ic)

    class HardDirichletConstraint(HardConstraint):
    # TODO: Needs some work.
    def __init__(
            self,
            domain: pinnjax.Domain,  # .min, .max
            condition: float,
            side: list[tuple[int, ...]] | tuple[int, ...],
            # e.g. (0,0,-1) or [(1,0,0),(-1,0,0)]
    ):
        # currently only works with the output of the model is scalar.
        # might be too programmatic to be considered valid.

        # also only worked for the -x, -y, or -z direction.

        self.domain = domain
        self.min = jnp.asarray(domain.min, dtype=_CONFIG.DTYPE)
        self.max = jnp.asarray(domain.max, dtype=_CONFIG.DTYPE)
        self.condition = jnp.asarray(condition, dtype=_CONFIG.DTYPE)

        self.idx = next(i for i, x in enumerate(side) if x)
        rng = jnp.asarray(domain.range(self.idx), dtype=_CONFIG.DTYPE)
        rng = jnp.where(jnp.abs(rng) < _CONFIG.EPS, _CONFIG.EPS, rng)
        phi95 = jnp.asarray(0.95, dtype=_CONFIG.DTYPE)
        self.tau = jnp.asarray(-rng / jnp.log1p(-phi95), dtype=_CONFIG.DTYPE)

    def __call__(self, inputs, output):
        val = self.point(inputs, jnp.squeeze(output))
        return jnp.atleast_1d(val)

    def point(self, inputs: jnp.ndarray, output: jnp.ndarray):
        q = inputs[self.idx]
        tau = self.tau.astype(inputs.dtype)
        phi = jnp.asarray(1.0, dtype=inputs.dtype) - jnp.exp(-q / tau)  # scalar

        bc = self.condition
        bc = jnp.asarray(bc, dtype=output.dtype)

        x = bc + phi * (output - bc)

        return x
