    class AdaptiveLossWeights(eqx.Module):
    """Gradient-based adaptive weighting over losses.

    Implements the method by Wang et al. for adaptively weighting losses
    during PINN training.

    """

    weights: jnp.ndarray
    mode: str = eqx.field(static=True)  # 'class' or 'type'
    alpha: float = eqx.field(static=True)
    verbosity: bool = eqx.field(static=True)

    def __init__(
        self,
        n_classes: int,
        mode: str,
        alpha: float = 0.9,
        *,
        verbosity: None | int = None,
    ) -> None:
        """Initialize an AdaptiveLossWeights instance.

        Args:
            n_classes: Number of classes or types to compute weights for.
            mode: 'class' or 'type' indicating which loss vector to use.
            alpha: Exponential smoothing factor for updating weights.
            verbosity: Optional verbosity flag; enables logging if not None.

        """
        self.alpha = alpha
        self.mode = mode
        self.weights = jnp.ones(n_classes, dtype=_CONFIG.DTYPE)
        self.verbosity = verbosity is not None

    def update_weights(
        self,
        model: pinnjax.NeuralNetwork,
        _geometry: pinnjax.ComplexShape,
        scale: Array | None = None,
    ) -> AdaptiveLossWeights:
        """Update the adaptive weights based on gradient norms of each loss.

        Args:
            model: The neural network model.
            _geometry: ComplexShape object containing the loss functions.
            scale: Optional array to scale losses before computing gradients.

        Returns:
            AdaptiveLossWeights: Updated object with new weights.

        """

        def _losses_vec(m: pinnjax.NeuralNetwork) -> Array:
            _, by_class, by_type = _geometry.residual_and_losses(
                m,
                pinnjax.DifferentialOperators(m),
            )
            return by_class if self.mode == "class" else by_type

        losses = _losses_vec(model)

        if scale is None:
            scale = jnp.ones_like(losses, dtype=_CONFIG.DTYPE)

        losses = losses * scale

        threshold = 1e-18
        non_zero_mask = losses > threshold

        if not bool(jnp.any(non_zero_mask)):
            return self

        grad_norms = []
        for k in range(losses.shape[0]):
            if not bool(non_zero_mask[k]):
                grad_norms.append(0.0)
                continue

            def _loss_k(m: pinnjax.NeuralNetwork, idx: int = k) -> Array:
                return scale[idx] * _losses_vec(m)[idx]

            grads = jax.grad(_loss_k)(model)
            leaves = [g for g in jax.tree_util.tree_leaves(grads) if g is not None]
            if len(leaves) == 0:
                grad_norms.append(0.0)
            else:
                flat = jnp.concatenate(
                    [g.reshape(-1) for g in leaves], dtype=_CONFIG.DTYPE
                )
                grad_norms.append(jnp.linalg.norm(flat))

        grad_norms = jnp.asarray(grad_norms, dtype=_CONFIG.DTYPE)
        active = grad_norms[non_zero_mask]
        total_g = jnp.sum(jnp.where(jnp.isfinite(active), active, 0.0))
        new_w_active = total_g / (active + _CONFIG.EPS)
        new_w_active = jnp.clip(
            new_w_active,
            0.0,
            10000.0,  # increased to 10000 from 100
            # decreased lower bound from 1.0 to 0.0
        )

        new_w = self.weights.copy()
        new_w = new_w.at[non_zero_mask].set(new_w_active)
        new_w = new_w.at[~non_zero_mask].set(1.0)

        updated_w = self.alpha * self.weights + (1.0 - self.alpha) * new_w

        if self.verbosity:
            _CONFIG.logger.info(
                "Updated %s Weights: %s",
                self.mode.title(),
                ", ".join([f"{float(w):.3f}" for w in updated_w.tolist()]),
            )

        return eqx.tree_at(lambda x: x.weights, self, updated_w)

    def train(
    model: pinnjax.NeuralNetwork,
    geometry: pinnjax.ComplexShape,
    training: pinnjax.Training,
    *,
    initial_loss_weights: Iterable | None = None,
    # Ground Truth
    ground_truth: str | Path | None = None,
    csv_coordinate_column_order: list | None = None,
    csv_truth_column: list | int = -1,
    ground_truth_slice: dict[str, float] | None = None,
    ground_truth_regularity: int = 10,
    ground_truth_full_grid: bool = True,
    # print
    verbosity: None | int = 10,
    print_mode: str = "class",
) -> pinnjax.NeuralNetwork:
    """Train a PINN model with optional adaptive loss weighting and ground truth.

    Args:

        model: Neural network model.
        geometry: ComplexShape defining residuals and loss functions.
        training: Training configuration object.
        initial_loss_weights: Optional iterable to override initial loss weights.
        ground_truth: Optional path to CSV with ground truth data.
        csv_coordinate_column_order: Optional column reordering for input coordinates.
        csv_truth_column: Column(s) in CSV corresponding to ground truth values.
        ground_truth_slice: Dictionary to fix specific coordinates during evaluation.
        ground_truth_regularity: Frequency (in epochs) for evaluating ground truth.
        ground_truth_full_grid: Whether the ComparisonGrid requires a full cartesian grid.
        verbosity: Logging frequency or None to suppress logs.
        print_mode: 'class' or 'type' to select which loss labels to print.

    Returns:
        pinnjax.NeuralNetwork: The best model observed during training.

    """
    if verbosity is not None:
        _CONFIG.logger.info("%s", training)

    if training.epochs <= 0:
        _CONFIG.logger.warning(
            "Epochs set to %d. No Training performed.",
            training.epochs,
        )
        return model

    is_lbfgs = training.optimizer.opt_.__name__.lower() == "lbfgs"
    is_optax = training.optimizer.opt_.__module__.startswith("optax")

    epochs = training.epochs
    if not is_optax and is_lbfgs:
        opt = training.optimizer.opt_(
            fun=lambda params: _loss(params, current_weights)[0],
            **training.optimizer.kwargs,
        )
        opt_state = None
    else:
        opt = training.optimizer.opt_(
            *training.optimizer.args,
            **training.optimizer.kwargs,
        )
        opt_state = opt.init(eqx.filter(model, eqx.is_inexact_array))

    n_classes = max(geometry.kinds_class) + 1
    n_types = max(geometry.kinds_type) + 1

    n_target = n_classes if training.train_mode == "class" else n_types

    use_adaptive_weights = training.use_adaptive_weights
    if is_lbfgs is True and use_adaptive_weights is True:
        msg = (
            "LBFGS is incompatible with Adaptive Weights.\n"
            'Press "Y" to continue without Adaptive Weights.\n'
            'Press "N" to stop training.'
        )
        user_input = input(msg)
        if user_input.casefold() == "y":
            # unfortunately, this change will not be reflected in the Training object
            # this must be done because the Training object is frozen.
            use_adaptive_weights = False
        else:
            sys.exit()

    # allow for the ability to dictate how important a given residual is to the
    # model training. defaults to 1.
    if training.priority is None:
        priority = jnp.ones(n_target, dtype=_CONFIG.DTYPE)
    else:
        priority = jnp.asarray(training.priority, dtype=_CONFIG.DTYPE)
        if priority.shape != (n_target,):
            msg = f"'priority' must have shape ({n_target},), got {priority.shape}"
            _CONFIG.logger.error(msg)
            raise ValueError(msg)

    weight_manager = (
        AdaptiveLossWeights(
            n_target,
            training.train_mode,
            alpha=training.adaptive_alpha,
            verbosity=verbosity,
        )
        if use_adaptive_weights
        else None
    )

    # GROUND TRUTH
    # ----------------------------------------------------------------------------------
    truth_grid: jnp.ndarray | None = None
    comparison_grid: pinnjax.ComparisonGrid | None = None
    closest_time: Array | float | None = None
    if ground_truth is not None:
        path = Path(ground_truth)
        if path.suffix.lower() != ".csv":
            msg = f"Expected a CSV file, got {path.suffix}"
            _CONFIG.logger.error(msg)
            raise ValueError(msg)
        truth = pl.read_csv(path).to_numpy().astype(float)

        coordinates = jnp.array(
            jnp.delete(truth, csv_truth_column, axis=1), copy=False, dtype=_CONFIG.DTYPE
        )
        if csv_coordinate_column_order is not None:
            coordinates = jnp.array(
                coordinates[:, csv_coordinate_column_order],
                copy=False,
                dtype=_CONFIG.DTYPE,
            )
        else:
            coordinates = jnp.array(coordinates, copy=False, dtype=_CONFIG.DTYPE)

        truth_values = jnp.array(
            truth[:, csv_truth_column], copy=False, dtype=_CONFIG.DTYPE
        )

        if coordinates.shape[1] != model.input_dim:
            msg = (
                f"Shape of CSV file [Columns: {coordinates.shape[1]}] is not "
                f"consistent with prediction inputs [Columns: {model.input_dim}] ",
            )
            _CONFIG.logger.error(msg)
            raise ValueError(msg)

        spatial_dimensions = len(model.domain.min)
        if model.domain.times is not None:
            times = coordinates[:, spatial_dimensions]  # column after the last spatial
            spatial_coordinates = coordinates[:, :spatial_dimensions]  # [x, y, z]
        else:
            times = None
            spatial_coordinates = coordinates

        comparison_grid = pinnjax.ComparisonGrid(
            domain=model.domain,
            coords=spatial_coordinates,
            fixed_coordinates=ground_truth_slice,
        )

        mask = slice(None)
        closest_time = None
        if times is not None:
            comparison_time = float(jnp.median(jnp.unique(times)))
            closest_time = jnp.asarray(
                times[jnp.argmin(jnp.abs(times - comparison_time))], dtype=_CONFIG.DTYPE
            )

            mask = jnp.isclose(times, closest_time, rtol=0.0, atol=1e-12)

        masked_spatial_coordinates = spatial_coordinates[mask, :]
        vals = truth_values[mask]

        truth_grid = comparison_grid.scatter(
            masked_spatial_coordinates,
            vals,
            require_full=ground_truth_full_grid,
        )

    # ----------------------------------------------------------------------------------

    if training.balance_magnitudes:
        magnitude = _initial_weights_from_magnitudes(
            model,
            geometry,
            mode=training.train_mode,
            magnitude_clip=training.magnitude_clip,
        )
    else:
        magnitude = jnp.ones(n_target, dtype=_CONFIG.DTYPE)

    @eqx.filter_jit
    def _loss(
        _model: pinnjax.NeuralNetwork,
        _weights: Array,
    ) -> tuple[Array, tuple[Array, Array, Array]]:
        _total_raw, _by_class, _by_type = geometry.residual_and_losses(
            _model,
            pinnjax.DifferentialOperators(_model),
        )
        eff = _weights * priority * magnitude
        if training.train_mode == "class":
            total = jnp.dot(eff, _by_class)
        else:
            total = jnp.dot(eff, _by_type)
        return total * training.loss_scale, (_total_raw, _by_class, _by_type)

    @eqx.filter_jit
    def step(
        _model: pinnjax.NeuralNetwork,
        _opt_state: ArrayTree,
        _weights: Array,
    ) -> tuple[
        pinnjax.NeuralNetwork,
        pinnjax.NeuralNetwork,
        ArrayTree,
        Array,
        Array,
        Array,
        Array,
    ]:
        _old_model = _model
        (_loss_val, (_total_raw, _by_class, _by_type)), _grads = (
            eqx.filter_value_and_grad(lambda m: _loss(m, _weights), has_aux=True)(
                _model,
            )
        )

        if is_lbfgs:
            if is_optax:
                updates, _opt_state = opt.update(
                    _grads,
                    _opt_state,
                    params=eqx.filter(_model, eqx.is_array),
                    value=_loss_val,
                    grad=_grads,
                    value_fn=lambda m: _loss(m, current_weights)[0],
                )

            else:
                out = opt.run(eqx.filter(_model, eqx.is_inexact_array))
                _model = eqx.tree_at(lambda m: m, _model, out.params)
                return (
                    _model,
                    _old_model,
                    _opt_state,
                    _loss_val,
                    _total_raw,
                    _by_class,
                    _by_type,
                )

        else:
            updates, _opt_state = opt.update(
                _grads,
                _opt_state,
                eqx.filter(_model, eqx.is_array),
            )

        _model = eqx.apply_updates(_model, updates)

        return (
            _model,
            _old_model,
            _opt_state,
            _loss_val,
            _total_raw,
            _by_class,
            _by_type,
        )

    best_model = model
    best_loss = jnp.inf
    best_epoch = 0
    last_epoch = training.epochs
    epochs_since_improvement = 0

    if initial_loss_weights is not None and len(list(initial_loss_weights)) != 0:
        if len(list(initial_loss_weights)) == n_target:
            current_weights = jnp.asarray(initial_loss_weights, dtype=_CONFIG.DTYPE)
        else:
            msg = (
                f"Inputted Initial Weights: {initial_loss_weights} "
                f"don't have the same number of indices as loss values: {n_target}",
            )
            _CONFIG.logger.error(msg)
            raise ValueError(msg)
    else:
        current_weights = jnp.ones(n_target, dtype=_CONFIG.DTYPE)

    if any(d.platform == "gpu" for d in jax.devices()) and not _CONFIG.FORCE_CPU:
        test_arr = jnp.array([1], dtype=_CONFIG.DTYPE)
        current_device = test_arr.device
        gpu_device = jax.devices("gpu")[0]
        if current_device != gpu_device:
            if verbosity:
                _CONFIG.logger.info(" - SWITCHING TO GPU - ")
            # Switch JAX to use GPU
            jax.config.update("jax_default_device", "gpu")

            # Auto-move JAX arrays to GPU
            geometry = _move_geometry_to_device(geometry, jax.devices("gpu")[0])
            model = _move_model_to_device(model, jax.devices("gpu")[0])

    scale = jnp.asarray(
        [m * p for m, p in zip(magnitude, priority, strict=True)], dtype=_CONFIG.DTYPE
    )
    for i in range(epochs):
        if (
            use_adaptive_weights
            and i % training.update_weights_every == 0
            and i > 0
            and i >= training.update_weights_after
        ) or (use_adaptive_weights and i == training.update_weights_after):
            weight_manager = weight_manager.update_weights(model, geometry, scale)

        if use_adaptive_weights:
            current_weights = weight_manager.weights

        model, old_model, opt_state, loss_val, total_raw, by_class, by_type = step(
            model,
            opt_state,
            current_weights,
        )

        # TODO(@mjryande): maybe recalculate the losses with the new model
        #   and save that, or something. Currently it shows the previous loss, not the
        #   loss the model achieved. - just what it was updated using.

        current = float(total_raw)
        if best_loss == jnp.inf:  # skip the first epoch to avoid division by infinity
            best_loss = current
            # TODO(@mjryande): this fixes the issue with jaxopt.LBFGS, but it's still
            #   possible the wrong. So, this change means if a second training loop
            #   is worse, it will be overwritten.
            best_model = model
            best_epoch = i + 1
            epochs_since_improvement = 0
        else:
            if current < best_loss:
                improvement = (best_loss - current) / abs(best_loss)
                best_loss = current
                best_model = old_model  # model which achieved the loss (before update)
                best_epoch = i + 1
                if improvement > training.rel_tolerance:
                    # return the best model, rather than just the last model
                    epochs_since_improvement = 0
                else:
                    epochs_since_improvement += 1
            else:
                epochs_since_improvement += 1

            if (
                training.early_stopping
                and epochs_since_improvement >= training.patience
            ):
                last_epoch = i + 1
                if verbosity is not None:
                    _CONFIG.logger.info(
                        "Early Stopping: Epoch %d\n"
                        "No Improvement greater than %.3f%% in %d epochs.",
                        i + 1,
                        training.rel_tolerance * 100,
                        training.patience,
                    )
                break

        # assign losses to loss_dictionary:
        for key, val in zip(
            geometry.loss_dictionary["class"].keys(),
            by_class,
            strict=True,
        ):
            geometry.loss_dictionary["class"][key].append(float(val))

        for key, val in zip(
            geometry.loss_dictionary["type"].keys(),
            by_type,
            strict=True,
        ):
            geometry.loss_dictionary["type"][key].append(float(val))

        if verbosity is not None and (
            (i + 1) % verbosity == 0 or i == training.epochs - 1 or i == 0
        ):
            labels = getattr(
                geometry,
                "labels_type" if print_mode == "type" else "labels_class",
                tuple(
                    str(i)
                    for i in range(n_types if print_mode == "type" else n_classes)
                ),
            )
            values = by_type if print_mode == "type" else by_class
            parts = " ".join(
                f"{lbl}={float(v):.4e}"
                for lbl, v in zip(labels, list(values), strict=True)
            )

            _CONFIG.logger.info(
                "Epoch: %d | weighted total: %.4e raw total: %.4e | %s",
                i + 1,
                float(loss_val),
                float(total_raw),
                parts,
            )
        # GROUND TRUTH
        if ground_truth is not None and (
            (i + 1) % ground_truth_regularity == 0 or i in [0, training.epochs - 1]
        ):
            rmse, relative_l2, r2_score = _compare_ground_truth(
                model=model,
                grid=comparison_grid,
                truth_grid=truth_grid,
                comparison_time=closest_time,
            )
            if verbosity is not None:
                _CONFIG.logger.info(
                    "Epoch: %d | Ground Truth: RMSE: %.4e - "
                    "[Relative L2]: %.4g - [R2]: %.4g",
                    i + 1,
                    rmse,
                    relative_l2,
                    r2_score,
                )

    # Wrap-Up
    (loss_val, (total_raw, by_class, by_type)), _ = eqx.filter_value_and_grad(
        lambda m: _loss(m, current_weights),
        has_aux=True,
    )(best_model)

    if verbosity is not None:
        labels = getattr(
            geometry,
            "labels_type" if print_mode == "type" else "labels_class",
            tuple(
                str(i) for i in range(n_types if print_mode == "type" else n_classes)
            ),
        )
        values = by_type if print_mode == "type" else by_class
        parts = " ".join(
            f"{lbl}={float(v):.4e}" for lbl, v in zip(labels, list(values), strict=True)
        )

        _CONFIG.logger.info(
            "Final | weighted total: %.4e raw total: %.4e | %s",
            float(loss_val),
            float(total_raw),
            parts,
        )

        if ground_truth is not None:
            rmse, relative_l2, r2_score = _compare_ground_truth(
                model=best_model,
                grid=comparison_grid,
                truth_grid=truth_grid,
                comparison_time=closest_time,
            )
            _CONFIG.logger.info(
                "Final  | Ground Truth: RMSE: %.4e - [Relative L2]: %.4g - [R2]: %.4g",
                rmse,
                relative_l2,
                r2_score,
            )

        _CONFIG.logger.info("The Best Model at Epoch %d", best_epoch)

    geometry.best_epoch += best_epoch
    geometry.last_epoch += last_epoch
    geometry.loss_weights = current_weights

    return best_model
