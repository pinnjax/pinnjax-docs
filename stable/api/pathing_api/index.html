    class Path(eqx.Module):
    """Abstract base class for all path definitions.

    Notes:
        Child classes must implement an `eval()` method that computes the spatial
            coordinates of the path at a given time `t`.

    Methods:
        validate: Check that path remains within given `domain` bounds.
        draw: Visualize path within its domain.

    """

    dimensions: int

    @abstractmethod
    def eval(self, t: Array) -> Array: ...

    @abstractmethod
    def offset(self, point: Array) -> Array:
        path_coords = self.eval(point[..., -1])
        return point[..., :-1] - path_coords

    def validate(self, domain: pinnjax.Domain) -> None:
        # TODO(@mjryande): Paths are defined as endless, so if the total time is
        #   too long, validate method wrongly assesses it.
        """Validate that the entire path stays within the domain bounds.

        Args:
            domain: A [`Domain`][pinnjax.Domain] object with spatial and
                temporal bounds.

        Raises:
            InputError: If `domain.times` is None (no temporal information).
            InputError: If path coordinates fall outside the domain's min/max bounds.

        """
        if domain.times is None:
            msg = "domain must have time information for path validation"
            _CONFIG.logger.error(msg)
            raise pinnjax.config.InputError(msg)

        time_steps = int((domain.times.max - domain.times.min) / domain.times.step)
        times = jnp.linspace(domain.times.min, domain.times.max, time_steps)
        coordinates = self.eval(times)  # Shape: (time_steps, spatial_dims)
        spatial_dims = coordinates.shape[1]
        # Check each spatial dimension
        for dim in range(spatial_dims):
            coords = coordinates[:, dim]
            coord_max, coord_min = jnp.max(coords), jnp.min(coords)

            if coord_min < domain.min[dim] or coord_max > domain.max[dim]:
                msg = "Path exceeds the bounds of the given domain."
                _CONFIG.logger.error(msg)
                raise pinnjax.config.InputError(msg)

    @staticmethod
    def _create_rotation(direction: jnp.ndarray) -> Rotation:
        r"""Calculate a rotation object aligning the global x-axis with a given direction.

        The rotation is constructed such that the local z-axis remains fixed to the
        global z-axis. This ensures a unique and consistent orientation.

        Args:
            direction: A 3D vector representing the path's direction of movement.

        Returns:
            A JAX `Rotation` object that maps a point from the global coordinate system
            to the path's local, x-aligned coordinate system.

        Raises:
            ValueError: If the input direction vector is of zero magnitude.

        Note:
            The rotation matrix $R$ is constructed from three orthonormal basis vectors
            for the local frame: \( \hat{x}_{local}$, $\hat{y}_{local}$, and
                $\hat{z}_{local}\).

            \( \hat{x}_{local} = \frac{\vec{d}}{||\vec{d}||} \)
            \( \hat{z}_{local} = (0, 0, 1) \)
            \( \hat{y}_{local} = \hat{z}_{local} \times \hat{x}_{local} \)
            \( R = \begin{bmatrix} \hat{x}_{local} & \hat{y}_{local} & \hat{z}_{local}
                \end{bmatrix} \)

        """
        norm = jnp.linalg.norm(direction)
        is_zero = norm < 1e-6

        # Build the quaternion for a zero-norm vector (identity rotation).
        zero_quat = Rotation.from_rotvec(jnp.zeros(3)).as_quat()

        # Build the quaternion for the normal case.
        # The where clause ensures direction is not zero for the calculation.
        safe_direction = jnp.where(
            is_zero,
            jnp.array([1.0, 0.0, 0.0]),
            direction / norm,
        )
        local_x_axis = safe_direction
        local_z_axis = jnp.array([0.0, 0.0, 1.0])
        local_y_axis = jnp.cross(local_z_axis, local_x_axis)
        local_x_axis_ortho = jnp.cross(local_y_axis, local_z_axis)
        rot_matrix = jnp.stack([local_x_axis_ortho, local_y_axis, local_z_axis], axis=1)
        normal_quat = Rotation.from_matrix(rot_matrix).as_quat()

        # Conditionally choose between the two quaternions. This is now valid.
        final_quaternion = jnp.where(is_zero, zero_quat, normal_quat)

        # Return the final Rotation object.
        return Rotation.from_quat(final_quaternion)

        self,

        domain: pinnjax.Domain = None,
        time: pinnjax.Time = None,
        fig: Figure = None,
        ax: Axes = None,
        show: bool | None = True,

            If only a Time object is provided only the Path will be plotted,

            !!! figure
                ![](assets/path_draw_example_3D.png){: style="display: block; margin: 0 auto;" }

            >>> # for a 2D domain and CIRCULAR path.

            !!! figure
                ![](assets/path_draw_circle_example_2D.png){: style="display: block; margin: 0 auto;" }

            >>> # for a 3D domain and CIRCULAR path.

            >>> _path = pinnjax.CirclePath(pinnjax.CirclePathConfig(center=_domain.center(), radius=3, speed=2))
            >>> _ = _path.draw(_domain)

            !!! figure
                ![](assets/path_draw_circle_example_3D.png){: style="display: block; margin: 0 auto;" }

        """
        if domain is not None:
            fig, ax = domain.draw(fig=fig, ax=ax, show=False)
            time = domain.times
        elif time is not None:
            pass
        else:  # Domain and Time are None
            msg = "Either Domain or Time object must be provided."
            _CONFIG.logger.error(msg)
            raise ValueError(msg)

        if fig is None:
            fig = plt.figure()

        # TODO(@mjryande): Need to be able to remove domain from this.
        time_steps = int((time.max - time.min) / time.step)
        times = jnp.linspace(time.min, time.max, time_steps)
        coordinates = self.eval(times)

        if ax is None:
            match len(coordinates[0]):
                case 2:
                    ax = fig.add_subplot(111)
                case 3:
                    ax = fig.add_subplot(111, projection="3d")
                case _:
                    msg = "Invalid domain size for path plotting."
                    _CONFIG.logger.error(msg)
                    raise pinnjax.config.InputError(msg)

        match len(coordinates[0]):
            case 2:
                ax.plot(*coordinates.T, color="red", linewidth=3)
            case 3:
                ax.plot3D(*coordinates.T, color="red", linewidth=3)
            case _:
                msg = "Invalid domain size for path plotting."
                _CONFIG.logger.error(msg)
                pinnjax.config.InputError(msg)

        if show:
            plt.show()

        return fig, ax

            vz = 0.0

            lambda a: Path._create_rotation(_get_velocity_from_angle(a)),

            msg = "Radius must be positive."
            _CONFIG.logger.error(msg)
            raise pinnjax.config.InputError(msg)
        if self.center.ndim != 1 or self.center[0] not in {
            TWO_DIMENSIONAL,
            THREE_DIMENSIONAL,

            _CONFIG.logger.error(msg)
            raise pinnjax.config.InputError(msg)

    def get_velocity(self, u: Array) -> Array:
        """Calculates the instantaneous velocity vector of the path."""
        t = u[..., -1]
        angle = self.angular_speed * t + self.initial_angle
        vx = -self.radius * self.angular_speed * jnp.sin(angle)
        vy = self.radius * self.angular_speed * jnp.cos(angle)
        vz = 0.0
        return jnp.array([vx, vy, vz])

    def get_rotation(self, u: jnp.ndarray) -> Rotation:
        """Returns the closest cached rotation object for a given point's time."""
        t = u[..., -1]
        current_angle = self.angular_speed * t + self.initial_angle
        normalized_angle = jnp.mod(current_angle, 2 * jnp.pi)
        idx = jnp.argmin(jnp.abs(self._cached_angles - normalized_angle))
        cached_quaternion = self._cached_rotations[idx]

        # Explicitly return a Rotation object
        return Rotation.from_quat(cached_quaternion)

    def offset(self, point: Array) -> Array:

    class ComplexLinearPath(Path, ABC):
    config: pinnjax.ComplexLinearPathConfig = eqx.field(static=True)
    cumulative_times: Array
    segment_times: Array
    active: Array
    velocities: Array
    rotations: Rotation

    segments: tuple[LinearPath]

    def __init__(self, config: pinnjax.ComplexLinearPathConfig) -> None:
        self.config = config
        self.cumulative_times = config.cumulative_times
        self.segment_times = config.segment_times
        self.active = jnp.asarray(config.active)
        self.dimensions = len(config.initials[0])
        segments = []

        for _i, (initial, speed, direction, active) in enumerate(
            zip(
                config.initials,
                config.speeds,
                config.directions,
                config.active,
                strict=False,
            ),
        ):
            cfg = pinnjax.LinearPathConfig(
                initial=initial,
                speed=speed,
                direction=direction,
                active=active,
            )
            segments.append(LinearPath(cfg))

        self.segments = tuple(segments)

        self.velocities = jnp.stack(
            [
                (direction / jnp.linalg.norm(direction)) * speed
                for direction, speed in zip(
                    config.directions,
                    config.speeds,
                    strict=False,
                )
            ],
        )

        self.rotations = jax.vmap(Path._create_rotation)(self.velocities)

    def _get_segment_index(self, t: Array | float) -> Array:
        idx = jnp.searchsorted(self.config.cumulative_times, t, side="right") - 1
        return jnp.clip(idx, 0, len(self.config.segment_times) - 1)

    def _get_segment(self, t: Array | float) -> LinearPath:
        idx = self._get_segment_index(t)
        return self.segments[int(idx)]

    def get_velocity(self, u: Array) -> Array:
        t = u[-1]
        idx = self._get_segment_index(t)
        return self.velocities[idx]

    def get_rotation(self, u: Array) -> Rotation:
        t = u[-1]
        idx = self._get_segment_index(t)
        return self.rotations[idx]

    def get_active(self, u: Array) -> Array:
        t = u[-1]
        idx = self._get_segment_index(t)
        return self.active[idx]

    @jax.jit
    def eval(self, t: Array) -> Array:
        """Evaluate position at time t (scalar or array) via interpolation."""
        t = jnp.asarray(t, dtype=_CONFIG.DTYPE)
        idx = self._get_segment_index(t)

        # local fraction within the chosen segment
        local_t = t - jnp.take(self.config.cumulative_times, idx, mode="clip")
