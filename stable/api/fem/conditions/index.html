    class MechanicalCondition(ABC):
    """A condition assembled into fixed degrees of freedom or an external force."""

    components: tuple[Component, ...] = ("x", "y", "z")

    def __init__(
        self,
        components: tuple[Component, ...] = ("x", "y", "z"),
    ) -> None:
        """Store the displacement components this condition acts on.

        Args:
            components: ``"x"``, ``"y"``, and ``"z"``, in any order.
                Default: all three.

        Raises:
            ValueError: If ``components`` is empty or contains another name.

        """
        if not components:
            raise ValueError("At least one component is required.")

        invalid = set(components) - {"x", "y", "z"}
        if invalid:
            raise ValueError(f"Unknown components: {sorted(invalid)}.")

        self.components = components

    @property
    @abstractmethod
    def APPLICATION(self) -> Application:
        """Report whether this condition selects nodes or elements."""
        ...

    @abstractmethod
    def elements(
        self,
        mesh: Mesh,
        cell_type: CellType,
    ) -> ArrayLike:
        """Return the nodes or elements this condition applies to.

        Args:
            mesh: Tatva mesh the condition is applied to.
            cell_type: Cell shape of ``mesh``.

        Returns:
            Node indices when ``APPLICATION`` is ``"node"``. Boundary-element
            connectivity when it is ``"element"``.

        """
        ...

    @abstractmethod
    def assemble(
        self,
        mesh: Mesh,
        operator: Operator,
        *,
        cell_type: CellType,
        n_dofs_per_node: int,

            cell_type: Cell shape of ``mesh``.
            n_dofs_per_node: Degrees of freedom stored at each node.

        Returns:
            The assembled contribution. Unused fields stay ``None``.

    class Dirichlet(MeshBoundaryCondition):
    """A prescribed nodal displacement on one boundary face.

    ``APPLICATION`` is ``"node"``. The same ``value`` is assigned to every
    selected degree of freedom.
    """

    APPLICATION = "node"
    value: float

    def __init__(
        self,
        value: float,
        location: Location | None = None,
        side: Side | None = None,
        components: tuple[Component, ...] = ("x", "y", "z"),
    ) -> None:
        """Store the prescribed displacement and the face it applies to.

        Args:
            value: Displacement assigned to each selected degree of freedom.
            location: Coordinates of the face. Mutually exclusive with ``side``.
            side: Named exterior face. Mutually exclusive with ``location``.
            components: Components held at ``value``. Default: all three.

        Raises:
            ValueError: If the face selector or ``components`` is invalid.

        """
        super().__init__(location=location, side=side, components=components)

        self.value = value

    def assemble(
        self,
        mesh: Mesh,
        operator: Operator,
        *,
        cell_type: CellType,
        n_dofs_per_node: int,
    ) -> MechanicalContribution:
        """Assemble prescribed displacements. ``operator`` is not used.

        Args:
            mesh: Tatva mesh the condition is applied to.
            operator: Ignored.
            cell_type: Cell shape of ``mesh``.
            n_dofs_per_node: Degrees of freedom stored at each node.

        Returns:
            ``fixed_dofs`` and ``fixed_values``. ``external_force`` stays ``None``.

        """
        _ = operator

        fixed_dofs: np.ndarray = self.dofs(
            mesh,
            cell_type=cell_type,
            n_dofs_per_node=n_dofs_per_node,
        )

        fixed_values: Array = jnp.full(
            fixed_dofs.shape,
            self.value,
        )

        return MechanicalContribution(
            fixed_dofs=fixed_dofs,
            fixed_values=fixed_values,
        )

    def dofs(
    self,
    mesh: Mesh,
    cell_type: CellType,
    *,
    n_dofs_per_node: int = 3,
) -> np.ndarray:
    """Return the degree-of-freedom indices on the selected face.

    Element selections are reduced to their unique nodes. Each node then
    contributes one index per component, at
    ``node * n_dofs_per_node + offset``.

    Args:
        mesh: Tatva mesh the condition is applied to.
        cell_type: Cell shape of ``mesh``.
        n_dofs_per_node: Degrees of freedom stored at each node.
            Default: ``3``.

    Returns:
        Indices into the global degree-of-freedom vector.

    """
    selection = np.asarray(
        self.elements(
            mesh,
            cell_type=cell_type,
        ),
    )

    if self.APPLICATION == "node":
        nodes = selection
    else:
        nodes = np.unique(selection)

    offsets = {
        "x": 0,
        "y": 1,
        "z": 2,
    }

    return np.concatenate(
        [nodes * n_dofs_per_node + offsets[component] for component in self.components],
    )

    class Neumann(MeshBoundaryCondition):
    """A traction on one boundary face.

    ``APPLICATION`` is ``"element"``. ``value`` has one entry per component, in
    the same order as ``components``. Each entry is a force per unit area. It is
    integrated over the selected face and added to the external-force vector.
    """

    APPLICATION = "element"
    value: tuple[float, ...]

    def __init__(
        self,
        value: tuple[float, ...],
        location: Location | None = None,
        side: Side | None = None,
        components: tuple[Component, ...] = ("x", "y", "z"),
    ) -> None:
        """Store the traction and the face it applies to.

        Args:
            value: One traction component for each entry in ``components``.
            location: Coordinates of the face. Mutually exclusive with ``side``.
            side: Named exterior face. Mutually exclusive with ``location``.
            components: Directions the traction entries correspond to.
                Default: all three.

        Raises:
            ValueError: If the face selector or ``components`` is invalid, or if
                ``value`` does not have one number per component.

        """
        super().__init__(location=location, side=side, components=components)

        self.value = value

        value: np.ndarray = np.asarray(self.value, dtype=float)
        if value.shape != (len(self.components),):
            raise ValueError(
                f"{type(self).__name__} values must contain exactly "
                f"{len(self.components)} components. "
                f"Got shape {value.shape}.",
            )

    def assemble(
        self,
        mesh: Mesh,
        operator: Operator,
        *,
        cell_type: CellType,
        n_dofs_per_node: int,
    ) -> MechanicalContribution:
        """Integrate the traction over the selected face.

        The boundary measure comes from the Jacobian of the boundary integral of a
        unit field. ``operator`` is not used. The boundary operator is built from
        ``mesh`` and ``cell_type``.

        Args:
            mesh: Tatva mesh the condition is applied to.
            operator: Ignored.
            cell_type: Cell shape of ``mesh``.
            n_dofs_per_node: Degrees of freedom stored at each node.

        Returns:
            ``external_force``. The fixed-degree-of-freedom fields stay ``None``.

        """
        _ = operator

        boundary_elements: np.ndarray = np.asarray(
            self.elements(
                mesh,
                cell_type=cell_type,
            ),
            dtype=np.int32,
        )

        op_boundary = Operator(
            Mesh(
                mesh.coords,
                elements=jnp.asarray(boundary_elements),
            ),
            boundary_element(cell_type),
        )

        n_nodes: int = mesh.coords.shape[0]
        n_dofs: int = n_nodes * n_dofs_per_node

        nodal_measure: Array = jax.jacrev(op_boundary.integrate)(jnp.ones(n_nodes))

        boundary_nodes: np.ndarray = np.unique(boundary_elements)

        external_force: Array = jnp.zeros(n_dofs)

        offsets: dict[Component, int] = {
            "x": 0,
            "y": 1,
            "z": 2,
        }

        for component, value in zip(
            self.components,
            self.value,
            strict=True,
        ):
            dofs: np.ndarray = n_dofs_per_node * boundary_nodes + offsets[component]

            external_force = external_force.at[dofs].add(
                value * nodal_measure[boundary_nodes],
            )

        return MechanicalContribution(
            external_force=external_force,
        )

    def __init__(

    """Store the displacement components this condition acts on.

    Args:
        components: ``"x"``, ``"y"``, and ``"z"``, in any order.
            Default: all three.

    Raises:
        ValueError: If ``components`` is empty or contains another name.

    """
    if not components:
        raise ValueError("At least one component is required.")

    invalid = set(components) - {"x", "y", "z"}
    if invalid:
        raise ValueError(f"Unknown components: {sorted(invalid)}.")

    self.components = components
