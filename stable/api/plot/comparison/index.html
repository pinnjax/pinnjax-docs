    def plot_comparison(
    prediction: jnp.ndarray,
    truth: jnp.ndarray,
    grid: SpatialGrid,
    *,
    sliced_axis: str | None = None,
    threshold: float | None = None,
    sparse_interior: bool = True,
    error_type: Literal["diff", "abs", "percent"] = "diff",
    title: str | None = None,
    x_label: str | None = None,
    y_label: str | None = None,
    fig_size: tuple[float, float] | None = None,
    ax: Axes | None = None,
    cbar_ax: Axes | None = None,
    norm_override: mpl.colors.Normalize | None = None,
    cbar_ticks: jnp.ndarray | None = None,
    add_colorbar: bool = True,
    cmap_override: str | None = None,
    cbar_label: str | None = None,
    show: bool = True,
    save: bool = False,
    save_location: str | Path | None = None,
    save_name: str | None = "comparison",
    save_format: Literal[".png", ".jpg", ".jpeg", ".tiff", "pdf"] = "pdf",
    dpi: int = 300,
):
    """Plot prediction vs truth: signed difference, absolute difference, or relative error (%).

    ``"diff"`` is truth minus prediction, drawn with a diverging scale about
    zero. ``"abs"`` is the absolute difference. ``"percent"`` is that
    absolute difference divided by ``abs(truth) + 1e-8``, in percent.
    A 2D field is an image. A 3D field is a scatter of the boundary, unless
    ``sparse_interior`` is false. A passed ``ax`` is used only for a 2D field.

    Args:
        prediction: Predicted field.
        truth: Reference field. Same shape as ``prediction`` after squeezing.
        grid: [`SpatialGrid`][pinnjax.SpatialGrid] that supplies the domain.
        sliced_axis: Axis removed from a 3D field before a 2D plot. One fixed
            grid coordinate is used when this is ``None``. Otherwise ``"z"``.
        threshold: Upper clip for both fields, applied before the error.
        sparse_interior: For a 3D field, draw only the six boundary faces.
        error_type: ``"diff"``, ``"abs"``, or ``"percent"``.
        title: Axes title. A new 2D figure grows by ``0.3`` to fit it.
        x_label: Horizontal axis label. When omitted, the axis name and ``[m]``.
        y_label: Vertical axis label. When omitted, the axis name and ``[m]``.
        fig_size: Figure size in inches. When omitted, from the 2D grid.
        ax: Existing axes for a 2D field.
        cbar_ax: Axes for the 2D colorbar.
        norm_override: Color scale used instead of the automatic one.
        cbar_ticks: Colorbar tick locations. When omitted, seven ticks across the
            automatic scale.
        add_colorbar: Draw the 2D colorbar. A 3D plot always draws one.
        cmap_override: Colormap name. When omitted, ``"RdBu_r"``, ``"turbo"``, or
            ``"BuPu"`` for the three error types.
        cbar_label: Colorbar label. When omitted, ``"Difference"``,
            ``"Absolute Difference"``, or ``"Relative Difference [%]"``.
        show: Display the figure.
        save: Write the figure to disk.
        save_location: Directory for the saved figure.
        save_name: File name.
        save_format: File format.
        dpi: Resolution used when saving.

    Returns:
        The Matplotlib figure and axes. ``show`` false closes a figure this
        function created.

    Raises:
        ValueError: If the squeezed fields differ in shape, are not 2D or 3D,
            or ``error_type`` is not one of the three names.

    """
    sliced_axis = _resolved_sliced_axis(grid, sliced_axis)
    pred = _squeeze_spatial_slice_for_image(prediction, grid, sliced_axis)
    tru = _squeeze_spatial_slice_for_image(truth, grid, sliced_axis)

    if pred.shape != tru.shape:
        raise ValueError(
            f"prediction and truth must have the same shape after squeeze. "
            f"Got pred={pred.shape}, truth={tru.shape}.",
        )

    dim = pred.ndim
    if dim not in (2, 3):
        raise ValueError(
            "plot_comparison expects 2D or 3D (or squeezable to 2D/3D) inputs.",
        )

    if threshold is not None:
        pred = jnp.clip(pred, min=None, max=threshold)
        tru = jnp.clip(tru, min=None, max=threshold)

    # --------------------
    # compute error field
    # --------------------
    et = error_type.lower()
    if et == "diff":
        err = tru - pred
        cmap = cmap_override or "RdBu_r"
        m = float(jnp.nanmax(jnp.abs(err)))
        m = max(m, 1e-12)  # ensure vmin < vcenter < vmax
        norm = TwoSlopeNorm(vmin=-m, vcenter=0.0, vmax=+m)
        ticks = jnp.linspace(-m, m, num=7)
        cbar_label = cbar_label or "Difference"
    elif et == "abs":
        err = jnp.abs(tru - pred)
        cmap = cmap_override or "turbo"
        vmin, vmax = 0.0, float(jnp.nanmax(err))
        norm = mpl.colors.Normalize(vmin=vmin, vmax=vmax)
        ticks = jnp.linspace(vmin, vmax, num=7)
        cbar_label = cbar_label or "Absolute Difference"
    elif et == "percent":
        err = jnp.abs(tru - pred) / (jnp.abs(tru) + 1e-8) * 100.0
        cmap = cmap_override or "BuPu"
        vmin, vmax = float(jnp.nanmin(err)), float(jnp.nanmax(err))
        norm = mpl.colors.Normalize(vmin=vmin, vmax=vmax)
        ticks = jnp.linspace(vmin, vmax, num=7)
        cbar_label = cbar_label or "Relative Difference [%]"
    else:
        raise ValueError(
            f"error_type must be 'diff', 'abs', or 'percent'. Got {error_type!r}.",
        )
    if norm_override is not None:
        norm = norm_override

    if cbar_ticks is not None:
        ticks = cbar_ticks
    # --------------------
    # plotting
    # --------------------
    owns_figure = ax is None
    if dim == 2:
        domain = grid.domain
        idx = AXIS_TO_IDX[sliced_axis or "z"]
        other_idxs = tuple(i for i in AXIS_TO_IDX.values() if i != idx)
        extent = (
            float(domain.min[other_idxs[0]]),
            float(domain.max[other_idxs[0]]),
            float(domain.min[other_idxs[1]]),
            float(domain.max[other_idxs[1]]),
        )

        fig_size = fig_size if fig_size is not None else _figsize_from_2d_grid(grid)
        if title is not None:
            fig_size = (fig_size[0], fig_size[1] + 0.3)

        if ax is None:
            fig, ax = plt.subplots(figsize=fig_size, constrained_layout=True)
        else:
            fig = ax.figure

        im = ax.imshow(
            err,
            origin="lower",
            extent=extent,
            aspect="equal",
            cmap=cmap,
            norm=norm,
        )

        cbar = None
        if add_colorbar:
            cbar = fig.colorbar(
                im,
                cax=cbar_ax,
                label=cbar_label,
                orientation="horizontal",
                pad=0.14,
                fraction=0.24,
                ticks=ticks,
            )
            cbar.set_ticks([float(tick) for tick in ticks])

        x_label = x_label or IDX_TO_AXIS[other_idxs[0]] + " [m]"
        y_label = y_label or IDX_TO_AXIS[other_idxs[1]] + " [m]"

        ax.set_xlabel(x_label)
        ax.set_ylabel(y_label)

        if title is not None:
            ax.set_title(title)

        _force_limit_ticks(ax)
        _style_2d_axes(ax, cbar)

    else:  # dim == 3
        fig = plt.figure()
        ax = fig.add_subplot(111, projection="3d")

        if sparse_interior:
            face = jnp.zeros(err.shape, dtype=bool)
            face = face.at[0, :, :].set(True)
            face = face.at[-1, :, :].set(True)
            face = face.at[:, 0, :].set(True)
            face = face.at[:, -1, :].set(True)
            face = face.at[:, :, 0].set(True)
            face = face.at[:, :, -1].set(True)
            mask = face.ravel()
        else:
            mask = jnp.ones(err.shape, dtype=bool).ravel()

        # gather boundary points
        x = grid.coordinates[..., 0].ravel()[mask]
        y = grid.coordinates[..., 1].ravel()[mask]
        z = grid.coordinates[..., 2].ravel()[mask]
        c = err.ravel()[mask]

        surf = ax.scatter3D(
            x,
            y,
            z,
            c=c,
            norm=norm,
            cmap=cmap,
            rasterized=True,
            depthshade=False,
        )

        fig.colorbar(
            surf,
            label=cbar_label,
            orientation="horizontal",
            pad=0.05,
            ticks=ticks,
            shrink=0.9,
        )

        fig, ax = grid.domain.draw(show=False, fig=fig, ax=ax)

        if title is not None:
            ax.set_title(title)

    if save:
        from pinnjax.plot.saving import save_plot

        save_plot(
            fig,
            save_location=save_location,
            save_name=save_name,
            save_format=save_format,
            dpi=dpi,
        )

    if show:
        plt.show()
    elif owns_figure and isinstance(fig, Figure):
        plt.close(fig)

    return fig, ax

    def plot_comparison_evolution(
    prediction_evolution: Array,
    truth_evolution: Array,
    grid: SpatialGrid,
    *,
    sliced_axis: str | None = None,
    error_type: Literal["diff", "abs", "percent"] = "diff",
    threshold: float | None = None,
    times: Array | None = None,
    fps: int | None = None,
    fig_size: tuple[float, float] | None = None,
    time: bool | None = False,
    show: bool | None = True,
    save: bool | None = False,
    save_location: str | Path | None = None,
    save_name: str | None = "comparison_evolution.gif",
    x_label: str | None = None,
    y_label: str | None = None,
    cbar_label: str | None = None,
    cmap: str | None = None,
) -> tuple[Figure, Axes, FuncAnimation]:
    """Animate comparison error over time for 2D spatial grids.

    The arrays are ``(num_times, nx, ny)`` and match
    ``grid.coordinates.shape[:-1]``. The color scale is taken from the whole
    series, not from each frame. ``"diff"``, ``"abs"``, and ``"percent"``
    match [`plot_comparison`][pinnjax.plot_comparison]. A ``.gif`` name is
    written with Pillow. Any other suffix uses ffmpeg.

    Args:
        prediction_evolution: Predicted frames.
        truth_evolution: Reference frames, same shape as the prediction.
        grid: [`SpatialGrid`][pinnjax.SpatialGrid] for the extent and labels.
        sliced_axis: Unused axis for the extent labels. ``None`` uses ``"z"``.
        error_type: ``"diff"``, ``"abs"``, or ``"percent"``.
        threshold: Upper clip for both fields, applied before the error.
        times: Time of each frame. ``None`` uses ``grid.times``.
        fps: Frames per second. ``None`` uses the domain time span when it
            exists, otherwise the span of ``times``, and otherwise ``10``.
            The result is at least ``1``.
        fig_size: Figure size in inches. When omitted, from the 2D grid.
        time: Overlay the frame time in the upper left. Requires ``times`` or
            ``grid.times``.
        show: Display the animation. ``False`` closes the figure.
        save: Write the animation.
        save_location: Directory for the saved file. When omitted, the working
            directory.
        save_name: File name, including the suffix.
        x_label: Horizontal axis label. When omitted, the axis name and ``[m]``.
        y_label: Vertical axis label. When omitted, the axis name and ``[m]``.
        cbar_label: Colorbar label. When omitted, the label for ``error_type``.
        cmap: Colormap name. When omitted, the colormap for ``error_type``.

    Returns:
        The Matplotlib figure, axes, and animation.

    Raises:
        ValueError: If the evolutions differ in shape, are not at least 3D, or
            do not match the grid.
        ValueError: If ``error_type`` is not one of the three names.
        ValueError: If ``time`` is true and no times are available.
        ValueError: If ``times`` does not have one entry per frame.

    """
    pred = jnp.asarray(prediction_evolution)
    tru = jnp.asarray(truth_evolution)

    if pred.shape != tru.shape:
        raise ValueError(
            f"prediction_evolution and truth_evolution must match shapes. Got {pred.shape=} {tru.shape=}",
        )

    if pred.ndim < 3:
        raise ValueError(
            "Expected evolution arrays shaped like (num_times, nx, ny). "
            f"Got pred.ndim={pred.ndim}, pred.shape={pred.shape}.",
        )

    spatial_shape = grid.coordinates.shape[:-1]
    if tuple(pred.shape[1:]) != tuple(spatial_shape):
        raise ValueError(
            f"Shape mismatch vs grid. Got evolution spatial {pred.shape[1:]} but grid expects {spatial_shape}.",
        )

    num_frames = pred.shape[0]
    et = error_type.lower()

    if threshold is not None:
        pred = jnp.clip(pred, min=None, max=threshold)
        tru = jnp.clip(tru, min=None, max=threshold)

    if et == "diff":
        err = tru - pred
        m = float(jnp.nanmax(jnp.abs(err)))
        m = max(m, 1e-12)
        cmap_use = cmap or "RdBu_r"
        norm = TwoSlopeNorm(vmin=-m, vcenter=0.0, vmax=+m)
        ticks = jnp.linspace(-m, m, num=7)
        cbar_use = cbar_label or "Difference"
    elif et == "abs":
        err = jnp.abs(tru - pred)
        vmin = 0.0
        vmax = float(jnp.nanmax(err))
        vmax = max(vmax, vmin + 1e-12)
        cmap_use = cmap or "turbo"
        norm = mpl.colors.Normalize(vmin=vmin, vmax=vmax)
        ticks = jnp.linspace(vmin, vmax, num=7)
        cbar_use = cbar_label or "Absolute Difference"
    elif et == "percent":
        err = jnp.abs(tru - pred) / (jnp.abs(tru) + 1e-8) * 100.0
        vmin = float(jnp.nanmin(err))
        vmax = float(jnp.nanmax(err))
        vmax = max(vmax, vmin + 1e-12)
        cmap_use = cmap or "BuPu"
        norm = mpl.colors.Normalize(vmin=vmin, vmax=vmax)
        ticks = jnp.linspace(vmin, vmax, num=7)
        cbar_use = cbar_label or "Relative Difference [%]"
    else:
        raise ValueError("error_type must be 'diff', 'abs', or 'percent'.")

    # Resolve times for overlay
    if times is None:
        times = getattr(grid, "times", None)
    if time and times is None:
        raise ValueError("time=True requires `times` or `grid.times`.")

    times_arr = None
    if times is not None:
        times_arr = jnp.asarray(times)
        if times_arr.shape[0] != num_frames:
            raise ValueError(
                f"times length must equal num_frames ({num_frames}). Got {times_arr.shape[0]}.",
            )

    # fps
    if fps is None:
        domain_times = getattr(getattr(grid, "domain", None), "times", None)
        if domain_times is not None:
            fps = max(1, int(num_frames / domain_times.range()))
        elif times_arr is not None:
            dt = float(jnp.max(times_arr) - jnp.min(times_arr))
            fps = max(1, int(num_frames / max(dt, config.get_EPS())))
        else:
            fps = 10

    fig, ax = plt.subplots(
        figsize=fig_size if fig_size is not None else _figsize_from_2d_grid(grid),
        constrained_layout=True,
    )

    idx = AXIS_TO_IDX[sliced_axis or "z"]
    other_idxs = tuple(i for i in AXIS_TO_IDX.values() if i != idx)
    im = ax.imshow(
        err[0],
        origin="lower",
        aspect="equal",
        extent=(
            grid.domain.min[other_idxs[0]],
            grid.domain.max[other_idxs[0]],
            grid.domain.min[other_idxs[1]],
            grid.domain.max[other_idxs[1]],
        ),
        cmap=cmap_use,
        norm=norm,
    )

    cbar = fig.colorbar(
        im,
        label=cbar_use,
        orientation="horizontal",
        pad=0.14,
        fraction=0.24,
        ticks=ticks,
    )
    x_label = x_label or IDX_TO_AXIS[other_idxs[0]] + " [m]"
    y_label = y_label or IDX_TO_AXIS[other_idxs[1]] + " [m]"

    ax.set_xlabel(x_label)
    ax.set_ylabel(y_label)
    _force_limit_ticks(ax)
    _style_2d_axes(ax, cbar)

    time_text = None
    if time:
        time_text = ax.text(
            0.02,
            0.98,
            "",
            transform=ax.transAxes,
            verticalalignment="top",
            fontsize=_TICK_FONTSIZE,
            bbox={"boxstyle": round, "pad": 0.3, "facecolor": "white", "alpha": 0.85},
        )

    def animate(frame: int):
        im.set_data(err[frame])
        if time_text is not None:
            if times_arr is None:
                msg = "time=True requires `times` or `grid.times`."
                raise ValueError(msg)
            time_text.set_text(f"t = {float(times_arr[frame]):.3f} s")
            return [im, time_text]

    fig.canvas.draw()
    fig.set_layout_engine(None)

    anim = FuncAnimation(
        fig,
        animate,
        frames=num_frames,
        interval=1000 // fps,
        blit=True,
        repeat=True,
    )

    if save:
        out_dir = Path(save_location) if save_location else Path.cwd()
        out_dir.mkdir(parents=True, exist_ok=True)
        out_path = out_dir / (save_name or "comparison_evolution.gif")

        ext = out_path.suffix.lower()
        if ext == ".gif":
            writer = PillowWriter(fps=fps, bitrate=-1)
            anim.save(out_path, writer=writer, dpi=min(150, _SAVE_DPI))
        else:
            writer = FFMpegWriter(fps=fps, extra_args=["-vcodec", "libx264"])
            anim.save(out_path, writer=writer, dpi=_SAVE_DPI)

    if show:
        plt.show()
    else:
        plt.close()

    return fig, ax, anim
