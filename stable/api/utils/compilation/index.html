    def force_compile(shape: ComplexShape, model: NeuralNetwork) -> None:
    """Warm up JIT compilation of `residual_and_losses`.

    This also replaces it with a compiled version. It triggers JAX JIT compilation
    for the [`residual_and_losses`][pinnjax.geometry.complex.ComplexShape.residual_and_losses] method
    of a `ComplexShape` object using a single call. After this, the method
    is replaced with its compiled version to avoid compilation overhead
    during training.

    Args:
        shape: [`ComplexShape`][pinnjax.ComplexShape] object containing the method.
        model: [`NeuralNetwork`][pinnjax.NeuralNetwork] object used as input.

    Raises:
        TypeError: If `residual_and_losses` is not defined or not callable
            in the `shape` object.

    """
    ops = DifferentialOperators(model)

    if not (hasattr(shape, "residual_and_losses") and callable(shape.residual_and_losses)):
        msg = f"{type(shape).__name__} must define a callable 'residual_and_losses' method."
        raise TypeError(msg)

    # Compile the function with JAX JIT
    compiled_fn = jax.jit(shape.residual_and_losses)

    # Call once to trigger compilation
    output = compiled_fn(model, ops)

    # Ensure all JAX arrays in the output are ready
    def _block_all(o: Array | tuple[Array, Array, Array]) -> None:
        if isinstance(o, jnp.ndarray):
            o.block_until_ready()
        elif isinstance(o, (tuple, list)):
            for v in o:
                _block_all(v)
        elif isinstance(o, dict):
            for v in o.values():
                _block_all(v)

    _block_all(output)

    # Replace the original method with the compiled version
    compiled_method: Any = compiled_fn
    shape.residual_and_losses = compiled_method
