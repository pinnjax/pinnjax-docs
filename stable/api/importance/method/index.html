    class MorrisMethod(Sensitivity):
    """Screen parameters with the Morris elementary-effects method.

    Each trajectory is a random walk on the unit hypercube ``[0, 1]``. One
    coordinate changes at each step, by ``+step_size`` or ``-step_size``.
    :meth:`to_real` maps those unit values back onto the ranges in the parameter
    space.

    A parameter space maps each name to ``(kind, bounds)``. ``kind`` is a
    :data:`~pinnjax.importance.types.ParamType`. ``bounds`` is a
    ``(low, high)`` pair for ``"continuous"`` and ``"log-cont"``, and a
    one-dimensional array of allowed values for ``"discrete"`` and ``"category"``.

    Attributes:
        step_size: Step length on the unit interval. At most ``0.5``.
        key: JAX random key. Sampling and trajectory generation replace it.
        trajectories: Walks from the last :meth:`generate_trajectories` call.
            ``None`` until that method runs.

    """

    trajectories: list[list[dict[str, jax.Array]]] | None

    def __init__(self, step_size: float = 0.25) -> None:
        """Store the step length and seed the random key.

        The key comes from :func:`pinnjax.config.get_seed`.

        Args:
            step_size: Step length on the unit interval. Default: ``0.25``.

        Raises:
            ValueError: If ``step_size`` is greater than ``0.5``.

        """
        if step_size <= 0.5:
            self.step_size = step_size
        else:
            msg = f"The step_size ({step_size}) is too LARGE for the Morris method."
            config.LOGGER.error(msg)
            raise ValueError(msg)

        key, _ = jax.random.split(jax.random.key(config.get_seed()))
        self.key = key

        self.trajectories = None

    def sample(
        self,
        parameter_space: dict[
            str,
            tuple[ParamType, ContinuousRange | DiscreteRange],
        ],
    ) -> dict[str, jax.Array]:
        """Draw one point in the unit hypercube.

        Each parameter receives one uniform value on ``[0, 1]``. The kinds and bounds
        in ``parameter_space`` are ignored. The draw replaces :attr:`key`.

        Args:
            parameter_space: Parameter name to ``(kind, bounds)``. Only the names are used.

        Returns:
            Parameter name to a scalar array on ``[0, 1]``.

        """
        # sample in [0, 1] space

        out: dict[str, jax.Array] = {}
        _keys = jax.random.split(self.key, len(parameter_space) + 1)
        self.key, *subkeys = _keys
        for (name, _), key in zip(parameter_space.items(), subkeys, strict=False):
            out[name] = jax.random.uniform(key, ())  # default between 0 and 1
        return out

    def _trajectory(
        self,
        parameter_space: dict[
            str,
            tuple[ParamType, ContinuousRange | DiscreteRange],
        ],
    ) -> list[dict[str, jax.Array]]:
        parameters = list(parameter_space.keys())

        self.key, (subkey) = jax.random.split(self.key, 2)
        permutations = jax.random.permutation(subkey, jnp.arange(len(parameters)))

        old = self.sample(parameter_space)
        trajectory: list[dict[str, jax.Array]] = [old]
        for perm in permutations:
            parameter = parameters[int(perm)]
            new = dict(old)
            new[parameter] = self._step(parameter, current_location=old)
            trajectory.append(new)
            old = new

        return trajectory

    def _determine_delta_sign(self, value: jax.Array) -> float:
        if value + self.step_size <= 1.0:
            return 1.0
        if value - self.step_size >= 0.0:
            return -1.0
        return -1.0 if value <= 0.5 else 1.0  # if this case triggers, the step_size is too large

    def _step(
        self,
        parameter: str,
        current_location: dict[str, jax.Array],
    ) -> jax.Array:
        p = current_location[parameter]
        _sign = self._determine_delta_sign(p)

        return p + (_sign * self.step_size)

    def generate_trajectories(
        self,
        parameter_space: dict[
            str,
            tuple[ParamType, ContinuousRange | DiscreteRange],
        ],
        n: int = 15,
    ) -> list[list[dict[str, jax.Array]]]:
        """Build elementary-effect trajectories and store them.

        Each trajectory starts at :meth:`sample` and then visits the parameters once,
        in random order. Each visit copies the previous point and moves that one
        parameter. The sign is positive when the step stays at most ``1``. Otherwise
        it is negative when the step stays at least ``0``. If neither step stays inside
        ``[0, 1]``, the sign is negative at values at most ``0.5`` and positive above
        that. A trajectory therefore has one more point than there are parameters.

        Args:
            parameter_space: Parameter name to ``(kind, bounds)``. Only the names are used.
            n: Number of trajectories. Default: ``15``.

        Returns:
            ``n`` trajectories. Each trajectory is a list of points, and each point maps
            a parameter name to a scalar array.

        """
        trajectories: list[list[dict[str, jax.Array]]] = []
        i = 0
        while i < n:
            trajectories.append(self._trajectory(parameter_space))
            i += 1
        self.trajectories = trajectories
        return self.trajectories

    def to_real(
        self,
        parameter_space: dict[
            str,
            tuple[ParamType, ContinuousRange | DiscreteRange],
        ],
    ) -> list[list[dict[str, jax.Array]]]:
        """Map the stored unit trajectories onto the parameter ranges.

        Args:
            parameter_space: Parameter name to ``(kind, bounds)``. Passed to
                :meth:`denormalize` for every coordinate.

        Returns:
            The stored trajectories, with each coordinate converted by :meth:`denormalize`.

        Raises:
            ValueError: If :meth:`generate_trajectories` has not been called.

        """
        trajectories = self.trajectories
        if trajectories is None:
            msg = "Call generate_trajectories() before to_real()."
            config.LOGGER.error(msg)
            raise ValueError(msg)

        real_trajectories: list[list[dict[str, jax.Array]]] = []
        for trajectory in trajectories:
            real_trajectory: list[dict[str, jax.Array]] = []
            for step in trajectory:
                _step: dict[str, jax.Array] = {}
                for name, value in step.items():
                    _step[name] = self.denormalize(name, value, parameter_space)
                real_trajectory.append(_step)
            real_trajectories.append(real_trajectory)
        return real_trajectories

    @staticmethod
    def normalize(value: jax.Array, low: jax.Array, high: jax.Array) -> jax.Array:
        """Map a value onto the unit interval.

        Args:
            value: Quantity to scale.
            low: Lower end of the original interval.
            high: Upper end of the original interval.

        Returns:
            ``(value - low) / (high - low)``.

        """
        return (value - low) / (high - low)

    @staticmethod
    def denormalize(
        name: str,
        norm_value: jax.Array,
        parameter_space: dict[
            str,
            tuple[ParamType, ContinuousRange | DiscreteRange],
        ],
    ) -> jax.Array:
        """Map one unit coordinate back onto its parameter range.

        ``"continuous"`` uses ``low + u * (high - low)``. ``"log-cont"`` uses the same
        fraction between ``log(low)`` and ``log(high)``, then exponentiates.
        ``"discrete"`` and ``"category"`` take index ``floor(u * n)``, clipped to the
        array, where ``n`` is the number of allowed values.

        Args:
            name: Parameter name in ``parameter_space``.
            norm_value: Coordinate on the unit interval.
            parameter_space: Parameter name to ``(kind, bounds)``.

        Returns:
            The coordinate on the original range or set of values.

        Raises:
            KeyError: If ``name`` is not in ``parameter_space``.
            TypeError: If a continuous kind does not have a ``(low, high)`` pair, or a
                discrete kind does not have an array of values.
            ValueError: If ``kind`` is not a known parameter type.

        """

        def _denormalize_continuous(
            _norm_value: jax.Array,
            low: jax.Array,
            high: jax.Array,
        ) -> jax.Array:
            return low + _norm_value * (high - low)

        def _denormalize_log_continuous(
            _norm_value: jax.Array,
            low: jax.Array,
            high: jax.Array,
        ) -> jax.Array:
            return jnp.exp(jnp.log(low) + _norm_value * (jnp.log(high) - jnp.log(low)))

        def _denormalize_discrete(
            _norm_value: jax.Array,
            values: jax.Array,
        ) -> jax.Array:
            n = int(values.shape[0])
            idx = jnp.floor(_norm_value * n).astype(jnp.int32)
            idx = jnp.clip(idx, 0, n - 1)
            return values[idx]

        def _denormalize_category(
            _norm_value: jax.Array,
            values: jax.Array,
        ) -> jax.Array:
            n = int(values.shape[0])
            idx = jnp.floor(_norm_value * n).astype(jnp.int32)
            idx = jnp.clip(idx, 0, n - 1)
            return values[idx]

        spec = parameter_space.get(name)
        if spec is None:
            msg = f"Unknown parameter: {name}"
            config.LOGGER.error(msg)
            raise KeyError(msg)
        ptype, bounds = spec

        match ptype.lower():
            case "continuous" | "log-cont":
                if not isinstance(bounds, tuple):
                    msg = f"{name} must have a (low, high) range."
                    config.LOGGER.error(msg)
                    raise TypeError(msg)
                low = jnp.asarray(bounds[0], dtype=config.get_dtype())
                high = jnp.asarray(bounds[1], dtype=config.get_dtype())
                if ptype.lower() == "continuous":
                    denorm = _denormalize_continuous(norm_value, low, high)
                else:
                    denorm = _denormalize_log_continuous(norm_value, low, high)
            case "discrete":
                if not isinstance(bounds, jax.Array):
                    msg = f"{name} must have an array of discrete values."
                    config.LOGGER.error(msg)
                    raise TypeError(msg)
                denorm = _denormalize_discrete(norm_value, bounds)
            case "category":
                if not isinstance(bounds, jax.Array):
                    msg = f"{name} must have an array of category values."
                    config.LOGGER.error(msg)
                    raise TypeError(msg)
                denorm = _denormalize_category(norm_value, bounds)
            case _:
                msg = f"Unknown parameter type: {ptype.lower()}"
                config.LOGGER.error(msg)
                raise ValueError(msg)

        return denorm

    def sample(
    self,
    parameter_space: dict[
        str,
        tuple[ParamType, ContinuousRange | DiscreteRange],
    ],
) -> dict[str, jax.Array]:
    """Draw one point in the unit hypercube.

    Each parameter receives one uniform value on ``[0, 1]``. The kinds and bounds
    in ``parameter_space`` are ignored. The draw replaces :attr:`key`.

    Args:
        parameter_space: Parameter name to ``(kind, bounds)``. Only the names are used.

    Returns:
        Parameter name to a scalar array on ``[0, 1]``.

    """
    # sample in [0, 1] space

    out: dict[str, jax.Array] = {}
    _keys = jax.random.split(self.key, len(parameter_space) + 1)
    self.key, *subkeys = _keys
    for (name, _), key in zip(parameter_space.items(), subkeys, strict=False):
        out[name] = jax.random.uniform(key, ())  # default between 0 and 1
    return out

    def to_real(
    self,
    parameter_space: dict[
        str,
        tuple[ParamType, ContinuousRange | DiscreteRange],
    ],
) -> list[list[dict[str, jax.Array]]]:
    """Map the stored unit trajectories onto the parameter ranges.

    Args:
        parameter_space: Parameter name to ``(kind, bounds)``. Passed to
            :meth:`denormalize` for every coordinate.

    Returns:
        The stored trajectories, with each coordinate converted by :meth:`denormalize`.

    Raises:
        ValueError: If :meth:`generate_trajectories` has not been called.

    """
    trajectories = self.trajectories
    if trajectories is None:
        msg = "Call generate_trajectories() before to_real()."
        config.LOGGER.error(msg)
        raise ValueError(msg)

    real_trajectories: list[list[dict[str, jax.Array]]] = []
    for trajectory in trajectories:
        real_trajectory: list[dict[str, jax.Array]] = []
        for step in trajectory:
            _step: dict[str, jax.Array] = {}
            for name, value in step.items():
                _step[name] = self.denormalize(name, value, parameter_space)
            real_trajectory.append(_step)
        real_trajectories.append(real_trajectory)
    return real_trajectories
