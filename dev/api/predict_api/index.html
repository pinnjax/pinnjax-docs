    def predict(
    model: pinnjax.NeuralNetwork,
    grid: pinnjax.SpatialGrid,
    time: float | None = None,
    output_index: int | None = 0,
) -> Array:
    """Predict model output on a spatial grid at a specific time.

    This function flattens the provided grid, optionally appends a scalar
    time coordinate, evaluates the
    [`NeuralNetwork`][pinnjax.NeuralNetwork] using a `vmap`-mapped
    forward pass, and reshapes the result back to the grid layout.

    Args:
        model: A [`NeuralNetwork`][pinnjax.NeuralNetwork] instance.
        grid: A [`Grid`][pinnjax.Grid] containing spatial coordinates.
        time: Optional scalar time. If provided, the model is evaluated on a
            spatiotemporal mesh. If ``None``, only spatial coordinates are
            used.
        output_index: Output component to extract when the model produces
            multi-output predictions.

    Returns:
        prediction: Prediction reshaped to match the spatial grid.

    Raises:
        ValueError: If ``output_index`` does not exist in the model output.

    """
    spatial_mesh = grid.coordinates
    spatial_dimensions = spatial_mesh.shape[-1]

    if time is not None:
        t = jnp.full(spatial_mesh[..., :1].shape, jnp.asarray(time))
        _mesh = jnp.concatenate([spatial_mesh, t], axis=-1).reshape(
            -1,
            spatial_dimensions + 1,
        )
    else:
        _mesh = spatial_mesh.reshape(-1, spatial_dimensions)

    _prediction = vmap(model)(_mesh)

    prediction_2d = 2
    if _prediction.ndim == prediction_2d and _prediction.shape[1] != 1:
        _prediction = _prediction[:, output_index]

    return _prediction.reshape(spatial_mesh.shape[:-1])

    def predict_at_times(
    model: pinnjax.NeuralNetwork,
    grid: pinnjax.SpatialGrid,
    times: Array,
) -> list[Array]:
    """Predict model outputs on a spatial grid across multiple times.

    Evaluates the [`NeuralNetwork`][pinnjax.NeuralNetwork] independently
    for each time value and returns a list of prediction arrays.

    Args:
        model: A [`NeuralNetwork`][pinnjax.NeuralNetwork] instance.
        grid: A [`Grid`][pinnjax.Grid] defining the spatial domain.
        times: A 1-D JAX array of time values.

    Returns:
        predictions: Prediction arrays for each time value.

    """
    return [predict(model, grid, time) for time in times]

    def predict_evolution(model: pinnjax.NeuralNetwork, grid: pinnjax.SpatialGrid) -> Array:
    """Predict full temporal evolution on a spatiotemporal grid.

    Evaluates the [`NeuralNetwork`][pinnjax.NeuralNetwork] on the full
    spatiotemporal mesh precomputed by the
    [`Grid`][pinnjax.Grid]. The grid must define a temporal range.

    Args:
        model: A [`NeuralNetwork`][pinnjax.NeuralNetwork] instance.
        grid: A [`Grid`][pinnjax.Grid] with temporal information
            (``has_time == True``).

    Returns:
        predictions: Predictions of shape
            ``(num_times, *spatial_shape)``.

    Raises:
        pinnjax.config.InputError: If the grid lacks temporal settings.

    """
    if not grid.has_time:
        msg = "Grid must have time_range and num_times for evolution prediction"
        config.LOGGER.error(msg)
        raise config.InputError(msg)

    coords_flat = grid.spatiotemporal_coordinates.reshape(
        -1,
        grid.spatial_dimensions + 1,
    )
    predictions: Array = jnp.asarray(vmap(model)(coords_flat))
    spatial_shape = grid.coordinates.shape[:-1]
    return predictions.reshape(grid.num_times, *spatial_shape)

    def run_prediction_at_time(
    model: pinnjax.NeuralNetwork,
    domain: pinnjax.SpatialDomain | None = None,
    time: float = 0.0,
    *,
    fixed_z: float | None = None,
    num_points: tuple[int, int] = DEFAULT_GRID_NUM_POINTS,
) -> tuple[Array, pinnjax.RectangularGrid2D]:
    """
    Predict on a 2D spatial slice at a single time (no evolution).

    Builds a spatial grid and calls :func:`predict` for one time value.

    Args:
        model: Loaded PINN model.
        domain: Domain (default: model.domain).
        time: Time value for the snapshot.
        fixed_z: Z value for the 2D slice (default: domain.max[2]).
        num_points: (nx, ny) grid resolution.

    Returns:
        (prediction, grid): prediction array shape (nx, ny), and the spatial grid.
    """
    if domain is None:
        domain = model.domain
    if domain is None:
        raise TypeError(
            "run_prediction_at_time requires domain or model.domain to be set."
        )
    domain = _as_spatial_domain(domain, context="run_prediction_at_time")
    if fixed_z is None:
        fixed_z = float(domain.max[2])
    grid = pinnjax.RectangularGrid2D(
        domain=domain,
        fixed_coordinates={"z": fixed_z},
        num_points=num_points,
    )
    prediction = predict(model, grid, time=float(time))
    return prediction, grid

    def run_prediction_at_times(
    model: pinnjax.NeuralNetwork,
    domain: pinnjax.SpatialDomain | None = None,
    times: Array | list[float] | tuple[float, ...] = (),
    *,
    fixed_z: float | None = None,
    num_points: tuple[int, int] = DEFAULT_GRID_NUM_POINTS,
) -> tuple[list[Array], pinnjax.RectangularGrid2D]:
    """
    Predict on a 2D spatial slice at multiple times.

    Builds a spatial grid and calls :func:`predict_at_times`.

    Args:
        model: Loaded PINN model.
        domain: Domain (default: model.domain).
        times: Time values (e.g. [0.0, 1.0, 2.0]).
        fixed_z: Z value for the 2D slice (default: domain.max[2]).
        num_points: (nx, ny) grid resolution.

    Returns:
        (predictions, grid): list of prediction arrays, each shape (nx, ny); and the spatial grid.
    """
    if domain is None:
        domain = model.domain
    if domain is None:
        raise TypeError(
            "run_prediction_at_times requires domain or model.domain to be set."
        )
    domain = _as_spatial_domain(domain, context="run_prediction_at_times")
    if fixed_z is None:
        fixed_z = float(domain.max[2])
    grid = pinnjax.RectangularGrid2D(
        domain=domain,
        fixed_coordinates={"z": fixed_z},
        num_points=num_points,
    )
    times_arr = jnp.asarray(times)
    predictions = predict_at_times(model, grid, times_arr)
    return predictions, grid
