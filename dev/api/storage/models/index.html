    def save_model(
    model: NeuralNetwork,
    path: Path | str | None = None,
    *,
    return_path: bool = False,
) -> Path | None:
    """Save a network's arrays and metadata.

    Array leaves are written to ``model.eqx``. ``to_meta`` is written to
    ``meta.json``, with the leaf dtype, the leaf shapes, and the equinox, JAX,
    and Python versions.

    Args:
        model: [`NeuralNetwork`][pinnjax.NeuralNetwork] to save.
        path: Output directory. When omitted, the current working directory.
        return_path: Return the output directory.

    Returns:
        The output directory when ``return_path`` is true. Otherwise ``None``.

    Raises:
        pinnjax.InputError: If the network has no ``to_meta``.

    """
    out_dir = Path(path) if path else Path.cwd()
    out_dir.mkdir(parents=True, exist_ok=True)

    # 1) arrays-only checkpoint (authoritative)
    arrays, _static = eqx.partition(model, eqx.is_array)
    eqx.tree_serialise_leaves(out_dir / "model.eqx", arrays)

    # 2) meta (plain primitives only; NO callables/objects)
    if not hasattr(model, "to_meta"):
        msg = "NeuralNetwork object needs a 'to_meta' function."
        config.LOGGER.error(msg)
        raise config.InputError(msg)
    meta = model.to_meta()

    optimization = getattr(model, "optimization", None)

    if isinstance(optimization, Optimization):
        meta["optimization_spec"] = optimization_to_spec(optimization)

    # Store dtype and array structure for correct reload and validation
    leaves = list(jax.tree_util.tree_leaves(arrays))
    dtype_str = str(leaves[0].dtype) if leaves else str(config.get_dtype())
    array_shapes = [[int(s) for s in arr.shape] for arr in leaves]
    meta = {
        **meta,
        "dtype": dtype_str,
        "array_shapes": array_shapes,
        "equinox_version": eqx.__version__,
        "jax_version": jax.__version__,
        "python_version": sys.version,
    }
    with Path.open(out_dir / "meta.json", "wb") as f:
        f.write(
            orjson.dumps(meta, option=orjson.OPT_INDENT_2 | orjson.OPT_SERIALIZE_NUMPY),
        )

    return out_dir if return_path else None

    def load_model(
    path: Path | str | None = None,
    *,
    project_name: str | None = None,
    model_number: int | None = None,
    models_base: Path | None = None,
    initialization: Initialization | None = None,
    optimization: Optimization | None = None,
) -> NeuralNetwork:
    """Load a trained model from a saved directory.

    Warning:
        This currently only works for [`FNN`][pinnjax.FNN] models.
            When using other model types, they must be loaded manually.

    Provide either:

    - **path**: Direct path to the model directory.
    - **project_name** and **model_number**: Resolve the model directory from the
    project structure.

    If both **initialization** and **optimization** are provided, they are used
    directly when reconstructing the network template.

    Otherwise, the saved ``optimization_spec`` is loaded from
    ``meta.json``

    Args:
        path: Direct path to the model directory.
        project_name: Project name. Use with ``model_number``.
        model_number: Model index. Use with ``project_name``.
        models_base: Base directory for ``models/``. When omitted, the current
            working directory.
        initialization: Initialization used to construct the loading template.
            Must be provided together with ``optimization``.
        optimization: Optimization used to construct the loading template.
            Must be provided together with ``initialization``.

    Returns:
        Loaded [`NeuralNetwork`][pinnjax.NeuralNetwork].

    Raises:
        ValueError: If the model path cannot be resolved, only one of
            ``initialization`` and ``optimization`` is provided, the checkpoint
            model class is unsupported, or no reconstructable
            ``optimization_spec`` is available.
        FileNotFoundError: If ``meta.json`` or ``model.eqx`` is missing.

    """
    model_path = resolve_model_path(
        model_path=Path(path) if path is not None else None,
        project_name=project_name,
        model_number=model_number,
        models_base=models_base,
    )

    if (initialization is None) != (optimization is None):
        raise ValueError(
            "'initialization' and 'optimization' must either both be provided or both be omitted."
        )

    if initialization is not None and optimization is not None:
        return _load_model_core(model_path, initialization, optimization)

    # High-level: build from config (or defaults)
    meta = _load_meta(model_path)
    if meta.get("model_class") != "FNN":
        raise ValueError(
            f"Only FNN is supported for prediction; got model_class={meta.get('model_class')!r}. "
            "Extend storage.TEMPLATE_BUILDERS for other architectures.",
        )
    domain = _build_domain_from_meta(meta)

    optimization_spec = meta.get("optimization_spec")
    if optimization_spec is None:
        raise ValueError(
            f"Checkpoint at {model_path} has no optimization_spec. "
            "Provide initialization and optimization explicitly when loading this checkpoint."
        )

    optimization = build_optimization_from_spec(
        optimization_spec,
        domain,
    )

    initialization = Initialization()

    return _load_model_core(model_path, initialization, optimization)

    def inspect_checkpoint(
    path: Path | str | None = None,
    *,
    project_name: str | None = None,
    model_number: int | None = None,
    models_base: Path | None = None,
) -> dict:
    """Report the array leaves a checkpoint loader would build.

    Only ``meta.json`` is read. The config and ``model.eqx`` are not loaded.

    Pass ``path``, or ``project_name`` and ``model_number``. ``models_base``
    is the parent of the ``models`` folder. Resolution matches
    [`load_model`][pinnjax.load_model].

    The template uses an empty [`Optimization`][pinnjax.Optimization].
    Input/output transformations and hard constraints do not affect the network
    array structure inspected here. Leaf shapes come from the topology and domain.

    A metadata dtype is used as given. Otherwise, the template is built as
    ``float32``, then ``float64``. The previous dtype and ``x64`` setting are
    restored.

    Args:
        path: Model directory. When omitted, resolve from the project and model
            number.
        project_name: Project directory name.
        model_number: Model index, as in ``Model-#{model_number}``.
        models_base: Parent of the ``models`` folder. When omitted, the current
            working directory.

    Returns:
        ``meta``, ``model_class``, ``topology``, and ``domain`` (``min``,
            ``max``, and ``times``). ``array_leaf_count`` and ``array_shapes``
            describe the template arrays. ``dtype`` is the first leaf's dtype.

    Raises:
        ValueError: If the path cannot be resolved, or the metadata is missing
            a required key, or ``model_class`` is not registered.
        FileNotFoundError: If ``meta.json`` is missing.

    """
    model_path = resolve_model_path(
        model_path=Path(path) if path is not None else None,
        project_name=project_name,
        model_number=model_number,
        models_base=models_base,
    )
    meta_file = model_path / "meta.json"
    if not meta_file.exists():
        raise FileNotFoundError(f"Model meta not found: {meta_file}")
    meta = orjson.loads(meta_file.read_bytes())
    _require_meta_keys(meta, model_path)

    dtype_str = meta.get("dtype")
    if dtype_str is not None:
        try:
            checkpoint_dtype = getattr(jnp, dtype_str)
        except AttributeError:
            checkpoint_dtype = jnp.dtype(dtype_str)
        dtypes_to_try = (checkpoint_dtype,)
    else:
        dtypes_to_try = (jnp.float32, jnp.float64)

    optimization = Optimization()
    initialization = Initialization()

    old_dtype = config.get_dtype()
    old_x64 = config.get_JAX_ENABLE_X64()
    try:
        for checkpoint_dtype in dtypes_to_try:
            config.set_dtype(checkpoint_dtype)
            if checkpoint_dtype == jnp.float64:
                config.set_JAX_ENABLE_X64(True)
                jax.config.update("jax_enable_x64", True)
            else:
                config.set_JAX_ENABLE_X64(False)
                jax.config.update("jax_enable_x64", False)
            template = _build_template(meta, initialization, optimization)
            template_arrays, _ = eqx.partition(template, eqx.is_array)
            leaves = list(jax.tree_util.tree_leaves(template_arrays))
            dom = meta["domain"]
            domain_summary = {
                "min": list(dom["min"]),
                "max": list(dom["max"]),
                "times": dom.get("times"),
            }
            return {
                "meta": meta,
                "model_class": meta.get("model_class"),
                "topology": list(meta["topology"]),
                "domain": domain_summary,
                "array_leaf_count": len(leaves),
                "array_shapes": [list(t.shape) for t in leaves],
                "dtype": str(leaves[0].dtype) if leaves else str(config.get_dtype()),
            }
    finally:
        config.set_dtype(old_dtype)
        if old_x64 is not None:
            config.set_JAX_ENABLE_X64(old_x64)
            jax.config.update("jax_enable_x64", old_x64)
    raise RuntimeError("inspect_checkpoint: no dtype succeeded")
