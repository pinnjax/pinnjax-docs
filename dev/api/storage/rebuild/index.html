    Build an Optimization instance from a loaded optimization_spec.

    The spec is the optimization_spec entry in Configurations and Parameters/_config.json.

    Input transformation

    • Missing, or "NormalizationBetweenNeg1and1": NormalizationBetweenNeg1and1. This is the default.
    • "NormalizationBetween0and1": map the domain onto [0, 1].
    • Any other name: identity input transform.

    Output transformation

    • Missing, or "ExpectedOutputRangeSoftplus": ExpectedOutputRangeSoftplus. This is the default.
    • "ExpectedOutputRangeSigmoid": sigmoid output shifted by the material.
    • Any other name: identity output transform.

    Hard constraints

    hard_constraints is a list. A missing or empty list becomes an identity hard constraint. Recognized entries are stored as a tuple. An unrecognized type is skipped.

    • "HardDirichletConstraint": condition defaults to the ambient temperature. side is used as stored. A positive entry is the maximum face and a negative entry is the minimum face. If side is missing and idx is set, the side is -1 on that axis, which is the minimum boundary. If both are missing, the side is (0, 0, -1).

    def build_optimization_from_spec(
    spec: dict,
    domain: Domain,
    material: ThermalMaterial,
) -> Optimization:
    """Build an Optimization instance from a loaded ``optimization_spec``.

    The spec is the ``optimization_spec`` entry in
    ``Configurations and Parameters/_config.json``.

    **Input transformation**

    - Missing, or ``"NormalizationBetweenNeg1and1"``:
      [`NormalizationBetweenNeg1and1`][pinnjax.NormalizationBetweenNeg1and1].
      This is the default.
    - ``"NormalizationBetween0and1"``: map the domain onto ``[0, 1]``.
    - Any other name: identity input transform.

    **Output transformation**

    - Missing, or ``"ExpectedOutputRangeSoftplus"``:
      [`ExpectedOutputRangeSoftplus`][pinnjax.ExpectedOutputRangeSoftplus].
      This is the default.
    - ``"ExpectedOutputRangeSigmoid"``: sigmoid output shifted by the material.
    - Any other name: identity output transform.

    **Hard constraints**

    ``hard_constraints`` is a list. A missing or empty list becomes an identity
    hard constraint. Recognized entries are stored as a tuple. An unrecognized
    ``type`` is skipped.

    - ``"HardScalarInitialConstraint"``: requires a
      [`SpatialDomain`][pinnjax.SpatialDomain] with a [`Time`][pinnjax.Time].
      Without that time range, the entry is skipped. ``initial_condition``
      defaults to the material ambient temperature. ``evolution_axis`` defaults
      to ``-1``.
    - ``"HardDirichletConstraint"``: ``condition`` defaults to the ambient
      temperature. ``side`` is used as stored. A positive entry is the maximum
      face and a negative entry is the minimum face. If ``side`` is missing and
      ``idx`` is set, the side is ``-1`` on that axis, which is the minimum
      boundary. If both are missing, the side is ``(0, 0, -1)``.

    Args:
        spec: Loaded ``optimization_spec``.
        domain: Domain the network was trained on.
        material: [`ThermalMaterial`][pinnjax.ThermalMaterial]. Supplies the
            ambient temperature and the output-range transforms.

    Returns:
        The [`Optimization`][pinnjax.Optimization] passed back into the loader.

    """
    in_name = spec.get("input_transformation", "NormalizationBetweenNeg1and1")
    out_name = spec.get("output_transformation", "ExpectedOutputRangeSoftplus")
    input_transformation = IdentityInputTransform()
    if in_name == "NormalizationBetweenNeg1and1":
        input_transformation = NormalizationBetweenNeg1and1(domain)
    elif in_name == "NormalizationBetween0and1":
        input_transformation = NormalizationBetween0and1(domain)
    output_transformation = IdentityOutputTransformation()
    if out_name == "ExpectedOutputRangeSoftplus":
        output_transformation = ExpectedOutputRangeSoftplus(material)
    elif out_name == "ExpectedOutputRangeSigmoid":
        output_transformation = ExpectedOutputRangeSigmoid(material)
    hc_specs = spec.get("hard_constraints", [])
    hard_constraints = []
    for hc in hc_specs:
        t = hc.get("type")
        if t == "HardScalarInitialConstraint":
            ic = hc.get("initial_condition", float(material.ambient_temperature))
            evolution_axis = int(hc.get("evolution_axis", -1))
            if not isinstance(domain, SpatialDomain) or domain.times is None:
                continue
            hard_constraints.append(
                HardScalarInitialConstraint(
                    times=domain.times,
                    initial_condition=ic,
                    evolution_axis=evolution_axis,
                ),
            )
        elif t == "HardDirichletConstraint":
            cond = hc.get("condition", float(material.ambient_temperature))
            side = hc.get("side")
            if side is None and "idx" in hc:
                side = [0, 0, 0]
                side[hc["idx"]] = -1
                side = tuple(side)
            if side is None:
                side = (0, 0, -1)
            hard_constraints.append(
                HardDirichletConstraint(
                    domain=domain,
                    condition=cond,
                    side=tuple(side),
                ),
            )
    if not hard_constraints:
        hard_constraint = IdentityHardConstraint()
    else:
        hard_constraint = tuple(hard_constraints)
    return Optimization(
        input_transformation=input_transformation,
        output_transformation=output_transformation,
        hard_constraint=hard_constraint,
    )

    def resolve_model_path(
    *,
    model_path: Path | None = None,
    project_name: str | None = None,
    model_number: int | None = None,
    models_base: Path | None = None,
) -> Path:
    """Resolve the model directory from a given path or from project name + model number.

    A set ``model_path`` is used, even when ``project_name`` and ``model_number``
    are also passed. ``Path()`` is treated as unset.

    When using project name and model number, the path is:
        {models_base} / models / {project_name} / Model-#{model_number}

    Args:
        model_path: Direct path to the model directory (takes precedence if set).
        project_name: Project name (e.g. ``"Comparison #2"``, ``"EXAMPLE"``).
        model_number: Model index (e.g. ``1``, ``2``).
        models_base: Base path for the ``models`` folder. When omitted,
            ``Path.cwd()``.

    Returns:
        Resolved path to the model directory.

    Raises:
        ValueError: If neither (``model_path``) nor (``project_name`` and
            ``model_number``) are provided.

    """
    if model_path is not None and model_path != Path():
        return Path(model_path).resolve()
    if project_name is not None and model_number is not None:
        base = Path(models_base) if models_base is not None else Path.cwd()
        path = base / "models" / project_name / f"Model-#{model_number}"
        return path.resolve()
    raise ValueError("Provide either model_path or both project_name and model_number.")
