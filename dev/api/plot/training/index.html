    def plot_loss(
    loss: dict,
    *,
    mode: str = "class",  # ["class", "type"]
    title: str | None = None,
    best_epoch: int | None = None,
    show_grid: bool = True,
    show: bool = True,
    save: bool = False,
    save_location: Path | str | None = None,
    save_name: str | None = None,
    save_format: str = "pdf",
    dpi: int = 300,
) -> tuple[Figure, Axes]:
    """Plot the class or type loss histories.

    The horizontal axis is the one-based step in each series, on a logarithmic
    vertical scale. ``best_epoch`` draws a dashed vertical line. When that
    position is within ``0.35`` of a tick step of an existing tick, the epoch
    is annotated instead of added as a tick. ``show`` false closes the figure
    after it is built.

    Args:
        loss: Histories keyed by ``"class"`` and ``"type"``. Each value maps a
            label to a series.
        mode: ``"class"`` or ``"type"``.
        title: Axes title.
        best_epoch: Epoch to mark.
        show_grid: Draw the axes grid.
        show: Display the figure before returning it.
        save: Write the figure to disk.
        save_location: Directory for the saved figure.
        save_name: File name without the extension. When omitted, the file is
            ``loss_plot_`` plus ``mode``.
        save_format: File extension.
        dpi: Resolution used when saving.

    Returns:
        The Matplotlib figure and axes.

    Raises:
        ValueError: If ``mode`` is not ``"class"`` or ``"type"``, or that
            entry has no series.
        KeyError: If ``loss`` has no entry for ``mode``.

    """
    if mode not in ("class", "type"):
        msg = f"mode must be 'class' or 'type'. Inputted mode: {mode}"
        config.LOGGER.error(msg)
        raise ValueError(msg)

    fig, ax = plt.subplots()

    _style(ax)

    for label, _loss in loss[mode].items():
        ax.semilogy(range(1, len(_loss) + 1), _loss, label=label)

    if best_epoch is not None:
        ax.axvline(best_epoch, color="k", linestyle="--")

        xticks = jnp.asarray(ax.get_xticks())
        step = jnp.median(jnp.diff(xticks)) if len(xticks) >= 2 else 1.0

        too_close = jnp.any(jnp.abs(xticks - best_epoch) < 0.35 * step)

        if not too_close:
            ax.set_xticks(jnp.sort(jnp.append(xticks, best_epoch)))
        else:
            ax.annotate(
                f"{best_epoch}",
                xy=(best_epoch, 1),
                xycoords=("data", "axes fraction"),
                xytext=(0, 4),
                textcoords="offset points",
                ha="center",
                va="bottom",
                fontsize=9,
                clip_on=False,
            )

    ax.set_xlabel("Epoch")
    ax.set_ylabel("Loss")

    # sometimes the best_epoch line changes the xlim
    max_epoch = max(len(v) for v in loss[mode].values())
    pad = 0.02 * max_epoch  # just slight padding for aesthetics
    ax.set_xlim(1 - pad, max_epoch + pad)

    if title is not None:
        ax.set_title(title)

    ax.grid(show_grid)
    ax.legend()

    if save:
        from pinnjax.plot.saving import save_plot

        save_plot(
            fig,
            save_location=save_location,
            save_name=save_name or f"loss_plot_{mode}",
            save_format=save_format,
            dpi=dpi,
        )

    if show:
        plt.show()
    else:
        plt.close()

    return fig, ax

    def plot_evaluation(
    evaluation_dictionary: dict,
    *,
    normalize: bool = False,
    scale: Literal["linear", "semilogy", "semilogx", "loglog"] = "linear",
    best_epoch: int | None = None,
    title: str | None = None,
    show_grid: bool = True,
    show: bool = True,
    save: bool = True,
    save_location: str | Path | None = None,
    save_name: str | None = "evaluation_plot",
    save_format: str = "pdf",
    dpi: int = 300,
) -> tuple[Figure, Axes]:
    """Plot each series in an evaluation dictionary.

    ``"Epochs"`` is the horizontal axis when that entry exists. Repeated epoch
    numbers are stitched so a restarted run keeps increasing. Otherwise the
    axis is the one-based iteration. ``normalize`` scales each series to its
    own minimum and maximum. A constant series is left unchanged. The vertical
    label is ``"Loss"``.

    Args:
        evaluation_dictionary: Metric histories. ``"Epochs"`` is optional.
            Every other entry is a series.
        normalize: Scale each series to the unit interval.
        scale: ``"linear"``, ``"semilogy"``, ``"semilogx"``, or ``"loglog"``.
        best_epoch: Draw a dashed vertical line at this epoch. When that
            position is within ``0.35`` of a tick step of an existing tick, the
            epoch is annotated instead of added as a tick.
        title: Axes title.
        show_grid: Draw the axes grid.
        show: Display the figure before returning it.
        save: Write the figure to disk.
        save_location: Directory for the saved figure.
        save_name: File name without the extension.
        save_format: File extension.
        dpi: Resolution used when saving.

    Returns:
        The Matplotlib figure and axes.

    Raises:
        ValueError: If there is no series other than ``"Epochs"``.

    """

    def _stitch_epochs(_epochs: Sequence[int | float]) -> list[float]:
        out: list[float] = []
        offset = 0.0
        prev: float | None = None
        for e in _epochs:
            e = float(e)
            if prev is not None and e < prev:  # reset detected
                offset += prev
            out.append(e + offset)
            prev = e
        return out

    fig, ax = plt.subplots(figsize=(6, 4))

    plot_fn_name = "plot" if scale == "linear" else scale
    plot_fn = getattr(ax, plot_fn_name)

    _style(ax)

    series = [(label, v) for label, v in evaluation_dictionary.items() if label != "Epochs"]
    if not series:
        msg = "evaluation_dictionary contains no series to plot."
        config.LOGGER.error(msg)
        raise ValueError(msg)

    # CHANGED: use stitched epochs if provided, else fall back to evaluation steps
    epochs_raw = evaluation_dictionary.get("Epochs")
    if epochs_raw is not None:
        epochs = _stitch_epochs(epochs_raw)
        x_all = epochs
        xlabel = "Epochs"
    else:
        max_len = max(len(v) for _, v in series)
        x_all = list(range(1, max_len + 1))
        xlabel = "Iterations"

    for label, _loss in series:
        y = _loss
        if normalize:
            lo, hi = min(_loss), max(_loss)
            y = [(x - lo) / (hi - lo) if hi > lo else x for x in _loss]

        x = x_all[: len(_loss)]
        plot_fn(x, y, label=label)

    if best_epoch is not None:
        ax.axvline(best_epoch, color="k", linestyle="--")

        xticks = jnp.asarray(ax.get_xticks())
        step = jnp.median(jnp.diff(xticks)) if len(xticks) >= 2 else 1.0

        too_close = jnp.any(jnp.abs(xticks - best_epoch) < 0.35 * step)

        if not too_close:
            ax.set_xticks(jnp.sort(jnp.append(xticks, best_epoch)))
        else:
            ax.annotate(
                f"{best_epoch}",
                xy=(best_epoch, 1),
                xycoords=("data", "axes fraction"),
                xytext=(0, 4),
                textcoords="offset points",
                ha="center",
                va="bottom",
                fontsize=9,
                clip_on=False,
            )

    ax.set_xlabel(xlabel)
    ax.set_ylabel("Loss")
    if title is not None:
        ax.set_title(title)

    if x_all:
        xmin, xmax = float(min(x_all)), float(max(x_all))
        pad = 0.02 * (xmax - xmin if xmax > xmin else 1.0)
        ax.set_xlim(xmin - pad, xmax + pad)

    ax.grid(show_grid)
    ax.legend()

    if save:
        from pinnjax.plot.saving import save_plot

        save_plot(
            fig,
            save_location=save_location,
            save_name=save_name or "evaluation_plot",
            save_format=save_format,
            dpi=dpi,
        )

    if show:
        plt.show()

    return fig, ax
