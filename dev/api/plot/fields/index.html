    def plot_prediction(
    prediction: jnp.ndarray,
    grid: SpatialGrid,
    *,
    sliced_axis: str | None = None,
    threshold: float | None = None,
    fig_size: tuple[float, float] | None = None,
    ax: Axes | None = None,
    cbar_ax: Axes | None = None,
    norm: mpl.colors.Normalize | None = None,
    cbar_ticks: jnp.ndarray | None = None,
    add_colorbar: bool = True,
    show: bool | None = True,
    save: bool | None = True,
    save_location: str | Path | None = None,
    save_name: str | None = "prediction",
    save_format: str = "pdf",
    dpi: int = 300,
    title: str | None = None,
    x_label: str | None = None,
    y_label: str | None = None,
    cbar_label: str | None = None,
    cmap: str | None = "turbo",
    sparse_interior: bool = True,
):
    """Plot a prediction defined on a SpatialGrid.

    A 2D field is drawn by [`plot_field`][pinnjax.plot.plot_field]. A 3D field
    is a scatter of the six boundary faces, unless ``sparse_interior`` is
    false, and the domain outline is added afterward. ``sliced_axis`` omitted
    uses one fixed grid coordinate, otherwise ``"z"``.

    Args:
        prediction: Predicted field.
        grid: [`SpatialGrid`][pinnjax.SpatialGrid] for the domain and labels.
        sliced_axis: Axis removed before a 2D plot.
        threshold: Upper clip.
        fig_size: Figure size in inches. When omitted, from the 2D grid, or
            unset for a 3D figure.
        ax: Existing axes.
        cbar_ax: Axes for the colorbar.
        norm: Color scale. When omitted, the plotted minimum and maximum.
        cbar_ticks: Colorbar tick locations. When omitted, five ticks.
        add_colorbar: Draw the colorbar.
        show: Display the figure.
        save: Write the figure to disk.
        save_location: Directory for the saved figure.
        save_name: File name.
        save_format: File format.
        dpi: Resolution used when saving.
        title: Axes title.
        x_label: Horizontal axis label. When omitted, the axis name and ``[m]``.
        y_label: Vertical axis label. When omitted, the axis name and ``[m]``.
        cbar_label: Colorbar label.
        cmap: Colormap name.
        sparse_interior: For a 3D field, draw only the boundary faces.

    Returns:
        The Matplotlib figure and axes. ``show`` false closes a 3D figure this
        function created.

    Raises:
        ValueError: If the squeezed field is not 2D or 3D.

    """
    sliced_axis = _resolved_sliced_axis(
        grid,
        sliced_axis,
    )

    prediction = _squeeze_spatial_slice_for_image(
        prediction,
        grid,
        sliced_axis,
    )

    dim = prediction.ndim

    if dim == TWO_DIMENSIONAL:
        idx = AXIS_TO_IDX[sliced_axis or "z"]

        other_idxs = tuple(i for i in AXIS_TO_IDX.values() if i != idx)

        extent = (
            grid.domain.min[other_idxs[0]],
            grid.domain.max[other_idxs[0]],
            grid.domain.min[other_idxs[1]],
            grid.domain.max[other_idxs[1]],
        )

        fig_size = fig_size if fig_size is not None else _figsize_from_2d_grid(grid)

        x_label = x_label or IDX_TO_AXIS[other_idxs[0]] + " [m]"

        y_label = y_label or IDX_TO_AXIS[other_idxs[1]] + " [m]"

        return plot_field(
            prediction,
            extent=extent,
            threshold=threshold,
            fig_size=fig_size,
            ax=ax,
            cbar_ax=cbar_ax,
            norm=norm,
            cbar_ticks=cbar_ticks,
            add_colorbar=add_colorbar,
            show=show,
            save=save,
            save_location=save_location,
            save_name=save_name,
            save_format=save_format,
            dpi=dpi,
            title=title,
            x_label=x_label,
            y_label=y_label,
            cbar_label=cbar_label,
            cmap=cmap,
        )

    if dim != THREE_DIMENSIONAL:
        msg = f"Unsupported grid dimension: {dim}"
        config.LOGGER.error(msg)
        raise ValueError(msg)

    # TODO(@mjryande): Convert this to PyVista or another volumetric plotter.

    if threshold is not None:
        prediction = jnp.clip(
            prediction,
            None,
            threshold,
        )

    owns_figure = ax is None

    if ax is None:
        fig = plt.figure(
            figsize=fig_size,
        )
        ax = fig.add_subplot(
            111,
            projection="3d",
        )
    else:
        fig = ax.figure

    if norm is None:
        norm = mpl.colors.Normalize(
            vmin=float(jnp.min(prediction)),
            vmax=float(jnp.max(prediction)),
        )

    if sparse_interior:
        face = jnp.zeros(
            prediction.shape,
            dtype=bool,
        )

        face = face.at[0, :, :].set(True)
        face = face.at[-1, :, :].set(True)
        face = face.at[:, 0, :].set(True)
        face = face.at[:, -1, :].set(True)
        face = face.at[:, :, 0].set(True)
        face = face.at[:, :, -1].set(True)

        mask = face.ravel()

    else:
        mask = jnp.ones(
            prediction.shape,
            dtype=bool,
        ).ravel()

    x = grid.coordinates[..., 0].ravel()[mask]
    y = grid.coordinates[..., 1].ravel()[mask]
    z = grid.coordinates[..., 2].ravel()[mask]
    values = prediction.ravel()[mask]

    if not isinstance(ax, Axes3D):
        msg = "A 3D prediction requires a 3D axes."
        config.LOGGER.error(msg)
        raise TypeError(msg)

    surface = ax.scatter3D(
        x,
        y,
        z,
        c=values,
        norm=norm,
        cmap=cmap,
        rasterized=True,
        depthshade=False,
    )

    ax.set_xlim(
        float(grid.domain.min[0]),
        float(grid.domain.max[0]),
    )
    ax.set_ylim(
        float(grid.domain.min[1]),
        float(grid.domain.max[1]),
    )
    ax.set_zlim(
        float(grid.domain.min[2]),
        float(grid.domain.max[2]),
    )

    _force_limit_ticks(ax)

    if add_colorbar:
        if cbar_ticks is None:
            tick_min = surface.norm.vmin
            tick_max = surface.norm.vmax
            if tick_min is None or tick_max is None:
                msg = "The color scale needs a minimum and a maximum."
                config.LOGGER.error(msg)
                raise ValueError(msg)
            cbar_ticks = jnp.linspace(
                float(jnp.asarray(tick_min)),
                float(jnp.asarray(tick_max)),
                num=5,
            )

        fig.colorbar(
            surface,
            cax=cbar_ax,
            label=cbar_label,
            orientation="horizontal",
            pad=0.05,
            ticks=cbar_ticks,
            shrink=0.9,
        )

    if not isinstance(fig, Figure):
        msg = "A 3D prediction requires a Matplotlib figure."
        config.LOGGER.error(msg)
        raise TypeError(msg)

    fig, ax = grid.domain.draw(
        show=False,
        fig=fig,
        ax=ax,
    )

    if title is not None:
        ax.set_title(title)

    if save:
        from pinnjax.plot.saving import save_plot

        save_plot(
            fig,
            save_location=save_location,
            save_name=save_name,
            save_format=save_format,
            dpi=dpi,
        )

    if show:
        plt.show()
    elif owns_figure:
        plt.close(fig)

    return fig, ax

    def plot_prediction_evolution(
    evolution: Array,
    grid: SpatialGrid,
    *,
    sliced_axis: str | None = None,
    threshold: float | None = None,
    fig_size: tuple[float, float] | None = None,
    fps: int | None = None,
    show: bool | None = True,
    time: bool | None = False,
    save: bool | None = False,
    save_location: str | Path | None = None,
    save_name: str | None = None,
    x_label: str | None = None,
    y_label: str | None = None,
    cbar_label: str | None = None,
    cmap: str | None = "turbo",
) -> tuple[Figure, Axes, FuncAnimation]:
    """Animate a predicted field over time.

    The first axis is the frame. The color scale uses the minimum and maximum
    of the whole series. ``sliced_axis`` omitted uses one fixed grid
    coordinate, otherwise ``"z"``. A ``.gif`` name is written with Pillow. Any
    other suffix uses ffmpeg. ``show`` false closes the figure.

    Args:
        evolution: Frames shaped ``(num_times, nx, ny)``.
        grid: [`SpatialGrid`][pinnjax.SpatialGrid] with sampled times.
        sliced_axis: Unused axis for the extent labels.
        threshold: Upper clip.
        fig_size: Figure size in inches. When omitted, from the 2D grid.
        fps: Frames per second. ``None`` divides the frame count by the domain
            time span. The result is at least ``1``.
        show: Display the animation.
        time: Overlay ``grid.times`` in the upper left.
        save: Write the animation.
        save_location: Directory for the saved file. When omitted, the working
            directory.
        save_name: File name, including the suffix.
        x_label: Horizontal axis label. When omitted, the axis name and ``[m]``.
        y_label: Vertical axis label. When omitted, the axis name and ``[m]``.
        cbar_label: Colorbar label.
        cmap: Colormap name.

    Returns:
        The Matplotlib figure, axes, and animation.

    Raises:
        ValueError: If the grid has no ``times``.

    """
    if threshold is not None:
        evolution = jnp.clip(evolution, None, threshold)

    if not hasattr(grid, "times") or grid.times is None:
        msg = "Grid must have temporal information for evolution plots"
        config.LOGGER.error(msg)
        raise ValueError(msg)

    if fps is None:
        domain_times = grid.domain.times
        if domain_times is None:
            msg = "Grid must have temporal information for evolution plots"
            config.LOGGER.error(msg)
            raise ValueError(msg)
        span = float(jnp.asarray(domain_times.range()))
        fps = max(1, int(evolution.shape[0] / span))

    # Get global min/max for consistent colorbar
    vmin = float(jnp.min(evolution))
    vmax = float(jnp.max(evolution))
    norm = mpl.colors.Normalize(vmin=vmin, vmax=vmax)

    # Create figure and axis
    fig, ax = plt.subplots(
        figsize=fig_size if fig_size is not None else _figsize_from_2d_grid(grid),
        constrained_layout=True,
    )

    sliced_axis = _resolved_sliced_axis(
        grid,
        sliced_axis,
    )

    # Initial plot
    idx = AXIS_TO_IDX[sliced_axis or "z"]
    other_idxs = tuple(i for i in AXIS_TO_IDX.values() if i != idx)
    im = ax.imshow(
        evolution[0],
        origin="lower",
        aspect="equal",
        extent=(
            grid.domain.min[other_idxs[0]],
            grid.domain.max[other_idxs[0]],
            grid.domain.min[other_idxs[1]],
            grid.domain.max[other_idxs[1]],
        ),
        cmap=cmap,
        norm=norm,
    )

    # Colorbar
    ticks = jnp.linspace(vmin, vmax, num=5)
    cbar = fig.colorbar(
        im,
        label=cbar_label,
        orientation="horizontal",
        pad=0.14,
        fraction=0.24,
        ticks=ticks,
    )
    x_label = x_label or IDX_TO_AXIS[other_idxs[0]] + " [m]"
    y_label = y_label or IDX_TO_AXIS[other_idxs[1]] + " [m]"

    ax.set_xlabel(x_label)
    ax.set_ylabel(y_label)
    _force_limit_ticks(ax)
    _style_2d_axes(ax, cbar)

    # Time text
    if time:
        time_text = ax.text(
            0.02,
            0.98,
            "",
            transform=ax.transAxes,
            verticalalignment="top",
            fontsize=_TICK_FONTSIZE,
            bbox={"boxstyle": round, "pad": 0.3, "facecolor": "white", "alpha": 0.85},
        )

    def animate(frame: int):
        """Animation function."""
        im.set_data(evolution[frame])
        if time:
            time_text.set_text(f"t = {grid.times[frame]:.3f} s")
            return [im, time_text]
        return [im]

    # Resolve constrained layout once before animation
    fig.canvas.draw()
    fig.set_layout_engine("none")

    # Create animation
    anim = FuncAnimation(
        fig,
        animate,
        frames=len(evolution),
        interval=1000 // fps,
        blit=True,
        repeat=True,
    )

    # Stop animation timer before the canvas is destroyed
    def on_close(_):
        if anim.event_source is not None:
            anim.event_source.stop()

    fig.canvas.mpl_connect("close_event", on_close)

    if save:
        # Resolve output path
        out_dir = Path(save_location) if save_location else Path.cwd()
        out_dir.mkdir(parents=True, exist_ok=True)
        out_path = out_dir / (save_name or "ani.gif")

        ext = out_path.suffix.lower()
        if ext == ".gif":
            writer = PillowWriter(fps=fps, bitrate=-1)
            anim.save(out_path, writer=writer, dpi=min(150, _SAVE_DPI))
        else:
            writer = FFMpegWriter(fps=fps, extra_args=["-vcodec", "libx264"])
            anim.save(out_path, writer=writer, dpi=_SAVE_DPI)

    if show:
        plt.show()
    else:
        if anim.event_source is not None:
            anim.event_source.stop()
        plt.close(fig)

    return fig, ax, anim

    plot_field(
    field: ndarray,
    *,
    extent: tuple[float, float, float, float] | None = None,
    threshold: float | None = None,
    fig_size: tuple[float, float] | None = None,
    ax: Axes | None = None,

    def plot_field(
    field: jnp.ndarray,
    *,
    extent: tuple[float, float, float, float] | None = None,
    threshold: float | None = None,
    fig_size: tuple[float, float] | None = None,
    ax: Axes | None = None,
    cbar_ax: Axes | None = None,
    norm: mpl.colors.Normalize | None = None,
    cbar_ticks: jnp.ndarray | None = None,
    add_colorbar: bool = True,
    show: bool | None = True,
    save: bool | None = False,
    save_location: str | Path | None = None,
    save_name: str | None = "field",
    save_format: str = "pdf",
    dpi: int = 300,
    title: str | None = None,
    x_label: str | None = None,
    y_label: str | None = None,
    cbar_label: str | None = None,
    cmap: str | None = "turbo",
):
    """Plot a two-dimensional scalar field.

    The image uses a lower origin and an equal aspect. ``threshold`` is an
    upper clip. An omitted color scale runs from the field minimum to its
    maximum, with five colorbar ticks. ``show`` false closes a figure this
    function created.

    Args:
        field: Array of shape ``(H, W)``.
        extent: ``(left, right, bottom, top)``.
        threshold: Upper clip.
        fig_size: Figure size in inches.
        ax: Existing axes.
        cbar_ax: Axes for the colorbar.
        norm: Color scale. When omitted, the field minimum and maximum.
        cbar_ticks: Colorbar tick locations. When omitted, five ticks.
        add_colorbar: Draw the colorbar.
        show: Display the figure.
        save: Write the figure to disk.
        save_location: Directory for the saved figure.
        save_name: File name.
        save_format: File format.
        dpi: Resolution used when saving.
        title: Axes title.
        x_label: Horizontal axis label.
        y_label: Vertical axis label.
        cbar_label: Colorbar label.
        cmap: Colormap name.

    Returns:
        The Matplotlib figure and axes.

    Raises:
        ValueError: If ``field`` is not two-dimensional.

    """
    field = jnp.asarray(field)

    if field.ndim != 2:
        raise ValueError(
            f"plot_field requires a two-dimensional scalar field. Got shape {field.shape}.",
        )

    if threshold is not None:
        field = jnp.clip(field, None, threshold)

    owns_figure = ax is None

    if ax is None:
        fig, ax = plt.subplots(
            figsize=fig_size,
            constrained_layout=True,
        )
    else:
        fig = ax.figure

    if norm is None:
        norm = mpl.colors.Normalize(
            vmin=float(jnp.min(field)),
            vmax=float(jnp.max(field)),
        )

    image = ax.imshow(
        field,
        origin="lower",
        aspect="equal",
        extent=extent,
        cmap=cmap,
        norm=norm,
    )

    cbar = None

    if add_colorbar:
        if cbar_ticks is None:
            vmin = image.norm.vmin
            vmax = image.norm.vmax
            if vmin is None or vmax is None:
                msg = "The color scale needs a minimum and a maximum."
                config.LOGGER.error(msg)
                raise ValueError(msg)
            cbar_ticks = jnp.linspace(
                float(jnp.asarray(vmin)),
                float(jnp.asarray(vmax)),
                num=5,
            )

        cbar = fig.colorbar(
            image,
            cax=cbar_ax,
            label=cbar_label,
            orientation="horizontal",
            pad=0.14,
            fraction=0.24,
            ticks=cbar_ticks,
        )

    if title is not None:
        ax.set_title(title)

    if x_label is not None:
        ax.set_xlabel(x_label)

    if y_label is not None:
        ax.set_ylabel(y_label)

    ax.grid(
        False,
        which="both",
        axis="both",
    )

    _force_limit_ticks(ax)
    _style_2d_axes(ax, cbar)

    if save:
        from pinnjax.plot.saving import save_plot

        save_plot(
            fig,
            save_location=save_location,
            save_name=save_name,
            save_format=save_format,
            dpi=dpi,
        )

    if show:
        plt.show()
    elif owns_figure and isinstance(fig, Figure):
        plt.close(fig)

    return fig, ax

    def plot_model_prediction(
    model,
    inputs: Array,
    *,
    sample: int = 0,
    channel: int | None = 0,
    prediction_transformation: Callable[
        [Array, Array],
        Array,
    ]
    | None = None,
    extent: tuple[
        float,
        float,
        float,
        float,
    ]
    | None = None,
    threshold: float | None = None,
    fig_size: tuple[float, float] | None = None,
    ax: Axes | None = None,
    cbar_ax: Axes | None = None,
    norm: mpl.colors.Normalize | None = None,
    cbar_ticks: jnp.ndarray | None = None,
    add_colorbar: bool = True,
    show: bool | None = True,
    save: bool | None = False,
    save_location: str | Path | None = None,
    save_name: str | None = "prediction",
    save_format: str = "pdf",
    dpi: int = 300,
    title: str | None = None,
    x_label: str | None = None,
    y_label: str | None = None,
    cbar_label: str | None = None,
    cmap: str | None = "turbo",
):
    """Predict one sample with a model and plot one output field.

    The model is called with ``inputs[sample]``. A ``(H, W)`` result is the
    field. A ``(C, H, W)`` result uses ``channel``. One channel and an omitted
    ``channel`` select channel 0. The field is drawn by
    [`plot_field`][pinnjax.plot.plot_field].

    Args:
        model: Callable model. It receives one sample.
        inputs: Batch of model inputs. The first axis is the sample.
        sample: Sample index.
        channel: Output channel. ``None`` or ``0`` for a ``(H, W)`` result.
        prediction_transformation: Called as ``(sample, prediction)`` before
            the field is selected.
        extent: ``(left, right, bottom, top)`` passed to the image.
        threshold: Upper clip applied by [`plot_field`][pinnjax.plot.plot_field].
        fig_size: Figure size in inches.
        ax: Existing axes.
        cbar_ax: Axes for the colorbar.
        norm: Color scale. When omitted, the field minimum and maximum.
        cbar_ticks: Colorbar tick locations. When omitted, five ticks.
        add_colorbar: Draw the colorbar.
        show: Display the figure.
        save: Write the figure to disk.
        save_location: Directory for the saved figure.
        save_name: File name.
        save_format: File format.
        dpi: Resolution used when saving.
        title: Axes title.
        x_label: Horizontal axis label.
        y_label: Vertical axis label.
        cbar_label: Colorbar label.
        cmap: Colormap name.

    Returns:
        The Matplotlib figure and axes from [`plot_field`][pinnjax.plot.plot_field].

    Raises:
        IndexError: If ``sample`` or ``channel`` is outside the array.
        ValueError: If a ``(H, W)`` result is asked for a channel other than
            ``None`` or ``0``.
        ValueError: If a multichannel result has no ``channel``.
        ValueError: If the result is not ``(H, W)`` or ``(C, H, W)``.

    """
    if not 0 <= sample < inputs.shape[0]:
        raise IndexError(f"sample={sample} is invalid for {inputs.shape[0]} samples.")

    model_input = inputs[sample]
    prediction = model(model_input)

    if prediction_transformation is not None:
        prediction = prediction_transformation(
            model_input,
            prediction,
        )

    prediction = jnp.asarray(prediction)

    if prediction.ndim == 2:
        if channel not in (None, 0):
            raise ValueError(
                "channel can only be None or 0 for "
                "a two-dimensional prediction. "
                f"Got channel={channel}.",
            )

        field = prediction

    elif prediction.ndim == 3:
        if channel is None:
            if prediction.shape[0] != 1:
                raise ValueError(
                    "A channel must be specified for "
                    "a multi-channel prediction. "
                    f"Got shape {prediction.shape}.",
                )

            channel = 0

        if not 0 <= channel < prediction.shape[0]:
            raise IndexError(
                f"channel={channel} is invalid for prediction shape {prediction.shape}.",
            )

        field = prediction[channel]

    else:
        raise ValueError(
            f"Expected model output with shape (H, W) or (C, H, W). Got {prediction.shape}.",
        )

    return plot_field(
        field,
        extent=extent,
        threshold=threshold,
        fig_size=fig_size,
        ax=ax,
        cbar_ax=cbar_ax,
        norm=norm,
        cbar_ticks=cbar_ticks,
        add_colorbar=add_colorbar,
        show=show,
        save=save,
        save_location=save_location,
        save_name=save_name,
        save_format=save_format,
        dpi=dpi,
        title=title,
        x_label=x_label,
        y_label=y_label,
        cbar_label=cbar_label,
        cmap=cmap,
    )
