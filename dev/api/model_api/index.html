    class FNN(NeuralNetwork, ABC):
    layers: list[eqx.nn.Linear]
    topology: tuple[int, ...] = eqx.field(static=True)

    activation: pinnjax.Activation = eqx.field(static=True)
    activation_function: Callable[[Array], Array] = eqx.field(static=True)
    initialization: pinnjax.Initialization = eqx.field(static=True)

    input_transformation: Callable[[Array], Array] | None = eqx.field(static=True)
    output_transformation: Callable[[Array], Array] | None = eqx.field(static=True)
    hard_constraints: tuple[Callable, ...] = eqx.field(static=True)

    differential_axes: tuple[int, ...] = eqx.field(static=True)
    evolution_axis: int | None = eqx.field(static=True)

    def __init__(
        self,
        domain: pinnjax.Domain | None,
        topology: tuple[int, ...],
        activation: pinnjax.Activation,
        *,
        initialization: pinnjax.Initialization = None,
        optimization: pinnjax.Optimization = None,
    ):
        super().__init__()
        keys = jax.random.split(jax.random.key(config.get_seed()), len(topology) - 1)
        self.topology = topology
        self.input_dim = self.topology[0]
        self.activation = activation
        self.activation_function = functools.partial(
            activation.act_,
            *activation.args,
            **activation.kwargs,
        )
        self.domain = domain

        if optimization is None:
            optimization = pinnjax.Optimization()

        self.input_transformation = (
            optimization.input_transformation or IdentityInputTransform()
        )
        self.output_transformation = (
            optimization.output_transformation or IdentityOutputTransformation()
        )
        hc = getattr(optimization, "hard_constraint", None)
        if hc is None:
            self.hard_constraints = (IdentityHardConstraint(),)
        elif isinstance(hc, tuple):
            self.hard_constraints = tuple(hc)
        else:
            self.hard_constraints = (hc,)

        self.layers = [
            eqx.nn.Linear(self.topology[i], self.topology[i + 1], key=keys[i])
            for i in range(len(self.topology) - 1)
        ]

        self.initialization = initialization or pinnjax.Initialization()

        for i, (layer, key) in enumerate(zip(self.layers, keys, strict=False)):
            weight_key, bias_key = jax.random.split(key, 2)
            if self.initialization.weights is not None:
                weight = self.initialization.weights(
                    weight_key,
                    layer.weight.shape,
                    layer.weight.dtype,
                )
                layer = eqx.tree_at(lambda L: L.weight, layer, weight)
            if self.initialization.bias is not None:
                bias = self.initialization.bias(
                    bias_key,
                    layer.bias.shape,
                    layer.bias.dtype,
                )
                layer = eqx.tree_at(lambda L: L.bias, layer, bias)
            self.layers[i] = layer

        if self.domain is None:
            msg = "FNN currently requires a domain for input dimension validation."
            config.LOGGER.error(msg)
            raise config.InputError(msg)
        n_domain = len(self.domain.min)
        has_time = (
            isinstance(self.domain, pinnjax.SpatialDomain)
            and self.domain.times is not None
        )
        input_dims = n_domain + (1 if has_time else 0)
        if input_dims != self.topology[0]:
            msg = (
                f"Mismatch between model input size ({self.topology[0]}) and "
                f"domain-derived input_dims ({input_dims})."
            )
            config.LOGGER.error(msg)
            raise config.InputError(msg)

        if isinstance(self.domain, pinnjax.SpatialDomain):
            self.differential_axes = tuple(range(n_domain))
            self.evolution_axis = -1 if has_time else None
        else:
            self.differential_axes = tuple(range(input_dims))
            self.evolution_axis = None

    @eqx.filter_jit
    def __call__(self, inputs: Array) -> Array:
        x = self.input_transformation(inputs)
        for layer in self.layers[:-1]:
            x = self.activation_function(layer(x))
        x = self.layers[-1](x)
        x = self.output_transformation(x)
        for hc in self.hard_constraints:
            x = hc(inputs, x)
        return x

    def to_meta(self) -> dict:
        t = (
            self.domain.times
            if isinstance(self.domain, pinnjax.SpatialDomain)
            else None
        )
        times_meta = (
            None
            if t is None
            else {
                "min": float(t.min),
                "max": float(t.max),
                "step": float(t.step),
            }
        )
        return {
            "model_class": "FNN",
            "topology": list(self.topology),
            "activation": {
                "func": self.activation.act_.__name__,
                "args": list(self.activation.args),
                "kwargs": dict(self.activation.kwargs),
            },
            "domain": {
                "min": list(self.domain.min) or None,
                "max": list(self.domain.max) or None,
                "times": times_meta,
            },
        }

    class CNN(NeuralNetwork, ABC):
    layers: list[Any]

    input_transformation: Callable[[Array], Array] | None = eqx.field(static=True)
    output_transformation: Callable[[Array], Array] | None = eqx.field(static=True)
    hard_constraints: tuple[Callable, ...] = eqx.field(static=True)

    domain: pinnjax.Domain | None = eqx.field(static=True, default=None)
    differential_axes: tuple[int, ...] | None = eqx.field(static=True, default=None)
    evolution_axis: int | None = eqx.field(static=True, default=None)
    input_dim: int | None = eqx.field(static=True, default=None)

    def __init__(
        self,
        *,
        layers: list[Any],
        input_transformation: Callable[[Array], Array] | None = None,
        output_transformation: Callable[[Array], Array] | None = None,
        hard_constraints: tuple[Callable, ...] | None = None,
        domain: pinnjax.Domain | None = None,
        differential_axes: tuple[int, ...] | None = None,
        evolution_axis: int | None = None,
        input_dim: int | None = None,
    ):
        super().__init__()
        self.layers = layers
        self.input_transformation = (
            input_transformation or pinnjax.IdentityInputTransform()
        )
        self.output_transformation = (
            output_transformation or pinnjax.IdentityOutputTransformation()
        )
        self.hard_constraints = hard_constraints or (pinnjax.IdentityHardConstraint(),)
        self.domain = domain
        self.differential_axes = differential_axes
        self.evolution_axis = evolution_axis
        self.input_dim = input_dim or self._infer_input_dim()

    @classmethod
    def from_layers(
        cls,
        layers: list[Any],
        *,
        optimization: pinnjax.Optimization | None = None,
        domain: pinnjax.Domain | None = None,
    ) -> CNN:
        """Build CNN directly from a list of layers.

        Layers can be:
        - Fully constructed Equinox modules (with keys already)
        - LayerSpec objects (constructor + args/kwargs, keys added automatically)
        - Callables (activations, jnp.ravel, etc.) - passed through unchanged

        Args:
            layers: List of layers (modules, LayerSpecs, or callables)
            optimization: Optional optimization config
            domain: Optional domain for spatial_dims/time_index inference

        Returns:
            CNN instance
        """
        if optimization is None:
            optimization = pinnjax.Optimization()

        input_transformation = (
            optimization.input_transformation or pinnjax.IdentityInputTransform()
        )
        output_transformation = (
            optimization.output_transformation or pinnjax.IdentityOutputTransformation()
        )
        hc = getattr(optimization, "hard_constraint", None)
        if hc is None:
            hard_constraints = (pinnjax.IdentityHardConstraint(),)
        elif isinstance(hc, tuple):
            hard_constraints = tuple(hc)
        else:
            hard_constraints = (hc,)

        # Process layers: construct LayerSpecs with keys, pass through others
        processed_layers = cls._process_layers(layers)

        # Infer spatial_dims and time_index from domain
        differential_axes = None
        evolution_axis = None
        if domain is not None:
            n = len(domain.min)
            has_time = (
                isinstance(domain, pinnjax.SpatialDomain) and domain.times is not None
            )
            if isinstance(domain, pinnjax.SpatialDomain):
                differential_axes = tuple(range(n))
                evolution_axis = -1 if has_time else None
            else:
                input_dims = (
                    n  # CNN may not append time in bounds; match your CNN convention
                )
                differential_axes = tuple(range(input_dims))
                evolution_axis = None

        return cls(
            layers=processed_layers,
            input_transformation=input_transformation,
            output_transformation=output_transformation,
            hard_constraints=hard_constraints,
            domain=domain,
            differential_axes=differential_axes,
            evolution_axis=evolution_axis,
        )

    @staticmethod
    def _process_layers(layers: list[Any]) -> list[Any]:
        """Process LayerSpecs with keys, pass through others unchanged."""
        processed = []

        # Determine which layers need keys
        key_layers = []
        for layer in layers:
            if (
                isinstance(layer, LayerSpec)
                and "key" in inspect.signature(layer.layer_fn).parameters
            ):
                key_layers.append(layer)

        # Generate keys for trainable layers
        keys = (
            jax.random.split(jax.random.PRNGKey(config.get_seed()), len(key_layers))
            if key_layers
            else []
        )
        key_idx = 0

        for layer in layers:
            if isinstance(layer, LayerSpec):
                sig = inspect.signature(layer.layer_fn)
                if "key" in sig.parameters:
                    constructed = layer.layer_fn(
                        *layer.args,
                        **layer.kwargs,
                        key=keys[key_idx],
                    )
                    key_idx += 1
                else:
                    constructed = layer.layer_fn(*layer.args, **layer.kwargs)
                processed.append(constructed)
            else:
                processed.append(layer)

        return processed

        layers = []

        if convolutional_activation is not None:

                convolutional_activation.act_,
                *convolutional_activation.args,
                **convolutional_activation.kwargs,
            )

            layers.append(convolution)  # Convolution is a LayerSpec
            layers.append(pooling)  # Pooling is a LayerSpec

                layers.append(conv_activation_fn)

        layers.append(jnp.ravel)

    @staticmethod
    def _calculate_flattened_size(
        input_shape: tuple[int, ...],
        convolution: Convolution,
        pooling: Pooling,
        num_convolutions: int,
    ) -> int:
        """Calculate the flattened size after conv/pool operations.

        Simulates the forward pass through conv/pool layers to determine
        the output size, then calculates flattened dimensions.

        Args:
            input_shape: Input tensor shape (height, width, channels) or
                (channels, height, width)
            convolution: Convolution specification
            pooling: Pooling specification
            num_convolutions: Number of conv-pool blocks

        Returns:
            Flattened size (product of all dimensions)
        """
        # Determine if channels-first or channels-last
        # Common patterns: (H, W, C) or (C, H, W)
        if len(input_shape) == THREE_DIMENSIONAL:
            # Assume (H, W, C) format for images
            h, w, c = input_shape
        elif len(input_shape) == TWO_DIMENSIONAL:
            # Grayscale: (H, W) -> treat as (H, W, 1)
            h, w = input_shape
            c = 1
        else:
            msg = f"Unsupported input_shape: {input_shape}. Expected 2D or 3D."
            config.LOGGER.error(msg)
            raise ValueError(msg)

        # Extract conv parameters
        conv_kwargs = convolution.kwargs.copy()
        kernel_size = conv_kwargs.get("kernel_size", 3)
        if isinstance(kernel_size, (tuple, list)):
            kh, kw = kernel_size
        else:
            kh = kw = kernel_size

        stride = conv_kwargs.get("stride", 1)
        if isinstance(stride, (tuple, list)):
            sh, sw = stride
        else:
            sh = sw = stride

        padding = conv_kwargs.get("padding", 0)
        if isinstance(padding, (tuple, list)):
            ph, pw = padding
        else:
            ph = pw = padding

        # Extract pool parameters
        pool_kwargs = pooling.kwargs.copy()
        pool_size = pool_kwargs.get("kernel_size", 2)
        if isinstance(pool_size, (tuple, list)):
            pkh, pkw = pool_size
        else:
            pkh = pkw = pool_size

        pool_stride = pool_kwargs.get("stride", pool_size)  # Default to kernel_size
        if isinstance(pool_stride, (tuple, list)):
            psh, psw = pool_stride
        else:
            psh = psw = pool_stride

        # Get output channels from conv args
        # conv args should be [in_channels, out_channels, ...]
        if len(convolution.args) >= 2:
            out_channels = convolution.args[1]
        else:
            msg = "Convolution args must include [in_channels, out_channels]"
            config.LOGGER.error(msg)
            raise ValueError(msg)

        # Simulate forward pass through conv/pool blocks
        current_h, current_w = h, w
        current_c = c

        for _ in range(num_convolutions):
            # Conv layer output size
            # Formula: (W - K + 2P) / S + 1
            current_h = (current_h - kh + 2 * ph) // sh + 1
            current_w = (current_w - kw + 2 * pw) // sw + 1
            current_c = out_channels  # Update channels

            # Pool layer output size
            current_h = (current_h - pkh) // psh + 1
            current_w = (current_w - pkw) // psw + 1
            # Channels unchanged by pooling

        # Flattened size
        flattened_size = current_h * current_w * current_c

        return flattened_size

    def _infer_input_dim(self) -> int | None:
        """Try to infer input_dim from the first layer.

        Note:
            This value may be inaccurate.
        """
        if len(self.layers) > 0:
            first_layer = self.layers[0]

            if hasattr(first_layer, "in_channels"):
                # For images, input_dim might be height * width * channels
                return first_layer.in_channels
            if hasattr(first_layer, "in_features"):
                return first_layer.in_features
        return None

    class HardScalarInitialConstraint(HardConstraint):
    def __init__(
        self,
        times: pinnjax.Time,
        initial_condition: float | tuple,
        evolution_axis: int = -1,
    ):
        self.times = times
        self.initial_condition = jnp.asarray(
            initial_condition, dtype=config.get_dtype()
        )
        self.evolution_axis = evolution_axis

        phi95 = jnp.asarray(0.95, dtype=config.get_dtype())
        rng = jnp.asarray(self.times.range(), dtype=config.get_dtype())
        rng = jnp.where(jnp.abs(rng) < config.get_EPS(), config.get_EPS(), rng)
        self.tau = jnp.asarray(-rng / jnp.log1p(-phi95), dtype=config.get_dtype())

    def __call__(self, inputs, output):
        if inputs.ndim == 1:
            val = self.point(inputs, jnp.squeeze(output))
            return jnp.atleast_1d(val)
        # multiple inputs
        fn = jax.vmap(self.point, in_axes=(0, 0))
        batched = fn(inputs, output)

        return batched[:, None] if batched.ndim == 1 else batched

    def point(self, inputs: jnp.ndarray, output: jnp.ndarray):
        t = evolution_coordinate(inputs, self.evolution_axis)
        tau = self.tau.astype(inputs.dtype)
        phi = jnp.asarray(1.0, dtype=inputs.dtype) - jnp.exp(-t / tau)  # scalar

        ic = self.initial_condition

        # --- validate initial_condition vs output ---
        if ic.ndim > 1:
            msg = f"initial_condition must be scalar or 1D, got shape {ic.shape}"
            config.LOGGER.error(msg)
            raise ValueError(msg)
        if output.ndim == 0:
            if ic.ndim == 1 and ic.size != 1:
                msg = (
                    f"Scalar output but initial_condition has size {ic.size} - "
                    f"Consider using 'HardArrayInitialConstraint instead."
                )
                config.LOGGER.error(msg)
                raise ValueError(msg)
        else:
            C = output.shape[-1]
            if ic.ndim == 1 and ic.size not in (1, C):
                msg = f"initial_condition length {ic.size} does not match outputs {C}."
                config.LOGGER.error(msg)
                raise ValueError(msg)
        # ---------------------------------------------

        ic = jnp.asarray(ic, dtype=output.dtype)
        return ic + phi * (output - ic)
