    """Base for mesh generators that can draw the last mesh they built."""

        self._mesh: MeshFEMBase | None = None

    def _points_and_tetrahedra(
        self,
        cells: tuple[int, int, int],
        /,
    ) -> tuple[NDArray[np.float64], NDArray[np.int32]]:
        raise NotImplementedError

    def draw(
        self,
        cells: tuple[int, int, int] | None = None,
        *,
        backend: Backend = "pyvista",
        ax=None,
        plotter=None,
        wireframe: bool = True,
        show_edges: bool = False,
        show: bool = True,
        save: bool = False,
        save_location: str | Path | None = None,
        save_name: str | None = "mesh",
        save_format: str = "pdf",
        dpi: int = 300,
    ):
        """Plot the stored tetrahedral mesh, or build a preview mesh.

        ``cells=None`` draws the mesh from the last ``generate`` call. Passing
        ``cells``, or calling ``draw`` before ``generate``, builds a temporary mesh
        with those cell counts. The default preview is ``(12, 6, 4)`` cells in
        ``x``, ``y``, and ``z``. The temporary mesh is not stored.

        Args:
            cells: Number of cells along ``x``, ``y``, and ``z``. Default: the
                stored mesh, or ``(12, 6, 4)`` when none is stored.
            backend: ``"matplotlib"`` or ``"pyvista"``. Default: ``"pyvista"``.
            ax: Existing matplotlib axes. A new figure is created when omitted.
            plotter: Existing PyVista plotter. A new plotter is created when omitted.
            wireframe: Draw the mesh as a wireframe. Default: ``True``.
            show_edges: Draw element edges. Default: ``False``.
            show: Display the figure before returning. Default: ``True``.
            save: Write the figure to disk. Default: ``False``.
            save_location: Directory for the saved figure. Default: the working
                directory.
            save_name: File name without the extension. Default: ``"mesh"``.
            save_format: File extension. Default: ``"pdf"``.
            dpi: Resolution used when saving a matplotlib figure. Default: ``300``.

        Returns:
            The matplotlib axes or the PyVista plotter that holds the plot.

        Raises:
            TypeError: If the stored mesh container does not hold a single mesh.

        """
        if self._mesh is not None and cells is None:
            meshes = self._mesh.meshes
            if not isinstance(meshes, Mesh):
                raise TypeError(f"Expected one mesh, got {type(meshes).__name__}.")
            mesh = meshes
        else:
            cells = (12, 6, 4) if cells is None else cells
            points, tetrahedra = self._points_and_tetrahedra(cells)
            mesh = Mesh(
                jnp.asarray(points),
                elements=jnp.asarray(tetrahedra),
            )

        return plot_mesh(
            mesh,
            cell_type="tetra",
            backend=backend,
            ax=ax,
            plotter=plotter,
            show_edges=show_edges,
            wireframe=wireframe,
            lighting=False,
            show=show,
            save=save,
            save_location=save_location,
            save_name=save_name,
            save_format=save_format,
            dpi=dpi,
        )

    class Dogbone(Generator):
    """A dogbone volume meshed as a warped box.

    The starting box has the grip width and the specimen thickness. Nodes are
    then moved in ``y`` to form the grips, cosine shoulders, and gauge, and in
    ``z`` to add a cosine weld cap. The bottom of the specimen is shifted to
    ``z = 0``.
    """

    def __init__(
        self,
        length: float,
        grip_width: float,
        grip_length: float,
        gauge_width: float,
        gauge_length: float,
        thickness: float,
        weld_width: float,
        weld_height: float,
        weld_center: float | None = None,
    ):
        """Store a validated dogbone specification.

        The arguments are checked by ``DogboneSpec``. ``weld_center=None`` becomes
        the midpoint of ``length``.

        Args:
            length: Total length along ``x``, in metres.
            grip_width: Full width of each grip.
            grip_length: Length of one grip, measured from the nearest end.
            gauge_width: Full width of the gauge section.
            gauge_length: Length of the gauge section.
            thickness: Specimen thickness before the weld cap is added.
            weld_width: Width of the weld cap along ``x``. ``0`` omits the weld.
            weld_height: Extra height of the weld cap. ``0`` omits the weld.
            weld_center: Center of the weld along ``x``. Default: the midpoint.

        Raises:
            InputError: If the lengths do not leave room for the shoulders, if
                ``gauge_width`` exceeds ``grip_width``, or if ``weld_center`` lies
                past ``length``.

        """
        super().__init__()

        # doubles the work in the .from_spec() classmethod, but ensures validation.
        spec = DogboneSpec(
            length=length,
            grip_width=grip_width,
            grip_length=grip_length,
            gauge_width=gauge_width,
            gauge_length=gauge_length,
            thickness=thickness,
            weld_width=weld_width,
            weld_height=weld_height,
            weld_center=weld_center,
        )

        self.length = spec.length
        self.grip_width = spec.grip_width
        self.grip_length = spec.grip_length
        self.gauge_width = spec.gauge_width
        self.gauge_length = spec.gauge_length
        self.thickness = spec.thickness
        self.weld_width = spec.weld_width
        self.weld_height = spec.weld_height
        weld_center = spec.weld_center
        if weld_center is None:
            weld_center = spec.length * 0.5
        self.weld_center = weld_center

    @classmethod
    def from_spec(cls, spec: DogboneSpec) -> Dogbone:
        """Build a dogbone from a validated specification.

        Args:
            spec: Specimen dimensions. ``weld_center`` may be ``None``.

        Returns:
            A ``Dogbone`` with those dimensions.

        """
        return cls(**attrs.asdict(spec))

    def _x_edges(
        self,
        n_cells: int,
        /,
        weld_refinement: int = 1,
        *,
        offset: float = 0.0,
    ) -> NDArray[np.float64]:
        if weld_refinement < 1:
            msg = f"'weld_refinement' must be >= 1, got {weld_refinement}."
            config.LOGGER.error(msg)
            raise InputError(msg)

        if offset < 0.0:
            msg = f"'offset' must be >= 0, got {offset}."
            config.LOGGER.error(msg)
            raise InputError(msg)

        edges: np.ndarray = np.linspace(0.0, self.length, n_cells + 1)

        if weld_refinement == 1 or self.weld_height <= 0.0 or self.weld_width <= 0.0:
            return edges

        half_width = 0.5 * self.weld_width
        weld_start = max(0.0, self.weld_center - half_width - offset)
        weld_end = min(self.length, self.weld_center + half_width + offset)

        cell_width = self.length / n_cells
        SNAP_TOLERANCE = 0.05 * cell_width

        # in cases where the closest cell edge is very close, don't add more cells.
        # instead, move the closest cell to the weld edge.
        for weld_edge in (weld_start, weld_end):
            nearest = np.argmin(np.abs(edges - weld_edge))

            if abs(edges[nearest] - weld_edge) <= SNAP_TOLERANCE:
                edges[nearest] = weld_edge
            else:
                edges = np.sort(np.append(edges, weld_edge))

        refined_edges = [edges[0]]

        # split cells in the weld region
        for left, right in pairwise(edges):
            if left >= weld_start and right <= weld_end:
                sub_edges = np.linspace(left, right, weld_refinement + 1)
                refined_edges.extend(sub_edges[1:])
            else:
                refined_edges.append(right)

        return np.asarray(refined_edges, dtype=np.float64)

    def _width_at_x(self, x: NDArray[np.floating] | float) -> NDArray[np.float64]:
        """Return the specimen width at each position along its length.

        The profile is the grip width near each end, the gauge width in the
        middle, and a cosine blend across the two shoulders.

        Args:
            x: Position along the specimen, in metres. A scalar or an array.

        Returns:
            Width at each ``x``, as ``float64``. A scalar input returns a
            zero-dimensional array.

        """
        x: np.ndarray = np.asarray(x, dtype=np.float64)
        is_scalar = x.ndim == 0
        x = np.atleast_1d(x)

        distance_from_end = np.minimum(x, self.length - x)

        widths: np.ndarray = np.empty_like(x)

        shoulder_length = 0.5 * (self.length - 2.0 * self.grip_length - self.gauge_length)

        in_grip = distance_from_end <= self.grip_length
        in_gauge = distance_from_end >= self.grip_length + shoulder_length
        in_shoulder = ~(in_grip | in_gauge)

        widths[in_grip] = self.grip_width
        widths[in_gauge] = self.gauge_width

        if np.any(in_shoulder):
            shoulder_position = (
                distance_from_end[in_shoulder] - self.grip_length
            ) / shoulder_length

            interpolation = 0.5 * (1.0 - np.cos(np.pi * shoulder_position))

            widths[in_shoulder] = (
                self.grip_width + (self.gauge_width - self.grip_width) * interpolation
            )

        if is_scalar:
            return np.asarray(widths[0], dtype=np.float64)

        return widths

    def _weld_height_at_x(
        self,
        x: NDArray[np.floating] | float,
    ) -> NDArray[np.float64]:
        """Return the weld-cap height at each x-coordinate."""
        x: np.ndarray = np.asarray(x, dtype=np.float64)
        is_scalar = x.ndim == 0
        x = np.atleast_1d(x)

        heights: np.ndarray = np.zeros_like(x)
        if self.weld_height <= 0.0 or self.weld_width <= 0.0:
            return np.asarray(heights[0], dtype=np.float64) if is_scalar else heights

        half_width = 0.5 * self.weld_width
        normalized_position = (x - self.weld_center) / half_width
        inside_weld = np.abs(normalized_position) < 1.0

        heights[inside_weld] = (
            self.weld_height * 0.5 * (1.0 + np.cos(np.pi * normalized_position[inside_weld]))
        )

        if is_scalar:
            return np.asarray(heights[0], dtype=np.float64)

        return heights

    def _warp_points(self, points: NDArray[np.floating]) -> NDArray[np.float64]:
        X, Y, Z = 0, 1, 2

        points: np.ndarray = np.asarray(points, dtype=np.float64)

        widths = self._width_at_x(points[:, X])
        weld_heights = self._weld_height_at_x(points[:, X])

        y_mid = 0.5 * self.grip_width
        z_mid = 0.5 * self.thickness

        warped = points.copy()

        width_scale = widths / self.grip_width
        warped[:, Y] = y_mid + (points[:, Y] - y_mid) * width_scale

        height_scale = 1.0 + 2.0 * weld_heights / self.thickness

        # expands the weld symmetrically
        warped[:, Z] = z_mid + (points[:, Z] - z_mid) * height_scale

        # shifts the Z-coordinates such that the minimum is zero
        warped[:, Z] -= warped[:, Z].min()

        return warped

    def _points_and_tetrahedra(
        self,
        cells: tuple[int, int, int],
        /,
        weld_refinement: int = 1,
        *,
        offset: float = 0.0,
    ) -> tuple[NDArray[np.float64], NDArray[np.int32]]:
        box = BoxMesh(
            length=self.length,
            width=self.grip_width,
            height=self.thickness,
        )

        # exists to add additional points at the weld.
        xs = self._x_edges(cells[0], weld_refinement, offset=offset)

        # noinspection PyProtectedMember
        points, tetrahedra = box._points_and_tetrahedra(cells, xs=xs)

        points = self._warp_points(points)

        _assert_positive_tetrahedra(points, tetrahedra)
        return points, tetrahedra

        self,
        /,
        cells: tuple[int, int, int],
        weld_refinement: int = 1,
        *,
        offset: float = 0.0,
    ) -> MeshFEMBase:

        ``cells`` is ``(n_x, n_y, n_z)`` on the starting box. ``weld_refinement``
        splits cells that cover the weld into that many pieces along ``x``.
        ``offset`` widens the refined interval past each side of the weld.

        Args:
            cells: Number of cells along ``x``, ``y``, and ``z`` before weld

            weld_refinement: Number of cells placed across the weld along ``x``.
