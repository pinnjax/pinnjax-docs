    class SobolSampler(SamplingMethod):
    """Generate sample points using a Sobol low-discrepancy sequence.

    This sampler approximates the number of necessary points in a uniform
    grid with resolution ``res`` in order to determine the number of samples.
    Internally, it draws Sobol points in the unit hypercube and maps them
    to the target domain via ``shape.transform``.

    Attributes:
        sampling_value: Default sampling resolution in meters, used if no
            resolution is passed at call time.
        multiplier: Artificially increase the number of points.
        scramble: Whether to enable digital scrambling for improved
            uniformity of the Sobol sequence.
        seed: Global seed used to initialize the JAX PRNG key.
        key: Internal key for generating new subkeys.

    """

    # TODO(@mjryande): Add z-warping
    def __init__(
        self,
        sampling_value: float = 1000,
        *,
        mode: Literal[SampleMode] = "point_count",
        multiplier: float = 1.0,
        scramble: bool = True,
    ) -> None:
        """Initialize the Sobol sampler.

        Args:
            sampling_value: Number of points based on the `mode`.
            mode: Determines the meaning of the sampling value.
                Either "grid_resolution", "point_count", or "density".
            multiplier: Artificially increase the number of points.
            scramble: Whether to enable digital scrambling for improved
                uniformity.

        """

        self.scramble = scramble
        self.mode = mode
        if self.mode == "point_count" and not isinstance(sampling_value, int):
            msg = (
                "When using `point_count`, `sampling_value` must be an integer. "
                f"Got {type(sampling_value)}."
            )
            config.LOGGER.error(msg)
            raise TypeError(msg)
        self.sampling_value = sampling_value  # default value -- can be overwritten
        self.multiplier = multiplier
        self.seed = config.get_seed()
        key, _ = jax.random.split(jax.random.key(self.seed))
        self.key = key

    def __call__(
        self,
        shape: pinnjax.Shape,
        sampling_value: float | None = None,
    ) -> Array:
        """Sample points using a Sobol low-discrepancy sequence and map them to `shape`.

        The number of draws is chosen to achieve similar accuracy to a uniform grid at
        the requested resolution, then rounded up to the next power of two.
        Samples are drawn in the unit hypercube and transformed with
        [`transform()`][pinnjax.Shape.transform].

        Args:
            shape: Domain descriptor with ``dims``, and methods
                [`estimate_nsamples()`][pinnjax.Shape.estimate_nsamples],
                [`bounds()`][pinnjax.Shape.bounds], and
                [`transform()`][pinnjax.Shape.transform].
            sampling_value: Optional sampling_value overriding
                the default set at initialization.

        Raises:
            TypeError: If `point_count` is used and `mode` is not an integer.
            TypeError: If the inputted `mode` is not valid.
            NotYetImplementedError: If the selected `mode` is `density`

        Returns:
            Sampled points with shape ``(n_samples, dims)`` and dtype ``pinnjax.DTYPE``.

        """
        if sampling_value is None:
            sampling_value = self.sampling_value
        else:
            if self.mode == "point_count" and not isinstance(sampling_value, int):
                msg = (
                    "When using 'point_count', 'sampling_value' must be "
                    f"an integer. Got {type(sampling_value)}."
                )
                config.LOGGER.error(msg)
                raise TypeError(msg)

        match self.mode.lower():
            case "grid_resolution":
                estimated_uniform_samples = shape.estimate_nsamples(sampling_value)

                n_samples = int(
                    jnp.round(
                        max(
                            self._sobol_equivalent_points(
                                estimated_uniform_samples,
                                shape.dims,
                            ),
                            1,
                        )
                        * self.multiplier,
                    ),
                )

            case "point_count":
                n_samples = int(sampling_value * self.multiplier)

            case "density":
                volume = shape.measure()
                if shape.times is not None:
                    n_samples = max(
                        int(
                            jnp.round(
                                (volume * sampling_value)
                                * shape.times.steps()
                                * self.multiplier,
                            ),
                        ),
                        1,
                    )
                else:
                    n_samples = max(
                        int(
                            jnp.round(volume * sampling_value * self.multiplier),
                        ),
                        1,
                    )

            case _:
                msg = f"Invalid 'mode': {self.mode}."
                config.LOGGER.error(msg)
                raise TypeError(msg)

        lower, _ = shape.bounds()
        next_power_of_2 = max(pinnjax.next_power_of_2(int(n_samples)), 2)
        key, subkey = jax.random.split(self.key, 2)
        self.key = key
        seed = np.random.default_rng(np.asarray(jax.random.key_data(subkey)))
        sampler = qmc.Sobol(d=lower.shape[0], scramble=self.scramble, rng=seed)
        sobol_samples = jnp.array(sampler.random(next_power_of_2)[:n_samples])

        samples = shape.transform(sobol_samples)

        return jnp.asarray(samples, dtype=config.get_dtype())

    @staticmethod
    def _sobol_equivalent_points(
        n_uniform: int,
        dim: int,
        smoothness_factor: float = 1.0,
    ) -> int:
        """Estimate the number of Sobol points required to achieve comparable accuracy.

        Based on a Uniform Grid.

        Args:
            n_uniform: Number of points in the equivalent uniform grid.
            dim: Sample space dimensionality (e.g., 3 for x,y,z; 4 for x,y,z,t).
            smoothness_factor: Adjustment factor reflecting PDE smoothness.
                Use values < 1.0 for smoother problems, > 1.0 for rougher boundaries.
                Defaults to 1.0.

        Returns:
            equivalent_sobol_points: Recommended number of Sobol points.

        """
        three_dimensions = 3
        four_dimensions = 4
        if dim <= three_dimensions:
            k = 2
        elif dim <= four_dimensions:
            k = 3
        else:
            k = 4
        return round(n_uniform / (k * smoothness_factor))
