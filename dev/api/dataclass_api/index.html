class Domain:
    """General N-dimensional axis-aligned domain"""

    min: tuple[float | int, ...]
    max: tuple[float | int, ...]
    axis_names: tuple[str, ...] | None = None

    UNITS: ClassVar[Mapping[str, str]] = {"min": "", "max": ""}
    VARIABLES: ClassVar[tuple[str, ...]] = ()

    def __attrs_post_init__(self) -> None:
        min_arr = jnp.atleast_1d(jnp.asarray(self.min, dtype=config.get_dtype()))
        max_arr = jnp.atleast_1d(jnp.asarray(self.max, dtype=config.get_dtype()))
        if min_arr.shape != max_arr.shape:
            msg = f"'min' and 'max' must have same shape, got {min_arr.shape} vs {max_arr.shape}."
            config.LOGGER.error(msg)
            raise InputError(msg)
        if jnp.any(max_arr < min_arr):
            msg = "'max' values must be >= 'min' values for all axes."
            config.LOGGER.error(msg)
            raise InputError(msg)

        # Normalize internal storage to tuples of python floats.
        min_tuple = tuple(float(x) for x in min_arr.tolist())
        max_tuple = tuple(float(x) for x in max_arr.tolist())
        object.__setattr__(self, "min", cast("tuple[float, ...]", min_tuple))
        object.__setattr__(self, "max", cast("tuple[float, ...]", max_tuple))
        # Default generic names if user does not provide axis names.
        if self.axis_names is None:
            axis_names = tuple(f"x{i}" for i in range(len(min_tuple)))
        else:
            axis_names = tuple(self.axis_names)
        if len(axis_names) != len(min_tuple):
            msg = (
                f"'axis_names' length must match domain dims ({len(min_tuple)}), "
                f"got {len(axis_names)}."
            )
            config.LOGGER.error(msg)
            raise InputError(msg)
        if len(set(axis_names)) != len(axis_names):
            msg = f"'axis_names' must be unique, got {axis_names}."
            config.LOGGER.error(msg)
            raise InputError(msg)
        object.__setattr__(self, "axis_names", axis_names)

    @property
    def dims(self) -> int:
        return len(self.min)

    @property
    def axis_to_idx(self) -> Mapping[str, int]:
        return dict(zip(self.axis_names, range(self.dims), strict=True))

    def range(self, axis: int | str) -> float:
        """Return the difference between the min and max value of dimension  `x`.

        Zero Indexed.

        Examples:
            >>> domain = pinnjax.Domain(min=(5,3,0),max=(10,15,3))
            >>> domain.range(0)
            5.0
            >>> domain.range(1)
            12.0

        """
        idx = self.axis_to_idx[axis] if isinstance(axis, str) else axis
        return float(self.max[idx] - self.min[idx])

    def _dim_center(self, idx: int) -> float:
        return float(self.min[idx] + self.range(idx) / 2)

    def center(self, fixed: dict[str, float] | None = None) -> tuple[float, ...]:
        """Return the central point of the domain.

        Args:
            fixed: Optional ``dict`` denoting the axis label ``x``, ``y``, ``z`` of
                the fixed dimension(s) and the fixed value(s).

        Examples:
            >>> domain = pinnjax.SpatialDomain(min=(0,0,0),max=(10,15,3),
            ... times=pinnjax.Time(min= 10, max = 100, step = 1))
            >>> domain.center()
            (5.0, 7.5, 1.5)

            >>> domain.center(fixed={'y': 10})
            (5.0, 10, 1.5)

            >>> domain.center(fixed={'y': 12, 'z': 1})
            (5.0, 12, 1)

        """
        fixed = fixed or {}
        out = []
        for i, axis_name in enumerate(self.axis_names):
            out.append(
                float(fixed[axis_name]) if axis_name in fixed else self._dim_center(i)
            )
        return tuple(out)

    @classmethod
    def from_data_shape(
        cls,
        data_shape: pinnjax.DataShape,
        *,
        padding: float = 0.0,
        axis_names: tuple[str, ...] | None = None,
    ) -> "Domain":
        """
        Build a Domain from DataShape input bounds.
        Args:
            data_shape: Object with `.inputs` array shaped (n_samples, n_features).
            padding: Optional symmetric padding added to each min/max bound.
                     If scalar, applied to every axis.
            axis_names: Optional axis names; defaults to x0, x1, ...
        Returns:
            Domain with min/max inferred from data_shape.inputs.
        """
        if not hasattr(data_shape, "inputs"):
            raise TypeError("`data_shape` must have an `inputs` attribute.")
        x = jnp.asarray(data_shape.inputs, dtype=config.get_dtype())
        if x.ndim != 2:
            raise ValueError(
                f"`data_shape.inputs` must be 2D (n_samples, n_features). Got shape {x.shape}."
            )
        if x.shape[0] == 0:
            raise ValueError("`data_shape.inputs` is empty.")
        mins = jnp.min(x, axis=0)
        maxs = jnp.max(x, axis=0)
        pad = jnp.asarray(padding, dtype=config.get_dtype())
        if pad.ndim == 0:
            pad = jnp.full_like(mins, pad)
        elif pad.shape != mins.shape:
            raise ValueError(
                f"`padding` must be scalar or shape {mins.shape}. Got {pad.shape}."
            )
        if jnp.any(pad < 0):
            raise ValueError("`padding` must be non-negative.")
        min_tuple = tuple(float(v) for v in (mins - pad).tolist())
        max_tuple = tuple(float(v) for v in (maxs + pad).tolist())
        return cls(min=min_tuple, max=max_tuple, axis_names=axis_names)

    def to_string(self, *, indent: int = 0) -> str:
        return _print_attrs_class(self, indent=indent)

    def __str__(self) -> str:
        return self.to_string(indent=0)

        data_shape: Object with `.inputs` array shaped (n_samples, n_features).
        padding: Optional symmetric padding added to each min/max bound.
                 If scalar, applied to every axis.

            f"`data_shape.inputs` must be 2D (n_samples, n_features). Got shape {x.shape}."
        )

        raise ValueError("`data_shape.inputs` is empty.")
    mins = jnp.min(x, axis=0)
    maxs = jnp.max(x, axis=0)
    pad = jnp.asarray(padding, dtype=config.get_dtype())
    if pad.ndim == 0:
        pad = jnp.full_like(mins, pad)
    elif pad.shape != mins.shape:
        raise ValueError(
            f"`padding` must be scalar or shape {mins.shape}. Got {pad.shape}."
        )
    if jnp.any(pad < 0):
        raise ValueError("`padding` must be non-negative.")
    min_tuple = tuple(float(v) for v in (mins - pad).tolist())
    max_tuple = tuple(float(v) for v in (maxs + pad).tolist())
    return cls(min=min_tuple, max=max_tuple, axis_names=axis_names)

class SpatialDomain(Domain):
    """Represent the spatial and temporal (optional) rectangular domain.

    In the form: (x, y, z) where the ``z`` component is optional depending on
    the problem dimensionality.

    Attributes:
        min: The minimum spatial dimensions [m].
        max: The maximum spatial dimensions [m].
        times: An optional [`Time`][pinnjax.Time] object.

    """

    # min: tuple[float | int, ...] # [m, m, m]
    # max: tuple[float | int, ...] # [m, m, m]
    times: Time | None = None

    UNITS: ClassVar[Mapping[str, str]] = {
        "min": "m",
        "max": "m",

    VARIABLES: ClassVar[tuple[str, ...]] = ("x", "y", "z")

    def __attrs_post_init__(self) -> None:
        Domain.__attrs_post_init__(self)

        # Enforce SpatialDomain dimension limits.
        n_dims = len(self.min)
        if n_dims not in (1, 2, 3):

            config.LOGGER.error(msg)

    @classmethod
    def from_obb(
        cls,
        origin: tuple[float | int, float | int, float | int] | Array,
        *edges: tuple[float | int, float | int]
        | tuple[float | int, float | int, float | int]
        | Array,
        times: Time | None = None,
    ) -> pinnjax.SpatialDomain:
        """Create a [`SpatialDomain`][pinnjax.SpatialDomain] object using a defined OBB.

        Notes:
            Only works for AABB (Axis-Aligned Bounded Boxes).

        Examples:
            >>> obb = ((0, 0, 0), (10, 0.0, 0.0), (0.0, 15, 0.0), (0.0, 0.0, 3))
            >>> domain = pinnjax.SpatialDomain.from_obb(*obb, times=None)
            >>> print(domain)
            SpatialDomain:
              min: (0, 0, 0) m
              max: (10, 15, 3) m
              times: None

            >>> domain = pinnjax.SpatialDomain.from_obb(
            ... (0, 0, 0),
            ... (10, 0.0, 0.0),
            ... (0.0, 15, 0.0),
            ... (0.0, 0.0, 3),
            ... times=None)
            >>> print(domain)
            SpatialDomain:
              min: (0, 0, 0) m
              max: (10, 15, 3) m
              times: None

        """
        # Check all edges are axis-aligned and orthogonal
        max_vec = list(origin)
        axis_used = set()
        for edge in edges:
            nonzero = [i for i, val in enumerate(edge) if abs(val) > config.get_EPS()]
            if len(nonzero) != 1:
                msg = f"Edge {edge} is not axis-aligned or is degenerate."
                config.LOGGER.error(msg)
                raise InputError(msg)
            axis = nonzero[0]
            if axis in axis_used:
                msg = f"Multiple edges along the same axis: {edges}"
                config.LOGGER.error(msg)
                raise InputError(msg)
            axis_used.add(axis)
            max_vec[axis] += edge[axis]
        max_tuple = tuple(max_vec[: len(origin)])
        return cls(min=tuple(origin), max=max_tuple, times=times)

    def to_obb(self) -> tuple[tuple[float, ...], ...]:
        """Convert the AABB SpatialDomain object to the OBB form for use in the creation
            of a [`ComplexShape`][pinnjax.ComplexShape].

        Notes:
            Only works for AABB (Axis-Aligned Bounded Boxes).

        Examples:
            >>> domain = pinnjax.SpatialDomain(min=(0,0,0),max=(10,15,3),
            ... times=pinnjax.Time(min= 10, max = 100, step = 1))
            >>> domain.to_obb()
            ((0.0, 0.0, 0.0), (10.0, 0.0, 0.0), (0.0, 15.0, 0.0), (0.0, 0.0, 3.0))
            >>> # [ origin   edge 1        edge 2         edge 3 ]

        """
        dim = len(self.min)

        if dim == 1:
            return (self.min[0],), (self.max[0],)

        edges = []
        for i in range(dim):
            edge = [0.0] * dim
            edge[i] = self.max[i] - self.min[i]
            if sum(edge) == 0:
                continue
            edges.append(tuple(edge))
        return self.min, *edges

    def draw(
        self,
        *,
        show: bool | None = True,
        fig: Figure | None = None,
        ax: Axes | None = None,

            fig = plt.figure()

        max_coords = jnp.array(self.max)
        spatial_dimensions = len(self.min)

        match spatial_dimensions:
            case 2:
                if ax is None:
                    ax = fig.add_subplot(111)

                    [
                        [min_coords[0], min_coords[1]],
                        [max_coords[0], min_coords[1]],
                        [max_coords[0], max_coords[1]],
                        [min_coords[0], max_coords[1]],
                    ],
                )

                edges = jnp.array([[0, 1], [1, 2], [2, 3], [3, 0]])

                for i in range(edges.shape[0]):
                    edge = edges[i]
                    points = corners[edge]
                    ax.plot(*points.T, "black", alpha=0.6)

                ax.set_xlabel("X (m)")
                ax.set_ylabel("Y (m)")
                ax.set_aspect("equal")

            case 3:
                if ax is None:
                    ax = fig.add_subplot(111, projection="3d")

                corners = jnp.array(
                    [
                        [min_coords[0], min_coords[1], min_coords[2]],
                        [max_coords[0], min_coords[1], min_coords[2]],
                        [max_coords[0], max_coords[1], min_coords[2]],
                        [min_coords[0], max_coords[1], min_coords[2]],
                        [min_coords[0], min_coords[1], max_coords[2]],
                        [max_coords[0], min_coords[1], max_coords[2]],
                        [max_coords[0], max_coords[1], max_coords[2]],
                        [min_coords[0], max_coords[1], max_coords[2]],
                    ],
                )

                # Define edges as JAX array directly
                edges = jnp.array(
                    [
                        [0, 1],
                        [1, 2],
                        [2, 3],
                        [3, 0],
                        [4, 5],
                        [5, 6],
                        [6, 7],
                        [7, 4],
                        [0, 4],
                        [1, 5],
                        [2, 6],
                        [3, 7],
                    ],
                )

                for i in range(edges.shape[0]):
                    edge = edges[i]
                    points = corners[edge]
                    if hasattr(ax, "plot3D"):
                        ax.plot3D(*points.T, "black", alpha=1, linewidth=1)
                    else:
                        msg = (
                            "3-dimensional axis 'ax' must have the attribute 'plot3D'. "
                        )
                        config.LOGGER.error(msg)
                        raise RuntimeError(msg)

                if hasattr(ax, "set_zticks"):
                    ax.set_zticks(jnp.linspace(min_coords[2], max_coords[2], 3))

                    [self.range(0), self.range(1), self.range(2)],
                    dtype=jnp.float32,

                ax.set_box_aspect(aspect.tolist())

            case _:
                msg = f"Invalid spatial dimensions: {spatial_dimensions}."
                config.LOGGER.error(msg)
                raise InputError(msg)

        if show:
            plt.show()

        assert ax is not None, "'ax' was not initialized."  # added to satisfy ty.
        return fig, ax

        fig = plt.figure()

    max_coords = jnp.array(self.max)

            if ax is None:
                ax = fig.add_subplot(111)

                    [max_coords[0], min_coords[1]],

                    [min_coords[0], max_coords[1]],

                edge = edges[i]
                points = corners[edge]
                ax.plot(*points.T, "black", alpha=0.6)

            ax.set_ylabel("Y (m)")
            ax.set_aspect("equal")

            if ax is None:
                ax = fig.add_subplot(111, projection="3d")

            corners = jnp.array(
                [
                    [min_coords[0], min_coords[1], min_coords[2]],
                    [max_coords[0], min_coords[1], min_coords[2]],
                    [max_coords[0], max_coords[1], min_coords[2]],
                    [min_coords[0], max_coords[1], min_coords[2]],
                    [min_coords[0], min_coords[1], max_coords[2]],
                    [max_coords[0], min_coords[1], max_coords[2]],
                    [max_coords[0], max_coords[1], max_coords[2]],
                    [min_coords[0], max_coords[1], max_coords[2]],
                ],
            )

            # Define edges as JAX array directly
            edges = jnp.array(
                [
                    [0, 1],
                    [1, 2],
                    [2, 3],
                    [3, 0],
                    [4, 5],
                    [5, 6],
                    [6, 7],
                    [7, 4],
                    [0, 4],
                    [1, 5],
                    [2, 6],
                    [3, 7],
                ],
            )

            for i in range(edges.shape[0]):
                edge = edges[i]
                points = corners[edge]
                if hasattr(ax, "plot3D"):
                    ax.plot3D(*points.T, "black", alpha=1, linewidth=1)
                else:
                    msg = (
                        "3-dimensional axis 'ax' must have the attribute 'plot3D'. "
                    )
                    config.LOGGER.error(msg)
                    raise RuntimeError(msg)

            ax.set_xticks(jnp.linspace(min_coords[0], max_coords[0], 3))
            ax.set_yticks(jnp.linspace(min_coords[1], max_coords[1], 3))
            if hasattr(ax, "set_zticks"):
                ax.set_zticks(jnp.linspace(min_coords[2], max_coords[2], 3))

            # attempt to make 3D plots look good when the dims are variable.
            spans = jnp.array(
                [self.range(0), self.range(1), self.range(2)],
                dtype=jnp.float32,
            )
            eps = jnp.finfo(spans.dtype).eps
            smin = jnp.maximum(jnp.min(spans), eps)
            smax = jnp.maximum(jnp.max(spans), eps)
            r = spans / smax  # normalize by max

            anisotropy = smax / smin

            # Map anisotropy to alpha in [alpha_min, 1]
            alpha_min = 0.75  # more compression when very anisotropic
            a_ref = 10.0  # anisotropy where we clamp to alpha_min
            t = jnp.clip(jnp.log(anisotropy) / jnp.log(a_ref), 0.0, 1.0)
            alpha = (1.0 - t) + alpha_min * t  # linear ramp from 1 -> alpha_min

            aspect = jnp.clip(r**alpha, 0.2, 1.0)

            pad_x = float(5 + 10 * (spans[0] / smax))
            pad_y = float(5 + 10 * (spans[1] / smax))
            pad_z = float(5 + 10 * (spans[2] / smax))

            ax.set_ylabel("Y (m)", labelpad=pad_y)
            if hasattr(ax, "set_zlabel"):
                ax.set_zlabel("Z (m)", labelpad=pad_z)

            ax.set_box_aspect(aspect.tolist())

        case _:
            msg = f"Invalid spatial dimensions: {spatial_dimensions}."
            config.LOGGER.error(msg)
            raise InputError(msg)

        plt.show()

    assert ax is not None, "'ax' was not initialized."  # added to satisfy ty.
    return fig, ax

    cls,

    *edges: tuple[float | int, float | int]

    | Array,

    Notes:
        Only works for AABB (Axis-Aligned Bounded Boxes).

    Examples:
        >>> obb = ((0, 0, 0), (10, 0.0, 0.0), (0.0, 15, 0.0), (0.0, 0.0, 3))
        >>> domain = pinnjax.SpatialDomain.from_obb(*obb, times=None)

        ... (0, 0, 0),
        ... (10, 0.0, 0.0),
        ... (0.0, 15, 0.0),
        ... (0.0, 0.0, 3),
        ... times=None)
        >>> print(domain)
        SpatialDomain:
          min: (0, 0, 0) m
          max: (10, 15, 3) m
          times: None

    """
    # Check all edges are axis-aligned and orthogonal
    max_vec = list(origin)
    axis_used = set()
    for edge in edges:
        nonzero = [i for i, val in enumerate(edge) if abs(val) > config.get_EPS()]
        if len(nonzero) != 1:
            msg = f"Edge {edge} is not axis-aligned or is degenerate."
            config.LOGGER.error(msg)
            raise InputError(msg)
        axis = nonzero[0]
        if axis in axis_used:
            msg = f"Multiple edges along the same axis: {edges}"
            config.LOGGER.error(msg)
            raise InputError(msg)
        axis_used.add(axis)
        max_vec[axis] += edge[axis]
    max_tuple = tuple(max_vec[: len(origin)])
    return cls(min=tuple(origin), max=max_tuple, times=times)

    def __attrs_post_init__(self) -> None:
        waypoint_array = jnp.asarray(self.waypoints)

        # --------------------------- TESTING  -----------------------------------------
        required_dimensions = 2
        if waypoint_array.ndim != required_dimensions:
            msg = "waypoints must be a 2D array-like: shape (N, D)."
            config.LOGGER.error(msg)
            raise InputError(msg)
        number, dims = waypoint_array.shape
        minimum_waypoints = 2
        if number < minimum_waypoints:
            msg = "At least two waypoints are required."
            config.LOGGER.error(msg)
            raise InputError(msg)

        if not jnp.isfinite(waypoint_array).all():
            msg = "waypoints contain NaN or Inf."
            config.LOGGER.error(msg)
            raise InputError(msg)

        if self.is_temporal:
            if dims not in (3, 4):
                msg = "Temporal path requires (x,y,t) or (x,y,z,t): D must be 3 or 4."
                config.LOGGER.error(msg)
                raise InputError(msg)
            # the last column is time
            t = waypoint_array[:, -1]
            if not jnp.isfinite(t).all():
                msg = "Time column contains NaN or Inf."
                config.LOGGER.error(msg)
                raise InputError(msg)
            dt = jnp.diff(t)
            if not (dt > 0).all():
                msg = "Time must be strictly increasing in the last column."
                config.LOGGER.error(msg)
                raise InputError(msg)
            if self._speed is not None:
                msg = (
                    "For temporal paths, 'speed' must be None. "
                    "Speeds are determined automatically."
                )
                config.LOGGER.error(msg)
                raise InputError(msg)
        else:
            if self._speed is None:
                msg = (
                    "If there is no time-component a speed or speeds must be provided."
                )
                config.LOGGER.error(msg)
                raise InputError(msg)
            if dims not in (2, 3):
                msg = (
                    "Non-temporal path supports only 2D or 3D spatial waypoints. "
                    "You might have meant to set is_temporal = True."
                )
                config.LOGGER.error(msg)
                raise InputError(msg)

        # ---------- Speed checks ----------
        if self._speed is not None:
            if isinstance(self._speed, tuple):
                # Per-segment speeds: length must be N-1
                if len(self._speed) not in (1, number - 1):
                    msg = (
                        "speed tuple length must be 1 (broadcast) "
                        "or N-1={number - 1} (per-segment)."
                    )
                    config.LOGGER.error(msg)
                    raise InputError(msg)
                if not jnp.isfinite(jnp.asarray(self._speed)).all():
                    msg = "speed contains NaN or Inf."
                    config.LOGGER.error(msg)
                    raise InputError(msg)
                if (jnp.asarray(self._speed) <= 0).any():
                    msg = "speed values must be positive."
                    config.LOGGER.error(msg)
                    raise InputError(msg)
            else:
                # Scalar speed
                if not jnp.isfinite(jnp.asarray(self._speed)):
                    msg = "speed contains NaN or Inf."
                    config.LOGGER.error(msg)
                    raise InputError(msg)
                if float(self._speed) <= 0:
                    msg = "speed values must be positive."
                    config.LOGGER.error(msg)
                    raise InputError(msg)

        # Ensure all rows have the same length
        if isinstance(self.waypoints, list):
            first_len = len(self.waypoints[0])
            if any(len(t) != first_len for t in self.waypoints):
                msg = "All waypoint tuples must have the same length."
                config.LOGGER.error(msg)
                raise InputError(msg)
            # When temporal, all tuples must be 3 or 4; otherwise 2 or 3.
            if self.is_temporal and first_len not in (3, 4):
                msg = "With is_temporal=True, tuples must be length 3 or 4."
                config.LOGGER.error(msg)
                raise InputError(msg)
            if not self.is_temporal and first_len not in (2, 3):
                msg = "With is_temporal=False, tuples must be length 2 or 3."
                config.LOGGER.error(msg)
                raise InputError(msg)

        # ------------------------------------------------------------------------------
        spatial_dimensions = dims - (1 if self.is_temporal else 0)
        spatial_coordinates = waypoint_array[:, :spatial_dimensions]

        distances = jnp.linalg.norm(jnp.diff(spatial_coordinates, axis=0), axis=1)
        number_segments = len(distances)
        if (distances == 0).any():
            msg = "Consecutive waypoints must not be identical (zero-length segment)."
            config.LOGGER.error(msg)
            raise InputError(msg)

        object.__setattr__(self, "initials", spatial_coordinates[:-1])
        object.__setattr__(
            self,
            "directions",
            spatial_coordinates[1:] - spatial_coordinates[:-1],
        )

        if self.is_temporal:
            time_deltas = jnp.diff(waypoint_array[:, -1])  # Δt > 0 already checked
            speeds = distances / time_deltas  # per-segment speeds
            if not jnp.isfinite(speeds).all() or (speeds <= 0).any():
                msg = "Derived speeds must be positive and finite."
                config.LOGGER.error(msg)
                raise InputError(msg)
            cumulative_times = waypoint_array[:, -1]
        else:
            speeds = (
                jnp.asarray(self._speed)
                if isinstance(self._speed, (tuple, list))
                else jnp.full((len(distances),), self._speed)
            )
            durations = distances / speeds
            cumulative_times = jnp.concatenate(
                [jnp.array([0.0]), jnp.cumsum(durations)],
            )

        segment_times = jnp.diff(cumulative_times)
        object.__setattr__(self, "cumulative_times", cumulative_times)
        object.__setattr__(self, "segment_times", segment_times)
        object.__setattr__(self, "speeds", tuple(map(float, speeds.tolist())))

        if isinstance(self.active, (bool, int, float)):
            activity = [bool(self.active)] * number_segments

        elif isinstance(self.active, (tuple, list)):
            if len(self.active) != number_segments:
                msg = "Active list Length MUST match number of segments."
                config.LOGGER.error(msg)
                raise InputError(msg)
            activity = [bool(i) for i in self.active]

        elif isinstance(self.active, dict):
            if set(self.active.keys()) != {"off"}:
                msg = 'Active Dictionary must only contain the key "off".'
                config.LOGGER.error(msg)
                raise InputError(msg)
            activity = [True] * number_segments

            off_segments = self.active["off"]
            if isinstance(off_segments, int):
                off_segments = [off_segments]
            elif isinstance(off_segments, tuple):
                off_segments = list(off_segments)

            for k in off_segments:
                if not (1 <= k <= number_segments):
                    msg = (
                        f"Invalid segment index {k} in active['off']. "
                        f"Segments are 1-indexed."
                    )
                    config.LOGGER.error(msg)
                    raise InputError(msg)
                activity[k - 1] = False

        else:
            msg = (
                "active must be boolean, scalar, list/tuple of booleans, "
                "or dict {'off': [indices]}."
            )
            config.LOGGER.error(msg)
            raise InputError(msg)

        object.__setattr__(self, "active", jnp.asarray(activity, dtype=bool))

        msg = "waypoints contain NaN or Inf."
        config.LOGGER.error(msg)
        raise InputError(msg)

    if self.is_temporal:
        if dims not in (3, 4):
            msg = "Temporal path requires (x,y,t) or (x,y,z,t): D must be 3 or 4."
            config.LOGGER.error(msg)
            raise InputError(msg)
        # the last column is time
        t = waypoint_array[:, -1]
        if not jnp.isfinite(t).all():
            msg = "Time column contains NaN or Inf."
            config.LOGGER.error(msg)
            raise InputError(msg)
        dt = jnp.diff(t)
        if not (dt > 0).all():
            msg = "Time must be strictly increasing in the last column."
            config.LOGGER.error(msg)
            raise InputError(msg)
        if self._speed is not None:
            msg = (
                "For temporal paths, 'speed' must be None. "
                "Speeds are determined automatically."
            )
            config.LOGGER.error(msg)
            raise InputError(msg)
    else:
        if self._speed is None:
            msg = (
                "If there is no time-component a speed or speeds must be provided."
            )
            config.LOGGER.error(msg)
            raise InputError(msg)
        if dims not in (2, 3):
            msg = (
                "Non-temporal path supports only 2D or 3D spatial waypoints. "
                "You might have meant to set is_temporal = True."
            )
            config.LOGGER.error(msg)
            raise InputError(msg)

    # ---------- Speed checks ----------
    if self._speed is not None:
        if isinstance(self._speed, tuple):
            # Per-segment speeds: length must be N-1
            if len(self._speed) not in (1, number - 1):
                msg = (
                    "speed tuple length must be 1 (broadcast) "
                    "or N-1={number - 1} (per-segment)."
                )
                config.LOGGER.error(msg)
                raise InputError(msg)
            if not jnp.isfinite(jnp.asarray(self._speed)).all():
                msg = "speed contains NaN or Inf."
                config.LOGGER.error(msg)
                raise InputError(msg)
            if (jnp.asarray(self._speed) <= 0).any():
                msg = "speed values must be positive."
                config.LOGGER.error(msg)
                raise InputError(msg)
        else:
            # Scalar speed
            if not jnp.isfinite(jnp.asarray(self._speed)):
                msg = "speed contains NaN or Inf."
                config.LOGGER.error(msg)
                raise InputError(msg)
            if float(self._speed) <= 0:
                msg = "speed values must be positive."
                config.LOGGER.error(msg)
                raise InputError(msg)

    # Ensure all rows have the same length
    if isinstance(self.waypoints, list):
        first_len = len(self.waypoints[0])
        if any(len(t) != first_len for t in self.waypoints):
            msg = "All waypoint tuples must have the same length."
            config.LOGGER.error(msg)
            raise InputError(msg)
        # When temporal, all tuples must be 3 or 4; otherwise 2 or 3.
        if self.is_temporal and first_len not in (3, 4):
            msg = "With is_temporal=True, tuples must be length 3 or 4."
            config.LOGGER.error(msg)
            raise InputError(msg)
        if not self.is_temporal and first_len not in (2, 3):
            msg = "With is_temporal=False, tuples must be length 2 or 3."
            config.LOGGER.error(msg)
            raise InputError(msg)

    # ------------------------------------------------------------------------------
    spatial_dimensions = dims - (1 if self.is_temporal else 0)
    spatial_coordinates = waypoint_array[:, :spatial_dimensions]

    distances = jnp.linalg.norm(jnp.diff(spatial_coordinates, axis=0), axis=1)
    number_segments = len(distances)
    if (distances == 0).any():
        msg = "Consecutive waypoints must not be identical (zero-length segment)."
        config.LOGGER.error(msg)
        raise InputError(msg)

    object.__setattr__(self, "initials", spatial_coordinates[:-1])
    object.__setattr__(
        self,
        "directions",
        spatial_coordinates[1:] - spatial_coordinates[:-1],
    )

    if self.is_temporal:
        time_deltas = jnp.diff(waypoint_array[:, -1])  # Δt > 0 already checked
        speeds = distances / time_deltas  # per-segment speeds
        if not jnp.isfinite(speeds).all() or (speeds <= 0).any():
            msg = "Derived speeds must be positive and finite."
            config.LOGGER.error(msg)
            raise InputError(msg)
        cumulative_times = waypoint_array[:, -1]
    else:
        speeds = (
            jnp.asarray(self._speed)
            if isinstance(self._speed, (tuple, list))
            else jnp.full((len(distances),), self._speed)
        )
        durations = distances / speeds
        cumulative_times = jnp.concatenate(
            [jnp.array([0.0]), jnp.cumsum(durations)],
        )

    segment_times = jnp.diff(cumulative_times)
    object.__setattr__(self, "cumulative_times", cumulative_times)
    object.__setattr__(self, "segment_times", segment_times)
    object.__setattr__(self, "speeds", tuple(map(float, speeds.tolist())))

    if isinstance(self.active, (bool, int, float)):
        activity = [bool(self.active)] * number_segments

    elif isinstance(self.active, (tuple, list)):
        if len(self.active) != number_segments:
            msg = "Active list Length MUST match number of segments."
            config.LOGGER.error(msg)
            raise InputError(msg)
        activity = [bool(i) for i in self.active]

    elif isinstance(self.active, dict):
        if set(self.active.keys()) != {"off"}:
            msg = 'Active Dictionary must only contain the key "off".'
            config.LOGGER.error(msg)
            raise InputError(msg)
        activity = [True] * number_segments

        off_segments = self.active["off"]
        if isinstance(off_segments, int):
            off_segments = [off_segments]
        elif isinstance(off_segments, tuple):
            off_segments = list(off_segments)

        for k in off_segments:
            if not (1 <= k <= number_segments):
                msg = (
                    f"Invalid segment index {k} in active['off']. "
                    f"Segments are 1-indexed."
                )
                config.LOGGER.error(msg)
                raise InputError(msg)
            activity[k - 1] = False

    else:
        msg = (
            "active must be boolean, scalar, list/tuple of booleans, "
            "or dict {'off': [indices]}."
        )
        config.LOGGER.error(msg)
        raise InputError(msg)

    object.__setattr__(self, "active", jnp.asarray(activity, dtype=bool))
