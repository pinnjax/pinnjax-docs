    class NeuralNetwork(eqx.Module):
    """Parent class for neural networks.

    Children implement `__call__` and `to_meta`. Operator models override `split_inputs`.
    `single_point` evaluates one coordinate vector.

    Attributes:
        input_dim: Number of input components. `None` until a child sets it.
        domain: [`Domain`][pinnjax.Domain] the network is defined on.
            `None` until a child sets it.
        differential_axes: Indices of the differentiable inputs.
            `None` until a child sets it.
        evolution_axis: Index of the time input. `None` until a child sets it.

    """

    input_dim: int | None = eqx.field(static=True)
    domain: Domain | None = eqx.field(static=True)
    differential_axes: tuple[int, ...] | None = eqx.field(static=True)
    evolution_axis: int | None = eqx.field(static=True)

    optimization: Optimization | None = eqx.field(static=True)

    def __init__(self):
        """Initialize common network state."""
        self.optimization = None

    def __call__(self, inputs):
        """Evaluate the network.

        Args:
            inputs: Network inputs.

        Returns:
            The network output.

        Raises:
            NotImplementedError: Always, unless a subclass overrides this method.

        """
        raise NotImplementedError

    def single_point(self, u: Array) -> Array:
        """Evaluate one coordinate vector.

        `u` is expanded to at least one dimension. A leading axis of length 1 is removed
        before `__call__`.

        Args:
            u: One point.

        Returns:
            The network output at `u`.

        """
        # ensure shape (d,)
        u = jnp.atleast_1d(u)
        if u.ndim > 1:
            u = jnp.squeeze(u, axis=0)
        return self(u)

    def split_inputs(self, inputs: Array) -> tuple[Array, Array]:
        """Split one concatenated operator input into descriptor and query.

        The input layout is ``[query..., descriptor...]``. The return order is
        ``(descriptor, query)``.

        Args:
            inputs: One concatenated point.

        Returns:
            The descriptor vector, then the query vector.

        Raises:
            NotImplementedError: Always, unless a subclass overrides this method.

        """
        raise NotImplementedError(
            f"{type(self).__name__} does not define split_inputs(), "
            "so operator residuals cannot separate query/descriptor inputs."
        )

    def to_meta(self) -> dict:
        """Return a dictionary describing this network.

        Raises:
            NotImplementedError: Always, unless a subclass overrides this method.

        """
        raise NotImplementedError

    def _optimization_meta(self) -> dict:
        if self.optimization is None:
            return {}

        return {
            "optimization_spec": optimization_to_spec(self.optimization),
        }

    class ImageNetwork(NeuralNetwork):
    """Base class for neural networks operating on single image tensors.

    `input_shape` is stored channel-first. `(H, W)` becomes `(1, H, W)`.
    `input_dim` is that leading channel count, not the number of domain axes.
    `out_channels` is the leading output channel when the output shape has length 3,
    and `None` otherwise.

    When `domain` is set, `differential_axes` is `tuple(range(len(domain.min)))`.
    `evolution_axis` is `-1` when `domain` is a [`SpatialDomain`][pinnjax.SpatialDomain]
    with time, and `None` otherwise.

    An omitted `optimization` stores [`IdentityInputTransform`][pinnjax.IdentityInputTransform],
    [`IdentityOutputTransformation`][pinnjax.IdentityOutputTransformation], and
    [`IdentityHardConstraint`][pinnjax.IdentityHardConstraint].

    Attributes:
        input_shape: Channel-first `(C, H, W)` image shape.
        output_shape: Output shape. Image shapes are channel-first. Other shapes are
            kept. `None` when the child does not fix one.
        in_channels: `input_shape[0]`.
        out_channels: `output_shape[0]` when `output_shape` has length 3.
            `None` otherwise.
        input_transformation: Map applied to inputs.
        output_transformation: Map applied to outputs.
        hard_constraints: Constraints applied after the output transformation.
            Each is called as `constraint(inputs, outputs)`.

    Args:
        input_shape: `(H, W)` or `(C, H, W)`.
        output_shape: Output shape. `(H, W)` and `(C, H, W)` are stored channel-first.
            Default: None.
        optimization: A [`Optimization`][pinnjax.Optimization] object supplying the
            transformations and hard constraint. Default: None.
        domain: A [`Domain`][pinnjax.Domain] object. Default: None.

    Raises:
        ValueError: If `input_shape` is not rank 2 or 3, or a dimension is not positive.
        ValueError: If an image `output_shape` is invalid, or a vector output dimension
            is not positive.

    """

    input_shape: tuple[int, int, int] = eqx.field(static=True)
    output_shape: tuple[int, ...] | None = eqx.field(static=True)
    in_channels: int = eqx.field(static=True)
    out_channels: int | None = eqx.field(static=True)

    input_transformation: Callable[[Array], Array] | pinnjax.InputTransformation = eqx.field(
        static=True
    )
    output_transformation: Callable[[Array], Array] | pinnjax.OutputTransformation = eqx.field(
        static=True
    )
    hard_constraints: tuple[Callable, ...] = eqx.field(static=True)

    def __init__(
        self,
        *,
        input_shape: tuple[int, ...],
        output_shape: tuple[int, ...] | None = None,
        optimization: Optimization | None = None,
        domain: Domain | None = None,
    ) -> None:
        """Store the image shape, domain axes, transformations, and hard constraints."""
        super().__init__()
        self.input_shape = self._parse_image_shape(input_shape)
        self.output_shape = self._parse_output_shape(output_shape)

        self.in_channels = self.input_shape[0]
        self.out_channels = (
            self.output_shape[0]
            if self.output_shape is not None and len(self.output_shape) == 3
            else None
        )

        # Retain the existing image-network convention for compatibility.
        # ``input_shape`` is the authoritative description of image inputs.
        self.input_dim = self.in_channels
        self.domain = domain

        if domain is None:
            self.differential_axes = None
            self.evolution_axis = None
        else:
            n = len(domain.min)
            has_time = isinstance(domain, SpatialDomain) and domain.times is not None
            self.differential_axes = tuple(range(n))
            self.evolution_axis = -1 if has_time else None

        self.optimization = optimization or Optimization()

        self.input_transformation = (
            self.optimization.input_transformation or IdentityInputTransform()
        )
        self.output_transformation = (
            self.optimization.output_transformation or IdentityOutputTransformation()
        )

        self.hard_constraints = combine_hard_constraints(self.optimization.hard_constraint)

    @staticmethod
    def _parse_image_shape(shape: tuple[int, ...]) -> tuple[int, int, int]:
        """Return a channel-first ``(C, H, W)`` image shape."""
        if len(shape) == 2:
            h, w = (int(shape[0]), int(shape[1]))
            parsed = (1, h, w)
        elif len(shape) == 3:
            parsed = tuple[int](int(dim) for dim in shape)  # ty: ignore[invalid-argument-type]
        else:
            msg = f"Unsupported image shape: {shape}. Expected (H, W) or (C, H, W)."
            config.LOGGER.error(msg)
            raise ValueError(msg)

        if any(dim < 1 for dim in parsed):
            msg = f"Image dimensions must be positive, got {shape}."
            config.LOGGER.error(msg)
            raise ValueError(msg)

        return parsed  # ty: ignore[invalid-return-type]

    @classmethod
    def _parse_output_shape(
        cls,
        shape: tuple[int, ...] | None,
    ) -> tuple[int, ...] | None:
        """Normalize image outputs to ``(C, H, W)`` and preserve vector outputs."""
        if shape is None:
            return None

        shape = tuple(int(dim) for dim in shape)
        if len(shape) in (2, 3):
            return cls._parse_image_shape(shape)

        if any(dim < 1 for dim in shape):
            msg = f"Output dimensions must be positive, got {shape}."
            config.LOGGER.error(msg)
            raise ValueError(msg)

        return shape

    def _apply_output_pipeline(self, inputs: Array, outputs: Array) -> Array:
        outputs = self.output_transformation(outputs)
        for constraint in self.hard_constraints:
            outputs = constraint(inputs, outputs)
        return outputs

    def _image_meta(self) -> dict:
        domain_meta = None
        if self.domain is not None:
            times_meta = None
            if isinstance(self.domain, SpatialDomain) and self.domain.times is not None:
                times = self.domain.times
                times_meta = {
                    "min": float(times.min),
                    "max": float(times.max),
                    "step": float(times.step),
                }

            domain_meta = {
                "min": list(self.domain.min) if self.domain.min else None,
                "max": list(self.domain.max) if self.domain.max else None,
                "axis_names": list(self.domain.axis_names)
                if getattr(self.domain, "axis_names", None)
                else None,
                "times": times_meta,
            }

        return {
            "input_shape": list(self.input_shape),
            "output_shape": list(self.output_shape) if self.output_shape is not None else None,
            "in_channels": self.in_channels,
            "out_channels": self.out_channels,
            "input_dim": self.input_dim,
            "differential_axes": list(self.differential_axes)
            if self.differential_axes is not None
            else None,
            "evolution_axis": self.evolution_axis,
            "domain": domain_meta,
            **self._optimization_meta(),
        }

    FNN #

    Bases: NeuralNetwork, ABC

    class FNN(NeuralNetwork, ABC):
    """Fully connected network of linear layers.

    ``topology`` is the width of each layer, from the input width to the
    output width. Consecutive widths become ``eqx.nn.Linear`` layers. The
    activation runs after every layer except the last. A one-dimensional
    input is one point. Any other rank is mapped over leading axes.

    ``differential_axes`` is ``tuple(range(len(domain.min)))``.
    ``evolution_axis`` is ``-1`` when ``domain`` is a
    [`SpatialDomain`][pinnjax.SpatialDomain] with time, and ``None``
    otherwise.

    An omitted ``optimization`` stores
    [`IdentityInputTransform`][pinnjax.IdentityInputTransform],
    [`IdentityOutputTransformation`][pinnjax.IdentityOutputTransformation],
    and ([`IdentityHardConstraint`][pinnjax.IdentityHardConstraint],).
    An omitted ``initialization`` keeps the equinox linear initialization.

    Args:
        domain: [`Domain`][pinnjax.Domain]. The input width is
            ``len(domain.min)``, plus one when a spatial domain has time.
        topology: Width of each layer, including the input and the output.
        activation: [`Activation`][pinnjax.Activation]. ``act_`` is called
            with ``args`` and ``kwargs``.
        initialization: [`Initialization`][pinnjax.Initialization] for the
            weights and biases. Default: ``None``.
        optimization: [`Optimization`][pinnjax.Optimization] supplying the
            transformations and hard constraint. Default: ``None``.

    Attributes:
        layers: One ``eqx.nn.Linear`` per consecutive pair in ``topology``.
        topology: Stored layer widths.
        activation: The stored [`Activation`][pinnjax.Activation].
        activation_function: ``act_`` bound to ``args`` and ``kwargs``.
        initialization: The stored [`Initialization`][pinnjax.Initialization].
        input_transformation: Map applied to the inputs.
        output_transformation: Map applied after the last linear layer.
        hard_constraints: Constraints applied after the output transformation.
            Each is called as ``constraint(inputs, outputs)``.
        differential_axes: Indices of the spatial inputs.
        evolution_axis: Index of the time input, or ``None``.
        input_dim: ``topology[0]``.
        domain: The stored [`Domain`][pinnjax.Domain].

    """

    layers: list[eqx.nn.Linear]
    topology: tuple[int, ...] = eqx.field(static=True)

    activation: Activation = eqx.field(static=True)
    activation_function: Callable[[Array], Array] = eqx.field(static=True)
    initialization: Initialization = eqx.field(static=True)
    optimization: Optimization = eqx.field(static=True)

    input_transformation: InputTransformation = eqx.field(static=True)
    output_transformation: OutputTransformation = eqx.field(static=True)
    hard_constraints: tuple[HardConstraint, ...] = eqx.field(static=True)

    differential_axes: tuple[int, ...] = eqx.field(static=True)
    evolution_axis: int | None = eqx.field(static=True)

    def __init__(
        self,
        domain: Domain | None,
        topology: tuple[int, ...],
        activation: Activation,
        *,
        initialization: Initialization | None = None,
        optimization: Optimization | None = None,
    ) -> None:
        """Store the layers, the activation, and the output pipeline.

        Raises:
            pinnjax.InputError: If ``domain`` is ``None``, or if the leading
                topology width is not the domain input width.

        """
        super().__init__()
        keys = jax.random.split(jax.random.key(config.get_seed()), len(topology) - 1)
        self.topology = topology
        self.input_dim = self.topology[0]
        self.activation = activation
        self.activation_function = functools.partial(
            activation.act_,
            *activation.args,
            **activation.kwargs,
        )
        self.domain = domain

        self.optimization = optimization or Optimization()
        (
            self.input_transformation,
            self.output_transformation,
            self.hard_constraints,
        ) = _pipeline_from_optimization(optimization)

        self.layers = [
            eqx.nn.Linear(self.topology[i], self.topology[i + 1], key=keys[i])
            for i in range(len(self.topology) - 1)
        ]

        self.initialization = initialization or Initialization()

        for i, (layer, key) in enumerate(zip(self.layers, keys, strict=False)):
            weight_key, bias_key = jax.random.split(key, 2)
            if self.initialization.weights is not None:
                weight = self.initialization.weights(
                    weight_key,
                    layer.weight.shape,
                    layer.weight.dtype,
                )
                layer = eqx.tree_at(lambda L: L.weight, layer, weight)
            if self.initialization.bias is not None:
                layer_bias = cast("Array", layer.bias)
                bias = self.initialization.bias(
                    bias_key,
                    layer_bias.shape,
                    layer_bias.dtype,
                )
                layer = eqx.tree_at(lambda L: L.bias, layer, bias)
            self.layers[i] = layer

        if self.domain is None:
            msg = "FNN currently requires a domain for input dimension validation."
            config.LOGGER.error(msg)
            raise config.InputError(msg)
        n_domain = len(self.domain.min)
        has_time = isinstance(self.domain, SpatialDomain) and self.domain.times is not None
        input_dims = n_domain + (1 if has_time else 0)
        if input_dims != self.topology[0]:
            msg = (
                f"Mismatch between model input size ({self.topology[0]}) and "
                f"domain-derived input_dims ({input_dims})."
            )
            config.LOGGER.error(msg)
            raise config.InputError(msg)

        self.differential_axes, self.evolution_axis = _axes_from_domain(self.domain)

    @eqx.filter_jit
    def __call__(self, inputs: Array) -> Array:
        """Evaluate one point or a batch.

        Wrapped with ``eqx.filter_jit``. The input transformation runs first.
        Hidden layers apply the activation. The last layer does not. The
        output transformation runs next, then each hard constraint. The
        constraints see the original ``inputs``.

        A rank above 1 is flattened over the leading axes and mapped, then
        restored with the last topology width in place of the input width.

        Args:
            inputs: One point, or a batch whose last axis is ``input_dim``.

        Returns:
            The constrained output. A batch keeps its leading shape, with the
            last axis replaced by ``topology[-1]``.

        Raises:
            ValueError: If the transformed input's last axis is not
                ``input_dim``.

        """
        x = self.input_transformation(inputs)

        if x.shape[-1] != self.input_dim:
            raise ValueError(
                f"FNN expects inputs with final dimension {self.input_dim}, got {x.shape}.",
            )

        def forward_point(point: Array) -> Array:
            for layer in self.layers[:-1]:
                point = self.activation_function(layer(point))
            return self.layers[-1](point)

        if x.ndim == 1:
            x = forward_point(x)
        else:
            input_shape = x.shape
            x = x.reshape(-1, self.input_dim)
            x = jax.vmap(forward_point)(x)
            x = x.reshape(*input_shape[:-1], self.topology[-1])

        x = self.output_transformation(x)

        for hc in self.hard_constraints:
            x = hc(inputs, x)

        return x

    def to_meta(self) -> dict:
        """Return a dictionary describing this FNN.

        The activation entry stores the function name, ``args``, and
        ``kwargs``. Domain ``min`` and ``max`` are lists, or ``None`` when
        that bound is empty. Time is ``min``, ``max``, and ``step`` when
        ``domain`` is a [`SpatialDomain`][pinnjax.SpatialDomain] with time.

        Returns:
            Metadata for the topology, activation, and domain.

        Raises:
            pinnjax.InputError: If ``domain`` is ``None``.

        """
        domain = self.domain
        if domain is None:
            msg = "FNN.to_meta requires a domain."
            config.LOGGER.error(msg)
            raise config.InputError(msg)
        t = domain.times if isinstance(domain, SpatialDomain) else None
        times_meta = (
            None
            if t is None
            else {
                "min": float(t.min),
                "max": float(t.max),
                "step": float(t.step),
            }
        )
        return {
            "model_class": "FNN",
            "topology": list(self.topology),
            "activation": {
                "func": cast("Any", self.activation.act_).__name__,
                "args": list(self.activation.args),
                "kwargs": dict(self.activation.kwargs),
            },
            "domain": {
                "min": list(domain.min) or None,
                "max": list(domain.max) or None,
                "times": times_meta,
            },
            **self._optimization_meta(),
        }

    """Convolutional [`ImageNetwork`][pinnjax.ImageNetwork].

    `from_layers` builds the network from a list of layers. `from_spec` repeats a
    convolution and a pooling layer, flattens the image, and appends linear layers.

    Attributes:
        layers: Constructed layers, applied in order.

    Args:
        layers: Constructed layers.
        input_shape: `(H, W)` or `(C, H, W)`.
        output_shape: Output shape. Default: None.
        optimization: A [`Optimization`][pinnjax.Optimization] object. Default: None.
        domain: A [`Domain`][pinnjax.Domain] object. Default: None.

    """

        self,
        *,
        layers: list[Any],
        input_shape: tuple[int, ...],
        output_shape: tuple[int, ...] | None = None,
        optimization: Optimization | None = None,

    ) -> None:

        processed_layers = process_layers(layers)

    @classmethod
    def from_spec(
        cls,
        *,
        input_shape: tuple[int, ...],
        output_shape: tuple[int, ...] | None = None,
        convolution: Convolution,
        pooling: Pooling,
        num_convolutions: int = 1,
        convolutional_activation: Activation | None = None,
        topology: tuple[int | None, ...],
        ffnn_activation: Activation | None = None,
        optimization: Optimization | None = None,
        domain: Domain | None = None,
        auto_topology: bool = False,
    ) -> CNN:
        """Build a CNN from a convolution, a pooling layer, and a linear topology.

        The convolution's input channels are set from `input_shape`. The spatial size
        after `num_convolutions` convolution and pooling pairs is flattened.
        `auto_topology=False` writes that size into a leading `None` in `topology`, and
        the flattened `output_shape` into a trailing `None`. `auto_topology=True`
        puts the flattened size before `topology` and the flattened output size after it.

        Each pair is the convolution, then `pooling`, then `convolutional_activation`
        when one is given. The channels into the next convolution are the previous
        convolution's output channels. `jnp.ravel` follows the pairs. Each step of
        `topology` becomes an `eqx.nn.Linear`, followed by `ffnn_activation` when one
        is given. An `output_shape` of more than one dimension is restored with a reshape.

        Args:
            input_shape: `(H, W)` or `(C, H, W)`.
            output_shape: Output shape. Default: None.
            convolution: Convolution specification. `args` starts with
                `[in_channels, out_channels]`.
            pooling: Pooling specification.
            num_convolutions: Number of convolution and pooling pairs. Default: 1.
            convolutional_activation: Activation after each pooling layer.
                Default: None.
            topology: Width of each linear layer. `None` at either end is filled as
                described above.
            ffnn_activation: Activation after each linear layer. Default: None.
            optimization: A [`Optimization`][pinnjax.Optimization] object. Default: None.
            domain: A [`Domain`][pinnjax.Domain] object. Default: None.
            auto_topology: Prepend and append the flattened sizes instead of replacing
                `None`. Default: False.

        Returns:
            The constructed network.

        Raises:
            ValueError: If `convolution.args` does not include the input and output channels.

        """
        input_shape = cls._parse_image_shape(input_shape)
        output_shape = cls._parse_output_shape(output_shape)

        convolution = cls._convolution_for_input_shape(convolution, input_shape)

        flattened_size = cls._calculate_flattened_size(
            input_shape=input_shape,
            convolution=convolution,
            pooling=pooling,
            num_convolutions=num_convolutions,
        )

        topology_list = list(topology)

        flattened_output_size = None
        if output_shape is not None:
            flattened_output_size = 1
            for dim in output_shape:
                flattened_output_size *= dim

        if auto_topology:
            topology_list = [flattened_size, *topology_list]
            if flattened_output_size is not None:
                topology_list.append(flattened_output_size)
        else:
            if topology_list[0] is None:
                topology_list[0] = flattened_size
            if flattened_output_size is not None and topology_list[-1] is None:
                topology_list[-1] = flattened_output_size

        topology = tuple(topology_list)
        layers = []

        conv_activation_fn = None
        if convolutional_activation is not None:
            conv_activation_fn = functools.partial(
                convolutional_activation.act_,
                *convolutional_activation.args,
                **convolutional_activation.kwargs,
            )

        out_channels = convolution.args[1]
        conv_extra = convolution.args[2:]
        block_in_channels = convolution.args[0]

        for _ in range(num_convolutions):
            block_conv = evolve(
                convolution,
                args=(block_in_channels, out_channels, *conv_extra),
            )
            layers.append(block_conv)
            layers.append(pooling)
            if conv_activation_fn is not None:
                layers.append(conv_activation_fn)
            block_in_channels = out_channels

        layers.append(jnp.ravel)

        if ffnn_activation is not None:
            ffnn_activation_fn = functools.partial(
                ffnn_activation.act_,
                *ffnn_activation.args,
                **ffnn_activation.kwargs,
            )
        else:
            ffnn_activation_fn = None

        for j in range(len(topology) - 1):
            layers.append(
                LayerSpec(
                    eqx.nn.Linear,
                    args=(topology[j], topology[j + 1]),
                ),
            )
            if ffnn_activation_fn is not None:
                layers.append(ffnn_activation_fn)

        if output_shape is not None and len(output_shape) > 1:
            target_shape = tuple(output_shape)

            def _reshape_output(
                x: Array,
                shape: tuple[int, ...] = target_shape,
            ) -> Array:
                return x.reshape(shape)

            layers.append(_reshape_output)

        return cls.from_layers(
            layers,
            input_shape=input_shape,
            output_shape=output_shape,
            optimization=optimization,
            domain=domain,
        )

    @classmethod
    def _convolution_for_input_shape(
        cls,
        convolution: Convolution,
        input_shape: tuple[int, ...],
    ) -> Convolution:
        """Set ``in_channels`` on the convolution spec from ``input_shape``."""
        if len(convolution.args) < 2:
            msg = "Convolution args must include [in_channels, out_channels]"
            config.LOGGER.error(msg)
            raise ValueError(msg)

        in_channels, _, _ = cls._parse_image_shape(input_shape)
        out_channels = convolution.args[1]
        if convolution.args[0] != in_channels:
            config.LOGGER.info(
                "Adjusting convolution in_channels from %s to %s for input_shape %s",
                convolution.args[0],
                in_channels,
                input_shape,
            )

        extra = convolution.args[2:]
        return evolve(convolution, args=(in_channels, out_channels, *extra))

    @classmethod
    def _calculate_flattened_size(
        cls,
        input_shape: tuple[int, ...],
        convolution: Convolution,
        pooling: Pooling,
        num_convolutions: int,
    ) -> int:
        """Calculate flattened size after the convolution/pooling blocks."""
        c, h, w = cls._parse_image_shape(input_shape)

        conv_kwargs = convolution.kwargs.copy()
        if len(convolution.args) >= 3:
            kernel_size = convolution.args[2]
        else:
            kernel_size = conv_kwargs.get("kernel_size", 3)

        if isinstance(kernel_size, (tuple, list)):
            kh, kw = kernel_size
        else:
            kh = kw = kernel_size

        if len(convolution.args) >= 4:
            stride = convolution.args[3]
        else:
            stride = conv_kwargs.get("stride", 1)
        if isinstance(stride, (tuple, list)):
            sh, sw = stride
        else:
            sh = sw = stride

        if len(convolution.args) >= 5:
            padding = convolution.args[4]
        else:
            padding = conv_kwargs.get("padding", 0)
        if isinstance(padding, (tuple, list)):
            ph, pw = padding
        else:
            ph = pw = padding

        pool_kwargs = pooling.kwargs.copy()
        if len(pooling.args) >= 1:
            pool_size = pooling.args[0]
        else:
            pool_size = pool_kwargs.get("kernel_size", 2)
        if isinstance(pool_size, (tuple, list)):
            pkh, pkw = pool_size
        else:
            pkh = pkw = pool_size

        if len(pooling.args) >= 2:
            pool_stride = pooling.args[1]
        elif "stride" in pool_kwargs:
            pool_stride = pool_kwargs["stride"]
        else:
            pool_stride = 1
        if isinstance(pool_stride, (tuple, list)):
            psh, psw = pool_stride
        else:
            psh = psw = pool_stride

        pool_padding = pool_kwargs.get("padding", 0)
        if isinstance(pool_padding, (tuple, list)):
            pph, ppw = pool_padding
        else:
            pph = ppw = pool_padding

        if len(convolution.args) >= 2:
            out_channels = convolution.args[1]
        else:
            msg = "Convolution args must include [in_channels, out_channels]"
            config.LOGGER.error(msg)
            raise ValueError(msg)

        current_h, current_w = h, w
        current_c = c

        for _ in range(num_convolutions):
            current_h = (current_h - kh + 2 * ph) // sh + 1
            current_w = (current_w - kw + 2 * pw) // sw + 1
            current_c = out_channels

            current_h = (current_h - pkh + 2 * pph) // psh + 1
            current_w = (current_w - pkw + 2 * ppw) // psw + 1

        return current_h * current_w * current_c

    @eqx.filter_jit
    def __call__(self, inputs: Array) -> Array:
        """Evaluate the network on one image.

        The input transformation is applied first, then each layer, then the output
        transformation and hard constraints. The constraints receive the original `inputs`.

        Utilizes `jax.jit` for computational efficiency.

        Args:
            inputs: One image tensor.

        Returns:
            The network output.

        """
        x = self.input_transformation(inputs)
        for layer in self.layers:
            x = layer(x)
        return self._apply_output_pipeline(inputs, x)

    def to_meta(self) -> dict:
        """Return a dictionary describing this network.

        The image fields come from [`ImageNetwork`][pinnjax.ImageNetwork].

        Returns:
            model_class: ``"CNN"``.
            num_layers: Number of stored layers.
            layer_types: Class name of each layer.

        """
        return {
            "model_class": "CNN",
            **self._image_meta(),
            "num_layers": len(self.layers),
            "layer_types": [type(layer).__name__ for layer in self.layers],
        }

        topology_list = [flattened_size, *topology_list]
        if flattened_output_size is not None:
            topology_list.append(flattened_output_size)
    else:
        if topology_list[0] is None:
            topology_list[0] = flattened_size
        if flattened_output_size is not None and topology_list[-1] is None:
            topology_list[-1] = flattened_output_size

    topology = tuple(topology_list)
    layers = []

    conv_activation_fn = None
    if convolutional_activation is not None:
        conv_activation_fn = functools.partial(
            convolutional_activation.act_,
            *convolutional_activation.args,
            **convolutional_activation.kwargs,
        )

    out_channels = convolution.args[1]
    conv_extra = convolution.args[2:]
    block_in_channels = convolution.args[0]

    for _ in range(num_convolutions):
        block_conv = evolve(
            convolution,
            args=(block_in_channels, out_channels, *conv_extra),
        )
        layers.append(block_conv)
        layers.append(pooling)
        if conv_activation_fn is not None:
            layers.append(conv_activation_fn)
        block_in_channels = out_channels

    layers.append(jnp.ravel)

    if ffnn_activation is not None:
        ffnn_activation_fn = functools.partial(
            ffnn_activation.act_,
            *ffnn_activation.args,
            **ffnn_activation.kwargs,
        )
    else:
        ffnn_activation_fn = None

    for j in range(len(topology) - 1):
        layers.append(
            LayerSpec(
                eqx.nn.Linear,
                args=(topology[j], topology[j + 1]),
            ),
        )
        if ffnn_activation_fn is not None:
            layers.append(ffnn_activation_fn)

    if output_shape is not None and len(output_shape) > 1:
        target_shape = tuple(output_shape)

        def _reshape_output(
            x: Array,
            shape: tuple[int, ...] = target_shape,
        ) -> Array:
            return x.reshape(shape)

        layers.append(_reshape_output)

    return cls.from_layers(
        layers,
        input_shape=input_shape,
        output_shape=output_shape,
        optimization=optimization,
        domain=domain,
    )

    """General image-to-image U-Net with encoder-decoder skip connections.

    ``from_spec`` builds the network. ``width`` is the channel count after
    the first block, and it doubles at each downsampling. ``depth`` is the
    number of stride-2 convolutions. The decoder resizes back to each skip,
    projects channels with a 1x1 convolution, and concatenates that skip on
    the channel axis. Skips are the encoder outputs above the bottleneck.

    Input and output are channel-first images with the same height and width.
    ``(H, W)`` is stored as ``(1, H, W)``. An omitted ``optimization`` stores
    [`IdentityInputTransform`][pinnjax.IdentityInputTransform],
    [`IdentityOutputTransformation`][pinnjax.IdentityOutputTransformation],
    and ([`IdentityHardConstraint`][pinnjax.IdentityHardConstraint],).

    Args:
        encoder: Encoder blocks. ``from_spec`` builds ``depth + 1`` of them.
        downs: Stride-2 convolutions, one per level.

            concatenation.
        decoder: Decoder blocks, one per upsampling.
        head: Final 1x1 convolution to the output channels.
        input_shape: ``(H, W)`` or ``(C, H, W)``.
        output_shape: ``(H, W)`` or ``(C, H, W)``, with the same spatial size
            as ``input_shape``.

        depth: Number of downsampling levels.
        optimization: [`Optimization`][pinnjax.Optimization] supplying the
            transformations and hard constraint. Default: ``None``.
        domain: [`Domain`][pinnjax.Domain]. Default: ``None``.

        downs: Stored downsampling convolutions.
        ups: Stored channel projections.
        decoder: Stored decoder blocks.
        head: Final convolution.
        width: Base channel count.
        depth: Number of downsampling levels.

    decoder: tuple

    @classmethod
    def from_spec(
        cls,
        *,
        input_shape: tuple[int, ...],
        output_shape: tuple[int, ...],
        width: int = 16,
        depth: int = 2,
        activation: Callable = jax.nn.gelu,
        optimization: Optimization | None = None,
        domain: Domain | None = None,
    ) -> UNet:
        """Build a U-Net from a base width and a number of levels.

        The first block maps the input channels to ``width``. Each level then
        applies a stride-2 convolution with kernel 2, doubles the channels,
        and applies a [`ConvBlock`][pinnjax.model.cnn.ConvBlock]. The decoder
        walks those levels in reverse. Each step resizes to the skip, applies
        a 1x1 convolution down to that skip's channel count, concatenates the
        skip, and applies a ``ConvBlock`` that returns the skip's channel
        count. A final 1x1 convolution maps ``width`` channels to the output
        channels. Activations and dtype come from ``activation`` and
        ``config.get_dtype()``. ``process_layers`` turns the specifications
        into equinox modules.

        Args:
            input_shape: ``(H, W)`` or ``(C, H, W)``.
            output_shape: ``(H, W)`` or ``(C, H, W)``, with the same spatial
                size as ``input_shape``.
            width: Channels after the first block. Default: ``16``.
            depth: Number of downsampling levels. Default: ``2``.
            activation: Activation inside each ``ConvBlock``.
                Default: ``jax.nn.gelu``.
            optimization: [`Optimization`][pinnjax.Optimization] supplying the
                transformations and hard constraint. Default: ``None``.
            domain: [`Domain`][pinnjax.Domain]. Default: ``None``.

        Returns:
            A [`UNet`][pinnjax.model.UNet].

        Raises:
            ValueError: If a shape is not an image, if a dimension is not
                positive, if the spatial sizes differ, or if ``width`` or
                ``depth`` is less than 1.

        """
        input_shape = cls._parse_image_shape(input_shape)
        parsed_output_shape = cls._parse_output_shape(output_shape)

        if parsed_output_shape is None or len(parsed_output_shape) != 3:
            raise ValueError("UNet requires an image output_shape.")
        output_shape = parsed_output_shape
        if input_shape[1:] != output_shape[1:]:
            raise ValueError(
                "UNet currently requires matching input/output spatial dimensions, "
                f"got {input_shape[1:]} and {output_shape[1:]}.",
            )
        if width < 1:
            raise ValueError(f"width must be at least 1, got {width}.")
        if depth < 1:
            raise ValueError(f"depth must be at least 1, got {depth}.")

        in_channels = input_shape[0]
        out_channels = output_shape[0]
        dtype = config.get_dtype()

        encoder = [
            ConvBlock(
                in_channels,
                width,
                activation=activation,
            ),
        ]
        downs = []
        channels = width

        for _ in range(depth):
            downs.append(
                Convolution(
                    eqx.nn.Conv2d,
                    args=(channels, 2 * channels, 2),
                    kwargs={
                        "stride": 2,
                        "dtype": dtype,
                    },
                ),
            )

            channels *= 2

            encoder.append(
                ConvBlock(
                    channels,
                    channels,
                    activation=activation,
                ),
            )

        ups = []
        decoder = []

        for stage in range(depth):
            skip_channels = width * 2 ** (depth - 1 - stage)
            source_channels = width * 2 ** (depth - stage)

            ups.append(
                Convolution(
                    eqx.nn.Conv2d,
                    args=(source_channels, skip_channels, 1),
                    kwargs={"dtype": dtype},
                ),
            )

            decoder.append(
                ConvBlock(
                    2 * skip_channels,
                    skip_channels,
                    activation=activation,
                ),
            )

        head = Convolution(
            eqx.nn.Conv2d,
            args=(width, out_channels, 1),
            kwargs={"dtype": dtype},
        )

        encoder, downs, ups, decoder, head = process_layers(
            (
                tuple(encoder),
                tuple(downs),
                tuple(ups),
                tuple(decoder),
                head,
            ),
        )

        return cls(
            encoder=encoder,
            downs=downs,
            ups=ups,
            decoder=decoder,
            head=head,
            input_shape=input_shape,
            output_shape=output_shape,
            width=width,
            depth=depth,
            optimization=optimization,
            domain=domain,
        )

    @eqx.filter_jit
    def __call__(self, inputs: Array) -> Array:
        """Evaluate one channel-first image.

        Wrapped with ``eqx.filter_jit``. The input transformation runs first.
        Encoder outputs above the bottleneck are kept as skips. Each decoder
        step resizes to the next skip, projects channels, concatenates that
        skip, and decodes. The output transformation and the hard constraints
        then run. The constraints see the original ``inputs``.

        Args:
            inputs: One channel-first image.

        Returns:
            The output image.

        """
        x = self.input_transformation(inputs)

        skips = []
        x = self.encoder[0](x)
        skips.append(x)

        for i, down in enumerate(self.downs):
            x = down(x)
            x = self.encoder[i + 1](x)
            if i < self.depth - 1:
                skips.append(x)

        for i, (up, decode) in enumerate(zip(self.ups, self.decoder, strict=True)):
            skip = skips[-(i + 1)]
            x = _match_spatial(x, skip)
            x = up(x)
            x = decode(jnp.concatenate([x, skip], axis=0))

        x = self.head(x)
        return self._apply_output_pipeline(inputs, x)

        """
        return {
            "model_class": "UNet",

def from_spec(
    cls,
    *,
    input_shape: tuple[int, ...],
    output_shape: tuple[int, ...],
    width: int = 16,
    depth: int = 2,
    activation: Callable = jax.nn.gelu,
    optimization: Optimization | None = None,
    domain: Domain | None = None,
) -> UNet:
    """Build a U-Net from a base width and a number of levels.

    The first block maps the input channels to ``width``. Each level then
    applies a stride-2 convolution with kernel 2, doubles the channels,
    and applies a [`ConvBlock`][pinnjax.model.cnn.ConvBlock]. The decoder
    walks those levels in reverse. Each step resizes to the skip, applies
    a 1x1 convolution down to that skip's channel count, concatenates the
    skip, and applies a ``ConvBlock`` that returns the skip's channel
    count. A final 1x1 convolution maps ``width`` channels to the output
    channels. Activations and dtype come from ``activation`` and
    ``config.get_dtype()``. ``process_layers`` turns the specifications
    into equinox modules.

    Args:
        input_shape: ``(H, W)`` or ``(C, H, W)``.
        output_shape: ``(H, W)`` or ``(C, H, W)``, with the same spatial
            size as ``input_shape``.
        width: Channels after the first block. Default: ``16``.
        depth: Number of downsampling levels. Default: ``2``.
        activation: Activation inside each ``ConvBlock``.
            Default: ``jax.nn.gelu``.
        optimization: [`Optimization`][pinnjax.Optimization] supplying the
            transformations and hard constraint. Default: ``None``.
        domain: [`Domain`][pinnjax.Domain]. Default: ``None``.

    Returns:
        A [`UNet`][pinnjax.model.UNet].

    Raises:
        ValueError: If a shape is not an image, if a dimension is not
            positive, if the spatial sizes differ, or if ``width`` or
            ``depth`` is less than 1.

    """
    input_shape = cls._parse_image_shape(input_shape)
    parsed_output_shape = cls._parse_output_shape(output_shape)

    if parsed_output_shape is None or len(parsed_output_shape) != 3:
        raise ValueError("UNet requires an image output_shape.")
    output_shape = parsed_output_shape
    if input_shape[1:] != output_shape[1:]:
        raise ValueError(
            "UNet currently requires matching input/output spatial dimensions, "
            f"got {input_shape[1:]} and {output_shape[1:]}.",
        )
    if width < 1:
        raise ValueError(f"width must be at least 1, got {width}.")
    if depth < 1:
        raise ValueError(f"depth must be at least 1, got {depth}.")

    in_channels = input_shape[0]
    out_channels = output_shape[0]
    dtype = config.get_dtype()

    encoder = [
        ConvBlock(
            in_channels,
            width,
            activation=activation,
        ),
    ]
    downs = []
    channels = width

    for _ in range(depth):
        downs.append(
            Convolution(
                eqx.nn.Conv2d,
                args=(channels, 2 * channels, 2),
                kwargs={
                    "stride": 2,
                    "dtype": dtype,
                },
            ),
        )

        channels *= 2

        encoder.append(
            ConvBlock(
                channels,
                channels,
                activation=activation,
            ),
        )

    ups = []
    decoder = []

    for stage in range(depth):
        skip_channels = width * 2 ** (depth - 1 - stage)
        source_channels = width * 2 ** (depth - stage)

        ups.append(
            Convolution(
                eqx.nn.Conv2d,
                args=(source_channels, skip_channels, 1),
                kwargs={"dtype": dtype},
            ),
        )

        decoder.append(
            ConvBlock(
                2 * skip_channels,
                skip_channels,
                activation=activation,
            ),
        )

    head = Convolution(
        eqx.nn.Conv2d,
        args=(width, out_channels, 1),
        kwargs={"dtype": dtype},
    )

    encoder, downs, ups, decoder, head = process_layers(
        (
            tuple(encoder),
            tuple(downs),
            tuple(ups),
            tuple(decoder),
            head,
        ),
    )

    return cls(
        encoder=encoder,
        downs=downs,
        ups=ups,
        decoder=decoder,
        head=head,
        input_shape=input_shape,
        output_shape=output_shape,
        width=width,
        depth=depth,
        optimization=optimization,
        domain=domain,
    )

    class PointNet(NeuralNetwork, ABC):
    """Permutation-invariant point-cloud network with the FNN construction contract.

    ``topology`` is the shared per-point encoder, ending in the global feature
    width. ``head`` is the per-point decoder after concatenating the local skip
    with the max-pooled global feature. ``__call__`` expects one cloud of shape
    ``(N, d)``.

    Every encoder layer is activated, including the last. The local skip is
    that activation at ``local_feature_index``. The global feature is the
    maximum of the last encoder activation over the points. The decoder
    activates every layer except its last.

    ``differential_axes`` is ``tuple(range(len(domain.min)))``.
    ``evolution_axis`` is ``-1`` when ``domain`` is a
    [`SpatialDomain`][pinnjax.SpatialDomain] with time, and ``None``
    otherwise.

    An omitted ``optimization`` stores
    [`IdentityInputTransform`][pinnjax.IdentityInputTransform],
    [`IdentityOutputTransformation`][pinnjax.IdentityOutputTransformation],
    and ([`IdentityHardConstraint`][pinnjax.IdentityHardConstraint],).
    An omitted ``initialization`` keeps the equinox linear initialization.

    Follows the work by Qi et al.
    https://doi.org/10.48550/arXiv.1612.00593

    Args:
        domain: [`Domain`][pinnjax.Domain]. The input width is
            ``len(domain.min)``, plus one when a spatial domain has time.
        topology: Encoder widths, from the point width to the global feature
            width. At least two entries.
        activation: [`Activation`][pinnjax.Activation]. ``act_`` is called
            with ``args`` and ``kwargs``.
        head: Decoder widths. ``head[0]`` is the local feature width plus
            the global width. At least two entries.
        local_feature_index: Encoder layer whose activation is the local
            skip. ``1`` is the first layer. Default: ``1``.
        initialization: [`Initialization`][pinnjax.Initialization] for the
            weights and biases. Default: ``None``.
        optimization: [`Optimization`][pinnjax.Optimization] supplying the
            transformations and hard constraint. Default: ``None``.

    Attributes:
        encoder: One ``eqx.nn.Linear`` per consecutive pair in ``topology``.
        decoder: One ``eqx.nn.Linear`` per consecutive pair in ``head``.
        topology: Stored encoder widths.
        head: Stored decoder widths.
        local_feature_index: Encoder layer used as the local skip.
        activation: The stored [`Activation`][pinnjax.Activation].
        activation_function: ``act_`` bound to ``args`` and ``kwargs``.
        initialization: The stored [`Initialization`][pinnjax.Initialization].
        input_transformation: Map applied to the point cloud.
        output_transformation: Map applied to the per-point outputs.
        hard_constraints: Constraints applied after the output transformation.
            Each is called as ``constraint(inputs, outputs)``.
        differential_axes: Indices of the spatial inputs.
        evolution_axis: Index of the time input, or ``None``.
        input_dim: ``topology[0]``.
        domain: The stored [`Domain`][pinnjax.Domain].

    """

    encoder: list[eqx.nn.Linear]
    decoder: list[eqx.nn.Linear]
    topology: tuple[int, ...] = eqx.field(static=True)
    head: tuple[int, ...] = eqx.field(static=True)
    local_feature_index: int = eqx.field(static=True)

    activation: Activation = eqx.field(static=True)
    activation_function: Callable[[Array], Array] = eqx.field(static=True)
    initialization: Initialization = eqx.field(static=True)
    optimization: Optimization = eqx.field(static=True)

    input_transformation: Callable[[Array], Array] = eqx.field(static=True)
    output_transformation: Callable[[Array], Array] = eqx.field(static=True)
    hard_constraints: tuple[Callable, ...] = eqx.field(static=True)

    differential_axes: tuple[int, ...] = eqx.field(static=True)
    evolution_axis: int | None = eqx.field(static=True)

    def __init__(
        self,
        domain: Domain | None,
        topology: tuple[int, ...],
        activation: Activation,
        *,
        head: tuple[int, ...],
        local_feature_index: int = 1,
        initialization: Initialization | None = None,
        optimization: Optimization | None = None,
    ) -> None:
        """Store the encoder, the decoder, and the output pipeline.

        Raises:
            pinnjax.InputError: If ``domain`` is ``None``, if ``topology`` or
                ``head`` has fewer than two entries, if
                ``local_feature_index`` is outside ``[1, len(topology) - 1]``,
                if ``head[0]`` is not the local width plus the global width,
                or if the leading topology width is not the domain input
                width.

        """
        super().__init__()

        if len(topology) < 2:
            msg = f"PointNet topology must have at least two entries, got {topology}."
            config.LOGGER.error(msg)
            raise config.InputError(msg)
        if len(head) < 2:
            msg = f"PointNet head must have at least two entries, got {head}."
            config.LOGGER.error(msg)
            raise config.InputError(msg)
        if not 1 <= local_feature_index <= len(topology) - 1:
            msg = (
                "local_feature_index must be a topology index in "
                f"[1, {len(topology) - 1}], got {local_feature_index}."
            )
            config.LOGGER.error(msg)
            raise config.InputError(msg)

        expected_head_in = topology[local_feature_index] + topology[-1]
        if head[0] != expected_head_in:
            msg = (
                f"Head input width ({head[0]}) must equal local feature width "
                f"({topology[local_feature_index]}) plus global width "
                f"({topology[-1]})."
            )
            config.LOGGER.error(msg)
            raise config.InputError(msg)

        self.topology = topology
        self.head = head
        self.local_feature_index = int(local_feature_index)
        self.input_dim = self.topology[0]
        self.activation = activation
        self.activation_function = functools.partial(
            activation.act_,
            *activation.args,
            **activation.kwargs,
        )
        self.domain = domain

        self.optimization = optimization or Optimization()
        (
            self.input_transformation,
            self.output_transformation,
            self.hard_constraints,
        ) = _pipeline_from_optimization(optimization)

        n_encoder = len(self.topology) - 1
        n_decoder = len(self.head) - 1
        keys = jax.random.split(
            jax.random.key(config.get_seed()),
            n_encoder + n_decoder,
        )

        self.initialization = initialization or Initialization()
        self.encoder = _linear_stack(
            self.topology,
            keys[:n_encoder],
            self.initialization,
        )
        self.decoder = _linear_stack(
            self.head,
            keys[n_encoder:],
            self.initialization,
        )
        if self.domain is None:
            msg = "PointNet currently requires a domain for input dimension validation."
            config.LOGGER.error(msg)
            raise config.InputError(msg)
        n_domain = len(self.domain.min)
        has_time = isinstance(self.domain, SpatialDomain) and self.domain.times is not None
        input_dims = n_domain + (1 if has_time else 0)
        if input_dims != self.topology[0]:
            msg = (
                f"Mismatch between model input size ({self.topology[0]}) and "
                f"domain-derived input_dims ({input_dims})."
            )
            config.LOGGER.error(msg)
            raise config.InputError(msg)

        self.differential_axes, self.evolution_axis = _axes_from_domain(self.domain)

    def _encode(self, points: Array) -> tuple[Array, Array]:
        def point_encoder(point: Array) -> tuple[Array, Array]:
            _local = point
            x = point
            for i, layer in enumerate(self.encoder):
                x = self.activation_function(layer(x))
                if i + 1 == self.local_feature_index:
                    _local = x
            return _local, x

        local, encoded = jax.vmap(point_encoder)(points)
        return local, jnp.max(encoded, axis=0)

    def _decode(self, local: Array, global_feature: Array) -> Array:
        def point_decoder(local_feature: Array) -> Array:
            x = jnp.concatenate([local_feature, global_feature], axis=-1)
            for layer in self.decoder[:-1]:
                x = self.activation_function(layer(x))
            return self.decoder[-1](x)

        return jax.vmap(point_decoder)(local)

    @eqx.filter_jit
    def __call__(self, inputs: Array) -> Array:
        """Evaluate one point cloud.

        Wrapped with ``eqx.filter_jit``. The cloud must have shape
        ``(N, input_dim)``. The input transformation runs on that cloud.
        Each point is encoded, the global feature is the max over points, and
        the decoder maps every local skip. The output transformation and the
        hard constraints then run. The constraints see the original
        ``inputs``.

        Args:
            inputs: One cloud, shape ``(N, input_dim)``.

        Returns:
            One output row per point.

        Raises:
            ValueError: If ``inputs`` is not rank 2, or its last axis is not
                ``input_dim``.

        """
        if inputs.ndim != 2 or inputs.shape[-1] != self.input_dim:
            msg = (
                f"PointNet expects a point cloud of shape (N, {self.input_dim}), "
                f"got {inputs.shape}."
            )
            config.LOGGER.error(msg)
            raise ValueError(msg)

        x = self.input_transformation(inputs)
        local, global_feature = self._encode(x)
        outputs = self._decode(local, global_feature)
        return _apply_output_pipeline(
            inputs,
            outputs,
            self.output_transformation,
            self.hard_constraints,
        )

    def to_meta(self) -> dict:
        """Return a dictionary describing this PointNet.

        The activation entry stores the function name, ``args``, and
        ``kwargs``. Domain ``min`` and ``max`` are lists, or ``None`` when
        that bound is empty. Time is ``min``, ``max``, and ``step`` when
        ``domain`` is a [`SpatialDomain`][pinnjax.SpatialDomain] with time.

        Returns:
            Metadata for the topology, head, activation, and domain.

        Raises:
            pinnjax.InputError: If ``domain`` is ``None``.

        """
        domain = self.domain
        if domain is None:
            msg = "PointNet.to_meta requires a domain."
            config.LOGGER.error(msg)
            raise config.InputError(msg)
        t = domain.times if isinstance(domain, SpatialDomain) else None
        times_meta = (
            None

        *activation.args,

    class NeuralOperator(NeuralNetwork):
    """Concatenation neural operator: ``G(u)(y) = network([y..., u...])``.

    Flat inputs follow [`OperatorGeometry`][pinnjax.OperatorGeometry]:

    ```
    [query..., descriptor...]
    ```

    ``query_dim`` includes time when the query has time. ``descriptor_dim`` is
    the network input width minus ``query_dim``. ``differential_axes`` indexes
    the spatial query axes. ``evolution_axis`` is the last query component
    when the query has time, and ``None`` otherwise. ``domain`` is the query.

    The wrapped network applies its own transformations and hard constraints.

    Args:
        operator: [`OperatorDomain`][pinnjax.OperatorDomain] with the query
            and the descriptor.
        network: Network whose input width is at least ``query_dim``. The
            extra width is the descriptor.

    Attributes:
        operator: The stored [`OperatorDomain`][pinnjax.OperatorDomain].
        network: The wrapped [`NeuralNetwork`][pinnjax.NeuralNetwork].
        query_dim: Query length, including time when ``query.times`` is set.
        descriptor_dim: ``input_dim - query_dim``.
        input_dim: The wrapped network's input width.
        differential_axes: ``tuple(range(len(query.min)))``.
        evolution_axis: ``query_dim - 1`` when the query has time, else
            ``None``.
        domain: The query [`SpatialDomain`][pinnjax.SpatialDomain].

    """

    network: NeuralNetwork

    query_dim: int = eqx.field(static=True)
    descriptor_dim: int = eqx.field(static=True)
    input_dim: int = eqx.field(static=True)

    differential_axes: tuple[int, ...] = eqx.field(static=True)

    def __init__(
        self,
        operator: OperatorDomain,
        network: NeuralNetwork,
    ) -> None:
        """Store the operator and the wrapped network.

        Raises:
            ValueError: If the network input width is shorter than
                ``query_dim``.
            TypeError: If the network has no ``input_dim`` and no
                ``topology``.

        """
        super().__init__()

        q = operator.query
        query_dim = _query_dim_from_domain(q)
        total_in = _network_input_dim(network)

        if total_in < query_dim:
            raise ValueError(
                f"network input_dim ({total_in}) must be at least query_dim ({query_dim}).",
            )

        descriptor_dim = total_in - query_dim

        self.operator = operator
        self.network = network

        self.query_dim = query_dim
        self.descriptor_dim = descriptor_dim
        self.input_dim = total_in

        self.differential_axes = tuple(range(len(q.min)))
        self.evolution_axis = (query_dim - 1) if q.times is not None else None
        self.domain = q

    @classmethod
    def parametric_from_fnn(
        cls,
        operator: OperatorDomain,
        topology: tuple[int, ...],
        activation: Activation,
        *,
        initialization: Initialization | None = None,
        optimization: Optimization | None = None,
    ) -> NeuralOperator:
        """Build a [`NeuralOperator`][pinnjax.NeuralOperator] around an [`FNN`][pinnjax.FNN].

        ``operator.combined_domain()`` is the FNN domain. That domain already
        includes query time when the query has time, then the descriptor
        bounds. ``topology[0]`` must equal the length of that combined
        ``min``.

        Args:
            operator: [`OperatorDomain`][pinnjax.OperatorDomain] with the query
                and the descriptor.
            topology: Width of each FNN layer, including the input and the
                output.
            activation: [`Activation`][pinnjax.Activation] for the FNN.
            initialization: [`Initialization`][pinnjax.Initialization] for the
                FNN weights and biases. Default: ``None``.
            optimization: [`Optimization`][pinnjax.Optimization] for the FNN.
                Default: ``None``.

        Returns:
            A neural operator whose network is that FNN.

        Raises:
            ValueError: If ``topology[0]`` is not the combined domain length.

        """
        input_domain = operator.combined_domain()

        if len(input_domain.min) != topology[0]:
            raise ValueError(
                f"input_domain dimension ({len(input_domain.min)}) must match "
                f"topology[0] ({topology[0]}).",
            )

        network = FNN(
            input_domain,
            topology,
            activation,
            initialization=initialization,
            optimization=optimization,
        )
        return cls(operator, network=network)

    def split_inputs(self, inputs: Array) -> tuple[Array, Array]:
        """Return ``(descriptor_input, query_input)`` from ``[query..., descriptor...]``.

        Args:
            inputs: One concatenated point.

        Returns:
            The descriptor vector, then the query vector.

        """
        y = inputs[: self.query_dim]
        u = inputs[self.query_dim :]
        return u, y

    def forward(self, descriptor_input: Array, query_input: Array) -> Array:
        """Evaluate the wrapped network on one query and one descriptor.

        Args:
            descriptor_input: One descriptor vector.
            query_input: One query vector.

        Returns:
            ``network(concatenate([query_input, descriptor_input]))``.

        """
        return self.network(jnp.concatenate([query_input, descriptor_input]))

    class DeepONet(NeuralNetwork):
    """DeepONet: ``G(u)(y) = dot(branch(u), trunk(y)) + bias``.

    The branch reads the descriptor. The trunk reads the query. One call
    passes the query coordinates first and the descriptor after them.
    ``differential_axes`` indexes the spatial query axes. ``evolution_axis``
    is the last query component when the query has time, and ``None``
    otherwise.

    An omitted ``optimization`` stores
    [`IdentityOutputTransformation`][pinnjax.IdentityOutputTransformation]
    and ([`IdentityHardConstraint`][pinnjax.IdentityHardConstraint],).
    The input transformation on ``optimization`` is unused.

    Args:
        operator: [`OperatorDomain`][pinnjax.OperatorDomain] with the query
            and the descriptor.
        branch: Network for the descriptor. Its input width is at least
            ``len(descriptor.min)``.
        trunk: Network for the query. Its input width equals the query
            length, including time when the query has time.
        bias: Added after the dot product. Default: ``0``.
        latent_dim: Shared branch and trunk output width. Default: the
            branch output width.
        optimization: [`Optimization`][pinnjax.Optimization] supplying the
            output transformation and hard constraint. Default: ``None``.

    Attributes:
        operator: The stored [`OperatorDomain`][pinnjax.OperatorDomain].
        branch: Descriptor network.
        trunk: Query network.
        bias: Value added after the dot product.
        query_dim: Query length, including time when ``query.times`` is set.
        descriptor_dim: ``len(descriptor.min)``.
        branch_input_dim: Branch input width.
        trunk_input_dim: Trunk input width.
        input_dim: ``query_dim + branch_input_dim``.
        latent_dim: Shared output width of the branch and the trunk.
        differential_axes: ``tuple(range(len(query.min)))``.
        evolution_axis: ``query_dim - 1`` when the query has time, else
            ``None``.
        domain: The query [`SpatialDomain`][pinnjax.SpatialDomain].
        output_transformation: Map applied to the dot product.
        hard_constraints: Constraints applied after that map by ``__call__``.

    """

    operator: OperatorDomain = eqx.field(static=True)

    branch: NeuralNetwork
    trunk: NeuralNetwork
    bias: Array

    query_dim: int = eqx.field(static=True)
    descriptor_dim: int = eqx.field(static=True)
    branch_input_dim: int = eqx.field(static=True)
    trunk_input_dim: int = eqx.field(static=True)
    input_dim: int = eqx.field(static=True)
    latent_dim: int = eqx.field(static=True)

    differential_axes: tuple[int, ...] = eqx.field(static=True)
    evolution_axis: int | None = eqx.field(static=True)
    domain: SpatialDomain | None = eqx.field(static=True)

    output_transformation: Callable[[Array], Array] = eqx.field(static=True)
    hard_constraints: tuple = eqx.field(static=True)

    def __init__(
        self,
        operator: OperatorDomain,
        branch: NeuralNetwork,
        trunk: NeuralNetwork,
        *,
        bias: float | Array | None = None,
        latent_dim: int | None = None,
        optimization: Optimization | None = None,
    ) -> None:
        """Store the branch, the trunk, and the output pipeline.

        Raises:
            ValueError: If the trunk input width is not ``query_dim``, if the
                branch input width is shorter than the descriptor, or if the
                branch output, trunk output, and ``latent_dim`` disagree.
            TypeError: If a network has no ``input_dim`` and no ``topology``.

        """
        super().__init__()

        q = operator.query
        query_dim = _query_dim_from_domain(q)
        descriptor_dim = len(operator.descriptor.min)

        branch_input_dim = _network_input_dim(branch)
        trunk_input_dim = _network_input_dim(trunk)

        if trunk_input_dim != query_dim:
            raise ValueError(
                f"trunk input_dim ({trunk_input_dim}) must match query_dim ({query_dim}).",
            )

        if branch_input_dim < descriptor_dim:
            raise ValueError(
                f"branch input_dim ({branch_input_dim}) must be at least "
                f"operator descriptor dim ({descriptor_dim}).",
            )

        branch_output_dim = _network_output_dim(branch, branch_input_dim)
        trunk_output_dim = _network_output_dim(trunk, trunk_input_dim)

        if branch_output_dim != trunk_output_dim:
            raise ValueError(
                f"branch output_dim ({branch_output_dim}) must match trunk "
                f"output_dim ({trunk_output_dim}).",
            )

        if latent_dim is not None and latent_dim != branch_output_dim:
            raise ValueError(
                f"latent_dim ({latent_dim}) must match branch/trunk output_dim "
                f"({branch_output_dim}).",
            )

        if bias is None:
            bias_arr = jnp.array(0.0, dtype=config.get_dtype())
        else:
            bias_arr = jnp.asarray(bias, dtype=config.get_dtype())

        self.optimization = optimization or Optimization()

        out_tf, hard_cs = _output_pipeline_from_optimization(self.optimization)

        self.operator = operator
        self.branch = branch
        self.trunk = trunk
        self.bias = bias_arr

        self.query_dim = query_dim
        self.descriptor_dim = descriptor_dim
        self.branch_input_dim = branch_input_dim
        self.trunk_input_dim = trunk_input_dim
        self.input_dim = query_dim + branch_input_dim
        self.latent_dim = branch_output_dim

        self.differential_axes = tuple(range(len(q.min)))
        self.evolution_axis = (query_dim - 1) if q.times is not None else None
        self.domain = q

        self.output_transformation = out_tf
        self.hard_constraints = hard_cs

    def split_inputs(self, inputs: Array) -> tuple[Array, Array]:
        """Return ``(descriptor_input, query_input)`` from ``[query..., descriptor...]``.

        Args:
            inputs: One concatenated point.

        Returns:
            The descriptor vector, then the query vector.

        """

        u = inputs[self.query_dim :]

    def branch_embedding(self, descriptor_input: Array) -> Array:
        """Return the raveled branch output.

        Args:
            descriptor_input: One descriptor vector.

        Returns:
            ``ravel(branch(descriptor_input))``.

        """
        return jnp.ravel(self.branch(descriptor_input))

    def trunk_embedding(self, query_input: Array) -> Array:
        """Return the raveled trunk output.

        Args:
            query_input: One query vector.

        Returns:
            ``ravel(trunk(query_input))``.

        """

    def forward_raw(self, descriptor_input: Array, query_input: Array) -> Array:
        """Return the dot product plus ``bias``, before the output pipeline.

        Args:
            descriptor_input: One descriptor vector.
            query_input: One query vector.

        Returns:
            ``dot(branch(u), trunk(y)) + bias``.

        """
        b = self.branch_embedding(descriptor_input)
        t = self.trunk_embedding(query_input)
        return jnp.dot(b, t) + self.bias

    def forward(self, descriptor_input: Array, query_input: Array) -> Array:
        """Return the transformed dot product.

        The raw value is squeezed, passed through ``output_transformation``,
        and expanded to at least one dimension. Hard constraints run in
        ``__call__``.

        Args:
            descriptor_input: One descriptor vector.
            query_input: One query vector.

        Returns:
            The transformed output, with at least one dimension.

        """
        out = jnp.squeeze(self.forward_raw(descriptor_input, query_input))
        out = self.output_transformation(out)
        return jnp.atleast_1d(out)

    @eqx.filter_jit
    def __call__(self, inputs: Array) -> Array:
        """Evaluate one concatenated point.

        Wrapped with ``eqx.filter_jit``. Splits ``inputs`` into the descriptor
        and the query, then applies the output transformation and each hard
        constraint. The constraints see the original ``inputs``.

        Args:
            inputs: Query coordinates, then the descriptor.

        Returns:
            The constrained output, with at least one dimension.

        """
        u, y = self.split_inputs(inputs)

        out = jnp.squeeze(self.forward_raw(u, y))

        out = _apply_output_pipeline(
            inputs,
            out,
            self.output_transformation,
            self.hard_constraints,
        )

        return jnp.atleast_1d(out)

    def evaluate_descriptor(
        self,
        descriptor_input: Array,
        query_inputs: Array,
    ) -> Array:
        """Evaluate one descriptor input at many query points.

        Uses ``forward`` for each query. Hard constraints run in ``__call__``.

        Args:
            descriptor_input: One descriptor vector.
            query_inputs: Query vectors, one per row.

        Returns:
            One transformed output per query.

        """
        return jax.vmap(lambda y: self.forward(descriptor_input, y))(query_inputs)

    Wrapped with ``eqx.filter_jit``. Splits ``inputs`` into the descriptor
    and the query, then applies the output transformation and each hard
    constraint. The constraints see the original ``inputs``.

    Args:
        inputs: Query coordinates, then the descriptor.

    Returns:
        The constrained output, with at least one dimension.

    """
    u, y = self.split_inputs(inputs)

    out = jnp.squeeze(self.forward_raw(u, y))

    out = _apply_output_pipeline(
        inputs,
        out,
        self.output_transformation,
        self.hard_constraints,
    )

    return jnp.atleast_1d(out)

    def __init__(
    self,
    operator: OperatorDomain,
    branch: NeuralNetwork,
    trunk: NeuralNetwork,
    *,
    bias: float | Array | None = None,
    latent_dim: int | None = None,
    optimization: Optimization | None = None,
) -> None:
    """Store the branch, the trunk, and the output pipeline.

    Raises:
        ValueError: If the trunk input width is not ``query_dim``, if the
            branch input width is shorter than the descriptor, or if the
            branch output, trunk output, and ``latent_dim`` disagree.
        TypeError: If a network has no ``input_dim`` and no ``topology``.

    """
    super().__init__()

    q = operator.query
    query_dim = _query_dim_from_domain(q)
    descriptor_dim = len(operator.descriptor.min)

    branch_input_dim = _network_input_dim(branch)
    trunk_input_dim = _network_input_dim(trunk)

    if trunk_input_dim != query_dim:
        raise ValueError(
            f"trunk input_dim ({trunk_input_dim}) must match query_dim ({query_dim}).",
        )

    if branch_input_dim < descriptor_dim:
        raise ValueError(
            f"branch input_dim ({branch_input_dim}) must be at least "
            f"operator descriptor dim ({descriptor_dim}).",
        )

    branch_output_dim = _network_output_dim(branch, branch_input_dim)
    trunk_output_dim = _network_output_dim(trunk, trunk_input_dim)

    if branch_output_dim != trunk_output_dim:
        raise ValueError(
            f"branch output_dim ({branch_output_dim}) must match trunk "
            f"output_dim ({trunk_output_dim}).",
        )

    if latent_dim is not None and latent_dim != branch_output_dim:
        raise ValueError(
            f"latent_dim ({latent_dim}) must match branch/trunk output_dim "
            f"({branch_output_dim}).",
        )

    if bias is None:
        bias_arr = jnp.array(0.0, dtype=config.get_dtype())
    else:
        bias_arr = jnp.asarray(bias, dtype=config.get_dtype())

    self.optimization = optimization or Optimization()

    out_tf, hard_cs = _output_pipeline_from_optimization(self.optimization)

    self.operator = operator
    self.branch = branch
    self.trunk = trunk
    self.bias = bias_arr

    self.query_dim = query_dim
    self.descriptor_dim = descriptor_dim
    self.branch_input_dim = branch_input_dim
    self.trunk_input_dim = trunk_input_dim
    self.input_dim = query_dim + branch_input_dim
    self.latent_dim = branch_output_dim

    self.differential_axes = tuple(range(len(q.min)))
    self.evolution_axis = (query_dim - 1) if q.times is not None else None
    self.domain = q

    self.output_transformation = out_tf
    self.hard_constraints = hard_cs

    def to_meta(self) -> dict:
    """Return a dictionary describing this DeepONet.

    Branch and trunk entries are those networks' ``to_meta`` dictionaries.
    A network without ``to_meta`` is stored as its class name. ``bias`` is
    stored with ``float``. Domain entries list ``min``, ``max``, and axis
    names. Query time is ``min``, ``max``, and ``step`` when the query has
    time.

    Returns:
        Metadata for the widths, domains, pipeline, branch, and trunk.

    Raises:
        TypeError: If ``bias`` cannot be converted with ``float``.

    """
    branch_meta = (
        self.branch.to_meta()
        if hasattr(self.branch, "to_meta")
        else {"model_class": type(self.branch).__name__}
    )
    trunk_meta = (
        self.trunk.to_meta()
        if hasattr(self.trunk, "to_meta")
        else {"model_class": type(self.trunk).__name__}
    )

    times = self.operator.query.times
    times_meta = (
        None
        if times is None
        else {
            "min": float(times.min),
            "max": float(times.max),
            "step": float(times.step),
        }
    )

    return {
        "model_class": "DeepONet",
        "latent_dim": self.latent_dim,
        "query_dim": self.query_dim,
        "descriptor_dim": self.descriptor_dim,
        "branch_input_dim": self.branch_input_dim,
        "trunk_input_dim": self.trunk_input_dim,
        "input_dim": self.input_dim,
        "differential_axes": list(self.differential_axes),
        "evolution_axis": self.evolution_axis,
        "bias": float(jnp.asarray(self.bias)),
        "query_domain": {
            "min": list(self.operator.query.min),
            "max": list(self.operator.query.max),
            "axis_names": list(self.operator.query.axis_names)
            if getattr(self.operator.query, "axis_names", None)
            else None,
            "times": times_meta,
        },
        "descriptor_domain": {
            "min": list(self.operator.descriptor.min),
            "max": list(self.operator.descriptor.max),
            "axis_names": list(self.operator.descriptor.axis_names)
            if getattr(self.operator.descriptor, "axis_names", None)
            else None,
        },
        "branch": branch_meta,
        "trunk": trunk_meta,
        "output_transformation": type(self.output_transformation).__name__,
        "hard_constraints": [type(hc).__name__ for hc in self.hard_constraints],
        **self._optimization_meta(),
    }
