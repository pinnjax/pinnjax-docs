class NormalizationBetweenNeg1and1(InputTransformation):
    """Map domain coordinates onto ``[-1, 1]``.

    ``(2 * ((inputs - min) / range)) - 1``. ``min`` and ``range`` come from
    the domain bounds. A [`Domain`][pinnjax.Domain] with time
    appends the time bounds as the last component.

    Args:
        domain: [`Domain`][pinnjax.Domain] whose bounds set the scale.

    Attributes:
        min: Lower bound of each input component.
        range: Width of each input component.

    """

    def __init__(self, domain: Domain):
        """Store the lower bound and the width of each input component."""
        mins, maxs = _domain_input_bounds(domain)
        rng = maxs - mins
        eps = jnp.asarray(config.get_EPS(), dtype=rng.dtype)

        self.min = mins
        self.range = jnp.where(rng == 0, eps, rng)

    def __call__(self, inputs: Array) -> Array:
        """Scale ``inputs`` onto ``[-1, 1]``.

        Args:
            inputs: Coordinates. The last axis matches the stored bounds.

        Returns:
            ``(2 * ((inputs - min) / range)) - 1``.

        """
        mins = self.min.astype(inputs.dtype)
        rng = self.range.astype(inputs.dtype)
        return (2 * ((inputs - mins) / rng)) - 1

    def to_spec(self) -> dict[str, Any]:  # noqa: D102
        return {}

    @classmethod
    def from_spec(  # noqa: D102
        cls,
        spec: dict[str, Any],
        *,
        domain: Domain | None = None,
    ) -> Self:
        if domain is None:
            raise ValueError(f"{cls.__name__} requires a domain for reconstruction.")

        return cls(domain)

    @register_optimization_component
class ExpectedOutputRangeSigmoid(OutputTransformation):
    """Map and scale sigmoid output into the expected output range.

    ``lower_bound + upper_bound * sigmoid(outputs)``.
    Sigmoid stays in ``(0, 1)``, so the result is then scaled between
    [lower_bound , upper_bound]

    Args:
        lower_bound:
        upper_bound:


    """

    def __init__(self, lower_bound: float, upper_bound: float) -> None:
        """Store the bounds."""
        self.lower_bound = jnp.asarray(
            lower_bound,
            dtype=config.get_dtype(),
        )
        self.upper_bound = jnp.asarray(
            upper_bound,
            dtype=config.get_dtype(),
        )

    def __call__(self, outputs: Array) -> Array:
        """Apply the sigmoid temperature map.

        Args:
            outputs: Network outputs.

        Returns:
            ``ambient_temperature + temperature_range * sigmoid(outputs)``.

        """
        lr = self.lower_bound.astype(outputs.dtype)
        up = self.upper_bound.astype(outputs.dtype)
        # Use sigmoid instead of softmax for single outputs
        scaled = jax.nn.sigmoid(outputs)
        return lr + up * scaled  # Maps to [lr, up]

    def to_spec(self) -> dict[str, Any]:  # noqa: D102
        return {
            "lower_bound": float(self.lower_bound),
            "upper_bound": float(self.upper_bound),
        }

    scale = self.scale.astype(outputs.dtype)
    # Use sigmoid instead of softmax for single outputs
    scaled = jax.nn.softplus(outputs)
    return lr + scale * scaled  # Maps to [lr, ?]
