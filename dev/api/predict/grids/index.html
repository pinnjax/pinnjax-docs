    def run_prediction(
    model: NeuralNetwork,
    domain: SpatialDomain,
    *,
    fixed_coordinates: dict[str, float] | None = None,
    num_points: tuple[int, int] = DEFAULT_GRID_NUM_POINTS,
    num_times: int = DEFAULT_GRID_NUM_TIMES,
) -> tuple[Array, RectangularGrid2D]:
    """Run an evolution prediction on a 2D slice.

    Builds a [`RectangularGrid2D`][pinnjax.RectangularGrid2D] over
    ``domain.times`` and evaluates
    [`predict_evolution`][pinnjax.predict_evolution]. An omitted slice holds
    ``z`` at the domain maximum. ``num_points`` is ``(nx, ny)`` along the free
    axes. The stored field is ``(num_times, ny, nx)``.

    Args:
        model: [`NeuralNetwork`][pinnjax.NeuralNetwork] to evaluate.
        domain: [`SpatialDomain`][pinnjax.SpatialDomain] with a positive time
            range.
        fixed_coordinates: Axis name to the constant value. When omitted,
            ``z`` is held at ``domain.max[2]``.
        num_points: Sample counts ``(nx, ny)`` for the free axes.
        num_times: Number of times, including both ends.

    Returns:
        The predictions, shaped ``(num_times, ny, nx)``, and the grid.

    Raises:
        TypeError: If ``domain`` is not a
            [`SpatialDomain`][pinnjax.SpatialDomain].
        ValueError: If ``domain`` has no time range, or the range is not
            positive.
        pinnjax.InputError: If the fixed axes do not leave two free axes.

    """
    if not isinstance(domain, SpatialDomain):
        raise TypeError(
            "run_prediction requires a SpatialDomain (evolution uses domain.times). "
            f"Got {type(domain).__name__}.",
        )
    if domain.times is None:
        raise ValueError("SpatialDomain has no times; cannot run evolution prediction.")
    t_min, t_max = float(domain.times.min), float(domain.times.max)
    if t_max <= t_min:
        raise ValueError(
            f"Domain time range is non-positive (min={t_min}, max={t_max}). "
            "Evolution prediction would be constant (e.g. initial condition only).",
        )
    time_range = (t_min, t_max)
    if fixed_coordinates is None:
        fixed_coordinates = {"z": float(domain.max[2])}
    grid = RectangularGrid2D(
        domain=domain,
        fixed_coordinates=fixed_coordinates,
        num_points=num_points,
        time_range=time_range,
        num_times=num_times,
    )
    predictions = predict_evolution(model, grid)
    return predictions, grid

    def run_prediction_at_time(
    model: NeuralNetwork,
    domain: SpatialDomain | None = None,
    time: float = 0.0,
    *,
    fixed_coordinates: dict[str, float] | None = None,
    num_points: tuple[int, int] = DEFAULT_GRID_NUM_POINTS,
) -> tuple[Array, RectangularGrid2D]:
    """Predict a 2D slice at one time.

    Builds a [`RectangularGrid2D`][pinnjax.RectangularGrid2D] and evaluates
    [`predict`][pinnjax.predict.base.predict]. An omitted domain uses ``model.domain``.
    An omitted slice holds ``z`` at the domain maximum. ``num_points`` is
    ``(nx, ny)`` along the free axes. The stored field is ``(ny, nx)``.

    Args:
        model: [`NeuralNetwork`][pinnjax.NeuralNetwork] to evaluate.
        domain: [`SpatialDomain`][pinnjax.SpatialDomain]. When omitted,
            ``model.domain`` is used.
        time: Time of the snapshot.
        fixed_coordinates: Axis name to the constant value. When omitted,
            ``z`` is held at ``domain.max[2]``.
        num_points: Sample counts ``(nx, ny)`` for the free axes.

    Returns:
        The prediction, shaped ``(ny, nx)``, and the spatial grid.

    Raises:
        TypeError: If neither ``domain`` nor ``model.domain`` is set, or the
            domain is not a [`SpatialDomain`][pinnjax.SpatialDomain].

    """
    resolved = domain if domain is not None else model.domain
    if resolved is None:
        raise TypeError(
            "run_prediction_at_time requires domain or model.domain to be set.",
        )
    domain = _as_spatial_domain(resolved, context="run_prediction_at_time")
    if fixed_coordinates is None:
        fixed_coordinates = {"z": float(domain.max[2])}
    grid = RectangularGrid2D(
        domain=domain,
        fixed_coordinates=fixed_coordinates,
        num_points=num_points,
    )
    prediction = predict(model, grid, time=float(time))
    return prediction, grid

        fixed_coordinates: Axis name to the constant value. When omitted,
            ``z`` is held at ``domain.max[2]``.
        num_points: Sample counts ``(nx, ny)`` for the free axes.

    Returns:
        Predictions of shape ``(n_times, ny, nx)``, and the spatial grid.

    Raises:
        TypeError: If neither ``domain`` nor ``model.domain`` is set, or the
            domain is not a [`SpatialDomain`][pinnjax.SpatialDomain].

    if resolved is None:
        raise TypeError(
            "run_prediction_at_times requires domain or model.domain to be set.",

        num_points=num_points,
    )

    predictions = predict_at_times(model, grid, times_arr)
    return predictions, grid
