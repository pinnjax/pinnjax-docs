    class Path(eqx.Module):
    """Abstract base class for all path definitions.

    Notes:
        Child classes must implement an `eval()` method that computes the spatial
            coordinates of the path at a given time `t`.

    Methods:
        validate: Check that path remains within given `domain` bounds.
        draw: Visualize path within its domain.
        animate: Animate the path over time.

    """

    dimensions: int

    @abstractmethod
    def eval(self, t: Array) -> Array:
        """Return the spatial coordinates of the path at time ``t``.

        Args:
            t: Time, or an array of times.

        Returns:
            Coordinates with shape ``(..., spatial_dims)``.

        """

    @abstractmethod
    def offset(self, point: Array) -> Array:
        """Return the displacement from the path to a point.

        The path is evaluated at the last component of ``point``. That position
        is subtracted from the point's spatial coordinates.

        Args:
            point: Spatial coordinates followed by time.

        Returns:
            ``point[..., :-1] - eval(point[..., -1])``.

        """
        path_coords = self.eval(point[..., -1])
        return point[..., :-1] - path_coords

    def validate(self, domain: SpatialDomain) -> None:
        # TODO(@mjryande): Paths are defined as endless, so if the total time is
        #   too long, validate method wrongly assesses it.
        """Validate that the entire path stays within the domain bounds.

        Args:
            domain: A [`SpatialDomain`][pinnjax.SpatialDomain] object with spatial and
                temporal bounds.

        Raises:
            InputError: If `domain.times` is None (no temporal information).
            InputError: If path coordinates fall outside the domain's min/max bounds.

        """
        if domain.times is None:
            msg = "domain must have time information for path validation"
            config.LOGGER.error(msg)
            raise config.InputError(msg)

        time_steps = int((domain.times.max - domain.times.min) / domain.times.step)
        times = jnp.linspace(domain.times.min, domain.times.max, time_steps)
        coordinates = self.eval(times)  # Shape: (time_steps, spatial_dims)
        spatial_dims = coordinates.shape[1]
        # Check each spatial dimension
        for dim in range(spatial_dims):
            coords = coordinates[:, dim]
            coord_max, coord_min = jnp.max(coords), jnp.min(coords)

            if coord_min < domain.min[dim] or coord_max > domain.max[dim]:
                msg = "Path exceeds the bounds of the given domain."
                config.LOGGER.error(msg)
                raise config.InputError(msg)

    @abstractmethod
    def get_velocity(self, t: float) -> Array:
        """Calculate the instantaneous velocity vector of the path at a given time.

        Args:
            t: The time coordinate at which to calculate the velocity.

        Returns:
            The velocity vector at ``t``. A 3D path returns ``[vx, vy, vz]``.

        """

    @abstractmethod
    def get_rotation(self, u: Array) -> Rotation:
        """Return the rotation into the path's local, x-aligned frame.

        The rotation takes a point from the global coordinate system into
        that frame.

        Args:
            u: A point with spatial and temporal coordinates. The last
                dimension is assumed to be time.

        Returns:
            A JAX Rotation object representing the required transformation.

        Note:
            Ensure the correct use of rotation object, Rotation.apply() or
                Rotation.inv().apply(), to meet your specific needs.

        """

    def offset_in_local_frame(self, u: Array) -> Array:
        """Return ``offset(u)`` in the path-local frame at ``u``.

        Default implementation uses [`get_rotation`][pinnjax.pathing.base.Path.get_rotation]
        and ``Rotation.inv().apply``. Subclasses may override with a cached
        inverse rotation matrix for a cheaper Goldak / source evaluation.

        Args:
            u: A point with spatial and temporal coordinates.

        Returns:
            Local-frame displacement, same spatial width as ``offset(u)``.

        """
        return self.get_rotation(u).inv().apply(self.offset(u))

    @abstractmethod
    def get_active(self, u: Array) -> Array:
        """Return whether the path is active at ``u``.

        Args:
            u: A point with spatial and temporal coordinates. The last
                dimension is assumed to be time.

        Returns:
            a single number: active = 1.0, inactive = 0.0

        """

    @staticmethod
    def _create_rotation(direction: jnp.ndarray) -> Rotation:
        r"""Calculate a rotation object aligning the global x-axis with a given direction.

        The rotation is constructed such that the local z-axis remains fixed to the
        global z-axis. This ensures a unique and consistent orientation. A direction
        with norm below ``1e-6`` returns the identity rotation.

        Args:
            direction: A 3D vector representing the path's direction of movement.

        Returns:
            A JAX `Rotation` object that maps a point from the global coordinate system
            to the path's local, x-aligned coordinate system.

        Note:
            The rotation matrix $R$ is constructed from three orthonormal basis vectors
            for the local frame: \( \hat{x}_{local}$, $\hat{y}_{local}$, and
                $\hat{z}_{local}\).

            \( \hat{x}_{local} = \frac{\vec{d}}{||\vec{d}||} \)
            \( \hat{z}_{local} = (0, 0, 1) \)
            \( \hat{y}_{local} = \hat{z}_{local} \times \hat{x}_{local} \)
            \( R = \begin{bmatrix} \hat{x}_{local} & \hat{y}_{local} & \hat{z}_{local}
                \end{bmatrix} \)

        """
        if direction.shape[-1] == 2:
            direction = jnp.append(direction, 0.0)
        elif direction.shape[-1] != 3:
            raise ValueError("Direction must be 2D or 3D.")

        norm = jnp.linalg.norm(direction)
        is_zero = norm < 1e-6

        # Build the quaternion for a zero-norm vector (identity rotation).
        zero_quat = Rotation.from_rotvec(jnp.zeros(3)).as_quat()

        # Build the quaternion for the normal case.
        # The where clause ensures direction is not zero for the calculation.
        safe_direction = jnp.where(
            is_zero,
            jnp.array([1.0, 0.0, 0.0]),
            direction / norm,
        )
        local_x_axis = safe_direction
        local_z_axis = jnp.array([0.0, 0.0, 1.0])
        local_y_axis = jnp.cross(local_z_axis, local_x_axis)
        local_x_axis_ortho = jnp.cross(local_y_axis, local_z_axis)
        rot_matrix = jnp.stack([local_x_axis_ortho, local_y_axis, local_z_axis], axis=1)
        normal_quat = Rotation.from_matrix(rot_matrix).as_quat()

        # Conditionally choose between the two quaternions. This is now valid.
        final_quaternion = jnp.where(is_zero, zero_quat, normal_quat)

        # Return the final Rotation object.
        return Rotation.from_quat(final_quaternion)

    def draw(
        self,
        *,
        domain: SpatialDomain | None = None,
        time: Time | None = None,
        show_angle_axes: bool = False,
        center: tuple[float, ...] | None = None,
        fig: Figure | None = None,
        ax: Axes | None = None,
        show: bool | None = True,
        save: bool | None = False,
        save_location: str | pathlib.Path | None = None,
        save_name: str | None = "path",
        save_format: str = "pdf",
        dpi: int = 300,
    ) -> tuple[Figure, Axes]:
        # TODO(@mjryande): Update for domain=None, as per Complex Case.
        """Plot the path within the specified domain.

        Either a SpatialDomain or Time must be provided. If both are provided,
            SpatialDomain rules. If only a Time object is provided, only the Path
            will be plotted, without the SpatialDomain shape.

        Args:
            domain: A [`SpatialDomain`][pinnjax.SpatialDomain] object.
            time: A [`Time`][pinnjax.Time] object.
            show_angle_axes: If True, draw a small protractor at 0°, 90°, 180°, 270°.
            center: Optional (x, y) or (x, y, z) for protractor center. If None and
                domain is set, use domain.center(); otherwise, use path bbox center.
            fig: An Optional Matplotlib Figure.
            ax: An Optional Matplotlib Axes.
            show: Boolean flag to immediately display the figure via ``plt.show()``.
            save: Whether to write the figure.
            save_location: Directory for the saved file.
            save_name: Output filename.
            save_format: File format passed to the saver.
            dpi: Resolution of the saved figure.

        Returns:
            fig, ax: Matplotlib figure and axes objects containing the plot.

        Raises:
            ValueError: If both ``domain`` and ``time`` are ``None``.
            InputError: If the path is not 2D or 3D while axes are created.
            RuntimeError: If ``domain`` is set and ``domain.times`` is ``None``.

        Examples:
            >>> # for a 2D domain and LINEAR path.
            >>> import pinnjax
            >>> _domain = pinnjax.SpatialDomain(min=(0,0), max=(12, 4), times=pinnjax.Time(0,3,0.05))
            >>> _path = pinnjax.LinearPath(pinnjax.LinearPathConfig(speed=2, direction=(3,1), initial=(1,1)))
            >>> _ = _path.draw(domain=_domain)

            !!! figure
                ![](../assets/path_draw_example_2D.png){: style="display: block; margin: 0 auto;" }

            >>> # for a 3D domain and LINEAR path.
            >>> _domain = pinnjax.SpatialDomain(min=(0, 0, 0), max=(12, 2, 2), times=pinnjax.Time(0, 3, 0.05))
            >>> _path = pinnjax.LinearPath(pinnjax.LinearPathConfig(speed=3, direction=(1, 0, 0), initial=(1, 1, 2)))
            >>> _ = _path.draw(domain=_domain)

            !!! figure
                ![](../assets/path_draw_example_3D.png){: style="display: block; margin: 0 auto;" }

            >>> # for a 2D domain and CIRCULAR path.
            >>> _domain = pinnjax.SpatialDomain(min=(0, 0), max=(12, 12), times=pinnjax.Time(0, 10, 0.05))
            >>> _path = pinnjax.CirclePath(pinnjax.CirclePathConfig(center=_domain.center(), radius=3, speed=2))
            >>> _ = _path.draw(domain=_domain)

            !!! figure
                ![](../assets/path_draw_circle_example_2D.png){: style="display: block; margin: 0 auto;" }

            >>> # for a 3D domain and CIRCULAR path.
            >>> _domain = pinnjax.SpatialDomain(min=(0, 0, 0), max=(12, 12, 2), times=pinnjax.Time(0, 10, 0.05))
            >>> _path = pinnjax.CirclePath(pinnjax.CirclePathConfig(center=_domain.center(fixed={'z': 2}), radius=3, speed=2))
            >>> _ = _path.draw(domain=_domain)

            !!! figure
                ![](../assets/path_draw_circle_example_3D.png){: style="display: block; margin: 0 auto;" }

        """
        if domain is not None:
            fig, ax = domain.draw(fig=fig, ax=ax, show=False)
            time = domain.times
        elif time is not None:
            pass
        else:
            msg = "Either SpatialDomain or Time object must be provided."
            config.LOGGER.error(msg)
            raise ValueError(msg)

        if fig is None:
            fig = plt.figure()

        # TODO(@mjryande): Need to be able to remove domain from this.
        if time is not None:
            time_steps = int((time.max - time.min) / time.step)
            times = jnp.linspace(time.min, time.max, time_steps)
            coordinates = self.eval(times)

            if ax is None:
                match len(coordinates[0]):
                    case 2:
                        ax = fig.add_subplot(111)
                    case 3:
                        ax = fig.add_subplot(111, projection="3d")
                    case _:
                        msg = "Invalid domain size for path plotting."
                        config.LOGGER.error(msg)
                        raise config.InputError(msg)

            match len(coordinates[0]):
                case 2:
                    ax.plot(*coordinates.T, color="red", linewidth=3, zorder=5)
                case 3:
                    if not isinstance(ax, Axes3D):
                        msg = "A 3D path requires a 3D axes."
                        config.LOGGER.error(msg)
                        raise TypeError(msg)
                    ax.plot3D(*coordinates.T, color="red", linewidth=3, zorder=5)
                case _:
                    msg = "Invalid domain size for path plotting."
                    config.LOGGER.error(msg)
                    raise config.InputError(msg)

            if show_angle_axes:
                protractor_center = (
                    center
                    if center is not None
                    else (domain.center() if domain is not None else None)
                )
                _draw_angle_axes(
                    ax,
                    coordinates,
                    ndim=len(coordinates[0]),
                    center=protractor_center,
                )

            if save:
                from pinnjax.plot.saving import save_plot

                save_plot(
                    fig,
                    save_location=save_location,
                    save_name=save_name,
                    save_format=save_format,
                    dpi=dpi,
                    bbox_inches="tight",
                )

            if show:
                plt.show()

            return fig, ax
        msg = (
            "'time' must be defined, either through a Time ",
            "or SpatialDomain object.",
        )
        config.LOGGER.error(msg)
        raise RuntimeError(msg)

    def animate(
        self,
        *,
        domain: SpatialDomain | None = None,
        time: Time | None = None,
        trail: bool = True,
        show_angle_axes: bool = False,
        center: tuple[float, ...] | None = None,
        fig: Figure | None = None,
        ax: Axes | None = None,
        show: bool | None = True,
        save: bool | None = True,
        save_location: str | pathlib.Path | None = None,
        save_name: str | None = "path.gif",
        n_frames: int | None = None,
        fps: int | None = 15,
        dpi: int = 300,
    ) -> tuple[Figure, Axes, FuncAnimation]:
        """Animate the path over time (path grows from start to current time).

        Either a [`SpatialDomain`][pinnjax.SpatialDomain] or [`Time`][pinnjax.Time] must be provided;
            [`SpatialDomain`][pinnjax.SpatialDomain] takes precedence.

        Args:
            domain: A [`SpatialDomain`][pinnjax.SpatialDomain] object.
            time: A [`Time`][pinnjax.Time] object.
            trail: If True, draw the path line and the circle at the head.
                If False, only the circle at the current time step is shown.
            show_angle_axes: If True, draw a small protractor at 0°, 90°, 180°, 270°.
            center: Optional (x, y) or (x, y, z) for protractor center.
                If None and domain is set, use domain.center();
                otherwise, use path bbox center.
            fig: Optional Matplotlib Figure.
            ax: Optional Matplotlib Axes.
            show: Whether to display the figure via ``plt.show()``.
            save: Whether to save the animation.
            save_location: Directory for saved file.
            save_name: Output filename.
            n_frames: Number of frames. When omitted, the time-step count is
                used, clamped to the range 10 through 80.
            fps: Frames per second for saved gif.
            dpi: Resolution of the saved gif.

        Returns:
            fig, ax, anim: Matplotlib figure, axes, and animation object.

        Raises:
            ValueError: If both ``domain`` and ``time`` are ``None``.
            InputError: If the path is not 2D or 3D.
            RuntimeError: If ``domain`` is set and ``domain.times`` is ``None``.

        """
        if domain is not None:
            fig, ax = domain.draw(fig=fig, ax=ax, show=False)
            time = domain.times
        elif time is not None:
            pass
        else:
            msg = "Either SpatialDomain or Time object must be provided."
            config.LOGGER.error(msg)
            raise ValueError(msg)

        if fig is None:
            fig = plt.figure()

        if time is None:
            msg = (
                "'time' must be defined, either through a Time or SpatialDomain ",
                "object.",
            )
            config.LOGGER.error(msg)
            raise RuntimeError(msg)

        time_steps = int((time.max - time.min) / time.step)
        if n_frames is None:
            n_frames = min(80, max(10, time_steps))
        frame_times = jnp.linspace(time.min, time.max, n_frames)

        if ax is None:
            # Infer dimensionality from path the first time
            coords_0 = self.eval(jnp.array([time.min]))
            ndim = coords_0.shape[1]
            if ndim == 2:
                ax = fig.add_subplot(111)
            elif ndim == 3:
                ax = fig.add_subplot(111, projection="3d")
            else:
                msg = "Invalid domain size for path plotting."
                config.LOGGER.error(msg)
                raise config.InputError(msg)

        ndim = int(self.eval(jnp.array([time.min])).shape[1])
        if show_angle_axes:
            full_coords = self.eval(jnp.linspace(time.min, time.max, 100))
            protractor_center = (
                center if center is not None else (domain.center() if domain is not None else None)
            )
            _draw_angle_axes(ax, full_coords, ndim=ndim, center=protractor_center)

        head_diameter_pt = 8.0
        head_s = 3.14159 * (head_diameter_pt / 2) ** 2
        if ndim == 2:
            head_scat = ax.scatter(
                [],
                [],
                s=head_s,
                c="red",
                edgecolors="white",
                linewidths=1.5,
                zorder=5,
            )
        else:
            if not isinstance(ax, Axes3D):
                msg = "A 3D path requires a 3D axes."
                config.LOGGER.error(msg)
                raise TypeError(msg)
            head_scat = ax.scatter(
                [],
                [],
                [],
                s=head_s,
                c="red",
                edgecolors="white",
                linewidths=1.5,
            )

        # Mutable list of path line artists; we redraw segments each frame
        # (active=solid, inactive=dashed)
        path_line_artists = []

        def _path_segments_by_active(
            coords: Array,
            path_times: Array,
        ) -> list[tuple[int, int, bool]]:
            """Return list of (start_idx, end_idx, is_active) for contiguous runs."""
            n = coords.shape[0]
            if n == 0:
                return []
            active = []
            for i in range(n):
                u = jnp.concatenate([coords[i], jnp.array([float(path_times[i])])])
                a = self.get_active(u)
                active.append(bool(float(a)))
            segments = []
            i = 0
            while i < n:
                j = i
                while j < n and active[j] == active[i]:
                    j += 1
                segments.append((i, j, active[i]))
                i = j
            return segments

        def init():
            if ndim == 2:
                head_scat.set_offsets(jnp.empty((0, 2)))
            else:
                head_scat.set_offsets(jnp.empty((0, 2)))
                cast("Any", head_scat).set_3d_properties(jnp.array([]), zdir="z")
            return (head_scat,)

        def update(frame_idx: int):
            t_max = float(frame_times[frame_idx])

    def animate(
    self,
    *,
    domain: SpatialDomain | None = None,
    time: Time | None = None,
    trail: bool = True,
    show_angle_axes: bool = False,
    center: tuple[float, ...] | None = None,
    fig: Figure | None = None,
    ax: Axes | None = None,
    show: bool | None = True,

    save_name: str | None = "path.gif",
    n_frames: int | None = None,
    fps: int | None = 15,
    dpi: int = 300,
) -> tuple[Figure, Axes, FuncAnimation]:
    """Animate the path over time (path grows from start to current time).

    Either a [`SpatialDomain`][pinnjax.SpatialDomain] or [`Time`][pinnjax.Time] must be provided;
        [`SpatialDomain`][pinnjax.SpatialDomain] takes precedence.

    Args:
        domain: A [`SpatialDomain`][pinnjax.SpatialDomain] object.
        time: A [`Time`][pinnjax.Time] object.
        trail: If True, draw the path line and the circle at the head.
            If False, only the circle at the current time step is shown.
        show_angle_axes: If True, draw a small protractor at 0°, 90°, 180°, 270°.
        center: Optional (x, y) or (x, y, z) for protractor center.
            If None and domain is set, use domain.center();
            otherwise, use path bbox center.
        fig: Optional Matplotlib Figure.
        ax: Optional Matplotlib Axes.
        show: Whether to display the figure via ``plt.show()``.
        save: Whether to save the animation.
        save_location: Directory for saved file.
        save_name: Output filename.
        n_frames: Number of frames. When omitted, the time-step count is
            used, clamped to the range 10 through 80.
        fps: Frames per second for saved gif.
        dpi: Resolution of the saved gif.

    Returns:
        fig, ax, anim: Matplotlib figure, axes, and animation object.

    Raises:
        ValueError: If both ``domain`` and ``time`` are ``None``.
        InputError: If the path is not 2D or 3D.
        RuntimeError: If ``domain`` is set and ``domain.times`` is ``None``.

    """
    if domain is not None:
        fig, ax = domain.draw(fig=fig, ax=ax, show=False)
        time = domain.times
    elif time is not None:
        pass
    else:
        msg = "Either SpatialDomain or Time object must be provided."
        config.LOGGER.error(msg)
        raise ValueError(msg)

    if fig is None:
        fig = plt.figure()

    if time is None:
        msg = (
            "'time' must be defined, either through a Time or SpatialDomain ",
            "object.",
        )
        config.LOGGER.error(msg)
        raise RuntimeError(msg)

    time_steps = int((time.max - time.min) / time.step)
    if n_frames is None:
        n_frames = min(80, max(10, time_steps))
    frame_times = jnp.linspace(time.min, time.max, n_frames)

    if ax is None:
        # Infer dimensionality from path the first time
        coords_0 = self.eval(jnp.array([time.min]))
        ndim = coords_0.shape[1]
        if ndim == 2:
            ax = fig.add_subplot(111)
        elif ndim == 3:
            ax = fig.add_subplot(111, projection="3d")
        else:
            msg = "Invalid domain size for path plotting."
            config.LOGGER.error(msg)
            raise config.InputError(msg)

    ndim = int(self.eval(jnp.array([time.min])).shape[1])
    if show_angle_axes:
        full_coords = self.eval(jnp.linspace(time.min, time.max, 100))
        protractor_center = (
            center if center is not None else (domain.center() if domain is not None else None)
        )
        _draw_angle_axes(ax, full_coords, ndim=ndim, center=protractor_center)

    head_diameter_pt = 8.0

    if ndim == 2:
        head_scat = ax.scatter(
            [],
            [],
            s=head_s,
            c="red",

            linewidths=1.5,

        )

        )

    # (active=solid, inactive=dashed)
    path_line_artists = []

    def _path_segments_by_active(
        coords: Array,
        path_times: Array,
    ) -> list[tuple[int, int, bool]]:
        """Return list of (start_idx, end_idx, is_active) for contiguous runs."""
        n = coords.shape[0]
        if n == 0:
            return []
        active = []
        for i in range(n):
            u = jnp.concatenate([coords[i], jnp.array([float(path_times[i])])])
            a = self.get_active(u)
            active.append(bool(float(a)))
        segments = []
        i = 0
        while i < n:
            j = i
            while j < n and active[j] == active[i]:
                j += 1
            segments.append((i, j, active[i]))
            i = j
        return segments

        else:
            head_scat.set_offsets(jnp.empty((0, 2)))
            cast("Any", head_scat).set_3d_properties(jnp.array([]), zdir="z")
        return (head_scat,)

        if trail:

            coords = self.eval(path_times)
        else:
            coords = self.eval(jnp.array([t_max]))

                if end - start < 2:

                seg_coords = coords[start:end]
                style = "-" if is_active else "--"
                width = 3 if is_active else 1
                if ndim == 2:

                        seg_coords[:, 0].tolist(),
                        seg_coords[:, 1].tolist(),
                        color="red",
                        linewidth=width,
                        linestyle=style,
                        zorder=5,
                    )
                else:
                    if not isinstance(ax, Axes3D):
                        msg = "A 3D path requires a 3D axes."
                        config.LOGGER.error(msg)
                        raise TypeError(msg)
                    (line,) = ax.plot3D(

                        seg_coords[:, 1].tolist(),
                        seg_coords[:, 2].tolist(),
                        color="red",
                        linewidth=width,
                        linestyle=style,
                        zorder=5,
                    )
                path_line_artists.append(line)

        head_active = bool(float(self.get_active(head_u)))
        head_color = "red" if head_active else "grey"
        cast("Any", head_scat).set_facecolors([head_color])

        if ndim == 2:
            head_scat.set_offsets([head_pt])
        else:
            head_scat.set_offsets([head_pt])
            cast("Any", head_scat).set_3d_properties([float(coords[-1, 2])], zdir="z")
        return (head_scat,)

    # blit=False so the head scatter updates every frame (blit can leave it stuck)
    playback_fps = 15 if fps is None else fps
    interval = 1000 / playback_fps
    anim = FuncAnimation(
        fig,
        update,
        frames=n_frames,
        init_func=init,
        interval=interval,
        blit=False,
        repeat=True,
    )

    if save:

        out_dir.mkdir(parents=True, exist_ok=True)

        anim.save(out_path, writer=writer, dpi=dpi)

    class LinearPath(Path, ABC):
    """Omnidirectional linear path.

    ``velocity`` is ``speed`` times the unit direction. ``None`` for
    ``active`` is stored as ``True``.

    Args:
        path_config: A [`LinearPathConfig`][pinnjax.LinearPathConfig] object containing
            `speed`, `initial` coordinates and `direction`.

    Attributes:
        initial: Initial coordinate location at ``t = 0``.
        velocity: Directional velocity as per the ``speed`` and normalized
            ``direction``.
        rotation: A ``jax.scipy.spatial.transform.Rotation`` for ``direction``.
        inv_rotation_matrix: Inverse of ``rotation`` as a ``(3, 3)`` matrix.
        active: Whether the heat source is on. ``None`` is stored as ``True``.
        config: The [`LinearPathConfig`][pinnjax.LinearPathConfig].
        dimensions: ``2`` or ``3``, from ``direction``.

    Raises:
        InputError: If the ``direction`` is 0.

    """

    initial: Array
    velocity: Array
    rotation: Rotation
    inv_rotation_matrix: Array
    active: bool
    config: LinearPathConfig = eqx.field(static=True)

    def __init__(self, path_config: LinearPathConfig) -> None:
        """Store the initial point, the velocity, and the rotation."""
        speed = jnp.asarray(path_config.speed, dtype=config.get_dtype())
        initial = jnp.asarray(path_config.initial, dtype=config.get_dtype())

        d = jnp.asarray(path_config.direction, dtype=config.get_dtype())
        norm = jnp.linalg.norm(d)
        if float(norm) == 0:
            msg = "Direction of 0 is not valid."
            config.LOGGER.error(msg)
            raise config.InputError(msg)
        direction = d / norm
        velocity = speed * direction

        self.dimensions = len(direction)

        self.initial = initial
        self.velocity = velocity
        self.config = path_config
        active = path_config.active
        self.active = True if active is None else active
        self.rotation = Path._create_rotation(direction)
        self.inv_rotation_matrix = self.rotation.inv().as_matrix()

    @jax.jit
    def eval(self, t: Array) -> Array:
        """Evaluate the path for the coordinate location of the moving object at time t.

        Wrapped with ``jax.jit``. The position is ``initial + velocity * t``.

        Args:
            t: Scalar time value as a JAX array, or an array of times.

        Returns:
            Coordinate location of the moving object. Shape
                ``(..., spatial_dims)``.

        """
        t = jnp.asarray(t, dtype=self.initial.dtype)
        return self.initial + self.velocity * t[..., None]

    @jax.jit
    def offset(self, point: Array) -> Array:
        """Return the offset displacement vector at the same time.

        The vector is the spatial coordinates of ``point`` minus the path
        position at the point's last component. Wrapped with ``jax.jit``.

        Args:
            point: JAX array of points in the form ``[x, y, t]`` or
                ``[x, y, z, t]``.

        Returns:
            Spatial displacement for each point, ``[dx, dy]`` or
                ``[dx, dy, dz]``.

        """
        p = point.astype(self.initial.dtype, copy=False)
        return p[..., :-1] - (self.initial + self.velocity * p[..., -1][..., None])

    def get_velocity(self, t: float) -> Array:
        """See [`Path.get_velocity`][pinnjax.pathing.base.Path.get_velocity].

        Note:
            For a linear path, the velocity vector is constant and
            pre-computed during initialization. ``t`` is unused.

        """
        return self.velocity

    @jax.jit
    def get_rotation(self, u: Array) -> Rotation:
        """See [`Path.get_rotation`][pinnjax.pathing.base.Path.get_rotation].

        Note:
            For a linear path, the rotation is constant and
            pre-computed during initialization. ``u`` is unused.

        """
        return self.rotation

    def offset_in_local_frame(self, u: Array) -> Array:
        """Return ``offset(u)`` using the cached inverse rotation matrix."""
        offset = self.offset(u)
        if offset.shape[-1] == 2:
            offset = jnp.concatenate([offset, jnp.zeros((1,), dtype=offset.dtype)])
        return self.inv_rotation_matrix @ offset

    def get_active(self, u: Array) -> Array:
        """Return the stored active flag.

        ``u`` is unused. The flag comes from the config.

        Args:
            u: Unused.

        Returns:
            ``1.0`` when the path is active, and ``0.0`` when it is inactive.

        """
        return jnp.asarray(float(self.active), dtype=config.get_dtype())

        msg = "Either SpatialDomain or Time object must be provided."
        config.LOGGER.error(msg)
        raise ValueError(msg)

    if fig is None:
        fig = plt.figure()

    if time is None:
        msg = (
            "'time' must be defined, either through a Time or SpatialDomain ",
            "object.",
        )
        config.LOGGER.error(msg)
        raise RuntimeError(msg)

    time_steps = int((time.max - time.min) / time.step)
    if n_frames is None:
        n_frames = min(80, max(10, time_steps))
    frame_times = jnp.linspace(time.min, time.max, n_frames)

    if ax is None:
        # Infer dimensionality from path the first time
        coords_0 = self.eval(jnp.array([time.min]))
        ndim = coords_0.shape[1]
        if ndim == 2:
            ax = fig.add_subplot(111)
        elif ndim == 3:
            ax = fig.add_subplot(111, projection="3d")
        else:
            msg = "Invalid domain size for path plotting."
            config.LOGGER.error(msg)
            raise config.InputError(msg)

    ndim = int(self.eval(jnp.array([time.min])).shape[1])
    if show_angle_axes:
        full_coords = self.eval(jnp.linspace(time.min, time.max, 100))
        protractor_center = (
            center if center is not None else (domain.center() if domain is not None else None)
        )
        _draw_angle_axes(ax, full_coords, ndim=ndim, center=protractor_center)

    head_diameter_pt = 8.0
    head_s = 3.14159 * (head_diameter_pt / 2) ** 2
    if ndim == 2:
        head_scat = ax.scatter(
            [],
            [],
            s=head_s,
            c="red",
            edgecolors="white",
            linewidths=1.5,
            zorder=5,
        )
    else:
        if not isinstance(ax, Axes3D):
            msg = "A 3D path requires a 3D axes."
            config.LOGGER.error(msg)
            raise TypeError(msg)
        head_scat = ax.scatter(
            [],
            [],
            [],
            s=head_s,
            c="red",
            edgecolors="white",
            linewidths=1.5,
        )

    # Mutable list of path line artists; we redraw segments each frame
    # (active=solid, inactive=dashed)
    path_line_artists = []

    def _path_segments_by_active(
        coords: Array,
        path_times: Array,
    ) -> list[tuple[int, int, bool]]:
        """Return list of (start_idx, end_idx, is_active) for contiguous runs."""
        n = coords.shape[0]
        if n == 0:
            return []
        active = []
        for i in range(n):
            u = jnp.concatenate([coords[i], jnp.array([float(path_times[i])])])
            a = self.get_active(u)
            active.append(bool(float(a)))
        segments = []
        i = 0
        while i < n:
            j = i
            while j < n and active[j] == active[i]:
                j += 1
            segments.append((i, j, active[i]))
            i = j
        return segments

    def init():
        if ndim == 2:
            head_scat.set_offsets(jnp.empty((0, 2)))
        else:
            head_scat.set_offsets(jnp.empty((0, 2)))
            cast("Any", head_scat).set_3d_properties(jnp.array([]), zdir="z")
        return (head_scat,)

    def update(frame_idx: int):
        t_max = float(frame_times[frame_idx])
        if trail:
            path_times = jnp.linspace(time.min, t_max, max(20, frame_idx + 2))
            coords = self.eval(path_times)
        else:
            coords = self.eval(jnp.array([t_max]))
        head_pt = (float(coords[-1, 0]), float(coords[-1, 1]))

        # Remove previous path segment lines
        for art in path_line_artists:
            art.remove()
        path_line_artists.clear()

        if trail and coords.shape[0] >= 2:
            segments = _path_segments_by_active(coords, path_times)
            for start, end, is_active in segments:
                if end - start < 2:
                    continue
                seg_coords = coords[start:end]
                style = "-" if is_active else "--"
                width = 3 if is_active else 1
                if ndim == 2:

                        seg_coords[:, 0].tolist(),
                        seg_coords[:, 1].tolist(),
                        color="red",
                        linewidth=width,
                        linestyle=style,
                        zorder=5,
                    )
                else:
                    if not isinstance(ax, Axes3D):
                        msg = "A 3D path requires a 3D axes."
                        config.LOGGER.error(msg)
                        raise TypeError(msg)
                    (line,) = ax.plot3D(
                        seg_coords[:, 0].tolist(),
                        seg_coords[:, 1].tolist(),
                        seg_coords[:, 2].tolist(),
                        color="red",
                        linewidth=width,
                        linestyle=style,
                        zorder=5,
                    )
                path_line_artists.append(line)

        # Head circle: red when path active, grey when off
        head_u = jnp.concatenate([coords[-1], jnp.array([t_max])])
        head_active = bool(float(self.get_active(head_u)))
        head_color = "red" if head_active else "grey"
        cast("Any", head_scat).set_facecolors([head_color])

        if ndim == 2:
            head_scat.set_offsets([head_pt])
        else:
            head_scat.set_offsets([head_pt])
            cast("Any", head_scat).set_3d_properties([float(coords[-1, 2])], zdir="z")
        return (head_scat,)

    # blit=False so the head scatter updates every frame (blit can leave it stuck)
    playback_fps = 15 if fps is None else fps
    interval = 1000 / playback_fps
    anim = FuncAnimation(
        fig,
        update,
        frames=n_frames,
        init_func=init,
        interval=interval,
        blit=False,
        repeat=True,
    )

    if save:
        out_dir = pathlib.Path(save_location) if save_location else pathlib.Path.cwd()
        out_dir.mkdir(parents=True, exist_ok=True)
        out_path = out_dir / (save_name or "path.gif")
        writer = PillowWriter(fps=playback_fps, bitrate=-1)
        anim.save(out_path, writer=writer, dpi=dpi)

    if show:
        plt.show()
    else:
        plt.close()

    return fig, ax, anim

    class CirclePath(Path, ABC):
    """Circle path around a fixed center.

    The circle is in the xy plane. A 3D center keeps its z coordinate.
    ``angular_speed`` is ``speed / radius``. ``initial_angle`` is added
    inside ``cos`` and ``sin``.

    Args:
        path_config: [`CirclePathConfig`][pinnjax.CirclePathConfig] with the
            center, radius, linear speed, initial angle, and active flag.
        num_cached_rotations: Number of tangent rotations stored around the
            circle. Default: ``1000``.

    Attributes:
        center: Circle center. Length 2 or 3.
        radius: Circle radius.
        angular_speed: ``speed / radius``.
        initial_angle: Starting angle, passed into ``cos`` and ``sin``.
        active: Whether the path is active. ``None`` is stored as ``True``.
        num_cached_rotations: Size of the rotation cache.
        _cached_inv_rotation_matrices: Inverse rotation matrices aligned with
            ``_cached_rotations``, shape ``(num_cached_rotations, 3, 3)``.
        config: The [`CirclePathConfig`][pinnjax.CirclePathConfig].
        dimensions: ``2`` or ``3``, from ``center``.

    Notes:
        It's recommended to create a [`ComplexLinearPath`][pinnjax.ComplexLinearPath]
        instead using [`circle_waypoint_generator`][pinnjax.circle_waypoint_generator].

    """

    center: Array
    radius: float
    angular_speed: float
    initial_angle: float
    active: bool

    _cached_angles: Array
    _cached_rotations: Array
    _cached_inv_rotation_matrices: Array
    num_cached_rotations: int

    config: CirclePathConfig = eqx.field(static=True)

    def __init__(
        self,
        path_config: CirclePathConfig,
        num_cached_rotations: int = 1000,
    ) -> None:
        """Store the circle, the active flag, and the rotation cache."""
        # TODO(@mjryande): I'm not sure about this implementation. It's almost certainly
        #  cleaner in CONFIG particularly, with num_cached_rotations as a second
        #  parameter. This implementation breaks create_path for the CirclePath
        self.center = jnp.asarray(path_config.center, dtype=config.get_dtype())
        self.radius = float(path_config.radius)
        self.angular_speed = float(path_config.speed) / self.radius
        self.initial_angle = float(path_config.initial_angle)
        active = path_config.active
        self.active = True if active is None else active
        self.config = path_config
        self.num_cached_rotations = num_cached_rotations

        self.dimensions = len(self.center)

        self._cached_angles = jnp.linspace(0, 2 * jnp.pi, num_cached_rotations)

        def _get_velocity_from_angle(angle: jnp.ndarray) -> jnp.ndarray:
            vx = -self.radius * jnp.sin(angle)
            vy = self.radius * jnp.cos(angle)
            vz = 0.0
            return jnp.array([vx, vy, vz])

        vmapped_rotation_fn = jax.vmap(
            lambda a: Path._create_rotation(_get_velocity_from_angle(a)),
        )
        rotations_batch = vmapped_rotation_fn(self._cached_angles)
        self._cached_rotations = rotations_batch.as_quat()
        self._cached_inv_rotation_matrices = jax.vmap(lambda r: r.inv().as_matrix())(
            rotations_batch,
        )

    def __check_init__(self) -> None:
        """Check that the radius is positive and the center is 2D or 3D.

        Raises:
            InputError: If the radius is not positive, or if the center is
                not a 1D coordinate of length 2 or 3.

        """
        if self.radius <= 0:
            msg = "Radius must be positive."
            config.LOGGER.error(msg)
            raise config.InputError(msg)

        if self.center.ndim != 1 or self.center.shape[0] not in {
            TWO_DIMENSIONAL,
            THREE_DIMENSIONAL,
        }:
            msg = "Center must be 2D or 3D coordinate"
            config.LOGGER.error(msg)
            raise config.InputError(msg)

    def get_velocity(self, t: float) -> Array:
        """Calculate the instantaneous velocity vector of the path.

        Args:
            t: Time.

        Returns:
            ``[vx, vy, vz]``. ``vz`` is ``0``.

        """
        angle = self.angular_speed * t + self.initial_angle
        vx = -self.radius * self.angular_speed * jnp.sin(angle)
        vy = self.radius * self.angular_speed * jnp.cos(angle)
        vz = 0.0
        return jnp.array([vx, vy, vz])

    def get_rotation(self, u: jnp.ndarray) -> Rotation:
        """Return the closest cached rotation for the point's time.

        The angle is wrapped onto ``[0, 2 * pi)`` and matched to the nearest
        cached sample.

        Args:
            u: A point with spatial coordinates and time in the last component.

        Returns:
            The cached JAX rotation for that angle.

        """
        t = u[..., -1]
        current_angle = self.angular_speed * t + self.initial_angle
        normalized_angle = jnp.mod(current_angle, 2 * jnp.pi)
        idx = jnp.argmin(jnp.abs(self._cached_angles - normalized_angle))
        cached_quaternion = self._cached_rotations[idx]

        # Explicitly return a Rotation object
        return Rotation.from_quat(cached_quaternion)

    def offset_in_local_frame(self, u: Array) -> Array:
        """Return ``offset(u)`` using the cached inverse rotation matrix."""
        t = u[..., -1]
        current_angle = self.angular_speed * t + self.initial_angle
        normalized_angle = jnp.mod(current_angle, 2 * jnp.pi)
        idx = jnp.argmin(jnp.abs(self._cached_angles - normalized_angle))
        offset = self.offset(u)
        if offset.shape[-1] == 2:
            offset = jnp.concatenate([offset, jnp.zeros((1,), dtype=offset.dtype)])
        return self._cached_inv_rotation_matrices[idx] @ offset

    def get_active(self, u: Array) -> Array:
        """Return the stored active flag.

        ``u`` is unused. The flag comes from the config.

        Args:
            u: Unused.

        Returns:
            ``1.0`` when the path is active, and ``0.0`` when it is inactive.

        """
        return jnp.asarray(float(self.active), dtype=config.get_dtype())

    @jax.jit
    def eval(self, time: Array) -> Array:
        """Return the circle coordinates at ``time``.

        Wrapped with ``jax.jit``. ``theta`` is
        ``angular_speed * time + initial_angle``. A 2D center returns
        ``[x, y]``. A 3D center returns ``[x, y, z]`` with ``z`` fixed at
        ``center[2]``.

        Args:
            time: Time, or an array of times.

        Returns:
            Coordinates with shape ``(..., 2)`` or ``(..., 3)``.

        """
        t = jnp.asarray(time, dtype=self.center.dtype)

        theta = self.angular_speed * t + self.initial_angle

        x = self.center[0] + self.radius * jnp.cos(theta)
        y = self.center[1] + self.radius * jnp.sin(theta)

        if self.center.shape[0] == TWO_DIMENSIONAL:
            coords = jnp.stack([x, y], axis=-1)
        else:  # 3D
            z = jnp.full_like(x, self.center[2])
            coords = jnp.stack([x, y, z], axis=-1)

        return coords

    @jax.jit
    def offset(self, point: Array) -> Array:
        """Return the displacement from the circle to a point.

        Wrapped with ``jax.jit``. The circle is evaluated at the last
        component of ``point``.

        Args:
            point: Spatial coordinates followed by time.

        Returns:
            ``point[..., :-1] - eval(point[..., -1])``.

        """
        path_coords = self.eval(point[..., -1])
        return point[..., :-1] - path_coords

    def animate(
    self,
    *,
    domain: SpatialDomain | None = None,
    time: Time | None = None,
    trail: bool = True,
    show_angle_axes: bool = False,
    center: tuple[float, ...] | None = None,
    fig: Figure | None = None,
    ax: Axes | None = None,
    show: bool | None = True,
    save: bool | None = True,
    save_location: str | pathlib.Path | None = None,
    save_name: str | None = "path.gif",
    n_frames: int | None = None,
    fps: int | None = 15,
    dpi: int = 300,
) -> tuple[Figure, Axes, FuncAnimation]:
    """Animate the path over time (path grows from start to current time).

    Either a [`SpatialDomain`][pinnjax.SpatialDomain] or [`Time`][pinnjax.Time] must be provided;
        [`SpatialDomain`][pinnjax.SpatialDomain] takes precedence.

    Args:
        domain: A [`SpatialDomain`][pinnjax.SpatialDomain] object.
        time: A [`Time`][pinnjax.Time] object.
        trail: If True, draw the path line and the circle at the head.
            If False, only the circle at the current time step is shown.
        show_angle_axes: If True, draw a small protractor at 0°, 90°, 180°, 270°.
        center: Optional (x, y) or (x, y, z) for protractor center.
            If None and domain is set, use domain.center();
            otherwise, use path bbox center.
        fig: Optional Matplotlib Figure.
        ax: Optional Matplotlib Axes.
        show: Whether to display the figure via ``plt.show()``.
        save: Whether to save the animation.
        save_location: Directory for saved file.
        save_name: Output filename.
        n_frames: Number of frames. When omitted, the time-step count is
            used, clamped to the range 10 through 80.
        fps: Frames per second for saved gif.
        dpi: Resolution of the saved gif.

    Returns:
        fig, ax, anim: Matplotlib figure, axes, and animation object.

    Raises:
        ValueError: If both ``domain`` and ``time`` are ``None``.
        InputError: If the path is not 2D or 3D.
        RuntimeError: If ``domain`` is set and ``domain.times`` is ``None``.

    """
    if domain is not None:
        fig, ax = domain.draw(fig=fig, ax=ax, show=False)
        time = domain.times
    elif time is not None:
        pass
    else:
        msg = "Either SpatialDomain or Time object must be provided."
        config.LOGGER.error(msg)
        raise ValueError(msg)

    if fig is None:
        fig = plt.figure()

    if time is None:
        msg = (
            "'time' must be defined, either through a Time or SpatialDomain ",
            "object.",
        )
        config.LOGGER.error(msg)
        raise RuntimeError(msg)

    time_steps = int((time.max - time.min) / time.step)
    if n_frames is None:
        n_frames = min(80, max(10, time_steps))
    frame_times = jnp.linspace(time.min, time.max, n_frames)

    if ax is None:
        # Infer dimensionality from path the first time
        coords_0 = self.eval(jnp.array([time.min]))
        ndim = coords_0.shape[1]
        if ndim == 2:
            ax = fig.add_subplot(111)
        elif ndim == 3:
            ax = fig.add_subplot(111, projection="3d")
        else:
            msg = "Invalid domain size for path plotting."
            config.LOGGER.error(msg)
            raise config.InputError(msg)

    ndim = int(self.eval(jnp.array([time.min])).shape[1])
    if show_angle_axes:
        full_coords = self.eval(jnp.linspace(time.min, time.max, 100))
        protractor_center = (
            center if center is not None else (domain.center() if domain is not None else None)
        )
        _draw_angle_axes(ax, full_coords, ndim=ndim, center=protractor_center)

    head_diameter_pt = 8.0
    head_s = 3.14159 * (head_diameter_pt / 2) ** 2
    if ndim == 2:
        head_scat = ax.scatter(
            [],
            [],
            s=head_s,
            c="red",
            edgecolors="white",
            linewidths=1.5,
            zorder=5,
        )
    else:
        if not isinstance(ax, Axes3D):
            msg = "A 3D path requires a 3D axes."
            config.LOGGER.error(msg)
            raise TypeError(msg)
        head_scat = ax.scatter(
            [],
            [],
            [],
            s=head_s,
            c="red",
            edgecolors="white",
            linewidths=1.5,
        )

    # Mutable list of path line artists; we redraw segments each frame
    # (active=solid, inactive=dashed)
    path_line_artists = []

    def _path_segments_by_active(
        coords: Array,
        path_times: Array,
    ) -> list[tuple[int, int, bool]]:
        """Return list of (start_idx, end_idx, is_active) for contiguous runs."""
        n = coords.shape[0]
        if n == 0:
            return []
        active = []
        for i in range(n):
            u = jnp.concatenate([coords[i], jnp.array([float(path_times[i])])])
            a = self.get_active(u)
            active.append(bool(float(a)))
        segments = []
        i = 0
        while i < n:
            j = i
            while j < n and active[j] == active[i]:
                j += 1
            segments.append((i, j, active[i]))
            i = j
        return segments

    def init():
        if ndim == 2:
            head_scat.set_offsets(jnp.empty((0, 2)))
        else:
            head_scat.set_offsets(jnp.empty((0, 2)))
            cast("Any", head_scat).set_3d_properties(jnp.array([]), zdir="z")
        return (head_scat,)

    def update(frame_idx: int):
        t_max = float(frame_times[frame_idx])
        if trail:
            path_times = jnp.linspace(time.min, t_max, max(20, frame_idx + 2))
            coords = self.eval(path_times)
        else:
            coords = self.eval(jnp.array([t_max]))
        head_pt = (float(coords[-1, 0]), float(coords[-1, 1]))

        # Remove previous path segment lines
        for art in path_line_artists:
            art.remove()
        path_line_artists.clear()

        if trail and coords.shape[0] >= 2:
            segments = _path_segments_by_active(coords, path_times)
            for start, end, is_active in segments:
                if end - start < 2:
                    continue
                seg_coords = coords[start:end]
                style = "-" if is_active else "--"
                width = 3 if is_active else 1
                if ndim == 2:
                    (line,) = ax.plot(
                        seg_coords[:, 0].tolist(),
                        seg_coords[:, 1].tolist(),
                        color="red",
                        linewidth=width,
                        linestyle=style,
                        zorder=5,
                    )
                else:
                    if not isinstance(ax, Axes3D):
                        msg = "A 3D path requires a 3D axes."

    """Return the circle coordinates at ``time``.

    Wrapped with ``jax.jit``. ``theta`` is
    ``angular_speed * time + initial_angle``. A 2D center returns
    ``[x, y]``. A 3D center returns ``[x, y, z]`` with ``z`` fixed at

    Args:

    Returns:
        Coordinates with shape ``(..., 2)`` or ``(..., 3)``.

    """
    t = jnp.asarray(time, dtype=self.center.dtype)

    theta = self.angular_speed * t + self.initial_angle

    x = self.center[0] + self.radius * jnp.cos(theta)
    y = self.center[1] + self.radius * jnp.sin(theta)

    if self.center.shape[0] == TWO_DIMENSIONAL:

    else:  # 3D
        z = jnp.full_like(x, self.center[2])
        coords = jnp.stack([x, y, z], axis=-1)

    return coords

    class ComplexLinearPath(Path, ABC):
    """Piecewise-linear path through a sequence of waypoints.

    Each segment is a [`LinearPath`][pinnjax.LinearPath]. Active segments
    are solid when drawn. Inactive segments are dashed. ``time`` on ``draw``
    is accepted and unused. The path uses its own waypoint times.

    Args:
        config: [`ComplexLinearPathConfig`][pinnjax.ComplexLinearPathConfig].

    Attributes:
        config: The stored configuration.
        cumulative_times: Waypoint times, one longer than the segment count.
        segment_times: Duration of each segment.
        active: One flag per segment.
        velocities: Unit direction times speed, one row per segment.
        rotations: One rotation per segment, from those velocities.
        inv_rotation_matrices: Inverse rotation matrices, shape
            ``(n_segments, 3, 3)``, matching ``rotations.inv().as_matrix()``.
        segments: One [`LinearPath`][pinnjax.LinearPath] per segment.
        dimensions: Spatial width of the first waypoint.

    """

    config: ComplexLinearPathConfig = eqx.field(static=True)
    cumulative_times: Array
    segment_times: Array
    active: Array
    velocities: Array
    rotations: Rotation
    inv_rotation_matrices: Array

    segments: tuple[LinearPath, ...]

    def __init__(self, config: ComplexLinearPathConfig) -> None:
        """Store the segments, the velocities, and one rotation per segment."""
        self.config = config
        active_flags = cast("Array", config.active)
        self.active = jnp.asarray(active_flags)
        self.cumulative_times = config.cumulative_times
        self.segment_times = config.segment_times
        self.active = jnp.asarray(config.active)
        self.dimensions = len(config.initials[0])
        segments = []

        for _i, (initial, speed, direction, active) in enumerate(
            zip(
                config.initials,
                config.speeds,
                config.directions,
                active_flags,
                strict=False,
            ),
        ):
            cfg = LinearPathConfig(
                initial=initial,
                speed=speed,
                direction=direction,
                active=active,
            )
            segments.append(LinearPath(cfg))

        self.segments = tuple(segments)

        self.velocities = jnp.stack(
            [
                (direction / jnp.linalg.norm(direction)) * speed
                for direction, speed in zip(
                    config.directions,
                    config.speeds,
                    strict=False,
                )
            ],
        )

        self.rotations = jax.vmap(Path._create_rotation)(self.velocities)
        self.inv_rotation_matrices = jax.vmap(lambda r: r.inv().as_matrix())(self.rotations)

    def _get_segment_index(self, t: Array | float) -> Array:
        idx = jnp.searchsorted(self.config.cumulative_times, t, side="right") - 1
        return jnp.clip(idx, 0, len(self.config.segment_times) - 1)

    def _get_segment(self, t: Array | float) -> LinearPath:
        idx = self._get_segment_index(t)
        return self.segments[int(idx)]

    def get_velocity(self, t: float) -> Array:
        """Return the constant velocity of the segment containing ``t``.

        Args:
            t: Time.

        Returns:
            The segment velocity. Its direction is the unit segment direction.

        """
        idx = self._get_segment_index(t)
        return self.velocities[idx]

    def get_rotation(self, u: Array) -> Rotation:
        """Return the rotation for the segment containing the point's time.

        Args:
            u: One point. Time is the last component.

        Returns:
            The segment's JAX rotation.

        """
        t = u[-1]
        idx = self._get_segment_index(t)
        return self.rotations[idx]

    def offset_in_local_frame(self, u: Array) -> Array:
        """Return ``offset(u)`` using the cached inverse rotation matrix."""
        offset = self.offset(u)
        if offset.shape[-1] == 2:
            offset = jnp.concatenate([offset, jnp.zeros((1,), dtype=offset.dtype)])
        idx = self._get_segment_index(u[-1])
        return self.inv_rotation_matrices[idx] @ offset

    def get_active(self, u: Array) -> Array:
        """Return the active flag for the segment containing the point's time.

        Args:
            u: One point. Time is the last component.

        Returns:
            The segment flag from ``active``.

        """
        t = u[-1]
        idx = self._get_segment_index(t)
        return self.active[idx]

    @jax.jit
    def eval(self, t: Array) -> Array:
        """Evaluate position at time ``t`` via interpolation.

        ``t`` may be a scalar or an array. Wrapped with ``jax.jit``. The
        segment is chosen from the cumulative waypoint times and clipped to
        the first and last segments. Position is the segment start plus the
        segment displacement times the fraction of that segment's duration.

        Args:
            t: Time, or an array of times.

        Returns:
            Coordinates with shape ``(..., spatial_dims)``.

        """
        t = jnp.asarray(t, dtype=config.get_dtype())
        idx = self._get_segment_index(t)

        # local fraction within the chosen segment
        local_t = t - jnp.take(self.config.cumulative_times, idx, mode="clip")
        seg_dt = jnp.take(self.config.segment_times, idx, mode="clip")
        frac = local_t / seg_dt

        # gather per-segment initial and direction, then interpolate
        p0 = jnp.take(self.config.initials, idx, axis=0, mode="clip")
        d = jnp.take(self.config.directions, idx, axis=0, mode="clip")
        return p0 + d * frac[..., None]

    @jax.jit
    def offset(self, point: Array) -> Array:
        """Return the displacement from the path to a point.

        Wrapped with ``jax.jit``. The path is evaluated at the last component
        of ``point``.

        Args:
            point: Spatial coordinates followed by time.

        Returns:
            ``point[..., :-1] - eval(point[..., -1])``.

        """
        p = jnp.asarray(point, dtype=config.get_dtype())
        return p[..., :-1] - self.eval(p[..., -1])

    def _path_segments_by_active(
        coords: Array,
        path_times: Array,
    ) -> list[tuple[int, int, bool]]:
        """Return list of (start_idx, end_idx, is_active) for contiguous runs."""
        n = coords.shape[0]
        if n == 0:
            return []
        active = []
        for i in range(n):
            u = jnp.concatenate([coords[i], jnp.array([float(path_times[i])])])
            a = self.get_active(u)
            active.append(bool(float(a)))
        segments = []
        i = 0
        while i < n:
            j = i
            while j < n and active[j] == active[i]:
                j += 1
            segments.append((i, j, active[i]))
            i = j
        return segments

    def init():
        if ndim == 2:
            head_scat.set_offsets(jnp.empty((0, 2)))
        else:
            head_scat.set_offsets(jnp.empty((0, 2)))
            cast("Any", head_scat).set_3d_properties(jnp.array([]), zdir="z")
        return (head_scat,)

    def update(frame_idx: int):
        t_max = float(frame_times[frame_idx])
        if trail:
            path_times = jnp.linspace(time.min, t_max, max(20, frame_idx + 2))
            coords = self.eval(path_times)
        else:
            coords = self.eval(jnp.array([t_max]))
        head_pt = (float(coords[-1, 0]), float(coords[-1, 1]))

        # Remove previous path segment lines
        for art in path_line_artists:
            art.remove()
        path_line_artists.clear()

        if trail and coords.shape[0] >= 2:
            segments = _path_segments_by_active(coords, path_times)
            for start, end, is_active in segments:
                if end - start < 2:
                    continue
                seg_coords = coords[start:end]
                style = "-" if is_active else "--"
                width = 3 if is_active else 1
                if ndim == 2:
                    (line,) = ax.plot(
                        seg_coords[:, 0].tolist(),
                        seg_coords[:, 1].tolist(),
                        color="red",
                        linewidth=width,
                        linestyle=style,
                        zorder=5,
                    )
                else:
                    if not isinstance(ax, Axes3D):
                        msg = "A 3D path requires a 3D axes."
                        config.LOGGER.error(msg)
                        raise TypeError(msg)
                    (line,) = ax.plot3D(
                        seg_coords[:, 0].tolist(),
                        seg_coords[:, 1].tolist(),
                        seg_coords[:, 2].tolist(),
                        color="red",
                        linewidth=width,
                        linestyle=style,
                        zorder=5,
                    )
                path_line_artists.append(line)

        # Head circle: red when path active, grey when off
        head_u = jnp.concatenate([coords[-1], jnp.array([t_max])])
        head_active = bool(float(self.get_active(head_u)))
        head_color = "red" if head_active else "grey"
        cast("Any", head_scat).set_facecolors([head_color])

        if ndim == 2:
            head_scat.set_offsets([head_pt])
        else:
            head_scat.set_offsets([head_pt])
            cast("Any", head_scat).set_3d_properties([float(coords[-1, 2])], zdir="z")
        return (head_scat,)

    # blit=False so the head scatter updates every frame (blit can leave it stuck)
    playback_fps = 15 if fps is None else fps
    interval = 1000 / playback_fps
    anim = FuncAnimation(
        fig,
        update,
        frames=n_frames,
        init_func=init,
        interval=interval,
        blit=False,
        repeat=True,
    )

    if save:
        out_dir = pathlib.Path(save_location) if save_location else pathlib.Path.cwd()
        out_dir.mkdir(parents=True, exist_ok=True)
        out_path = out_dir / (save_name or "path.gif")
        writer = PillowWriter(fps=playback_fps, bitrate=-1)
        anim.save(out_path, writer=writer, dpi=dpi)

    if show:
        plt.show()
    else:
        plt.close()

    return fig, ax, anim

    @jax.jit
def eval(self, t: Array) -> Array:
    """Evaluate position at time ``t`` via interpolation.

    ``t`` may be a scalar or an array. Wrapped with ``jax.jit``. The
    segment is chosen from the cumulative waypoint times and clipped to
    the first and last segments. Position is the segment start plus the
    segment displacement times the fraction of that segment's duration.

    Args:
        t: Time, or an array of times.

    Returns:
        Coordinates with shape ``(..., spatial_dims)``.

    """
    t = jnp.asarray(t, dtype=config.get_dtype())
    idx = self._get_segment_index(t)

    # local fraction within the chosen segment
    local_t = t - jnp.take(self.config.cumulative_times, idx, mode="clip")
    seg_dt = jnp.take(self.config.segment_times, idx, mode="clip")
    frac = local_t / seg_dt

    # gather per-segment initial and direction, then interpolate
    p0 = jnp.take(self.config.initials, idx, axis=0, mode="clip")
    d = jnp.take(self.config.directions, idx, axis=0, mode="clip")
    return p0 + d * frac[..., None]

    def validate(self, domain: SpatialDomain) -> None:
    # TODO(@mjryande): Paths are defined as endless, so if the total time is
    #   too long, validate method wrongly assesses it.
    """Validate that the entire path stays within the domain bounds.

    Args:
        domain: A [`SpatialDomain`][pinnjax.SpatialDomain] object with spatial and
            temporal bounds.

    Raises:
        InputError: If `domain.times` is None (no temporal information).
        InputError: If path coordinates fall outside the domain's min/max bounds.

    """
    if domain.times is None:
        msg = "domain must have time information for path validation"
        config.LOGGER.error(msg)
        raise config.InputError(msg)

    time_steps = int((domain.times.max - domain.times.min) / domain.times.step)
    times = jnp.linspace(domain.times.min, domain.times.max, time_steps)
    coordinates = self.eval(times)  # Shape: (time_steps, spatial_dims)
    spatial_dims = coordinates.shape[1]
    # Check each spatial dimension
    for dim in range(spatial_dims):
        coords = coordinates[:, dim]
        coord_max, coord_min = jnp.max(coords), jnp.min(coords)

        if coord_min < domain.min[dim] or coord_max > domain.max[dim]:
            msg = "Path exceeds the bounds of the given domain."
            config.LOGGER.error(msg)
            raise config.InputError(msg)
