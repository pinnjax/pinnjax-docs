    class AdaptiveLossWeights(eqx.Module):
    """Gradient-based adaptive weighting over losses.

    Implements the method by Wang et al. for adaptively weighting losses
    during PINN training.

    """

    weights: jnp.ndarray
    mode: str = eqx.field(static=True)  # 'class' or 'type'
    alpha: float = eqx.field(static=True)
    verbosity: bool = eqx.field(static=True)

    def __init__(
        self,
        n_classes: int,
        mode: str,
        alpha: float = 0.9,
        *,
        verbosity: None | int = None,
    ) -> None:
        """Initialize an AdaptiveLossWeights instance.

        Args:
            n_classes: Number of classes or types to compute weights for.
            mode: 'class' or 'type' indicating which loss vector to use.
            alpha: Exponential smoothing factor for updating weights.
            verbosity: Optional verbosity flag; enables logging if not None.

        """
        self.alpha = alpha
        self.mode = mode
        self.weights = jnp.ones(n_classes, dtype=config.get_dtype())
        self.verbosity = verbosity is not None

    def update_weights(
        self,
        model: pinnjax.NeuralNetwork,
        _geometry: pinnjax.ComplexShape,
        scale: Array | None = None,
    ) -> AdaptiveLossWeights:
        """Update the adaptive weights based on gradient norms of each loss.

        Args:
            model: The neural network model.
            _geometry: ComplexShape object containing the loss functions.
            scale: Optional array to scale losses before computing gradients.

        Returns:
            AdaptiveLossWeights: Updated object with new weights.

        """

        def _losses_vec(m: pinnjax.NeuralNetwork) -> Array:
            _, by_class, by_type = _geometry.residual_and_losses(
                m,
                pinnjax.DifferentialOperators(m),
            )
            return by_class if self.mode == "class" else by_type

        losses = _losses_vec(model)

        if scale is None:
            scale = jnp.ones_like(losses, dtype=config.get_dtype())

        losses = losses * scale

        threshold = 1e-18
        non_zero_mask = losses > threshold

        if not bool(jnp.any(non_zero_mask)):
            return self

        n_losses = losses.shape[0]
        active_indices = jnp.where(non_zero_mask)[0]  # Get indices of active losses

        if len(active_indices) == 0:
            grad_norms = jnp.zeros(n_losses, dtype=config.get_dtype())
        else:
            # Helper function to compute gradient norm for a single loss component
            def _compute_grad_norm(idx: int) -> float:
                def _loss_k(m: pinnjax.NeuralNetwork) -> Array:
                    loss_vec = _losses_vec(m)
                    return scale[idx] * loss_vec[idx]

                grads = jax.grad(_loss_k)(model)
                leaves = [g for g in jax.tree_util.tree_leaves(grads) if g is not None]
                if len(leaves) == 0:
                    return 0.0

                flat = jnp.hstack([g.reshape(-1) for g in leaves])
                return float(jnp.linalg.norm(flat))

            # Compute gradients only for active losses
            active_grad_norms = jnp.array(
                [_compute_grad_norm(int(idx)) for idx in active_indices],
                dtype=config.get_dtype(),
            )

            # Build full grad_norms array with zeros for inactive losses
            grad_norms = jnp.zeros(n_losses, dtype=config.get_dtype())
            grad_norms = grad_norms.at[active_indices].set(active_grad_norms)

        active = grad_norms[non_zero_mask]
        total_g = jnp.sum(jnp.where(jnp.isfinite(active), active, 0.0))
        new_w_active = total_g / (active + config.get_EPS())
        new_w_active = jnp.clip(
            new_w_active,
            0.0,
            10000.0,  # increased to 10000 from 100
            # decreased lower bound from 1.0 to 0.0
        )

        new_w = jnp.array(self.weights)
        new_w = new_w.at[non_zero_mask].set(new_w_active)
        new_w = new_w.at[~non_zero_mask].set(1.0)

        updated_w = self.alpha * self.weights + (1.0 - self.alpha) * new_w

        if self.verbosity:
            config.LOGGER.info(
                "Updated %s Weights: %s",
                self.mode.title(),
                ", ".join([f"{float(w):.3f}" for w in updated_w]),
            )

        return eqx.tree_at(lambda x: x.weights, self, updated_w)

    def train(
    model: pinnjax.NeuralNetwork,
    geometry: pinnjax.ComplexShape,
    training: pinnjax.Training,
    *,
    initial_loss_weights: Iterable | None = None,
    # Ground Truth
    evaluation: pinnjax.Evaluation | None = None,
    # print
    verbosity: None | int = 10,
    print_mode: str = "class",
) -> pinnjax.NeuralNetwork:
    """Train a PINN model with optional adaptive loss weighting and ground truth.

    Args:
        model: pinnjax.NeuralNetwork object.
        geometry: ComplexShape defining residuals and loss functions.
        training: Training configuration object.
        initial_loss_weights: Optional iterable to override initial loss weights.
        evaluation: pinnjax.Evaluation object.

        verbosity: Logging frequency or None to suppress logs.
        print_mode: 'class' or 'type' to select which loss labels to print.


    When ``training.batch_size`` is set, each epoch runs multiple optimizer steps
    over leaf shapes. Logged ``total_raw`` and
    per-class / per-type values are means over mini-batches in that epoch.
    LBFGS requires ``batch_size=None``.

    Returns:
        pinnjax.NeuralNetwork: The best model observed during training.

    """
    if verbosity is not None and training.epochs != 0:
        config.LOGGER.info("%s", training)

    if training.epochs <= 0:
        config.LOGGER.warning(
            "Epochs set to %d. No Training performed.",
            training.epochs,
        )
        return model

    if training.batch_size is not None:
        if training.batch_size < 1:
            msg = (
                f"training.batch_size must be >= 1 or None, got {training.batch_size}."
            )
            config.LOGGER.error(msg)
            raise ValueError(msg)

    is_lbfgs = training.optimizer.opt_.__name__.lower() == "lbfgs"
    is_optax = training.optimizer.opt_.__module__.startswith("optax")

    if is_lbfgs and training.batch_size is not None:
        msg = (
            "Mini-batch training (training.batch_size is set) is not supported with "
            "LBFGS; use batch_size=None for full-batch second-order optimization."
        )
        config.LOGGER.error(msg)
        raise ValueError(msg)

    match training.selection_criterion:
        case "loss":
            pass
        case "evaluate":
            if evaluation is None:
                msg = (
                    "training.selection_criterion is 'evaluate' but evaluation is None."
                )
                config.LOGGER.error(msg)
                raise ValueError(msg)
        case _:
            msg = (
                f"Invalid training.selection_criterion: {training.selection_criterion!r}. "
                "Expected 'loss' or 'evaluate'."
            )
            config.LOGGER.error(msg)
            raise ValueError(msg)

    epochs = training.epochs

    if hasattr(geometry, "_n_class") and geometry.n_class > 0:
        n_classes = geometry.n_class
        n_types = geometry.n_type
    else:
        n_classes = max(geometry.kinds_class) + 1
        n_types = max(geometry.kinds_type) + 1
    n_target = n_classes if training.train_mode == "class" else n_types

    if initial_loss_weights is not None:
        try:
            iter(initial_loss_weights)
            if len(initial_loss_weights) == n_target:  # type: ignore
                current_weights = jnp.asarray(
                    initial_loss_weights, dtype=config.get_dtype()
                )
            else:
                msg = (
                    f"Inputted Initial Weights: {initial_loss_weights} "
                    f"don't have the same number of indices as loss values: {n_target}",
                )
                config.LOGGER.error(msg)
                raise ValueError(msg)
        except (TypeError, AttributeError):
            # Not iterable or no length
            msg = f"Initial loss weights must be iterable, got {type(initial_loss_weights)}"
            config.LOGGER.error(msg)
            raise ValueError(msg)
    else:
        current_weights = jnp.ones(n_target, dtype=config.get_dtype())

    use_adaptive_weights = training.use_adaptive_weights
    if is_lbfgs is True and use_adaptive_weights is True:
        msg = (
            "LBFGS is incompatible with Adaptive Weights.\n"
            'Press "Y" to continue without Adaptive Weights.\n'
            'Press "N" to stop training.'
        )
        user_input = input(msg)
        if user_input.casefold() == "y":
            # unfortunately, this change will not be reflected in the Training object
            # this must be done because the Training object is frozen.
            use_adaptive_weights = False
        else:
            sys.exit()

    # allow for the ability to dictate how important a given residual is to the
    # model training. defaults to 1.
    if training.priority is None:
        priority = jnp.ones(n_target, dtype=config.get_dtype())
    else:
        priority = jnp.asarray(training.priority, dtype=config.get_dtype())
        if priority.shape != (n_target,):
            msg = f"'priority' must have shape ({n_target},), got {priority.shape}"
            config.LOGGER.error(msg)
            raise ValueError(msg)

    weight_manager = (
        AdaptiveLossWeights(
            n_target,
            training.train_mode,
            alpha=training.adaptive_alpha,
            verbosity=verbosity,
        )
        if use_adaptive_weights
        else None
    )

    if training.balance_magnitudes:
        magnitude = _initial_weights_from_magnitudes(
            model,
            geometry,
            mode=training.train_mode,
            magnitude_clip=training.magnitude_clip,
        )
    else:
        magnitude = jnp.ones(n_target, dtype=config.get_dtype())

    @eqx.filter_jit
    def _loss(
        _model: pinnjax.NeuralNetwork,
        _weights: Array,
        _samples_override: tuple[Array, ...] | None,
    ) -> tuple[Array, tuple[Array, Array, Array]]:
        ops = pinnjax.DifferentialOperators(_model)
        _total_raw, _by_class, _by_type = geometry.residual_and_losses(
            _model,
            ops,
            samples_list_override=_samples_override,
        )
        eff = _weights * priority * magnitude
        if training.train_mode == "class":
            total = jnp.dot(eff, _by_class)
        else:
            total = jnp.dot(eff, _by_type)
        return total * training.loss_scale, (_total_raw, _by_class, _by_type)

    @eqx.filter_jit
    def step(
        _model: pinnjax.NeuralNetwork,
        _opt_state: ArrayTree,
        _weights: Array,
        _samples_override: tuple[Array, ...] | None,
    ) -> tuple[
        pinnjax.NeuralNetwork,
        pinnjax.NeuralNetwork,
        ArrayTree,
        Array,
        Array,
        Array,
        Array,
    ]:
        _old_model = _model
        (_loss_val, (_total_raw, _by_class, _by_type)), _grads = (
            eqx.filter_value_and_grad(
                lambda m: _loss(m, _weights, _samples_override),
                has_aux=True,
            )(
                _model,
            )
        )

        if is_lbfgs:
            if is_optax:
                updates, _opt_state = opt.update(
                    _grads,
                    _opt_state,
                    params=eqx.filter(_model, eqx.is_array),
                    value=_loss_val,
                    grad=_grads,
                    value_fn=lambda m: _loss(m, current_weights, None)[0],
                )

            else:
                out = opt.run(eqx.filter(_model, eqx.is_inexact_array))
                _model = eqx.tree_at(lambda m: m, _model, out.params)
                return (
                    _model,
                    _old_model,
                    _opt_state,
                    _loss_val,
                    _total_raw,
                    _by_class,
                    _by_type,
                )

        else:
            updates, _opt_state = opt.update(
                _grads,
                _opt_state,
                eqx.filter(_model, eqx.is_array),
            )

        _model = eqx.apply_updates(_model, updates)

        return (
            _model,
            _old_model,
            _opt_state,
            _loss_val,
            _total_raw,
            _by_class,
            _by_type,
        )

    if not is_optax and is_lbfgs:
        opt = training.optimizer.opt_(
            fun=lambda params: _loss(params, current_weights, None)[0],
            **training.optimizer.kwargs,
        )
        opt_state = None
    else:
        opt = training.optimizer.opt_(
            *training.optimizer.args,
            **training.optimizer.kwargs,
        )
        opt_state = opt.init(eqx.filter(model, eqx.is_inexact_array))

    if training.batch_size is None:
        steps_per_epoch = 1
    else:
        steps_per_epoch = _training_steps_per_epoch(
            geometry,
            training.batch_size,
            drop_last=training.drop_last_batch,
        )

    if evaluation is not None and not geometry.evaluation_dictionary:
        geometry.evaluation_dictionary = {
            "Epochs": [],
            **{k: [] for k in evaluation.DICTIONARY},
        }

    best_model = model
    best_monitor: float | None = None
    best_epoch = 0
    last_epoch = training.epochs
    epochs_since_improvement = 0

    test_arr = jnp.array([1], dtype=config.get_dtype())
    current_device = test_arr.device

    if is_lbfgs and current_device.platform == "gpu":
        # LBFGS is slower on GPU, switch to CPU
        if verbosity:
            config.LOGGER.info(" - SWITCHING TO CPU (LBFGS optimizer) - ")
        # Switch JAX to use CPU
        jax.config.update("jax_default_device", "cpu")
        cpu_device = jax.devices("cpu")[0]

        # Auto-move JAX arrays to CPU
        geometry = _move_geometry_to_device(geometry, cpu_device)
        model = _move_model_to_device(model, cpu_device)
        if opt_state is not None:
            opt_state = _move_to_device(opt_state, cpu_device)
        current_weights = jax.device_put(current_weights, cpu_device)
        magnitude = jax.device_put(magnitude, cpu_device)
        priority = jax.device_put(priority, cpu_device)

        best_model = model

    elif any(d.platform == "gpu" for d in jax.devices()) and not config.get_FORCE_CPU():
        # For non-LBFGS optimizers, prefer GPU if available
        gpu_device = jax.devices("gpu")[0]
        if current_device != gpu_device:
            if verbosity:
                config.LOGGER.info(" - SWITCHING TO GPU - ")
            # Switch JAX to use GPU
            jax.config.update("jax_default_device", "gpu")

            # Auto-move JAX arrays to GPU
            geometry = _move_geometry_to_device(geometry, gpu_device)
            model = _move_model_to_device(model, gpu_device)
            if opt_state is not None:
                opt_state = _move_to_device(opt_state, gpu_device)
            current_weights = jax.device_put(current_weights, gpu_device)
            magnitude = jax.device_put(magnitude, gpu_device)
            priority = jax.device_put(priority, gpu_device)

            best_model = model

    scale = jnp.multiply(magnitude, priority).astype(config.get_dtype())
    for i in range(epochs):
        if (
            use_adaptive_weights
            and i % training.update_weights_every == 0
            and i > 0
            and i >= training.update_weights_after
        ) or (use_adaptive_weights and i == training.update_weights_after):
            weight_manager = weight_manager.update_weights(model, geometry, scale)

        if use_adaptive_weights:
            current_weights = weight_manager.weights

        if training.batch_size is None:
            model, old_model, opt_state, loss_val, total_raw, by_class, by_type = step(
                model,
                opt_state,
                current_weights,
                None,
            )
        else:
            epoch_key = jax.random.fold_in(jax.random.key(int(config.get_seed())), i)
            perms = _epoch_permutations(geometry, epoch_key,
                                        shuffle=training.shuffle_batches)
            batch_totals: list[float] = []
            batch_loss_vals: list[float] = []
            batch_by_class: list[Array] = []
            batch_by_type: list[Array] = []
            for t in range(steps_per_epoch):
                so = _training_batch_samples(
                    geometry,
                    training.batch_size,
                    drop_last=training.drop_last_batch,
                    perms=perms,
                    step_index=t,
                )
                (
                    model,
                    old_model,
                    opt_state,
                    loss_val,
                    total_raw,
                    by_class,
                    by_type,
                ) = step(
                    model,
                    opt_state,
                    current_weights,
                    so,
                )
                batch_totals.append(float(total_raw))
                batch_loss_vals.append(float(loss_val))
                batch_by_class.append(by_class)
                batch_by_type.append(by_type)
            total_raw = jnp.mean(jnp.asarray(batch_totals))
            loss_val = jnp.mean(jnp.asarray(batch_loss_vals))
            by_class = jnp.mean(jnp.stack(batch_by_class), axis=0)
            by_type = jnp.mean(jnp.stack(batch_by_type), axis=0)

        # TODO(@mjryande): maybe recalculate the losses with the new model
        #   and save that, or something. Currently it shows the previous loss, not the
        #   loss the model achieved. - just what it was updated using.

        current = float(total_raw)

        match training.selection_criterion:
            case "loss":
                if best_monitor is None:
                    best_monitor = current
                    best_model = model
                    best_epoch = i + 1
                    epochs_since_improvement = 0
                else:
                    if current < best_monitor:
                        improvement = (best_monitor - current) / abs(best_monitor)
                        best_monitor = current
                        best_model = (
                            old_model  # model which achieved the loss (before update)
                        )
                        best_epoch = i + 1
                        if improvement > training.rel_tolerance:
                            epochs_since_improvement = 0
                        else:
                            epochs_since_improvement += 1
                    else:
                        epochs_since_improvement += 1

                    if (
                        training.early_stopping
                        and epochs_since_improvement >= training.patience
                    ):
                        last_epoch = i + 1
                        if verbosity is not None:
                            config.LOGGER.info(
                                "Early Stopping: Epoch %d\n"
                                "No Improvement greater than %.3f%% in %d epochs.",
                                i + 1,
                                training.rel_tolerance * 100,
                                training.patience,
                            )
                        break
            case "evaluate":
                assert evaluation is not None
                sel_key = evaluation.metric_key
                if sel_key not in evaluation.metric_direction:
                    msg = (
                        f"metric_key {sel_key!r} is missing from "
                        "evaluation.metric_direction."
                    )
                    config.LOGGER.error(msg)
                    raise ValueError(msg)
                direction = evaluation.metric_direction[sel_key]
                higher_is_better = direction == "maximize"
                monitor = float(evaluation.evaluate_selection_metric(old_model))

                if best_monitor is None:
                    best_monitor = monitor
                    best_model = model
                    best_epoch = i + 1
                    epochs_since_improvement = 0
                else:
                    if higher_is_better:
                        improved = monitor > best_monitor
                    else:
                        improved = monitor < best_monitor
                    if improved:
                        if higher_is_better:
                            improvement = (monitor - best_monitor) / abs(best_monitor)
                        else:
                            improvement = (best_monitor - monitor) / abs(best_monitor)
                        best_monitor = monitor
                        best_model = old_model
                        best_epoch = i + 1
                        if improvement > training.rel_tolerance:
                            epochs_since_improvement = 0
                        else:
                            epochs_since_improvement += 1
                    else:
                        epochs_since_improvement += 1

                    if (
                        training.early_stopping
                        and epochs_since_improvement >= training.patience
                    ):
                        last_epoch = i + 1
                        if verbosity is not None:
                            config.LOGGER.info(
                                "Early Stopping: Epoch %d\n"
                                "No Improvement greater than %.3f%% in %d epochs.",
                                i + 1,
                                training.rel_tolerance * 100,
                                training.patience,
                            )
                        break

        # assign losses to loss_dictionary:
        by_class_list = by_class.tolist()
        by_type_list = by_type.tolist()

        for key, val in zip(
            geometry.loss_dictionary["class"].keys(),
            by_class_list,
            strict=True,
        ):
            geometry.loss_dictionary["class"][key].append(val)

        for key, val in zip(
            geometry.loss_dictionary["type"].keys(),
            by_type_list,
            strict=True,
        ):
            geometry.loss_dictionary["type"][key].append(val)

        if verbosity is not None and (
            (i + 1) % verbosity == 0 or i == training.epochs - 1 or i == 0
        ):
            labels = getattr(
                geometry,
                "labels_type" if print_mode == "type" else "labels_class",
                tuple(
                    str(i)
                    for i in range(n_types if print_mode == "type" else n_classes)
                ),
            )
            values = by_type if print_mode == "type" else by_class
            parts = " ".join(
                f"{lbl}={float(v):.4e}" for lbl, v in zip(labels, values, strict=True)
            )

            config.LOGGER.info(
                "Epoch: %d | weighted total: %.4e raw total: %.4e | %s",
                i + 1,
                float(loss_val),
                float(total_raw),
                parts,
            )
        # Evaluation
        if (
            evaluation is not None
            and evaluation.regularity is not None
            and ((i + 1) % evaluation.regularity == 0 or i in [0, training.epochs - 1])
        ):
            values = evaluation.evaluate(model)

            geometry.evaluation_dictionary["Epochs"].append(i + 1)
            for k in evaluation.DICTIONARY:
                geometry.evaluation_dictionary[k].append(float(values[k]))

            if verbosity is not None:
                if (i + 1) % evaluation.verbosity == 0 or i in [0, training.epochs - 1]:
                    metrics = evaluation.log_metrics(values)
                    config.LOGGER.info("Epoch: %d | %s", i + 1, metrics)

    # Wrap-Up
    (loss_val, (total_raw, by_class, by_type)) = _loss(
        best_model,
        current_weights,
        None,
    )

    if verbosity is not None:
        labels = getattr(
            geometry,
            "labels_type" if print_mode == "type" else "labels_class",
            tuple(
                str(i) for i in range(n_types if print_mode == "type" else n_classes)
            ),
        )
        values = by_type if print_mode == "type" else by_class
        parts = " ".join(
            f"{lbl}={float(v):.4e}" for lbl, v in zip(labels, values, strict=True)
        )

        config.LOGGER.info(
            "Final | weighted total: %.4e raw total: %.4e | %s",
            float(loss_val),
            float(total_raw),
            parts,
        )

        if evaluation is not None:
            values = evaluation.evaluate(best_model)
            if verbosity is not None:
                metrics = evaluation.log_metrics(values)
                config.LOGGER.info("Final | %s", metrics)

        config.LOGGER.info("The Best Model at Epoch %d", best_epoch)

    geometry.best_epoch += best_epoch
    geometry.last_epoch += last_epoch
    geometry.loss_weights = current_weights

    return best_model
