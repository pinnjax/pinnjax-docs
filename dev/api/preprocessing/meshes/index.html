    def from_stl(
    to: FileType,
    file: str | Path,
    *,
    folder: str | Path | None = None,
    name: str | None = None,
    verbose: bool = False,
) -> None:
    """Convert an STL file or directory to a smaller mesh file.

    ``"hdf5"`` writes ``float32`` points and ``uint32`` faces to a
    gzip-compressed ``.h5`` file. ``"npz"`` writes those arrays to a compressed
    ``.npz`` archive. ``"ply"`` writes a binary PLY mesh. A directory is
    searched recursively for ``.stl`` files and written in numeric filename
    order. ``verbose`` logs that progress.

    Args:
        to: ``"hdf5"``, ``"ply"``, or ``"npz"``.
        file: STL file, or a directory of STL files.
        folder: Output directory. A directory input keeps each file's relative
            folder. When omitted, the source file's folder.
        name: Output file name. A different suffix warns and is replaced.
            Only the file name is kept. When omitted, the source stem plus the
            format suffix.
        verbose: Log directory progress.

    Returns:
        ``None``.

    Raises:
        ValueError: If ``name`` is set while ``file`` is a directory.
        ValueError: If ``file`` is not an STL file.
        ValueError: If ``to`` is not ``"hdf5"``, ``"ply"``, or ``"npz"``.

    """
    file = Path(file)

    if file.is_dir():
        if name is not None:
            raise ValueError("'name' cannot be used when converting a directory.")

        stl_files = sorted(
            (path for path in file.rglob("*") if path.is_file() and path.suffix.lower() == ".stl"),
            key=lambda path: [
                int(part) if part.isdigit() else part.lower()
                for part in re.split(r"(\d+)", str(path.relative_to(file)))
            ],
        )
        total = len(stl_files)
        for index, stl_file in enumerate(stl_files, start=1):
            if folder is None:
                output_folder = None
            else:
                relative_folder = stl_file.parent.relative_to(file)
                output_folder = Path(folder) / relative_folder

            from_stl(
                to,
                stl_file,
                folder=output_folder,
            )

            if verbose:
                width = len(str(total))
                percent = index / total * 100

                config.LOGGER.info(f"[{index:>{width}}/{total}] {percent:5.1f}% {stl_file.name}")

        return

    if file.suffix.lower() != ".stl":
        raise ValueError(f"Expected an STL file, got {file.suffix!r}")

    mesh = trimesh.load_mesh(file)

    points = np.asarray(mesh.vertices, dtype=np.float32)
    faces = np.asarray(mesh.faces, dtype=np.uint32)

    suffixes = {
        "npz": ".npz",
        "hdf5": ".h5",
        "ply": ".ply",
    }
    suffix = suffixes[to]

    output_folder = Path(folder) if folder is not None else file.parent
    output_folder.mkdir(parents=True, exist_ok=True)

    if name is None:
        output_name = file.stem + suffix
    else:
        output_name = Path(name)

        if output_name.suffix:
            if output_name.suffix.lower() != suffix:
                warnings.warn(
                    f"Expected suffix {suffix!r}, got {output_name.suffix!r}. Replacing it.",
                    stacklevel=2,
                )

            output_name = output_name.with_suffix(suffix)
        else:
            output_name = output_name.with_suffix(suffix)

        output_name = output_name.name

    output_path = output_folder / output_name

    match to:
        case "npz":
            npz_path = cast("str", output_path)

            np.savez_compressed(
                npz_path,
                points=points,
                faces=faces,
            )

        case "hdf5":
            import h5py

            with h5py.File(output_path, "w") as f:
                f.create_dataset(
                    "points",
                    data=points,
                    compression="gzip",
                    shuffle=True,
                )
                f.create_dataset(
                    "faces",
                    data=faces,
                    compression="gzip",
                    shuffle=True,
                )

        case "ply":
            mesh.export(
                output_path,
                file_type="ply",
                encoding="binary",
            )

        case _:
            raise ValueError(
                f"File type must be in FileType {get_args(FileType)}. Got '{to}'.",
            )
