class Training:
    """Container for the network training parameters.

    Attributes:
        optimizer: [`Optimizer`][pinnjax.Optimizer] object.
        epochs: Maximum number of training iterations.

        train_mode: Either 'class' or 'type'. Determines if the losses are split
            by type [PDE, BC, IC] or specifically by class.
        loss_scale: Multiplies the loss -- useful for avoiding overflow errors.
        priority: Sets the priority of each of the losses.
            Must be a tuple of equal length to the number of losses.
        normalize_weights: Normalize the effective loss weights to have a mean of one.
            This preserves their relative weighting without changing
            the overall loss scale.

        use_adaptive_weights: Adaptively update the weights at
            given intervals using a gradient-based approach
        adaptive_alpha: Relative importance of the new weights to existing.
        update_weights_every: How often (epochs) to update the weights.
        update_weights_after: After how many epochs should the weights
            start to be updated.

        balance_magnitudes: Different weight categories can be at wildly different
            magnitudes, resulting in certain losses being completely neglected.
            This boolean aims to balance the magnitudes once at the start.
        magnitude_clip: Since, completely balancing the magnitudes would remove any
            relative importance between losses.
            The clip is the maximum amount a loss can be reduced by.

        early_stopping: Often, given particular parameter values,
            training can continue long after convergence, or into overfitting.
            Early stopping, therefore, terminates the training given a satisfied
            `rel_tolerance` for a given number of epochs (`patience`).
        patience: How many epochs without a satisfactory improvement before
            premature termination.
        rel_tolerance: Relative Percent improvement (as a decimal) required
            between each epoch.

        batch_size: If set, each epoch runs multiple mini-batch gradient steps.
            If ``None``, one full-batch step per epoch (default).
        shuffle_batches: When batching, whether to shuffle indices each epoch.
        drop_last_batch: When batching, whether to drop the last batch if it would
            be smaller than ``batch_size`` (remainder). Incompatible with
            quasi-Newton optimizers (e.g. L-BFGS); use ``batch_size=None``.

        microbatch_size: Rows per chunk for a large MSE or MAE term.
            ``None`` evaluates that term in one pass. Evaluation residuals
            stay in one pass. This does not add optimizer steps.
            Default: ``None``.
        gradient_checkpoint: Recompute each residual forward during the
            backward pass. Uses less memory and takes longer. The weights
            are the same either way. Default: ``False``.

        selection_criterion: Either ``"loss"`` or ``"evaluate"``.
            ``"loss"`` keeps best model and early stopping on training loss.
            ``"evaluate"`` uses
            [`evaluate_selection_metric`][pinnjax.loss.evaluation.Evaluation.evaluate_selection_metric]
            each epoch (requires ``evaluation``).

    """

    optimizer: Optimizer
    epochs: int

    train_mode: str = "class"

    selection_criterion: Literal["loss", "evaluate"] = "loss"

    loss_scale: float = 1.0
    priority: tuple[int | float, ...] | None = None
    normalize_weights: bool = True

    # Adaptive Weights
    use_adaptive_weights: bool = False
    adaptive_alpha: float = 0.9
    update_weights_every: int = 100
    update_weights_after: int = 100

    # Balance Magnitudes
    balance_magnitudes: bool = False
    magnitude_clip: float = 1e-2

    # Early Stopping
    early_stopping: bool = False
    patience: int = 50
    rel_tolerance: float = 0.005

    # Batching
    batch_size: int | None = None
    shuffle_batches: bool = True
    drop_last_batch: bool = False

    # Within-term chunk size for large MSE/MAE terms (``jax.lax.scan``; None = full batch).
    microbatch_size: int | None = None
    gradient_checkpoint: bool = False

    def to_string(self, *, indent: int = 0) -> str:
        """Return a string of the [`Training`][pinnjax.Training] object.

        Args:
            indent: Indentation level used for the formatted output.
                Each level corresponds to two spaces.
                Default: 0.

        Examples:
            >>> import pinnjax

class EvaluationResults:
    """Container for evaluation metrics comparing model predictions to ground truth.

    This class can store any evaluation metrics returned by an Evaluation object,
    making it generic for all evaluation types.

    Main point of this dataclass is for convenient and predicable output in save_configuration

    Attributes:
        metrics: Dictionary of metric names to their float values.
            Keys are metric names (e.g., "RMSE", "Relative L2", "R2"),
            values are the corresponding metric values as floats.

    """

    metrics: dict[str, float] = field(factory=dict)

    @classmethod
    def from_dict(cls, eval_dict: dict[str, Array]) -> EvaluationResults:
        """Create EvaluationResults from evaluation.evaluate() output.

        Args:
            eval_dict: Dictionary returned by evaluation.evaluate() with keys
                as metric names and Array values (typically 0-dimensional scalars).

        Returns:
            EvaluationResults instance with float values extracted from Arrays.

        """
        converted_metrics: dict[str, float] = {}
        for key, value in eval_dict.items():
            arr = jnp.asarray(value)

            if arr.ndim != 0:
                msg = f"Evaluation metric '{key}' must be scalar, got shape {arr.shape}."
                raise ValueError(msg)

            converted_metrics[key] = float(arr.item())
        return cls(metrics=converted_metrics)

    def to_string(self, *, indent: int = 0) -> str:
        """Return a string representation of the EvaluationResults object.

        Args:
            indent: Indentation level used for the formatted output.
                Each level corresponds to two spaces.
                Default: 0.

        Examples:
            >>> import pinnjax
            >>> import jax.numpy as jnp
            >>> # From any evaluation.evaluate() dict result
            >>> eval_dict = {
            ...     "RMSE": jnp.array(0.1),
            ...     "Relative L2": jnp.array(0.05),
            ...     "R2": jnp.array(0.95),
            ...     "Custom Metric": jnp.array(0.42)
            ... }
            >>> results = pinnjax.EvaluationResults.from_dict(eval_dict)
            >>> print(results)
            EvaluationResults:
              RMSE: 0.1
              Relative L2: 0.05
              R2: 0.95
              Custom Metric: 0.42

        """
        # custom printer because of an unwrapped dictionary.

        pad = "  " * indent
        subpad = "  " * (indent + 1)

        lines = [f"{pad}{self.__class__.__name__}:"]

        for name, value in self.metrics.items():
            lines.append(f"{subpad}{name}: {value:.6g}")

        return "\n".join(lines)

    def __str__(self) -> str:  # noqa: D105
        return self.to_string(indent=0)

def from_dict(cls, eval_dict: dict[str, Array]) -> EvaluationResults:
    """Create EvaluationResults from evaluation.evaluate() output.

    Args:
        eval_dict: Dictionary returned by evaluation.evaluate() with keys
            as metric names and Array values (typically 0-dimensional scalars).

    Returns:
        EvaluationResults instance with float values extracted from Arrays.

    """
    converted_metrics: dict[str, float] = {}
    for key, value in eval_dict.items():
        arr = jnp.asarray(value)

        if arr.ndim != 0:
            msg = f"Evaluation metric '{key}' must be scalar, got shape {arr.shape}."
            raise ValueError(msg)

        converted_metrics[key] = float(arr.item())
    return cls(metrics=converted_metrics)

    def to_string(self, *, indent: int = 0) -> str:
    """Return a string representation of the EvaluationResults object.

    Args:
        indent: Indentation level used for the formatted output.
            Each level corresponds to two spaces.
            Default: 0.

    Examples:
        >>> import pinnjax
        >>> import jax.numpy as jnp
        >>> # From any evaluation.evaluate() dict result
        >>> eval_dict = {
        ...     "RMSE": jnp.array(0.1),
        ...     "Relative L2": jnp.array(0.05),
        ...     "R2": jnp.array(0.95),
        ...     "Custom Metric": jnp.array(0.42)
        ... }
        >>> results = pinnjax.EvaluationResults.from_dict(eval_dict)
        >>> print(results)
        EvaluationResults:
          RMSE: 0.1
          Relative L2: 0.05
          R2: 0.95
          Custom Metric: 0.42

    """
    # custom printer because of an unwrapped dictionary.

    pad = "  " * indent
    subpad = "  " * (indent + 1)

    lines = [f"{pad}{self.__class__.__name__}:"]

    for name, value in self.metrics.items():
        lines.append(f"{subpad}{name}: {value:.6g}")

    return "\n".join(lines)
