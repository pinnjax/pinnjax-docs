    """Container for ```jax.nn``` activation functions.

    Attributes:

        kwargs: Relevant kwargs of given activation function. Provided as a ``Dict``.

    Notes:
        See [JAX documentation](https://docs.jax.dev/en/latest/jax.nn.html) for
            available activation functions, arguments (args)
            and keyword arguments (kwargs).

    """

        """Return a string of the [`Activation`][pinnjax.Activation] object.

                Default: 0.

            >>> import pinnjax
            >>> activation = pinnjax.Activation(jax.nn.gelu,
            ...                                 kwargs={"approximate": False})
            >>> print(activation)
            Activation:
              act_: <function gelu>
              args: ()
              kwargs: {'approximate': False}

        """
        return _print_attrs_class(self, indent=indent)

    """Container for ```jax.nn.initializer``` initialization functions.

    Attributes:
        weights: Callable ````jax.nn.initializer``` function which determines
            the initial weights.
        bias: Callable ````jax.nn.initializer``` function which determines
            the initial biases.

    Notes:
        See [JAX documentation](https://docs.jax.dev/en/latest/jax.nn.initializers.html)
            for available initialization functions for weights and bias.

    """

    weights: jinit.Initializer | None = None
    bias: jinit.Initializer | None = None

    def to_string(self, *, indent: int = 0) -> str:
        """Return a string of the [`Initialization`][pinnjax.Initialization] object.

        Args:
            indent: Indentation level used for the formatted output.

                Default: 0.

            >>> import pinnjax
            >>> weights = jinit.variance_scaling(scale=1.48**2, mode="fan_in",
            ...                                  distribution="truncated_normal")
            >>> bias = jinit.zeros
            >>> initialization = pinnjax.Initialization(weights=weights, bias=bias)
            >>> print(initialization)
            Initialization:
              weights: <function init>
              bias: <function zeros>

        """

    def __str__(self) -> str:  # noqa: D105

class Optimization:
    """Container for Input/Output Transformations and Hard Constraints used in creation of [`NeuralNetwork`][pinnjax.NeuralNetwork] objects.

    Attributes:
        input_transformation: Optional
            [`InputTransformation`][pinnjax.InputTransformation] object.
        output_transformation: Optional
            [`OutputTransformation`][pinnjax.OutputTransformation] object.
        hard_constraint: Optional
            [`HardConstraint`][pinnjax.HardConstraint] object or ``tuple`` of objects.
            A tuple of compatible
            [`HardValueConstraint`][pinnjax.HardValueConstraint] objects is combined
            when a network is built, at the position of the first of them. Other
            hard constraints stay in their written order. The tuple stored here
            is left unchanged.

    """

    input_transformation: InputTransformation | None = None
    output_transformation: OutputTransformation | None = None
    hard_constraint: HardConstraint | tuple[HardConstraint, ...] | None = None

    def to_string(self, *, indent: int = 0) -> str:
        """Return a string of the [`Optimization`][pinnjax.Optimization] object.

        Args:
            indent: Indentation level used for the formatted output.
                Each level corresponds to two spaces.
                Default: 0.

        Examples:
            >>> import pinnjax
            >>> optimization = pinnjax.Optimization(
            ...     pinnjax.IdentityInputTransform(),
            ...     pinnjax.IdentityOutputTransformation(),
            ...     pinnjax.IdentityHardConstraint()
            ...    )
            >>> # this is the equivalent of ``optimization = pinnjax.Optimization()``
                 once applied to a NeuralNetwork object.
            >>> print(optimization)
            Optimization:
              input_transformation: <function IdentityInputTransform>
              output_transformation: <function IdentityOutputTransformation>
              hard_constraint: <function IdentityHardConstraint>

        """
        return _print_attrs_class(self, indent=indent)

    def __str__(self) -> str:  # noqa: D105
        return self.to_string(indent=0)
