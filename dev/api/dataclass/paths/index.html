class LinearPathConfig(PathConfig):
    """Configuration of LINEAR paths to be used in [`Path`][pinnjax.Path].

    Attributes:
        speed: Speed [m/s] of the moving object following the path,
            i.e., heat-source [m].
        direction: Direction of the moving object as a vector-tuple.
        initial: Initial location of a moving object at t=0 [m].
        active: Whether the [`Heat`][pinnjax.Heat] source is active.
            Exists for compatibility with
            [`ComplexLinearPathConfig`][pinnjax.ComplexLinearPathConfig].

    Raises:
        InputError: If the `direction` and `initial` don't have the
            same number of dimensions.
        InputError: If the number of dimensions is less than 2 or greater than 3.

    """

    speed: float | int
    direction: tuple[float | int, float | int] | tuple[float | int, float | int, float | int]
    initial: tuple[float | int, float | int] | tuple[float | int, float | int, float | int]

    active: bool | None = True

    UNITS: ClassVar[Mapping[str, str]] = {"speed": "m/s", "initial": "m"}

    def __attrs_post_init__(self) -> None:  # noqa: D105
        n_dir, n_init = len(self.direction), len(self.initial)
        if n_dir != n_init:
            msg = "direction and initial must have same dimension."
            config.LOGGER.error(msg)
            raise InputError(msg)
        if n_dir not in (2, 3):
            msg = f"dimension must be 2 or 3, got {n_dir}"
            config.LOGGER.error(msg)
            raise InputError(msg)

    def to_string(self, *, indent: int = 0) -> str:
        """Return a string of the [`LinearPathConfig`][pinnjax.LinearPathConfig] object.

        Args:
            indent: Indentation level used for the formatted output.
                Each level corresponds to two spaces.
                Default: 0.

        Examples:
            >>> path_config = LinearPathConfig(
            ...     speed=0.01, direction=(1, 0, 0), initial=(0, 0.003, 0.003)
            ... )
            >>> print(path_config)
            LinearPathConfig:
              speed: 0.01 m/s
              direction: (1, 0, 0)
              initial: (0, 0.003, 0.003) m
              active: True

        """
        return _print_attrs_class(self, indent=indent)

    def __str__(self) -> str:  # noqa: D105
        return self.to_string(indent=0)

    @frozen
class ComplexLinearPathConfig(PathConfig):
    """Configuration of COMPLEX LINEAR paths to be used in [`Path`][pinnjax.Path].

    Attributes:
        waypoints: Sequence of spatial or spatial-temporal coordinates.
            - (x, y)
            - (x, y, z) or (x, y, t)
            - (x, y, z, t)
        _speed: Optional scalar or per-segment [m/s]. [alias : speed]
            - Must be ``None`` if `is_temporal=True`.
                Speed is then computed internally.
        is_temporal: Boolean flag indicating whether the last coordinate dimension
            represents time.
        initials: Array of initial points for each path segment -
            Computed internally.
        directions: Array of direction vectors for each path segment -
            Computed internally.
        speeds: Array of per-segment speeds [m/s] -
            Computed internally.
        segment_times: Array of per-segment durations [s] -
            Computed internally.
        cumulative_times: Array of cumulative durations at each waypoint [s] -
            Computed internally.

    Raises:
        InputError: If fewer than two waypoints are provided.
        InputError: If waypoint dimensionality is inconsistent or unsupported.
        InputError: If waypoints contain NaN, Inf, or zero-length segments.
        InputError: If temporal waypoints are not strictly increasing in time.
        InputError: If `speed` is invalid (negative, non-finite, or incorrectly sized).

    Notes:
        - When `is_temporal=True`, time is interpreted as the last column of
            each waypoint.
        - Speeds for temporal paths are derived automatically from distances
            and time deltas.
        - For non-temporal paths, a positive speed (scalar or per-segment)
            must be explicitly provided.

    """

    waypoints: (
        list[tuple[float, float] | tuple[float, float, float] | tuple[float, float, float, float]]
        | Array
    )

    _speed: float | tuple[float, ...] | list[float] | None = field(
        init=True,
        repr=False,
        eq=False,
        default=None,
        alias="speed",
    )

    is_temporal: bool = field(init=True, repr=False, default=False)

    UNITS: ClassVar[Mapping[str, str]] = {
        "waypoints": "m",
        "initials": "m",
        "speeds": "m/s",
        "cumulative_times": "s",
        "segment_times": "s",
    }

    initials: Array = field(init=False)
    directions: Array = field(init=False)
    speeds: Array = field(init=False)
    segment_times: Array = field(init=False)
    cumulative_times: Array = field(init=False)
    active: bool | int | float | tuple | list | dict[str, list | int] | None = field(
        init=True,
        eq=False,
        default=True,
    )

    def __attrs_post_init__(self) -> None:  # noqa: D105
        waypoint_array = jnp.asarray(self.waypoints)

        # --------------------------- TESTING  -----------------------------------------
        required_dimensions = 2
        if waypoint_array.ndim != required_dimensions:
            msg = "waypoints must be a 2D array-like: shape (N, D)."
            config.LOGGER.error(msg)
            raise InputError(msg)
        number, dims = waypoint_array.shape
        minimum_waypoints = 2
        if number < minimum_waypoints:
            msg = "At least two waypoints are required."
            config.LOGGER.error(msg)
            raise InputError(msg)

        if not jnp.isfinite(waypoint_array).all():
            msg = "waypoints contain NaN or Inf."
            config.LOGGER.error(msg)
            raise InputError(msg)

        if self.is_temporal:
            if dims not in (3, 4):
                msg = "Temporal path requires (x,y,t) or (x,y,z,t): D must be 3 or 4."
                config.LOGGER.error(msg)
                raise InputError(msg)
            # the last column is time
            t = waypoint_array[:, -1]
            if not jnp.isfinite(t).all():
                msg = "Time column contains NaN or Inf."
                config.LOGGER.error(msg)
                raise InputError(msg)
            dt = jnp.diff(t)
            if not (dt > 0).all():
                msg = "Time must be strictly increasing in the last column."
                config.LOGGER.error(msg)
                raise InputError(msg)
            if self._speed is not None:
                msg = (
                    "For temporal paths, 'speed' must be None. Speeds are determined automatically."
                )
                config.LOGGER.error(msg)
                raise InputError(msg)
        else:
            if self._speed is None:
                msg = "If there is no time-component a speed or speeds must be provided."
                config.LOGGER.error(msg)
                raise InputError(msg)
            if dims not in (2, 3):
                msg = (
                    "Non-temporal path supports only 2D or 3D spatial waypoints. "
                    "You might have meant to set is_temporal = True."
                )
                config.LOGGER.error(msg)
                raise InputError(msg)

        # ---------- Speed checks ----------
        if self._speed is not None:
            if isinstance(self._speed, (tuple, list)):
                # Per-segment speeds: length must be N-1
                if len(self._speed) not in (1, number - 1):
                    msg = (
                        "speed tuple length must be 1 (broadcast) "
                        "or N-1={number - 1} (per-segment)."
                    )
                    config.LOGGER.error(msg)
                    raise InputError(msg)
                if not jnp.isfinite(jnp.asarray(self._speed)).all():
                    msg = "speed contains NaN or Inf."
                    config.LOGGER.error(msg)
                    raise InputError(msg)
                if (jnp.asarray(self._speed) <= 0).any():
                    msg = "speed values must be positive."
                    config.LOGGER.error(msg)
                    raise InputError(msg)
            else:
                # Scalar speed
                if not jnp.isfinite(jnp.asarray(self._speed)):
                    msg = "speed contains NaN or Inf."
                    config.LOGGER.error(msg)
                    raise InputError(msg)
                if float(self._speed) <= 0:
                    msg = "speed values must be positive."
                    config.LOGGER.error(msg)
                    raise InputError(msg)

        # Ensure all rows have the same length
        if isinstance(self.waypoints, list):
            first_len = len(self.waypoints[0])
            if any(len(t) != first_len for t in self.waypoints):
                msg = "All waypoint tuples must have the same length."
                config.LOGGER.error(msg)
                raise InputError(msg)
            # When temporal, all tuples must be 3 or 4; otherwise 2 or 3.
            if self.is_temporal and first_len not in (3, 4):
                msg = "With is_temporal=True, tuples must be length 3 or 4."
                config.LOGGER.error(msg)
                raise InputError(msg)
            if not self.is_temporal and first_len not in (2, 3):
                msg = "With is_temporal=False, tuples must be length 2 or 3."
                config.LOGGER.error(msg)
                raise InputError(msg)

        # ------------------------------------------------------------------------------
        spatial_dimensions = dims - (1 if self.is_temporal else 0)
        spatial_coordinates = waypoint_array[:, :spatial_dimensions]

        distances = jnp.linalg.norm(jnp.diff(spatial_coordinates, axis=0), axis=1)
        number_segments = len(distances)
        if (distances == 0).any():
            msg = "Consecutive waypoints must not be identical (zero-length segment)."
            config.LOGGER.error(msg)
            raise InputError(msg)

        object.__setattr__(self, "initials", spatial_coordinates[:-1])
        object.__setattr__(
            self,
            "directions",
            spatial_coordinates[1:] - spatial_coordinates[:-1],
        )

        if self.is_temporal:
            time_deltas = jnp.diff(waypoint_array[:, -1])  # Δt > 0 already checked
            speeds = distances / time_deltas  # per-segment speeds
            if not jnp.isfinite(speeds).all() or (speeds <= 0).any():
                msg = "Derived speeds must be positive and finite."
                config.LOGGER.error(msg)
                raise InputError(msg)
            cumulative_times = waypoint_array[:, -1]
        else:
            assert self._speed is not None
            speeds = (
                jnp.asarray(self._speed)
                if isinstance(self._speed, (tuple, list))
                else jnp.full((len(distances),), self._speed)
            )
            durations = distances / speeds
            cumulative_times = jnp.concatenate(
                [jnp.array([0.0]), jnp.cumsum(durations)],
            )

        segment_times = jnp.diff(cumulative_times)
        object.__setattr__(self, "cumulative_times", cumulative_times)
        object.__setattr__(self, "segment_times", segment_times)
        object.__setattr__(self, "speeds", tuple(map(float, speeds.tolist())))

        if isinstance(self.active, (bool, int, float)):
            activity = [bool(self.active)] * number_segments

        elif isinstance(self.active, (tuple, list)):
            if len(self.active) != number_segments:
                msg = "Active list Length MUST match number of segments."
                config.LOGGER.error(msg)
                raise InputError(msg)
            activity = [bool(i) for i in self.active]

        elif isinstance(self.active, dict):
            if set(self.active.keys()) != {"off"}:
                msg = 'Active Dictionary must only contain the key "off".'
                config.LOGGER.error(msg)
                raise InputError(msg)
            activity = [True] * number_segments

            off_segments = self.active["off"]
            if isinstance(off_segments, int):
                off_segments = [off_segments]
            elif isinstance(off_segments, tuple):
                off_segments = list(off_segments)

            for k in off_segments:
                if not (1 <= k <= number_segments):
                    msg = f"Invalid segment index {k} in active['off']. Segments are 1-indexed."
                    config.LOGGER.error(msg)
                    raise InputError(msg)
                activity[k - 1] = False

        else:
            msg = (
                "active must be boolean, scalar, list/tuple of booleans, "
                "or dict {'off': [indices]}."
            )
            config.LOGGER.error(msg)
            raise InputError(msg)

        object.__setattr__(self, "active", jnp.asarray(activity, dtype=bool))
