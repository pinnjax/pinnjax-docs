class ElasticMaterial(Material):
    """Material with isotropic linear-elastic properties."""

    youngs_modulus: float
    poissons_ratio: float
    lam: float = field(init=False)
    lam_2d: float = field(init=False)
    mu: float = field(init=False)

    lam_hat: float = field(init=False)
    lam_hat_2d: float = field(init=False)
    mu_hat: float = field(init=False)

    UNITS: ClassVar[Mapping[str, str]] = {
        "youngs_modulus": "Pa",
        "poissons_ratio": "-",
        "lam": "Pa",
        "lam_2d": "Pa",
        "mu": "Pa",
        "lam_hat": "-",
        "lam_hat_2d": "-",
        "mu_hat": "-",
    }

    def __attrs_post_init__(self) -> None:  # noqa: D105
        e, nu = self.youngs_modulus, self.poissons_ratio

        lam = (e * nu) / ((1.0 + nu) * (1.0 - 2.0 * nu))
        lam_2d = (e * nu) / (1.0 - nu**2)
        mu = e / (2.0 * (1.0 + nu))

        object.__setattr__(self, "lam", lam)
        object.__setattr__(self, "lam_2d", lam_2d)
        object.__setattr__(self, "mu", mu)

        object.__setattr__(self, "lam_hat", lam / e)
        object.__setattr__(self, "lam_hat_2d", lam_2d / e)
        object.__setattr__(self, "mu_hat", mu / e)

    def to_string(self, *, indent: int = 0) -> str:
        """Return a string representation of the [`Material`][pinnjax.Material] object.

        Args:
            indent: Indentation level used for the formatted output.
                Each level corresponds to two spaces.
                Default: 0.

        Examples:
            >>> import pinnjax
            >>> material = pinnjax.ElasticMaterial(
            ...    youngs_modulus=210e9,
            ...    poissons_ratio=0.3,
            ...   )

              youngs_modulus: 2.1e+11 Pa
              poissons_ratio: 0.3 -
              lam: 1.21154e+11 Pa
              lam_2d: 6.92308e+10 Pa
              mu: 8.07692e+10 Pa
              lam_hat: 0.576923 -
              lam_hat_2d: 0.32967 -
              mu_hat: 0.384615 -

        """
        return _print_attrs_class(self, indent=indent)

    def __str__(self) -> str:  # noqa: D105
        return self.to_string(indent=0)

class Goldak(Heat):
    """Child class of [`Heat`][pinnjax.Heat] specifically for the Goldak heat source.

    Attributes:
        width: Width of the weld pool (perpendicular to welding path) [m].
            Value is halved during computation.
        front_ellipsoid_length: Front length component of weld pool [m].
        rear_ellipsoid_length: Rear length component of weld pool [m].
        depth: Depth of weld pool [m].
        front_ellipsoid_distribution_factor: Fraction of power is deposited
            in the front-half.
        rear_ellipsoid_distribution_factor: Fraction of power is deposited
            in the rear-half.

    See Also:
        [`Heat`][pinnjax.Heat]: Parent class which defines `power` and `efficiency`.

    Raises:
        InputError: If front and rear distribution factors != 2.
        InputError: If the combination of distribution factors and
            ellipsoid lengths results in a discontinuous heat flux at the interface.

    """

    width: float | int  # m
    front_ellipsoid_length: float | int  # m
    rear_ellipsoid_length: float | int  # m
    depth: float | int  # m
    front_ellipsoid_distribution_factor: float | int  # -
    rear_ellipsoid_distribution_factor: float | int  # -

    UNITS: ClassVar[Mapping[str, str]] = {
        "power": "W",
        "efficiency": "-",
        "width": "m",
        "front_ellipsoid_length": "m",
        "rear_ellipsoid_length": "m",
        "depth": "m",
        "front_ellipsoid_distribution_factor": "-",
        "rear_ellipsoid_distribution_factor": "-",
    }

    def __attrs_post_init__(self) -> None:  # noqa: D105
        total_distribution_factor = 2.0
        if (
            self.front_ellipsoid_distribution_factor + self.rear_ellipsoid_distribution_factor
            != total_distribution_factor
        ):
            msg = "Distribution Factors MUST sum to 2.0."
            config.LOGGER.error(msg)
            raise InputError(msg)

        tolerance = 1e-1  # 0.1% Tolerance
        if (
            abs(
                self.front_ellipsoid_distribution_factor / self.front_ellipsoid_length
                - self.rear_ellipsoid_distribution_factor / self.rear_ellipsoid_length,
            )
            / (
                (
                    self.front_ellipsoid_distribution_factor / self.front_ellipsoid_length
                    + self.rear_ellipsoid_distribution_factor / self.rear_ellipsoid_length
                )
                / 2
            )
            * 100
            > tolerance
        ):
            msg = "Modify Distribution Factors or Lengths. Heat Flux won't be CONTINUOUS."
            config.LOGGER.error(msg)
            raise InputError(msg)

    def to_string(self, *, indent: int = 0) -> str:
        """Return a string of the [`Goldak`][pinnjax.Goldak] object.

        Args:
            indent: Indentation level used for the formatted output.
                Each level corresponds to two spaces.
                Default: 0.

        Examples:
            >>> import pinnjax
            >>> goldak = pinnjax.Goldak(
            ...    power=2.45e3,
            ...    efficiency=0.9,
            ...    width=0.005,
            ...    front_ellipsoid_length=0.00257,
            ...    rear_ellipsoid_length=0.006,
            ...    depth=0.004,
            ...    front_ellipsoid_distribution_factor=0.6,
            ...    rear_ellipsoid_distribution_factor=1.4,
            ...   )
            >>> print(goldak)
            Goldak:
              power: 2450 W
              efficiency: 0.9 -
              width: 0.005 m
              front_ellipsoid_length: 0.00257 m
              rear_ellipsoid_length: 0.006 m
              depth: 0.004 m
              front_ellipsoid_distribution_factor: 0.6 -
              rear_ellipsoid_distribution_factor: 1.4 -

        """
