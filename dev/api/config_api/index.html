    def relocate_log(
    relocation_folder_location: Path | str,
    logger_name: str = "pinnjax",
) -> None:
    """Relocate an existing log file to a new folder, keeping the original filename.

    Args:
        relocation_folder_location: Folder where the log file should be moved
        logger_name: Name of the logger to relocate

    """
    _logger = logging.getLogger(logger_name)

    # Find the current file handler and path
    file_handler = None
    old_log_path = None

    for handler in _logger.handlers:
        if isinstance(handler, logging.FileHandler):
            file_handler = handler
            old_log_path = Path(handler.baseFilename)
            break

    if not file_handler or not old_log_path:
        _logger.warning("No file handler found to relocate")
        return

    # Extract just the filename from the old path
    original_filename = old_log_path.name

    # Create paths - keep folder and file separate!
    target_folder = Path(relocation_folder_location)
    target_file_path = target_folder / original_filename

    if old_log_path == target_file_path:
        _logger.info("Log already at target location")
        return

    # Close current file handler
    file_handler.close()
    _logger.removeHandler(file_handler)

    # Create the FOLDER (not the file!)
    target_folder.mkdir(parents=True, exist_ok=True)

    # Handle case where file path exists as directory (cleanup from previous bug)
    if target_file_path.exists() and target_file_path.is_dir():
        shutil.rmtree(target_file_path)  # Remove the incorrect directory

    # Move the file
    if old_log_path.exists():
        shutil.move(str(old_log_path), str(target_file_path))

    # Create new file handler pointing to the FILE path
    new_file_handler = logging.FileHandler(
        target_file_path,
        mode="a",
        encoding="utf-8",
    )
    new_file_handler.setFormatter(file_handler.formatter)
    _logger.addHandler(new_file_handler)

    _logger.info(
        "Log relocated from %s to %s",
        old_log_path,
        target_file_path,
    )

    def set_dtype(dtype: jnp.dtype) -> None:
    try:
        dtype = jnp.dtype(dtype)
    except TypeError:
        msg = f"dtype must be convertible to jnp.dtype. Got {type(dtype).__name__}"
        LOGGER.error(msg)
        raise TypeError(msg)
    if dtype == jnp.dtype(jnp.float64):
        _CONFIG.JAX_ENABLE_X64 = True
        jax.config.update("jax_enable_x64", _CONFIG.JAX_ENABLE_X64)

    _CONFIG.DTYPE = dtype
    _CONFIG.EPS = jnp.asarray(_CONFIG.EPS, dtype=dtype)
    _CONFIG.BIG = jnp.asarray(_CONFIG.BIG, dtype=dtype)

    def setup_clean_errors() -> None:
    """Replace Python's excepthook to print InputError messages cleanly.

    Sets `sys.excepthook` so that `InputError` exceptions are printed
    in a user-friendly format instead of a full traceback.
    """

    def custom_excepthook(
        exc_type: type[BaseException],
        exc_value: BaseException,
        exc_traceback: TracebackType | None,
    ) -> None:
        if exc_type is InputError:
            # Use logger or print replacement here if needed
            LOGGER.info("%s: %s", exc_type.__name__, exc_value)
        else:
            sys.__excepthook__(exc_type, exc_value, exc_traceback)

    sys.excepthook = custom_excepthook
