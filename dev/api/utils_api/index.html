    def create_folder(base_path: str | Path, name: str) -> Path:
    """Create a folder inside a base path.

    Args:
        base_path (str | Path): Base directory in which to create the folder.
        name (str): Name of the folder to create.

    Returns:
        Path: The full path of the created folder.

    """
    base = Path(base_path)
    target = base / name
    target.mkdir(parents=True, exist_ok=True)
    config.LOGGER.info("Folder Created at: %s", target)
    return target

    def create_path_folder(
    starting_path: str | Path | None = None,
    *folders: str,
) -> Path:
    """Create a nested folder structure.

    Starting from a base path, creates a folder named ``models`` and then
    creates each folder provided in ``folders`` inside it, in order.

    Args:
        starting_path (str | Path): Root directory to start from.
        *folders (str): Additional nested folders to create.

    Returns:
        Path: Path to the deepest created folder.

    """
    # create folders if not already created.
    if starting_path is None:
        starting_path = Path.cwd()

    folder_location = create_folder(starting_path, "models")

    for folder in folders:
        folder_location = create_folder(folder_location, folder)

    return folder_location

    def create_model_path(starting_path: str | Path = Path.cwd(), *folders: str) -> Path:
    """Create a new model directory with an incremented model index.

    Creates the folder structure using :func:`create_path_folder`, then scans
    the resulting directory for existing ``Model-#<n>`` folders and creates a
    new one with the next available index.

    Args:
        starting_path (Path): Directory to start building the model path under.
        *folders (str): Additional folders that determine the experiment path.

    Returns:
        Path: The newly created model directory.

    """
    # extension of create_path() to find if a combination of parameter folders
    # has already been evaluated, and makes new folder.

    path = create_path_folder(starting_path, *folders)

    # folder system is such that the additional test runs are called 'Model #1', etc...
    models_list = [p.name.split("#")[1] for p in Path(path).iterdir() if "#" in p.name]

    model_number = max([int(x) for x in models_list], default=0) + 1
    name_of_folder: str = f"Model-#{model_number}"

    return create_folder(path, name_of_folder)

    def duration_converter_str(duration: float) -> str:
    """Convert an int/float seconds into a string "Y:D:H:M:S" (with labels).

    Args:
        duration: given time (in seconds)

    Returns:
        Printable string equivalent to the given 'duration' extrapolated to
            years, months, days, hours, minutes and seconds.

    Examples:
        >>> x = duration_converter_str(100000000)
        >>> print(x)
        3y: 62d: 9h: 46m: 40s

    """
    times = []
    units = []

    seconds_in_minute = 60
    minutes_in_hour = 60
    hours_in_day = 24
    days_in_year = 365

    if duration >= seconds_in_minute:
        seconds = round(duration % seconds_in_minute, 3)  # seconds rounded to 3dp
        times.append(seconds)
        units.append("s")

        duration = int(duration // minutes_in_hour)  # cast to int
        if duration >= minutes_in_hour:
            minutes = int(duration % minutes_in_hour)  # int minutes
            times.append(minutes)
            units.append("m")
            duration = int(duration // minutes_in_hour)  # cast to int

            if duration >= hours_in_day:
                hours = int(duration % hours_in_day)  # int hours
                times.append(hours)
                units.append("h")
                duration = int(duration // hours_in_day)  # cast to int

                if duration >= days_in_year:
                    days = int(duration % days_in_year)  # int days
                    times.append(days)
                    units.append("d")
                    duration = int(duration // days_in_year)  # cast to int

                    if duration >= 1:
                        years = int(duration)  # int years
                        times.append(years)
                        units.append("y")
                else:
                    days = int(duration)  # int days
                    times.append(days)
                    units.append("d")
            else:
                hours = int(duration)  # int hours
                times.append(hours)
                units.append("h")
        else:
            minutes = int(duration)  # int minutes
            times.append(minutes)
            units.append("m")
    else:
        times.append(round(duration, 3))  # seconds only, 3dp
        units.append("s")

    times.reverse()
    units.reverse()
    return ": ".join(f"{t}{u}" for t, u in zip(times, units, strict=True))

    def force_compile(shape: pinnjax.ComplexShape, model: pinnjax.NeuralNetwork) -> None:
    """Warm up JIT compilation of `residual_and_losses`.

    This also replaces it with a compiled version. It triggers JAX JIT compilation
    for the [`residual_and_losses`][pinnjax.ComplexShape.residual_and_losses] method
    of a `ComplexShape` object using a single call. After this, the method
    is replaced with its compiled version to avoid compilation overhead
    during training.

    Args:
        shape: [`ComplexShape`][pinnjax.ComplexShape] object containing the method.
        model: [`NeuralNetwork`][pinnjax.NeuralNetwork] object used as input.

    Raises:
        TypeError: If `residual_and_losses` is not defined or not callable
            in the `shape` object.

    """
    ops = pinnjax.DifferentialOperators(model)

    if not (
        hasattr(shape, "residual_and_losses") and callable(shape.residual_and_losses)
    ):
        msg = (
            f"{type(shape).__name__} must define a callable "
            f"'residual_and_losses' method."
        )
        raise TypeError(msg)

    # Compile the function with JAX JIT
    compiled_fn = jax.jit(shape.residual_and_losses)

    # Call once to trigger compilation
    output = compiled_fn(model, ops)

    # Ensure all JAX arrays in the output are ready
    def _block_all(o: Array | tuple[Array, Array, Array]) -> None:
        if isinstance(o, jnp.ndarray):
            o.block_until_ready()
        elif isinstance(o, (tuple, list)):
            for v in o:
                _block_all(v)
        elif isinstance(o, dict):
            for v in o.values():
                _block_all(v)

    _block_all(output)

    # Replace the original method with the compiled version
    shape.residual_and_losses = compiled_fn

    def index_shuffle(
    source: int | MutableSequence[int],
    *,
    portion: float = 1.0,
    seed: int | None = None,
) -> list[int]:
    if seed is None:
        seed = config.get_seed()

    indices = list(range(source)) if isinstance(source, int) else source

    num_data = len(indices)
    if portion != 1.0:
        if 0 < portion <= 1 and isinstance(portion, float):
            num_data = max(1, int(len(indices) * portion))
        else:
            msg = (
                "`portion` must be an float between 0 (exclusive) and 1. "
                f"Got {portion} ({type(portion).__name__})."
            )
            config.LOGGER.error(msg)
            raise ValueError(msg)

    random.seed(seed)
    random.shuffle(indices)
    return indices[:num_data]

    def is_power_of_2(x: int) -> bool:
    """Return if the input is power of 2.

    Args:
        x: value to be checked

    Returns:
        True if `x` is power of 2.

    """
    return (x & (x - 1) == 0) and x > 0

    def next_power_of_2(n: int) -> int:
    """Return the next power of 2 greater than or equal to `n`.

    Args:
        n: value to be checked

    Returns:
        Next power of 2 greater than or equal to `n`.

    """
    return 1 << (n - 1).bit_length()
