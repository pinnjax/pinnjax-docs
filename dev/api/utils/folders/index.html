    def create_folder(base_path: str | Path, name: str) -> Path:
    """Create a folder inside a base path.

    Parent directories are created when missing. An existing folder is kept.
    The path is written to the log.

    Args:
        base_path: Directory in which to create the folder.
        name: Name of the folder to create.

    Returns:
        Path: The full path of the created folder.

    """
    base = Path(base_path)
    target = base / name
    target.mkdir(parents=True, exist_ok=True)
    config.LOGGER.info("Folder Created at: %s", target)
    return target

    def create_path_folder(
    starting_path: str | Path | None = None,
    initial_folder_name: str = "models",
    *folders: str,
) -> Path:
    """Create a nested folder structure.

    ``starting_path`` defaults to the current working directory. The first
    created folder is ``initial_folder_name``, then each entry of ``*folders``
    is created inside the previous one. Existing folders are kept.

    Args:
        starting_path: Root directory. ``None`` uses the current working
            directory. Default: ``None``.
        initial_folder_name: First folder created under ``starting_path``.
            Default: ``"models"``.
        *folders: Further folder names, created one inside the next.

    Returns:
        Path: Path to the deepest created folder.

    """
    # create folders if not already created.
    if starting_path is None:
        starting_path = Path.cwd()

    folder_location = create_folder(starting_path, initial_folder_name)

    for folder in folders:
        folder_location = create_folder(folder_location, folder)

    return folder_location

    def create_model_path(
    starting_path: str | Path | None = None,
    initial_folder_name: str = "models",
    counting_name: str = "Model",
    *folders: str,
) -> Path:
    # TODO: This function, and adjacent folders, need help.
    """Create a new model directory with an incremented model index.

    The parent folders come from ``create_path_folder``. Child names that
    contain ``#`` supply the existing indexes: the text after ``#`` is read
    as an integer. The new folder is ``{counting_name}-#{n}``, where ``n``
    is one more than the largest index, or 1 when none exist.

    Args:
        starting_path: Root directory. ``None`` uses the current working
            directory. Default: ``None``.
        initial_folder_name: First folder created under ``starting_path``.
            Default: ``"models"``.
        counting_name: Prefix of the numbered model folder.
            Default: ``"Model"``.
        *folders: Further folders that determine the experiment path.

    Returns:
        Path: The newly created model directory.

    """
    # extension of create_path() to find if a combination of parameter folders
    # has already been evaluated, and makes new folder.

    if starting_path is None:
        starting_path = Path.cwd()

    path = create_path_folder(starting_path, initial_folder_name, *folders)

    # folder system is such that the additional test runs are called 'Model #1', etc...
    models_list = [p.name.split("#")[1] for p in Path(path).iterdir() if "#" in p.name]

    model_number = max([int(x) for x in models_list], default=0) + 1
    name_of_folder: str = f"{counting_name}-#{model_number}"

    return create_folder(path, name_of_folder)
