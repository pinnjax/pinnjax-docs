    def save_model(
    model: pinnjax.NeuralNetwork,
    path: Path | str | None = None,
    *,
    return_path: bool = False,
) -> None | Path:
    out_dir = Path(path) if path else Path.cwd()
    out_dir.mkdir(parents=True, exist_ok=True)

    # 1) arrays-only checkpoint (authoritative)
    arrays, static = eqx.partition(model, eqx.is_array)
    eqx.tree_serialise_leaves(out_dir / "model.eqx", arrays)

    # 2) meta (plain primitives only; NO callables/objects)
    if not hasattr(model, "to_meta"):
        msg = "NeuralNetwork object needs a 'to_meta' function."
        config.LOGGER.error(msg)
        raise config.InputError(msg)
    meta = model.to_meta()
    # Store dtype and array structure for correct reload and validation
    leaves = list(jax.tree_util.tree_leaves(arrays))
    dtype_str = str(leaves[0].dtype) if leaves else str(config.get_dtype())
    array_shapes = [list(int(s) for s in arr.shape) for arr in leaves]
    meta = {
        **meta,
        "dtype": dtype_str,
        "array_shapes": array_shapes,
        "equinox_version": eqx.__version__,
        "jax_version": jax.__version__,
        "python_version": sys.version,
    }
    with Path.open(out_dir / "meta.json", "wb") as f:
        f.write(
            orjson.dumps(meta, option=orjson.OPT_INDENT_2 | orjson.OPT_SERIALIZE_NUMPY),
        )

    return out_dir if return_path else None

    def save_configuration(
    path: str | Path = Path.cwd(),
    *,
    overwrite: bool = True,
    **configs,
) -> str:
    # TODO(@mjryande): Allow for multiple Training loops.
    # TODO(@mjryande): Save more like Time and Device.
    parameter_folder = pinnjax.create_folder(
        Path(path),
        "Configurations and Parameters",
    )
    txt_path = Path(parameter_folder) / "_config.txt"
    json_path = Path(parameter_folder) / "_config.json"

    header = "Configurations and Parameters"
    divider_length = len(header) * 3
    header_footer_sep = "=" * divider_length
    section_sep = "-" * divider_length

    # JSON
    plain = {name: _to_plain(cfg) for name, cfg in configs.items()}
    # Add machine-readable optimization_spec for model reload (e.g. prediction.py).
    # Loaded models must use the same optimization (transforms, constraints) to match training.
    if "optimization" in configs:
        opt = configs["optimization"]
        if getattr(opt, "__class__", None) and opt.__class__.__name__ == "Optimization":
            try:
                plain["optimization_spec"] = _optimization_to_spec(opt)
            except Exception as e:
                config.LOGGER.warning(
                    "Could not build optimization_spec for reload; load will infer from plain. %s",
                    e,
                )
    with Path.open(json_path, "w") as jf:
        json.dump(plain, jf, indent=2)

    # TXT
    mode = (
        "w" if overwrite or not txt_path.exists() else "a"
    )  # TODO(@mjryande): confirm 'overwrite' works correctly.
    with Path.open(txt_path, mode) as f:
        if mode == "w":
            f.write(
                f"{header_footer_sep}\n{header:^{divider_length}}\n"
                f"{header_footer_sep}\n",
            )

        def get_package_version() -> str:
            try:
                return version("pinnjax")
            except PackageNotFoundError:
                pyproject_path = Path(__file__).parent.parent / "pyproject.toml"
                if pyproject_path.exists():
                    with Path.open(pyproject_path, "rb") as _f:
                        pyproject_data = tomllib.load(_f)
                    return pyproject_data["project"]["version"]
                return "unknown"

        _version = get_package_version()
        dt = str(datetime.now().date())
        str_version = f"Version: {_version}"
        f.write(f"{str_version:^{divider_length}}\n")
        f.write(f"{dt:^{divider_length}}\n")

        for name, cfg in configs.items():
            section = _format_header(name) or (
                cfg.__class__.__name__ if _is_attrs_instance(cfg) else "Item"
            )
            f.write(f"\n{section_sep}\n{section}:\n{section_sep}\n")

            if isinstance(cfg, ComplexShape):
                _fmt_geometry(
                    section,
                    cfg,
                    lambda k, v, ind, kw: f.write("\t" * ind + f"{k:<{kw}} = {v}\n"),
                    lambda k, ind: f.write("\t" * ind + f"{k}:\n"),
                    indent=1,
                    key_width=0,
                )
                f.write(f"{section_sep}\n")
                continue
            # --------------------------------------------------------------------------

            if _is_attrs_instance(cfg):
                cls = cfg.__class__
                units_map = getattr(cls, "UNITS", None)
                keys = [fld.name for fld in attrs.fields(cls)]
                key_width = max((len(_format_header(k)) for k in keys), default=0)

                def write_line(k, v, ind, kw):
                    if v != "":
                        f.write("\t" * ind + f"{k:<{kw}} = {v}\n")
                    else:
                        f.write("\t" * ind + f"{k:<{kw}}\n")

                def write_block(k, ind):
                    f.write("\t" * ind + f"{k}:\n")

                for field_name in keys:
                    units = (
                        units_map.get(field_name)
                        if isinstance(units_map, dict)
                        else None
                    )
                    value = getattr(cfg, field_name)
                    # Use _format_value so lists/tuples remain multi-line;
                    # units propagate to each [i] line
                    _format_value(
                        _format_header(field_name),
                        value,
                        write_line,
                        write_block,
                        indent=1,
                        key_width=key_width,
                        units=units,
                    )

                f.write(f"{section_sep}\n")
                continue

            if isinstance(cfg, dict):
                data = {_format_header(k): v for k, v in cfg.items()}
            else:
                data = {
                    _format_header(k): v
                    for k, v in (
                        vars(cfg) if hasattr(cfg, "__dict__") else {"Value": cfg}
                    ).items()
                }

            key_width = max((len(k) for k in data), default=0)
            for k, v in data.items():
                _write_txt_value(f, k, v, key_width, indent=1)

            f.write(f"{section_sep}\n")

        f.write(f"{header_footer_sep}\n")

    return str(txt_path)

    def load_model(
    path: Path | str | None = None,
    *,
    project_name: str | None = None,
    model_number: int | None = None,
    models_base: Path | None = None,
    initialization: pinnjax.Initialization | None = None,
    optimization: pinnjax.Optimization | None = None,
    ambient_temperature: float | None = None,
    temperature_range: float | None = None,
) -> pinnjax.NeuralNetwork:
    """
    Load a trained model from a saved directory.

    **Resolving the model directory**
    Provide either:
    - **path**: direct path to the model directory (e.g. ``"models/Comparison #2/Model-#19"``), or
    - **project_name** and **model_number** (and optionally **models_base**), which resolve to
      ``{models_base}/models/{project_name}/Model-#{model_number}``.

    **How the model is built**
    - If you pass both **initialization** and **optimization**: they are used as-is and weights
      are loaded from ``meta.json`` + ``model.eqx`` (low-level, full control).
    - Otherwise: the function loads ``Configurations and Parameters/_config.json`` (if present),
      rebuilds material and optimization from it (or uses defaults for older runs), uses a default
      initialization, then loads the weights. Optional **ambient_temperature** and
      **temperature_range** override the config when building material/optimization.

    Args:
        path: Direct path to the model directory. Optional if project_name and model_number given.
        project_name: Project name (e.g. ``"Comparison #2"``). Use with model_number.
        model_number: Model index (e.g. 19). Use with project_name.
        models_base: Base directory for ``models/`` when using project_name/model_number (default: cwd).
        initialization: If given together with optimization, use these instead of config/defaults.
        optimization: If given together with initialization, use these instead of config/defaults.
        ambient_temperature: Override when building from config (ignored if init+opt provided).
        temperature_range: Override when building from config (ignored if init+opt provided).

    Returns:
        Loaded model (e.g. FNN) ready for prediction or further training.

    Raises:
        ValueError: If neither path nor (project_name and model_number) are provided.
    """
    model_path = resolve_model_path(
        model_path=Path(path) if path is not None else None,
        project_name=project_name,
        model_number=model_number,
        models_base=models_base,
    )

    if initialization is not None and optimization is not None:
        return _load_model_core(model_path, initialization, optimization)

    # High-level: build from config (or defaults)
    meta = _load_meta(model_path)
    if meta.get("model_class") != "FNN":
        raise ValueError(
            f"Only FNN is supported for prediction; got model_class={meta.get('model_class')!r}. "
            "Extend storage.TEMPLATE_BUILDERS for other architectures."
        )
    domain = _build_domain_from_meta(meta)
    configuration = load_configuration(model_path)

    if configuration.get("material") is not None:
        m = configuration["material"]
        material = pinnjax.Material(
            heat_capacity=m.heat_capacity,
            density=m.density,
            radiation_coefficient=m.radiation_coefficient,
            thermal_conductivity=m.thermal_conductivity,
            convection_coefficient=m.convection_coefficient,
            ambient_temperature=(
                ambient_temperature
                if ambient_temperature is not None
                else m.ambient_temperature
            ),
            temperature_range=(
                temperature_range
                if temperature_range is not None
                else m.temperature_range
            ),
        )
    else:
        material = _default_material(
            ambient_temperature=ambient_temperature or DEFAULT_AMBIENT_TEMPERATURE,
            temperature_range=temperature_range or DEFAULT_TEMPERATURE_RANGE,
        )

    if configuration.get("optimization_spec") is not None:
        optimization = build_optimization_from_spec(
            configuration["optimization_spec"],
            domain,
            material,
        )
    else:
        optimization = _build_optimization_fallback(domain, material)

    initialization = pinnjax.Initialization(
        weights=jinit.variance_scaling(
            scale=1.48**2,
            mode="fan_in",
            distribution="truncated_normal",
        ),
        bias=jinit.zeros,
    )

    return _load_model_core(model_path, initialization, optimization)

    def load_configuration(path: Path | str) -> dict:
    """
    Load configuration from a saved model directory (Configurations and Parameters/_config.json).

    Returns a dict with keys: material (pinnjax.Material), goldak (pinnjax.Goldak),
    optimization_spec (dict or None). Keys are omitted if not present in the JSON.
    """
    path = Path(path)
    config_dir = path / "Configurations and Parameters"
    json_path = config_dir / "_config.json"
    if not json_path.exists():
        return {}
    try:
        data = json.loads(json_path.read_text())
    except (json.JSONDecodeError, OSError):
        return {}
    out = {}
    if "material" in data and isinstance(data["material"], dict):
        try:
            out["material"] = pinnjax.Material(
                **{k: float(v) for k, v in data["material"].items()}
            )
        except (TypeError, KeyError):
            pass
    if "goldak" in data and isinstance(data["goldak"], dict):
        try:
            out["goldak"] = pinnjax.Goldak(
                **{k: float(v) for k, v in data["goldak"].items()}
            )
        except (TypeError, KeyError):
            pass
    if "optimization_spec" in data and isinstance(data["optimization_spec"], dict):
        out["optimization_spec"] = data["optimization_spec"]
    elif "optimization" in data and isinstance(data["optimization"], dict):
        # Backward compatibility: infer spec from string reprs in old _config.json
        out["optimization_spec"] = _infer_optimization_spec_from_plain(
            data["optimization"]
        )
    return out

    def resolve_model_path(
    *,
    model_path: Path | None = None,
    project_name: str | None = None,
    model_number: int | None = None,
    models_base: Path | None = None,
) -> Path:
    """
    Resolve the model directory from a given path or from project name + model number.

    When using project name and model number, the path is:
        {models_base} / models / {project_name} / Model-#{model_number}

    Args:
        model_path: Direct path to the model directory (takes precedence if set).
        project_name: Project name (e.g. "Comparison #2", "EXAMPLE").
        model_number: Model index (e.g. 1, 2).
        models_base: Base path for the ``models`` folder (default: Path.cwd()).

    Returns:
        Resolved path to the model directory.

    Raises:
        ValueError: If neither (model_path) nor (project_name and model_number) are provided.
    """
    if model_path is not None and model_path != Path():
        return Path(model_path).resolve()
    if project_name is not None and model_number is not None:
        base = Path(models_base) if models_base is not None else Path.cwd()
        path = base / "models" / project_name / f"Model-#{model_number}"
        return path.resolve()
    raise ValueError("Provide either model_path or both project_name and model_number.")

    def inspect_checkpoint(
    path: Path | str | None = None,
    *,
    project_name: str | None = None,
    model_number: int | None = None,
    models_base: Path | None = None,
) -> dict:
    """
    Inspect what the loader expects for a checkpoint (array count and shapes).
    Uses only meta.json; does not require config or model.eqx.

    Use this to verify that the load path will build the correct template for
    your saved models (e.g. across many checkpoints).

    Resolve path the same way as load_model: either pass **path** or
    **project_name** and **model_number** (and optionally **models_base**).

    Returns:
        dict with keys:
        - meta: raw meta dict from meta.json
        - model_class: from meta
        - topology: from meta
        - domain: domain summary (min, max, times if present)
        - array_leaf_count: number of array leaves the template will have
        - array_shapes: list of shapes (one per leaf)
        - dtype: dtype string the loader will use (from meta or default)
    """
    model_path = resolve_model_path(
        model_path=Path(path) if path is not None else None,
        project_name=project_name,
        model_number=model_number,
        models_base=models_base,
    )
    meta_file = model_path / "meta.json"
    if not meta_file.exists():
        raise FileNotFoundError(f"Model meta not found: {meta_file}")
    meta = orjson.loads(meta_file.read_bytes())
    _require_meta_keys(meta, model_path)

    dtype_str = meta.get("dtype")
    if dtype_str is not None:
        try:
            checkpoint_dtype = getattr(jnp, dtype_str)
        except AttributeError:
            checkpoint_dtype = jnp.dtype(dtype_str)
        dtypes_to_try = (checkpoint_dtype,)
    else:
        dtypes_to_try = (jnp.float32, jnp.float64)

    domain = _build_domain_from_meta(meta)
    material = _default_material()
    optimization = _build_optimization_fallback(domain, material)
    initialization = pinnjax.Initialization(
        weights=jinit.variance_scaling(
            scale=1.48**2,
            mode="fan_in",
            distribution="truncated_normal",
        ),
        bias=jinit.zeros,
    )

    old_dtype = config.get_dtype()
    old_x64 = config.get_JAX_ENABLE_X64()
    try:
        for checkpoint_dtype in dtypes_to_try:
            config.set_dtype(checkpoint_dtype)
            if checkpoint_dtype == jnp.float64:
                config.set_JAX_ENABLE_X64(True)
                jax.config.update("jax_enable_x64", True)
            else:
                config.set_JAX_ENABLE_X64(False)
                jax.config.update("jax_enable_x64", False)
            template = _build_template(meta, initialization, optimization)
            template_arrays, _ = eqx.partition(template, eqx.is_array)
            leaves = list(jax.tree_util.tree_leaves(template_arrays))
            dom = meta["domain"]
            domain_summary = {
                "min": list(dom["min"]),
                "max": list(dom["max"]),
                "times": dom.get("times"),
            }
            return {
                "meta": meta,
                "model_class": meta.get("model_class"),
                "topology": list(meta["topology"]),
                "domain": domain_summary,
                "array_leaf_count": len(leaves),
                "array_shapes": [list(t.shape) for t in leaves],
                "dtype": str(leaves[0].dtype) if leaves else str(config.get_dtype()),
            }
    finally:
        config.set_dtype(old_dtype)
        if old_x64 is not None:
            config.set_JAX_ENABLE_X64(old_x64)
            jax.config.update("jax_enable_x64", old_x64)
    raise RuntimeError("inspect_checkpoint: no dtype succeeded")
