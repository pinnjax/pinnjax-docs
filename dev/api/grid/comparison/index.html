    class SpatialComparisonGrid(SpatialGrid):
    """A Cartesian grid whose axes are the unique coordinates of a point set.

    The input rows are not stored as the mesh. Each column is reduced to its
    sorted unique values, and ``coordinates`` is the Cartesian product of those
    values. A fixed axis keeps only the unique value that matches the requested
    number.

    Attributes:
        domain: Spatial bounds passed in at construction.
        spatial_dimensions: Number of columns, taken from ``len(domain.min)``.
        fixed_coordinates: Axis name to the value held constant. Empty when omitted.
        column_values: Sorted unique values on each axis, in ``x``, ``y``, ``z`` order.
            A fixed axis holds one value.
        coordinates: Mesh of those column values. The last axis is one entry per
            spatial axis.

    """

    def __init__(
        self,
        domain: SpatialDomain,
        coords: Array,
        fixed_coordinates: dict[str, float] | None = None,
    ) -> None:
        """Build the mesh from the unique value in each coordinate column.

        A fixed axis is kept only when one unique value lies within ``1e-9`` of the
        requested number. The stored value is the one from ``coords``, not the number
        that was requested. The comparison does not use a relative tolerance.

        Args:
            domain: Spatial bounds. ``len(domain.min)`` is the number of columns read
                from ``coords``.
            coords: Point rows, shaped ``(n, spatial_dimensions)``, in ``x``, ``y``, ``z``
                order.
            fixed_coordinates: Axis name to the value held constant. Default: no fixed axes.

        Raises:
            ValueError: If a fixed value is not present in that column.

        """
        super().__init__()
        self.domain = domain
        self.spatial_dimensions = len(domain.min)
        self.fixed_coordinates = fixed_coordinates or {}

        unique_per_column = [
            jnp.sort(jnp.unique(coords[:, i])) for i in range(self.spatial_dimensions)
        ]

        # Apply fixed coordinates by masking the uniques
        self.column_values = []
        for i in range(self.spatial_dimensions):
            axis_name = list(SpatialGrid.AXIS_TO_IDX.keys())[i]  # "x", "y", "z", ...
            if axis_name in self.fixed_coordinates:
                # Only keep the fixed value
                # (match by tolerance so model float matches CSV)
                fixed_val = float(self.fixed_coordinates[axis_name])
                uniques = unique_per_column[i]
                close = jnp.isclose(uniques, fixed_val, rtol=0.0, atol=1e-9)
                if not jnp.any(close):
                    msg = f"Fixed coordinate {axis_name}={fixed_val} not in column {i}"
                    config.LOGGER.error(msg)
                    raise ValueError(msg)
                # Use the CSV's value so scatter indexing matches
                self.column_values.append(jnp.array([uniques[jnp.argmax(close)]]))
            else:
                self.column_values.append(unique_per_column[i])

        # build the coordinate mesh
        if len(self.column_values) == 1:
            mesh = [self.column_values[0]]
        else:
            mesh = jnp.meshgrid(*self.column_values, indexing="xy")
        self.coordinates = jnp.stack(mesh, axis=-1)

    def shape(self) -> tuple[int, ...]:
        """Return the unique-value count on each spatial axis.

        The order is ``x``, ``y``, ``z``. This is not the shape of ``coordinates``.
        Two axes stored with ``xy`` indexing have mesh shape ``(ny, nx, 2)``, while
        this method returns ``(nx, ny)``. Time is not included.

        Returns:
            One count per spatial axis.

        """
        return tuple(int(v.size) for v in self.column_values)

    def indices_for(self, coords: Array) -> list[Array]:
        """Return the column index of each coordinate.

        Each column is located with a binary search on ``column_values``. The search
        returns an insertion index and does not check that the coordinate is present.

        Args:
            coords: Point rows, shaped ``(n, spatial_dimensions)``, in ``x``, ``y``, ``z``
                order.

        Returns:
            One index array per spatial axis, in ``x``, ``y``, ``z`` order.

        """
        n_spatial = self.coordinates.shape[-1]
        return [jnp.searchsorted(self.column_values[i], coords[:, i]) for i in range(n_spatial)]

    def add_time(
        self,
        times: Array,
        *,
        time_range: tuple[float, float] | None = None,
    ) -> None:
        """Append these time samples to the mesh.

        ``times`` is flattened. ``time_range`` defaults to the first and last stored
        time after that flatten. The parent builder then repeats the spatial mesh once
        per time and appends time as the last coordinate.

        Args:
            times: Time values, one entry per time sample.
            time_range: First and last time to record. Default: the minimum and maximum
                of ``times``.

        """
        times = jnp.asarray(times, dtype=config.get_dtype()).reshape(-1)
        self.has_time = True
        self.times = times
        self.num_times = int(times.shape[0])
        self.time_range = time_range or (float(times.min()), float(times.max()))
        self._build_spatiotemporal_coordinates()

    def scatter(
        self,
        coords: Array,
        values: Array,
        *,
        require_full: bool = True,
    ) -> Array:
        """Place sample values onto the Cartesian mesh.

        ``coords`` uses ``x``, ``y``, ``z`` order. For two or more axes the mesh is
        stored with ``xy`` indexing, so the output shape is ``(ny, nx, ...)`` and the
        ``x`` and ``y`` indices are swapped to match. One axis keeps shape ``(nx,)``.

        ``require_full`` checks that the number of rows equals the number of mesh
        points. It does not check that every mesh point appears exactly once.

        Args:
            coords: Point rows, shaped ``(n, spatial_dimensions)``.
            values: One value per row of ``coords``.
            require_full: Reject a point set that is not the same length as the mesh.
                Default: ``True``.

        Returns:
            The values on the mesh.

        Raises:
            ValueError: If ``require_full`` is true and the row count is not the mesh size.

        """
        # coords: (n_samples, n_spatial) in (x, y, [z]) column order
        n_spatial = self.coordinates.shape[-1]

        # indices along each axis as per column_values order: [x, y, (z)]
        idxs = self.indices_for(coords)

        # meshgrid(indexing="xy") => grid shape is (ny, nx, [nz]), i.e. Y first, then X
        if n_spatial >= TWO_DIMENSIONAL:
            shp = (
                int(self.column_values[1].size),  # ny
                int(self.column_values[0].size),  # nx
                *(int(v.size) for v in self.column_values[2:]),
            )  # [nz, ...]
            idxs = [idxs[1], idxs[0], *idxs[2:]]  # swap X<->Y for XY indexing
        else:
            shp = (int(self.column_values[0].size),)

        if require_full and int(coords.shape[0]) != int(jnp.prod(jnp.array(shp))):
            config.LOGGER.info(f"{coords.shape[0]=}")
            config.LOGGER.info(f"{int(jnp.prod(jnp.array(shp)))=}")
            msg = "CSV does not form a full Cartesian grid."
            config.LOGGER.error(msg)
            raise ValueError(msg)

        out = jnp.empty(shp, dtype=values.dtype)
        return out.at[tuple(idxs)].set(values)

    def slice(
    self,
    fixed_coordinates: dict[str, float] | None = None,
    *,
    atol: float = 1e-9,
) -> SpatialComparisonGrid:
    """Return the grid points that sit on the requested fixed coordinates.

    The spatial coordinates are flattened to one row per point. With no fixed
    coordinates, every row is kept and the grid's own ``fixed_coordinates`` are
    passed through when that attribute exists. Each requested axis must be within
    ``atol`` of the given value. The comparison does not use a relative tolerance.

    Args:
        fixed_coordinates: Axis name to the value that row must match. Default: no
            constraint.
        atol: Absolute tolerance for each comparison. Default: ``1e-9``.

    Returns:
        A [`SpatialComparisonGrid`][pinnjax.SpatialComparisonGrid] of the matching rows.

    Raises:
        pinnjax.InputError: If an axis name is outside this grid's spatial axes.
        ValueError: If no row matches.

    """
    from .comparison import SpatialComparisonGrid

    fixed_coordinates = fixed_coordinates or {}

    coords_flat = self.coordinates.reshape(-1, self.spatial_dimensions)

    if not fixed_coordinates:
        return SpatialComparisonGrid(
            domain=self.domain,
            coords=coords_flat,
            fixed_coordinates=getattr(self, "fixed_coordinates", None),
        )

    valid_axes = set(["x", "y", "z"][: self.spatial_dimensions])
    bad = [k for k in fixed_coordinates if k not in valid_axes]
    if bad:
        msg = f"Invalid fixed axis/axes: {bad}. Valid: {sorted(valid_axes)}"
        config.LOGGER.error(msg)
        raise config.InputError(msg)

    mask = jnp.ones((coords_flat.shape[0],), dtype=bool)
    for axis, val in fixed_coordinates.items():
        idx = self.AXIS_TO_IDX[axis]
        mask = mask & jnp.isclose(
            coords_flat[:, idx],
            float(val),
            rtol=0.0,
            atol=atol,
        )

    if not bool(jnp.any(mask)):
        msg = f"Slice produced no points for {fixed_coordinates}."
        config.LOGGER.error(msg)
        raise ValueError(msg)

    sliced_coords = coords_flat[mask, :]

    return SpatialComparisonGrid(
        domain=self.domain,
        coords=sliced_coords,
        fixed_coordinates=fixed_coordinates,
    )
