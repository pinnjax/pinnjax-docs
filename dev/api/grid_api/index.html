    class LineGrid(SpatialGrid):
    """1D line grid through domain"""

    def __init__(
        self,
        domain: pinnjax.SpatialDomain,
        fixed_coordinates: dict[str, float] | None = None,
        num_points: int = 100,
        axis: str | None = None,
        time_range: tuple[float, float] | None = None,
        num_times: int | None = None,
        *,
        periodic: bool = False,
    ) -> None:
        super().__init__()
        self.domain = domain
        self.spatial_dimensions = len(domain.min)

        self.periodic = periodic
        self.fixed_coordinates = fixed_coordinates or {}
        self.num_points = (num_points,)

        if axis is None:
            free_axes = self._get_free_axes(self.fixed_coordinates, expected_count=1)
            self.free_axis = free_axes[0]
        else:
            self.free_axis = axis

        self.free_axes = [self.free_axis]

        # Build spatial coordinates
        self.coordinates = self._build_coordinates(
            self.free_axes,
            self.fixed_coordinates,
            self.num_points,
        )

        # Setup temporal dimension if specified
        if time_range is not None and num_times is not None:
            self._setup_temporal(time_range, num_times)

        self,
        domain: pinnjax.SpatialDomain,
        fixed_coordinates: dict[str, float] | None = None,
        num_points: tuple[int, int] = (100, 100),
        axes: tuple[str, str] | None = None,
        time_range: tuple[float, float] | None = None,
        num_times: int | None = None,
        *,
        periodic: bool = False,
    ) -> None:
        super().__init__()
        self.domain = domain
        self.spatial_dimensions = len(domain.min)

        self.periodic = periodic
        self.fixed_coordinates = fixed_coordinates or {}
        self.num_points = num_points

        if axes is None:

    class RectangularGrid3D(SpatialGrid):
    """3D rectangular grid for full domain"""

    def __init__(
        self,
        domain: pinnjax.SpatialDomain,
        num_points: tuple[int, int, int] = (100, 100, 100),
        time_range: tuple[float, float] | None = None,
        num_times: int | None = None,
        *,
        periodic: bool = False,
    ) -> None:
        super().__init__()
        self.domain = domain
        self.spatial_dimensions = len(domain.min)

        self.periodic = periodic

        if self.spatial_dimensions != THREE_DIMENSIONAL:
            msg = "RectangularGrid3D requires 3D domain"
            config.LOGGER.error(msg)
            raise config.InputError(msg)

        self.num_points = num_points
        self.free_axes = ["x", "y", "z"]
        self.fixed_coordinates = {}

        # Build spatial coordinates
        self.coordinates = self._build_coordinates(
            self.free_axes,
            self.fixed_coordinates,
            self.num_points,
        )

        # Setup temporal dimension if specified
        if time_range is not None and num_times is not None:
            self._setup_temporal(time_range, num_times)
