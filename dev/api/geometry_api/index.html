    def sample(
        self,
        method: pinnjax.SamplingMethod,
        sampling_value: float | None = None,
    ) -> tuple[Array, int]:
        """Generate collocation points within the Shape.
            The number of points collocated is determined by the sampling_value
            and varies by [`pinnjax.SamplingMethod`][pinnjax.SamplingMethod] `mode`.

        Args:
            method: A [`pinnjax.SamplingMethod`][pinnjax.SamplingMethod] object.
            sampling_value: Determines the number of points to sample based
                on the `mode` provided at initialization of the `SamplingMethod`.
                Default `mode` is `point_count`.
        Returns:
            samples: Array of collocation points.
            n_points: Number of collocation points.

        """
        if sampling_value is None:
            sampling_value = 1000

        self.samples = jnp.asarray(
            method(self, sampling_value), dtype=config.get_dtype()
        )
        self.n_points = len(self.samples)

        return self.samples, self.n_points

    def residuals(self, ops: pinnjax.DifferentialOperators) -> Array:
        """Calculate the residual values for all collocation points using the bound
            [`pinnjax.Residual`][pinnjax.Residual].

        Args:
            ops: A [`DifferentialOperators`][pinnjax.DifferentialOperators] object.


        Returns:
            residuals: Array of residual values for each collocation point.

        Raises:
            InputError: If no collocation points have been generated for this Shape.
            InputError: If no [`pinnjax.Residual`][pinnjax.Residual]
                has been bound to this Shape.

        """

            msg = (
                "There are no samples associated with this shape. Call .sample() first."
            )
            config.LOGGER.error(msg)
            raise config.InputError(msg)

        if not self.has_residual():
            msg = (
                "There is no residual function associated with this shape. "
                "Call .set_residual() first."
            )
            config.LOGGER.error(msg)
            raise config.InputError(msg)
        return self.residual_function(self.samples, ops)

            bool: Whether the Shape has a [`pinnjax.Loss`][pinnjax.Loss].

            [`pinnjax.Loss`][pinnjax.Loss].

        Args:
            residuals: Array of residual values for each collocation point.

        """

            msg = (
                "There is no loss function associated with this shape. "
                "Call .set_loss first."
            )
            config.LOGGER.error(msg)
            raise config.InputError(msg)

        self.loss_val = self.loss_function(residuals)
        return self.loss_val

    def residual_and_losses(
        self,

        ops: pinnjax.DifferentialOperators,
    ) -> Array:
        """Calculate the loss and residuals. Return the loss."""
        residuals = self.residuals(ops)
        return self.loss(residuals)

    class Dot(Shape):
    def __init__(
        self,
        a: float
        | tuple[float]
        | tuple[float, float]
        | tuple[float, float, float]
        | Array,
        times: pinnjax.Time | float | None = None,
    ) -> None:
        super().__init__()
        self.a = jnp.atleast_1d(jnp.asarray(a, dtype=config.get_dtype()))
        self.spatial_dims = self.a.size
        self.dims = self.a.size

        if isinstance(times, (int, float)):
            # Single time instant
            self.single_time = float(times)
            self.times = None
            self.dims += 1
        elif isinstance(times, pinnjax.Time):
            # Time range
            self.single_time = None
            self.times = times
            self.tmin = times.min
            self.tmax = times.max
            self.tstep = times.step
            self.dims += 1
        else:
            # No time dimension
            self.single_time = None
            self.times = None

    def estimate_nsamples(self, res: float) -> int:
        # 1 is not enough samples.
        if self.times is None:
            return 1
        return int(jnp.round(self.times.range() / self.tstep))

    def measure(self) -> Array:
        msg = "Not implemented."
        config.LOGGER.error(msg)
        raise NotImplementedError

    def normal(self) -> Array:
        if self.a.shape[0] != 1:
            msg = "Dot.normal() only implemented for 1D"
            config.LOGGER.error(msg)
            raise NotImplementedError(msg)

        x = float(self.a[0])
        if jnp.isclose(x, -1.0):
            return jnp.array([-1.0])
        if jnp.isclose(x, 1.0):
            return jnp.array([1.0])
        return jnp.sign(jnp.array([x], dtype=config.get_dtype()))

    def bounds(self) -> tuple[Array, Array]:
        a = jnp.expand_dims(self.a, axis=0)  # shape (1, ndim)
        return a, a

    def transform(self, unit_samples: Array) -> Array:
        spatial = self.a

        if self.single_time is not None:
            point = jnp.concatenate([spatial, jnp.array([self.single_time])])
        elif self.times is not None:
            t_param = unit_samples[:, -1]
            time = self.tmin + t_param * (self.tmax - self.tmin)

    def __init__(
        self,
        origin: float
        | tuple[float]
        | tuple[float, float]
        | tuple[float, float, float]
        | Array,
        end: float
        | tuple[float]
        | tuple[float, float]
        | tuple[float, float, float]
        | Array,
        times: pinnjax.Time | float | None = None,
    ) -> None:
        super().__init__()
        self.origin = jnp.atleast_1d(
            jnp.array(origin, dtype=config.get_dtype()),
        )  # Ensure proper dtype
        self.end = jnp.atleast_1d(jnp.array(end, dtype=config.get_dtype()))
        self.spatial_dims = self.origin.size
        self.dims = self.origin.size

        # Handle different time specifications
        if isinstance(times, (int, float)):
            # Single time instant
            self.single_time = float(times)
            self.times = None
            self.dims += 1
        elif isinstance(times, pinnjax.Time):
            # Time range
            self.single_time = None
            self.times = times
            self.tmin = times.min
            self.tmax = times.max
            self.tstep = times.step
            self.dims += 1
        else:
            # No time dimension
            self.single_time = None
            self.times = None

    def measure(self) -> Array:
        return jnp.linalg.norm(self.end - self.origin)

    def normal(self) -> Array:
        """Normal vector to the line (perpendicular)."""
        if self.origin.shape[0] == TWO_DIMENSIONAL:
            direction = self.end - self.origin
            # Perpendicular in 2D: rotate 90 degrees counterclockwise
            normal_2d = jnp.array(
                [-direction[1], direction[0]], dtype=config.get_dtype()
            )
            norm_magnitude = jnp.linalg.norm(normal_2d)
            return jnp.where(
                norm_magnitude > 1e-12,
                normal_2d / norm_magnitude,
                jnp.array([0.0, 1.0], dtype=config.get_dtype()),
            )  # Fallback
        msg = "Normal for 3D lines not implemented."
        config.LOGGER.error("Normal for 3D lines not implemented.")
        raise NotImplementedError(msg)

    def bounds(self) -> tuple[Array, Array]:
        lower = jnp.minimum(self.origin, self.end)
        upper = jnp.maximum(self.origin, self.end)

        if self.single_time is not None:
            # Single time point
            lower = jnp.concatenate([lower, jnp.array([self.single_time])])
            upper = jnp.concatenate([upper, jnp.array([self.single_time])])
        elif self.times is not None:
            # Time range
            lower = jnp.concatenate([lower, jnp.array([self.tmin])])
            upper = jnp.concatenate([upper, jnp.array([self.tmax])])

        # unit_samples shape (..., 1) for spatial parameter
        t_param = unit_samples[..., 0:1]  # Parameter along line [0,1]

        spatial = self.origin + t_param * (self.end - self.origin)

    class Box(Shape, ABC):
    def __init__(
        self,
        origin: tuple[float, float, float] | Array,
        edge1: tuple[float, float, float] | Array,
        edge2: tuple[float, float, float] | Array,
        edge3: tuple[float, float, float] | Array,
        times: pinnjax.Time | float | None = None,
    ) -> None:
        super().__init__()
        self.origin = jnp.array(origin)
        self.edge1 = jnp.array(edge1)
        self.edge2 = jnp.array(edge2)
        self.edge3 = jnp.array(edge3)
        self.spatial_dims = self.origin.size
        self.dims = self.origin.size

        # Handle different time specifications
        if isinstance(times, (int, float)):
            # Single time instant
            self.single_time = float(times)
            self.times = None
            self.dims += 1
        elif isinstance(times, pinnjax.Time):
            # Time range
            self.single_time = None
            self.times = times
            self.tmin = times.min
            self.tmax = times.max
            self.tstep = times.step
            self.dims += 1
        else:
            # No time dimension
            self.single_time = None
            self.times = None

    def measure(self) -> Array:
        return jnp.abs(jnp.dot(self.edge1, jnp.cross(self.edge2, self.edge3)))

    def bounds(self) -> tuple[Array, Array]:
        # Compute all 8 corners, then min/max
        corners = jnp.stack(
            [
                self.origin,
                self.origin + self.edge1,
                self.origin + self.edge2,
                self.origin + self.edge3,
                self.origin + self.edge1 + self.edge2,
                self.origin + self.edge1 + self.edge3,
                self.origin + self.edge2 + self.edge3,
                self.origin + self.edge1 + self.edge2 + self.edge3,
            ],
        )
        lower = jnp.min(corners, axis=0)
        upper = jnp.max(corners, axis=0)

        if self.single_time is not None:
            # Single time point
            lower = jnp.concatenate([lower, jnp.array([self.single_time])])
            upper = jnp.concatenate([upper, jnp.array([self.single_time])])
        elif self.times is not None:
            # Time range
            lower = jnp.concatenate([lower, jnp.array([self.tmin])])
            upper = jnp.concatenate([upper, jnp.array([self.tmax])])

        return lower, upper

    def draw(
        self,
        *,
        show_normals: bool = False,
        normal_scale: float = 1.0,
        fig: Figure | None = None,
        ax: Axes | None = None,
        show: bool | None = True,
    ) -> tuple[Figure, Axes]:
        domain = pinnjax.SpatialDomain.from_obb(
            self.origin,
            self.edge1,
            self.edge2,
            self.edge3,
            times=self.times,
        )

        fig, ax = domain.draw(show=show, fig=fig, ax=ax)

        return fig, ax

    def transform(self, unit_samples: Array) -> Array:
        # Spatial transformation
        pt = (
            self.origin
            + unit_samples[..., 0:1] * self.edge1
            + unit_samples[..., 1:2] * self.edge2
            + unit_samples[..., 2:3] * self.edge3
        )

        if self.single_time is not None:
            # Add constant time dimension
            t = jnp.full((*unit_samples.shape[:-1], 1), self.single_time)
            pt = jnp.concatenate([pt, t], axis=-1)
        elif self.times is not None:
            # Add time range
            t = self.tmin + unit_samples[..., 3:4] * (self.tmax - self.tmin)
            pt = jnp.concatenate([pt, t], axis=-1)

        return pt

    def estimate_nsamples(self, res: float) -> int:
        len1, len2, len3 = (
            jnp.linalg.norm(self.edge1),
            jnp.linalg.norm(self.edge2),
            jnp.linalg.norm(self.edge3),
        )
        n_space = (
            jnp.maximum(jnp.round(len1 / res), 2)
            * jnp.maximum(jnp.round(len2 / res), 2)
            * jnp.maximum(jnp.round(len3 / res), 2)
        )

        if self.single_time is not None:
            # Single time instant - only spatial samples needed
            return int(n_space)
        if self.times is not None:
            # Time range - multiply by time samples
            n_time = jnp.maximum(jnp.round((self.tmax - self.tmin) / self.tstep), 2)
            return int(n_space * n_time)
        return int(n_space)

    been_split: bool = False

    train_indices: Array
    train_inputs: Array
    train_outputs: Array

    test_indices: Array
    test_inputs: Array
    test_outputs: Array

    _shape: dict[str, tuple[int, ...]] | None = None

    validation_outputs: Array | None = None

    _full_outputs: Array | None = None

    def __init__(
        self,
        inputs: Array,
        outputs: Array,
        *,
        indices: Array | Sequence[int] | None = None,
        portion: float = 1.0,
        materialize_validation_from_portion: bool = True,
    ) -> None:
        super().__init__()

        inputs = jnp.asarray(inputs, dtype=config.get_dtype())
        outputs = jnp.asarray(outputs, dtype=config.get_dtype())

        if indices is None:
            indices = jnp.asarray(
                pinnjax.index_shuffle(len(inputs), portion=portion),
                dtype=int,
            )
        else:
            if not isinstance(indices, Array):
                if not isinstance(indices, Sequence):
                    msg = f"'indices' must be a Sequence. Got {type(indices).__name__}"
                    config.LOGGER.error(msg)
                    raise TypeError(msg)
                indices = jnp.asarray(indices, dtype=int)

        indices = jnp.sort(indices)
        n_full = int(inputs.shape[0])

        used = jnp.zeros(n_full, dtype=bool)
        used = used.at[indices].set(True)
        validation_indices = jnp.sort(jnp.where(~used)[0])

        self.validation_inputs = None
        self.validation_outputs = None
        self._validation_indices = None
        self._full_inputs = None
        self._full_outputs = None

        if validation_indices.size > 0:
            if materialize_validation_from_portion:
                self.validation_inputs = inputs[validation_indices]
                self.validation_outputs = outputs[validation_indices]
            else:
                self._validation_indices = validation_indices
                self._full_inputs = inputs
                self._full_outputs = outputs

        self.inputs = inputs[indices]
        self.outputs = outputs[indices]
        self.indices = indices

        if len(self.inputs) == len(self.outputs):
            self.num_data = len(self.inputs)
        else:
            msg = (
                f"Length of Inputs: {len(self.inputs)} and "
                f"Length of Outputs: {len(self.outputs)} must be equal."
            )
            config.LOGGER.error(msg)
            raise ValueError(msg)

    def materialize_validation_from_portion(self) -> None:
        """Copy deferred held-out rows into ``validation_inputs`` / ``validation_outputs``.

        Only applies when the shape was built with ``portion < 1.0`` and
        ``materialize_validation_from_portion=False``, so row indices were set aside
        at construction. Calling this again after validation is already filled is a
        no-op. If no rows were ever held out (e.g. ``portion=1.0``), there is nothing
        to copy; logs a warning and returns.
        """
        if self.validation_inputs is not None:
            return
        if self._validation_indices is None or self._validation_indices.size == 0:
            config.LOGGER.warning(
                "All DATA was used in training/testing. Use a portion less than 1.0.",
            )
            return
        assert self._full_inputs is not None and self._full_outputs is not None
        self.validation_inputs = self._full_inputs[self._validation_indices]
        self.validation_outputs = self._full_outputs[self._validation_indices]
        self._validation_indices = None
        self._full_inputs = None
        self._full_outputs = None

    def clone(self, *, copy_split: bool = True) -> DataShape:
        # Pass identity indices so `inputs`/`outputs` are already the working rows
        # (global `self.indices` are not valid row picks into the subset tensor).
        n = len(self.inputs)
        ds = DataShape(
            self.inputs,
            self.outputs,
            indices=jnp.arange(n, dtype=int),
            portion=1.0,
            materialize_validation_from_portion=True,
        )
        ds.validation_inputs = self.validation_inputs
        ds.validation_outputs = self.validation_outputs
        if self._validation_indices is not None:
            ds._validation_indices = self._validation_indices
            ds._full_inputs = self._full_inputs
            ds._full_outputs = self._full_outputs
        if copy_split and self.been_split:
            ds.train_indices = self.train_indices
            ds.test_indices = self.test_indices
            ds.train_inputs = self.train_inputs
            ds.test_inputs = self.test_inputs
            ds.train_outputs = self.train_outputs
            ds.test_outputs = self.test_outputs
            ds.samples = self.samples
            ds.n_points = self.n_points
            ds.been_split = True
        return ds

    def from_table(
        cls,
        table: str | Path | pl.DataFrame,
        *,
        input_columns: str | int | Sequence[str] | Sequence[int],
        output_columns: str | int | Sequence[str] | Sequence[int],
        portion: float = 1.0,
        drop_nulls: bool = True,

        def _normalize_columns(
            spec: str | int | Sequence[str] | Sequence[int],
            available: Sequence[str],
            *,
            arg_name: str,
        ) -> list[str]:
            cols = list(available)

            if isinstance(spec, (str, int)):
                items = [spec]
            elif isinstance(spec, Sequence):
                if len(spec) == 0:
                    raise ValueError(f"`{arg_name}` cannot be empty.")
                items = list(spec)
            else:
                raise TypeError(f"`{arg_name}` has invalid type: {type(spec).__name__}")

            has_str = any(isinstance(x, str) for x in items)
            has_int = any(isinstance(x, int) for x in items)
            if has_str and has_int:
                raise TypeError(f"`{arg_name}` cannot mix str and int in one sequence.")

            if has_str:
                out = [str(x) for x in items]
                missing = [c for c in out if c not in cols]
                if missing:
                    raise ValueError(f"`{arg_name}` unknown columns: {missing}")
                return out

            out: list[str] = []
            for idx in items:
                if not isinstance(idx, int):
                    raise TypeError(f"`{arg_name}` must contain only str or only int.")
                if idx < 0 or idx >= len(cols):
                    raise IndexError(
                        f"`{arg_name}` index {idx} out of range [0, {len(cols) - 1}]"
                    )
                out.append(cols[idx])
            return out

        if isinstance(table, pl.DataFrame):
            df = table
        else:
            path = Path(table)
            if path.suffix.lower() != ".csv":
                raise ValueError(f"Expected .csv, got {path.suffix}")
            df = pl.read_csv(path)

        in_cols = _normalize_columns(
            input_columns, df.columns, arg_name="input_columns"
        )
        out_cols = _normalize_columns(
            output_columns, df.columns, arg_name="output_columns"
        )

        overlap = sorted(set(in_cols).intersection(out_cols))
        if overlap:
            raise ValueError(f"Input/output columns overlap: {overlap}")

        selected = in_cols + out_cols
        df = df.select(selected)
        if drop_nulls:
            df = df.drop_nulls(subset=selected)

        x = df.select(in_cols).to_numpy().astype(float)
        y = df.select(out_cols).to_numpy().astype(float)
        if y.ndim == 1:
            y = y[:, None]

        indices = None
        if portion != 1.0:
            indices = jnp.asarray(
                pinnjax.index_shuffle(len(x), portion=portion),
                dtype=int,
            )

        return cls(
            jnp.asarray(x, dtype=config.get_dtype()),
            jnp.asarray(y, dtype=config.get_dtype()),
            indices=indices,
            portion=1.0,
            materialize_validation_from_portion=materialize_validation_from_portion,
        )

    def test_train_split(
        self,
        split_percentage: float = 0.5,
        *,
        overwrite: bool = False,
        seed: int | None = None,  # default: config.get_seed() # 100
        mode: Literal["shuffle", "stratified"] = "shuffle",
        # stratified mode params
        consecutive_samples: int = 5,  # group size
        train_per_group: Optional[int] = 3,  # if None: derived from split_percentage
        # what to sort by (grout paper sorts by output/fatigue life)
        stratify_on: Literal["outputs", "inputs"] = "outputs",
        output_column: int = 0,  # if outputs is (N, k), which column to sort by
    ) -> None:
        if self.been_split is not True or overwrite is True:
            if overwrite:
                msg = "This DataShape was previously split. Overwriting."
                config.LOGGER.warning(msg)
            if split_percentage <= 0 or split_percentage >= 1:
                msg = (
                    "`split_percentage` must be between 0 and 1 (non-exclusive)."
                    f" Got {split_percentage}."
                )
                config.LOGGER.error(msg)
                raise ValueError(msg)

            if seed is None:
                seed = pinnjax.config.get_seed()

            if mode == "shuffle":
                indices = pinnjax.index_shuffle(self.num_data, seed=seed)
                split = max(int(self.num_data * split_percentage), 1)
                self.train_indices = jnp.asarray(sorted(indices[:split]), dtype=int)
                self.test_indices = jnp.asarray(sorted(indices[split:]), dtype=int)
            elif mode == "stratified":
                if consecutive_samples < 2:
                    msg = "`consecutive_samples` must be >= 2 for stratified mode."
                    config.LOGGER.error(msg)
                    raise ValueError(msg)
                # determine sort key (paper uses outputs / fatigue life)
                if stratify_on == "outputs":
                    y = jnp.asarray(self.outputs)
                    if y.ndim == 1:
                        key = y
                    elif y.ndim == 2:
                        if output_column < 0 or output_column >= y.shape[1]:
                            msg = (
                                f"`output_column` out of range for outputs shape {y.shape}. "
                                f"Got {output_column}."
                            )
                            config.LOGGER.error(msg)
                            raise ValueError(msg)
                        key = y[:, output_column]
                    else:
                        msg = (
                            f"Unsupported outputs ndim={y.ndim} for stratification. "
                            "Expected 1D or 2D outputs."
                        )
                        config.LOGGER.error(msg)
                        raise ValueError(msg)
                else:  # stratify_on == "inputs"
                    x = jnp.asarray(self.inputs)
                    if x.ndim != 2:
                        msg = (
                            f"Unsupported inputs ndim={x.ndim} for stratification on inputs. "
                            "Expected 2D inputs."
                        )
                        config.LOGGER.error(msg)
                        raise ValueError(msg)
                    if output_column < 0 or output_column >= x.shape[1]:
                        msg = (
                            f"`output_column` out of range for inputs shape {x.shape}. "
                            f"Got {output_column}."
                        )
                        config.LOGGER.error(msg)
                        raise ValueError(msg)
                    key = x[:, output_column]
                # sort by key ascending
                sorted_idx = list(map(int, jnp.argsort(key).tolist()))
                # choose train count per group
                if train_per_group is None:
                    tpg = int(round(consecutive_samples * split_percentage))
                else:
                    tpg = int(train_per_group)
                # keep valid and non-degenerate
                tpg = max(1, min(tpg, consecutive_samples - 1))
                rng = random.Random(seed)
                train_idx: list[int] = []
                test_idx: list[int] = []
                # split each consecutive block
                for start in range(0, len(sorted_idx), consecutive_samples):
                    block = sorted_idx[start : start + consecutive_samples]
                    if len(block) == 1:
                        # degenerate final block: put lone sample in train
                        train_idx.extend(block)
                        continue
                    rng.shuffle(block)
                    # handle partial final block robustly
                    k = min(max(1, tpg), len(block) - 1)
                    train_idx.extend(block[:k])
                    test_idx.extend(block[k:])
                # sort for stable indexing order
                self.train_indices = jnp.asarray(sorted(train_idx), dtype=int)
                self.test_indices = jnp.asarray(sorted(test_idx), dtype=int)
            else:
                msg = f"Invalid split mode: {mode}. Use 'shuffle' or 'stratified'."
                config.LOGGER.error(msg)
                raise ValueError(msg)
            self.train_inputs, self.test_inputs = (
                self.inputs[self.train_indices],
                self.inputs[self.test_indices],
            )
            self.train_outputs, self.test_outputs = (
                self.outputs[self.train_indices],
                self.outputs[self.test_indices],
            )

            self.samples = self.train_inputs

    def __init__(self, shapes: list[Shape], *, times: pinnjax.Time = None) -> None:
        super().__init__()

        self.kinds_type: tuple[int, ...] = ()
        self.labels_class: tuple[str, ...] = ()
        self.labels_type: tuple[str, ...] = ()

        self.evaluation_dictionary = {}
        self.final_loss_dictionary = {}

        self.last_epoch: int = 0

    def assign_residuals(
        self,
        residuals: pinnjax.Residual | list[pinnjax.Residual],
        *,
        class_map: dict | None = None,
        type_map: dict | None = None,
        labels_class: list | None = None,
        labels_type: list | None = None,
        kinds_class: list | None = None,
        kinds_type: list | None = None,
    ) -> None:
        if isinstance(residuals, pinnjax.Residual):
            residuals = [residuals] * len(self.shapes)

        if len(residuals) != len(self.shapes):
            msg = "Length of Residuals must EQUAL length of Shapes"
            config.LOGGER.error(msg)
            raise config.InputError(msg)

        if class_map is None or type_map is None:
            class_map, type_map = {}, {}
        if labels_class is None or labels_type is None:
            labels_class, labels_type = [], []
        if kinds_class is None or kinds_type is None:
            kinds_class, kinds_type = [], []

        for _i, (shape, residual) in enumerate(
            zip(self.shapes, residuals, strict=True),
        ):
            if isinstance(shape, ComplexShape):
                shape.assign_residuals(
                    residual,
                    class_map=class_map,
                    type_map=type_map,
                    labels_class=labels_class,
                    labels_type=labels_type,
                    kinds_class=kinds_class,
                    kinds_type=kinds_type,
                )
                class_map, type_map = shape.class_map, shape.type_map
                labels_class, labels_type = shape.labels_class, shape.labels_type
                kinds_class, kinds_type = shape.kinds_class, shape.kinds_type
            else:
                shape.set_residual(residual)

                # class-level tag (exact residual class)
                cls = residual.__class__
                if cls not in class_map:
                    class_map[cls] = len(class_map)
                    labels_class.append(cls.__name__)
                kinds_class.append(class_map[cls])

                # type-level tag (overarching category)
                if isinstance(residual, pinnjax.BoundaryCondition):
                    type_cls = pinnjax.BoundaryCondition
                elif isinstance(residual, pinnjax.PartialDifferentialEquation):
                    type_cls = pinnjax.PartialDifferentialEquation
                elif isinstance(residual, pinnjax.InitialCondition):
                    type_cls = pinnjax.InitialCondition
                elif isinstance(residual, pinnjax.DataResidual):
                    type_cls = pinnjax.DataResidual
                else:
                    msg = f"Unsupported residual type: {type(residual)}"
                    config.LOGGER.error(msg)
                    raise TypeError(msg)

                if type_cls not in type_map:
                    type_map[type_cls] = len(type_map)
                    labels_type.append(type_cls.__name__)
                kinds_type.append(type_map[type_cls])

        self.class_map = class_map
        self.type_map = type_map
        self.kinds_class = tuple(kinds_class)
        self.kinds_type = tuple(kinds_type)
        self.labels_class = tuple(labels_class)
        self.labels_type = tuple(labels_type)

        self.n_class = int(max(self.kinds_class) + 1) if self.kinds_class else 0
        self.n_type = int(max(self.kinds_type) + 1) if self.kinds_type else 0

        if not self.loss_dictionary:
            self.loss_dictionary = {
                "class": {lab: [] for lab in self.labels_class},
                "type": {lab: [] for lab in self.labels_type},
            }

    def unpack(self, *, flatten: bool = True) -> None:
        # The purpose of this function is unpack internal ComplexShapes
        # into a single ComplexShape.
        for _i, shape in enumerate(list(self.shapes)):
            if isinstance(shape, ComplexShape):
                self.shapes.remove(shape)
                if flatten:
                    valid, _shapes, _residual, _loss = shape.verify_repetition(
                        set_return=True,
                    )
                    if valid:
                        _shape = next(iter(_shapes), None)
                        if _shape is not None:
                            domain = shape.encompassing_domain()
                            obb = domain.to_obb()
                            flattened_shape = _shape(*obb, times=domain.times)
                            if _residual:
                                flattened_shape.set_residual(_residual)
                            if _loss:
                                flattened_shape.set_loss(_loss)
                            samples = []
                            n_points = 0
                            for s in shape.shapes:
                                if s.samples is not None:
                                    samples.append(s.samples)
                                    n_points += round(s.n_points)

                            if len(samples) > 0:
                                flattened_shape.samples = jnp.vstack(samples)
                            else:
                                flattened_shape.samples = None

                            flattened_shape.n_points = round(n_points)
                            self.shapes.append(flattened_shape)
                    else:
                        config.LOGGER.warning(
                            "Complex Shape Identified, but repetition not verified.",
                        )
                else:
                    for s in shape.shapes:
                        self.shapes.append(s)

    def encompassing_domain(self) -> pinnjax.SpatialDomain:
        # returns a SpatialDomain which encompasses all shapes.
        mins = []
        maxs = []
        for shape in self.shapes:
            bmin, bmax = shape.bounds()
            mins.append(bmin)
            maxs.append(bmax)

        mins = jnp.stack(mins)
        maxs = jnp.stack(maxs)

        # compute overall min/max per column
        mins = jnp.min(mins, axis=0)
        maxs = jnp.max(maxs, axis=0)

        if self.times is not None:
            return pinnjax.SpatialDomain(
                min=tuple(mins[:-1]),
                max=tuple(maxs[:-1]),
                times=self.times,
            )
        times = pinnjax.Time(min=mins[-1], max=maxs[-1], step=0.1)
        return pinnjax.SpatialDomain(
            min=tuple(mins[:-1]),
            max=tuple(maxs[:-1]),
            times=times,
        )

    def verify_repetition(
        self,
        *,
        set_return: bool = False,
    ) -> bool | tuple[bool, set, set, set]:
        shapes = set()
        residual = set()
        loss = set()
        for shape in self.shapes:
            shapes.add(type(shape))
            if shape.has_residual():
                residual.add(shape.residual_function)
            if shape.has_loss():
                loss.add(shape.loss_function)

        valid = all(len(s) <= 1 for s in (shapes, residual, loss))
        if set_return:
            return valid, shapes, residual, loss
        return valid

    class BoundedBox(ComplexShape, ABC):
    def __init__(
        self,
        origin: tuple[float, float, float],
        edge1: tuple[float, float, float],
        edge2: tuple[float, float, float],
        edge3: tuple[float, float, float],
        times: pinnjax.Time | float | None = None,
        *,
        initial: bool = False,
    ) -> None:
        # Create interior box
        interior = Box(
            origin=origin,
            edge1=edge1,
            edge2=edge2,
            edge3=edge3,
            times=times,
        )

        # Create 6 boundary faces
        o = jnp.array(origin)
        e1 = jnp.array(edge1)
        e2 = jnp.array(edge2)
        e3 = jnp.array(edge3)

        # LEFT (-X)
        left_face = Rectangle(origin=o, edge1=e3, edge2=e2, times=times)
        # RIGHT (+X)
        right_face = Rectangle(origin=o + e1, edge1=e2, edge2=e3, times=times)

        # FRONT (-Y)
        front_face = Rectangle(origin=o, edge1=e1, edge2=e3, times=times)
        # BACK (+Y)
        back_face = Rectangle(origin=o + e2, edge1=e3, edge2=e1, times=times)

        # BOTTOM (-Z)
        bottom_face = Rectangle(origin=o, edge1=e2, edge2=e1, times=times)
        # TOP (+Z)
        top_face = Rectangle(origin=o + e3, edge1=e1, edge2=e2, times=times)

        shapes = [
            interior,
            left_face,
            right_face,
            front_face,
            back_face,
            bottom_face,
            top_face,
        ]
        super().__init__(shapes)

        self.initials_set = False
        if initial:
            self.add_initial()

        self.path_set = False

        # Store references for easy access
        self.interior = interior
        self.left_face = left_face
        self.right_face = right_face
        self.back_face = back_face
        self.front_face = front_face
        self.bottom_face = bottom_face
        self.top_face = top_face

        # Store the original parameters
        self.origin = o
        self.edge1 = e1
        self.edge2 = e2
        self.edge3 = e3
        self.times = times

        # ensure all faces and normals face outwards.
        faces_and_normals = [
            (
                self.left_face,
                jnp.cross(
                    jnp.array(self.left_face.edge1),
                    jnp.array(self.left_face.edge2),
                ),
            ),
            (
                self.right_face,
                jnp.cross(
                    jnp.array(self.right_face.edge1),
                    jnp.array(self.right_face.edge2),
                ),
            ),
            (
                self.front_face,
                jnp.cross(
                    jnp.array(self.front_face.edge1),
                    jnp.array(self.front_face.edge2),
                ),
            ),
            (
                self.back_face,
                jnp.cross(
                    jnp.array(self.back_face.edge1),
                    jnp.array(self.back_face.edge2),
                ),
            ),
            (
                self.bottom_face,
                jnp.cross(
                    jnp.array(self.bottom_face.edge1),
                    jnp.array(self.bottom_face.edge2),
                ),
            ),
            (
                self.top_face,
                jnp.cross(
                    jnp.array(self.top_face.edge1),
                    jnp.array(self.top_face.edge2),
                ),
            ),
        ]

        # Normalize
        faces_and_normals = [
            (face, n / (jnp.linalg.norm(n) + 1e-12)) for face, n in faces_and_normals
        ]

        for face, n in faces_and_normals:
            face._outward_normal = n
            face.normal = lambda nn=n: nn

    def add_initial(self) -> None:
        if not self.initials_set:
            self.shapes.append(
                Box(
                    origin=self.origin,
                    edge1=self.edge1,
                    edge2=self.edge2,
                    edge3=self.edge3,
                    times=self.times.min,
                ),
            )
            self.initials_set = True
            return
        config.LOGGER.warning(
            "Initial Domain has already been added to this ComplexShape.",
        )
        return

    def add_path_shape(
        self,
        path: pinnjax.Path,
        radius: float | Iterable = 1.0,
    ) -> None:
        if isinstance(radius, (int, float)):
            radius = len(self.origin) * (radius,)

        if any(r <= 0.0 for r in radius):
            msg = f"{radius} is invalid. Must be greater than zero."
            config.LOGGER.error(msg)
            raise ValueError(msg)

        if len(radius) != len(self.origin):
            msg = (
                f"radius: {radius} must have the same number of "
                f"dims as spatial-domain: {len(self.origin)} "
            )
            config.LOGGER.error(msg)
            raise ValueError(msg)

        if not self.path_set:
            match type(path):
                case pinnjax.LinearPath:
                    end_point = path.initial + path.velocity * self.times.max
                    initial = tuple(
                        float(max(x - r, o))
                        for x, o, r in zip(
                            path.initial,
                            self.origin,
                            tuple(radius),
                            strict=True,
                        )
                    )
                    end = tuple(
                        float(min(x + r, o))
                        for x, o, r in zip(
                            end_point,
                            self.origin + self._get_domain_dimensions(),
                            tuple(radius),
                            strict=True,
                        )
                    )
                    path_domain = pinnjax.SpatialDomain(
                        min=initial,
                        max=end,
                        times=self.times,
                    )
                    obb = path_domain.to_obb()
                    self.shapes.append(Box(*obb, times=self.times))
                    # this adds a shape on the boundary at the +z
                    # generalization in the future might be smart
                    origin, edge1, edge2, edge3 = obb
                    o = jnp.array(origin)
                    e1 = jnp.array(edge1)
                    e2 = jnp.array(edge2)
                    e3 = jnp.array(edge3)
                    self.shapes.append(
                        Rectangle(origin=o + e3, edge1=e1, edge2=e2, times=self.times),
                    )
                    self.path = path
                    self.path_set = True
                case pinnjax.ComplexLinearPath:
                    internal_shapes = []
                    boundary_shapes = []
                    for i, segment in enumerate(path.segments):
                        end_point = (
                            segment.initial + segment.velocity * path.segment_times[i]
                        )
                        initial = tuple(
                            float(max(x - r, o))
                            for x, o, r in zip(
                                segment.initial,
                                self.origin,
                                tuple(radius),
                                strict=True,
                            )
                        )
                        end = tuple(
                            float(min(x + r, o))
                            for x, o, r in zip(
                                end_point,
                                self.origin + self._get_domain_dimensions(),
                                tuple(radius),
                                strict=True,
                            )
                        )
                        path_domain = pinnjax.SpatialDomain(min=initial, max=end)

                        time = pinnjax.Time(
                            min=path.cumulative_times[i],
                            max=path.cumulative_times[i + 1],
                            step=self.times.step,
                        )
                        obb = path_domain.to_obb()
                        internal_shapes.append(pinnjax.Box(*obb, times=time))

                        origin, edge1, edge2, edge3 = obb
                        o = jnp.array(origin)
                        e1 = jnp.array(edge1)
                        e2 = jnp.array(edge2)
                        e3 = jnp.array(edge3)
                        boundary_shapes.append(
                            Rectangle(origin=o + e3, edge1=e1, edge2=e2, times=time),
                        )

                    self.shapes.append(
                        pinnjax.ComplexShape(internal_shapes, times=self.times),
                    )
                    self.shapes.append(
                        pinnjax.ComplexShape(boundary_shapes, times=self.times),
                    )

                    self.path = path
                    self.path_set = True
                case _:
                    # TODO(@mjryande): Add the CirclePath
                    config.LOGGER.warning(
                        "Additional Points have not yet been "
                        "implemented for the %s path type.",
                        type(path).__name__,
                    )
                    config.LOGGER.warning("Try using a Linear or ComplexLinear path.")
                    config.LOGGER.warning(
                        "Evaluation will continue as normal, "
                        "but no additional points have been added.",
                    )
            return
        config.LOGGER.warning(
            "Additional Points have already been added to this ComplexShape",
        )
        return

    def draw(

    def _collect_samples_with_time(self) -> Array | None:
        """Collect all samples from shapes into (N, 4) array [x, y, z, t].


        Returns None if no samples.

        """
        parts = []
        for shape in self.shapes:
            if isinstance(shape, ComplexShape):
                for s in shape.shapes:
                    if s.samples is not None:
                        parts.append(jnp.array(s.samples))
            else:
                if shape.samples is not None:
                    parts.append(jnp.array(shape.samples))
        if not parts:
            return None
        return jnp.vstack(parts)

    def _draw_face_normal(
        self,
        ax: Axes,
        face: pinnjax.Rectangle,
        color: str = "blue",
        scale: float = 0.5,
    ) -> None:
        """Draw face normal arrows."""
        # Get face center
        o = jnp.array(face.origin)
        e1, e2 = jnp.array(face.edge1), jnp.array(face.edge2)
        center = o + 0.5 * (e1 + e2)

        try:
            normal = face.normal()
            dimensions = self._get_domain_dimensions()

            scaled_arrow = normal * dimensions * scale

            ax.quiver(
                center[0],
                center[1],
                center[2],
                scaled_arrow[0],
                scaled_arrow[1],
                scaled_arrow[2],
                color=color,
                alpha=0.9,
                arrow_length_ratio=0.15,
                linewidth=2,
            )

        except (ValueError, ArithmeticError, IndexError) as e:
            config.LOGGER.warning(
                "Could not draw normal due to mathematical error: %s",
                e,
            )

    def _get_domain_dimensions(self) -> Array:
        """Calculate the actual dimensions of the domain."""
        min_corner = self.origin
        max_corner = self.origin + self.edge1 + self.edge2 + self.edge3
        return abs(max_corner - min_corner)

    def __init__(
        self,
        inputs: Array,
        outputs: Array,
        *,
        indices: Array | Sequence[int] | None = None,
        portion: float = 1.0,
        materialize_validation_from_portion: bool = True,
    ) -> None:
        data_shape = DataShape(
            inputs,
            outputs,
            indices=indices,
            portion=portion,
            materialize_validation_from_portion=materialize_validation_from_portion,
        )
        super().__init__([data_shape])
        self.data_shape = data_shape

    @classmethod
    def _from_data_shape(cls, data_shape: DataShape) -> DataGeometry:
        self = object.__new__(cls)
        ComplexShape.__init__(self, [data_shape])
        self.data_shape = data_shape
        return self

    def duplicate_data_shape(
        self,
        times: int = 1,
        *,
        copy_split: bool = True,
    ) -> list[pinnjax.DataShape]:
        if times < 1:
            raise ValueError(f"`times` must be >= 1, got {times}.")
        duplicates: list[pinnjax.DataShape] = []
        for _ in range(times):
            ds = self.data_shape.clone(copy_split=copy_split)
            self.shapes.append(ds)
            duplicates.append(ds)
        return duplicates

    def validate(
        self,
        model: pinnjax.NeuralNetwork,
        evaluation: pinnjax.Evaluation | None = None,
    ) -> dict[str, float]:
        """Metrics on rows held out by ``portion`` (not in the main tabular pool).

        If validation was deferred at construction, materializes it when deferred
        indices exist. If nothing was held out at construction (e.g. ``portion=1.0``),
        logs a warning and returns ``{}`` — calling
        [`materialize_validation_from_portion`][pinnjax.DataShape.materialize_validation_from_portion]
        cannot create rows after the fact.

        With ``evaluation=None``, builds a
        [`TabularPointwiseEvaluation`][pinnjax.TabularPointwiseEvaluation] and binds validation
        rows. If you pass an ``evaluation`` (e.g. the same
        [`TabularPointwiseEvaluation`][pinnjax.TabularPointwiseEvaluation] or another tabular
        subclass used during training), it is **shallow-copied**, ``.inputs`` and ``.truth`` are
        set to the validation arrays, then ``evaluate(model)`` is called so the original instance
        is unchanged.
        """
        from pinnjax.loss import TabularPointwiseEvaluation

        ds = self.data_shape
        if ds.validation_inputs is None and (
            ds._validation_indices is not None and ds._validation_indices.size > 0
        ):
            ds.materialize_validation_from_portion()
        if ds.validation_inputs is None or ds.validation_inputs.shape[0] == 0:
            config.LOGGER.warning(
                "All DATA was used in training/testing. Use a portion less than 1.0. "
                "Returning empty metrics.",
            )
            return {}
        if evaluation is None:
            evaluation = TabularPointwiseEvaluation()

    def validate(
    self,
    model: pinnjax.NeuralNetwork,
    evaluation: pinnjax.Evaluation | None = None,
) -> dict[str, float]:
    """Metrics on rows held out by ``portion`` (not in the main tabular pool).

    If validation was deferred at construction, materializes it when deferred
    indices exist. If nothing was held out at construction (e.g. ``portion=1.0``),
    logs a warning and returns ``{}`` — calling
    [`materialize_validation_from_portion`][pinnjax.DataShape.materialize_validation_from_portion]
    cannot create rows after the fact.

    With ``evaluation=None``, builds a
    [`TabularPointwiseEvaluation`][pinnjax.TabularPointwiseEvaluation] and binds validation
    rows. If you pass an ``evaluation`` (e.g. the same
    [`TabularPointwiseEvaluation`][pinnjax.TabularPointwiseEvaluation] or another tabular
    subclass used during training), it is **shallow-copied**, ``.inputs`` and ``.truth`` are
    set to the validation arrays, then ``evaluate(model)`` is called so the original instance
    is unchanged.
    """
    from pinnjax.loss import TabularPointwiseEvaluation

    ds = self.data_shape
    if ds.validation_inputs is None and (
        ds._validation_indices is not None and ds._validation_indices.size > 0
    ):
        ds.materialize_validation_from_portion()
    if ds.validation_inputs is None or ds.validation_inputs.shape[0] == 0:
        config.LOGGER.warning(
            "All DATA was used in training/testing. Use a portion less than 1.0. "
            "Returning empty metrics.",
        )
        return {}
    if evaluation is None:
        evaluation = TabularPointwiseEvaluation()
        evaluation.bind_tabular_data(ds.validation_inputs, ds.validation_outputs)
    else:
        ev = copy.copy(evaluation)
        if not hasattr(ev, "inputs") or not hasattr(ev, "truth"):
            msg = (
                "When passing evaluation, it must have .inputs and .truth so "
                "validation rows can be bound (e.g. TabularPointwiseEvaluation or "
                "a tabular Evaluation with the same attributes)."
            )
            config.LOGGER.error(msg)
            raise TypeError(msg)
        ev.inputs = ds.validation_inputs
        ev.truth = ds.validation_outputs
        evaluation = ev
    metrics = evaluation.evaluate(model)
    return {k: float(v) for k, v in metrics.items()}
