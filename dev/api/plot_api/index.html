    def plot_prediction(
    prediction: jnp.ndarray,
    grid: pinnjax.SpatialGrid,
    *,
    threshold: float | None = None,
    show: bool | None = True,
    save: bool | None = True,
    save_location: str | Path | None = None,
    save_name: str | None = "prediction.pdf",
    save_format: str = "pdf",
    dpi: int = 300,
    title: str | None = None,
    cbar_label: str | None = None,
    cmap: str | None = "turbo",
    sparse_interior: bool = True,
):
    # Treat sliced 3D (e.g. shape (ny, nx, 1)) as 2D for plotting
    prediction = jnp.squeeze(prediction)
    dim = len(prediction.shape)

    if threshold is not None and isinstance(threshold, float):
        prediction = jnp.clip(prediction, None, threshold)

    if dim == TWO_DIMENSIONAL:
        figsize = _figsize_from_2d_grid(grid)
        fig, ax = plt.subplots(figsize=figsize, constrained_layout=True)
        norm = mpl.colors.Normalize(
            vmin=float(jnp.min(prediction)),
            vmax=float(jnp.max(prediction)),
        )

        im = ax.imshow(
            prediction,
            origin="lower",
            aspect="equal",
            extent=(
                grid.domain.min[0],
                grid.domain.max[0],
                grid.domain.min[1],
                grid.domain.max[1],
            ),
            cmap=cmap,
            norm=norm,
        )
        ticks = jnp.linspace(jnp.min(prediction), jnp.max(prediction), num=5)
        cbar = fig.colorbar(
            im,
            label=cbar_label,
            orientation="horizontal",
            pad=0.14,
            fraction=0.24,
            ticks=ticks,
        )

        ax.set_xlabel("x [m]")
        ax.set_ylabel("y [m]")
        _style_2d_axes(ax, cbar)

    elif dim == THREE_DIMENSIONAL:
        # TODO(@mjryande): Convert to PyVista or some other volumetric plotter.
        fig = plt.figure()
        ax = fig.add_subplot(111, projection="3d")

        norm = mpl.colors.Normalize(
            vmin=float(jnp.min(prediction)),
            vmax=float(jnp.max(prediction)),
        )

        if sparse_interior:
            face = jnp.zeros(prediction.shape, dtype=bool)
            face = face.at[0, :, :].set(True)
            face = face.at[-1, :, :].set(True)
            face = face.at[:, 0, :].set(True)
            face = face.at[:, -1, :].set(True)
            face = face.at[:, :, 0].set(True)
            face = face.at[:, :, -1].set(True)
            mask = face.ravel()
        else:
            mask = jnp.ones(prediction.shape, dtype=bool)

        # gather boundary points
        x = grid.coordinates[..., 0].ravel()[mask]
        y = grid.coordinates[..., 1].ravel()[mask]
        z = grid.coordinates[..., 2].ravel()[mask]
        c = prediction.ravel()[mask]

        surf = ax.scatter3D(
            x,
            y,
            z,
            c=c,
            norm=norm,
            cmap=cmap,
            rasterized=True,
            depthshade=False,
        )

        ticks = jnp.linspace(jnp.min(prediction), jnp.max(prediction), num=5)
        fig.colorbar(
            surf,
            label=cbar_label,
            orientation="horizontal",
            pad=0.05,
            ticks=ticks,
            shrink=0.9,
        )

        fig, ax = grid.domain.draw(show=False, fig=fig, ax=ax)

    else:
        msg = f"Unsupported grid dimension: {dim}"
        config.LOGGER.error(msg)
        raise ValueError(msg)

    if title is not None:
        ax.set_title(title)

    if save:
        out_dir = Path(save_location) if save_location else Path.cwd()
        out_dir.mkdir(parents=True, exist_ok=True)
        out_path = out_dir / (save_name or "prediction.pdf")
        ext = out_path.suffix.lower()
        fmt = (
            ext.lstrip(".")
            if ext
            else (save_format.lstrip(".") if isinstance(save_format, str) else "pdf")
        )
        fig.savefig(out_path, format=fmt or "pdf", dpi=dpi)

    if show:
        plt.show()
    else:
        plt.close()

    return fig, ax

    def plot_prediction_evolution(
    evolution: Array,
    grid: pinnjax.SpatialGrid,
    *,
    threshold: float | None = None,
    fig_size: tuple[float, float] | None = None,
    fps: int | None = None,
    show: bool | None = True,
    time: bool | None = False,
    save: bool | None = False,
    save_location: str | Path | None = None,
    save_name: str | None = None,
    cbar_label: str | None = None,
    cmap: str | None = "turbo",
) -> tuple[Figure, Axes, FuncAnimation]:
    if threshold is not None and isinstance(threshold, float):
        evolution = jnp.clip(evolution, None, threshold)

    if not hasattr(grid, "times") or grid.times is None:
        msg = "Grid must have temporal information for evolution plots"
        config.LOGGER.error(msg)
        raise ValueError(msg)

    if fps is None:
        fps = max(1, int(evolution.shape[0] / grid.domain.times.range()))

    # Get global min/max for consistent colorbar
    vmin = float(jnp.min(evolution))
    vmax = float(jnp.max(evolution))
    norm = mpl.colors.Normalize(vmin=vmin, vmax=vmax)

    # Create figure and axis
    fig, ax = plt.subplots(
        figsize=fig_size if fig_size is not None else _figsize_from_2d_grid(grid),
        constrained_layout=True,
    )

    # Initial plot
    im = ax.imshow(
        evolution[0],
        origin="lower",
        aspect="equal",
        extent=(
            grid.domain.min[0],
            grid.domain.max[0],
            grid.domain.min[1],
            grid.domain.max[1],
        ),
        cmap=cmap,
        norm=norm,
    )

    # Colorbar
    ticks = jnp.linspace(vmin, vmax, num=5)
    cbar = fig.colorbar(
        im,
        label=cbar_label,
        orientation="horizontal",
        pad=0.14,
        fraction=0.24,
        ticks=ticks,
    )
    ax.set_xlabel("x [m]")
    ax.set_ylabel("y [m]")
    _style_2d_axes(ax, cbar)

    # Time text
    if time:
        time_text = ax.text(
            0.02,
            0.98,
            "",
            transform=ax.transAxes,
            verticalalignment="top",
            fontsize=_TICK_FONTSIZE,
            bbox=dict(boxstyle="round,pad=0.3", facecolor="white", alpha=0.85),
        )

    def animate(frame: int):
        """Animation function."""
        im.set_data(evolution[frame])
        if time:
            time_text.set_text(f"t = {grid.times[frame]:.3f} s")
            return [im, time_text]
        return [im]

    fig.canvas.draw()
    fig.set_constrained_layout = False

    # Create animation
    anim = FuncAnimation(
        fig,
        animate,
        frames=len(evolution),
        interval=1000 // fps,
        blit=True,
        repeat=True,
    )

    if save:
        # Resolve output path
        out_dir = Path(save_location) if save_location else Path.cwd()
        out_dir.mkdir(parents=True, exist_ok=True)
        out_path = out_dir / (save_name or "ani.gif")

        ext = out_path.suffix.lower()
        if ext == ".gif":
            writer = PillowWriter(fps=fps, bitrate=-1)
            anim.save(out_path, writer=writer, dpi=min(150, _SAVE_DPI))
        else:
            writer = FFMpegWriter(fps=fps, extra_args=["-vcodec", "libx264"])
            anim.save(out_path, writer=writer, dpi=_SAVE_DPI)

    if show:
        plt.show()
    else:
        plt.close()

    return fig, ax, anim

    def plot_comparison_evolution(
    prediction_evolution: Array,
    truth_evolution: Array,
    grid: pinnjax.SpatialGrid,
    *,
    error_type: Literal["diff", "abs", "percent"] = "diff",
    threshold: float | None = None,
    times: Array | None = None,
    fps: int | None = None,
    fig_size: tuple[float, float] | None = None,
    time: bool | None = False,
    show: bool | None = True,
    save: bool | None = False,
    save_location: str | Path | None = None,
    save_name: str | None = "comparison_evolution.gif",
    cbar_label: str | None = None,
    cmap: str | None = None,
) -> tuple[Figure, Axes, FuncAnimation]:
    """
    Animate comparison error over time for 2D spatial grids.

    Args:
        prediction_evolution: (num_times, *spatial_shape) model predictions.
        truth_evolution: (num_times, *spatial_shape) ground-truth values.
        grid: Grid providing domain min/max for plot extent.
        error_type: 'diff' (truth-pred), 'abs', or 'percent' relative error [%].
        times: Optional time vector for overlay text (length == num_times).
              If None and `time=True`, tries `grid.times`.
    """
    pred = jnp.asarray(prediction_evolution)
    tru = jnp.asarray(truth_evolution)

    if pred.shape != tru.shape:
        raise ValueError(
            f"prediction_evolution and truth_evolution must match shapes. Got {pred.shape=} {tru.shape=}"
        )

    if pred.ndim < 3:
        raise ValueError(
            "Expected evolution arrays shaped like (num_times, nx, ny). "
            f"Got pred.ndim={pred.ndim}, pred.shape={pred.shape}."
        )

    spatial_shape = grid.coordinates.shape[:-1]
    if tuple(pred.shape[1:]) != tuple(spatial_shape):
        raise ValueError(
            f"Shape mismatch vs grid. Got evolution spatial {pred.shape[1:]} but grid expects {spatial_shape}."
        )

    num_frames = pred.shape[0]
    et = error_type.lower()

    if threshold is not None:
        pred = jnp.clip(pred, min=None, max=threshold)
        tru = jnp.clip(tru, min=None, max=threshold)

    if et == "diff":
        err = tru - pred
        m = float(jnp.nanmax(jnp.abs(err)))
        m = max(m, 1e-12)
        cmap_use = cmap or "RdBu_r"
        norm = TwoSlopeNorm(vmin=-m, vcenter=0.0, vmax=+m)
        ticks = jnp.linspace(-m, m, num=7)
        cbar_use = cbar_label or "Difference"
    elif et == "abs":
        err = jnp.abs(tru - pred)
        vmin = 0.0
        vmax = float(jnp.nanmax(err))
        vmax = max(vmax, vmin + 1e-12)
        cmap_use = cmap or "turbo"
        norm = mpl.colors.Normalize(vmin=vmin, vmax=vmax)
        ticks = jnp.linspace(vmin, vmax, num=7)
        cbar_use = cbar_label or "Absolute Difference"
    elif et == "percent":
        err = jnp.abs(tru - pred) / (jnp.abs(tru) + 1e-8) * 100.0
        vmin = float(jnp.nanmin(err))
        vmax = float(jnp.nanmax(err))
        vmax = max(vmax, vmin + 1e-12)
        cmap_use = cmap or "BuPu"
        norm = mpl.colors.Normalize(vmin=vmin, vmax=vmax)
        ticks = jnp.linspace(vmin, vmax, num=7)
        cbar_use = cbar_label or "Relative Difference [%]"
    else:
        raise ValueError("error_type must be 'diff', 'abs', or 'percent'.")

    # Resolve times for overlay
    if times is None:
        times = getattr(grid, "times", None)
    if time and times is None:
        raise ValueError("time=True requires `times` or `grid.times`.")

    if times is not None:
        times_arr = jnp.asarray(times)
        if times_arr.shape[0] != num_frames:
            raise ValueError(
                f"times length must equal num_frames ({num_frames}). Got {times_arr.shape[0]}."
            )

    # fps
    if fps is None:
        domain_times = getattr(getattr(grid, "domain", None), "times", None)
        if domain_times is not None:
            fps = max(1, int(num_frames / domain_times.range()))
        elif times is not None:
            dt = float(jnp.max(times_arr) - jnp.min(times_arr))
            fps = max(1, int(num_frames / max(dt, 1e-12)))
        else:
            fps = 10

    fig, ax = plt.subplots(
        figsize=fig_size if fig_size is not None else _figsize_from_2d_grid(grid),
        constrained_layout=True,
    )

    im = ax.imshow(
        err[0],
        origin="lower",
        aspect="equal",
        extent=(
            grid.domain.min[0],
            grid.domain.max[0],
            grid.domain.min[1],
            grid.domain.max[1],
        ),
        cmap=cmap_use,
        norm=norm,
    )

    cbar = fig.colorbar(
        im,
        label=cbar_use,
        orientation="horizontal",
        pad=0.14,
        fraction=0.24,
        ticks=ticks,
    )
    ax.set_xlabel("x [m]")
    ax.set_ylabel("y [m]")
    _style_2d_axes(ax, cbar)

    time_text = None
    if time:
        time_text = ax.text(
            0.02,
            0.98,
            "",
            transform=ax.transAxes,
            verticalalignment="top",
            fontsize=_TICK_FONTSIZE,
            bbox=dict(boxstyle="round,pad=0.3", facecolor="white", alpha=0.85),
        )

    def animate(frame: int):
        im.set_data(err[frame])
        if time_text is not None:
            time_text.set_text(f"t = {float(times_arr[frame]):.3f} s")
            return [im, time_text]
        return [im]

    fig.canvas.draw()
    fig.set_constrained_layout = False

    anim = FuncAnimation(
        fig,
        animate,
        frames=num_frames,
        interval=1000 // fps,
        blit=True,
        repeat=True,
    )

    if save:
        out_dir = Path(save_location) if save_location else Path.cwd()
        out_dir.mkdir(parents=True, exist_ok=True)
        out_path = out_dir / (save_name or "comparison_evolution.gif")

        ext = out_path.suffix.lower()
        if ext == ".gif":
            writer = PillowWriter(fps=fps, bitrate=-1)
            anim.save(out_path, writer=writer, dpi=min(150, _SAVE_DPI))
        else:
            writer = FFMpegWriter(fps=fps, extra_args=["-vcodec", "libx264"])
            anim.save(out_path, writer=writer, dpi=_SAVE_DPI)

    if show:
        plt.show()
    else:
        plt.close()

    return fig, ax, anim

    def plot_loss(
    loss: dict,
    *,
    mode: str = "class",  # ["class", "type"]
    title: str | None = None,
    best_epoch: int | None = None,
    show_grid: bool = True,
    show: bool = True,
    save: bool = False,
    save_location: Path | str | None = None,
    save_name: str | None = None,
    save_format: str = "pdf",
) -> tuple[Figure, Axes]:
    if mode not in ("class", "type"):
        msg = f"mode must be 'class' or 'type'. Inputted mode: {mode}"
        config.LOGGER.error(msg)
        raise ValueError(msg)

    fig, ax = plt.subplots()

    def plot_comparison(
    prediction: jnp.ndarray,
    truth: jnp.ndarray,
    grid: pinnjax.SpatialGrid,
    *,
    threshold: float | None = None,
    sparse_interior: bool = True,
    error_type: Literal["diff", "abs", "percent"] = "diff",
    title: str | None = None,
    fig_size: tuple[float, float] | None = None,
    cmap_override: str = None,
    cbar_label: str | None = None,
    show: bool = True,
    save: bool = False,
    save_location: str | Path | None = None,
    save_name: str | None = "comparison.pdf",
    save_format: Literal[".png", ".jpg", ".jpeg", ".tiff", "pdf"] = "pdf",
    dpi: int = 300,
):
    """Plot prediction vs truth: signed difference, absolute difference, or relative error (%)."""
    pred = jnp.squeeze(prediction)
    tru = jnp.squeeze(truth)

    if pred.shape != tru.shape:
        raise ValueError(
            f"prediction and truth must have the same shape after squeeze. "
            f"Got pred={pred.shape}, truth={tru.shape}."
        )

    dim = pred.ndim
    if dim not in (2, 3):
        raise ValueError(
            "plot_comparison expects 2D or 3D (or squeezable to 2D/3D) inputs."
        )

    if threshold is not None and isinstance(threshold, float):
        pred = jnp.clip(pred, min=None, max=threshold)
        tru = jnp.clip(tru, min=None, max=threshold)

    # --------------------
    # compute error field
    # --------------------
    et = error_type.lower()
    if et == "diff":
        err = tru - pred
        cmap = "RdBu_r" or cmap_override
        m = float(jnp.nanmax(jnp.abs(err)))
        m = max(m, 1e-12)  # ensure vmin < vcenter < vmax
        norm = TwoSlopeNorm(vmin=-m, vcenter=0.0, vmax=+m)
        ticks = jnp.linspace(-m, m, num=7)
        cbar_label = cbar_label or "Difference"
    elif et == "abs":
        err = jnp.abs(tru - pred)
        cmap = "turbo" or cmap_override
        vmin, vmax = 0.0, float(jnp.nanmax(err))
        norm = mpl.colors.Normalize(vmin=vmin, vmax=vmax)
        ticks = jnp.linspace(vmin, vmax, num=7)
        cbar_label = cbar_label or "Absolute Difference"
    elif et == "percent":
        err = jnp.abs(tru - pred) / (jnp.abs(tru) + 1e-8) * 100.0
        cmap = "BuPu" or cmap_override
        vmin, vmax = float(jnp.nanmin(err)), float(jnp.nanmax(err))
        norm = mpl.colors.Normalize(vmin=vmin, vmax=vmax)
        ticks = jnp.linspace(vmin, vmax, num=7)
        cbar_label = cbar_label or "Relative Difference [%]"
    else:
        raise ValueError(
            f"error_type must be 'diff', 'abs', or 'percent'. Got {error_type!r}."
        )

    # --------------------
    # plotting
    # --------------------
    if dim == 2:
        domain = grid.domain
        extent = (
            float(domain.min[0]),
            float(domain.max[0]),
            float(domain.min[1]),
            float(domain.max[1]),
        )
        fig_size = fig_size if fig_size is not None else _figsize_from_2d_grid(grid)
        if title is not None:
            fig_size = (fig_size[0], fig_size[1] + 0.3)

        fig, ax = plt.subplots(
            figsize=fig_size,
            constrained_layout=True,
        )
        im = ax.imshow(
            err,
            origin="lower",
            extent=extent,
            aspect="equal",
            cmap=cmap,
            norm=norm,
        )
        cbar = fig.colorbar(
            im,
            label=cbar_label,
            orientation="horizontal",
            pad=0.14,
            fraction=0.24,
        )
        cbar.set_ticks(ticks)
        ax.set_xlabel("x [m]")
        ax.set_ylabel("y [m]")

        if title is not None:
            ax.set_title(title)

        _style_2d_axes(ax, cbar)

    else:  # dim == 3
        fig = plt.figure()
        ax = fig.add_subplot(111, projection="3d")

        if sparse_interior:
            face = jnp.zeros(err.shape, dtype=bool)
            face = face.at[0, :, :].set(True)
            face = face.at[-1, :, :].set(True)
            face = face.at[:, 0, :].set(True)
            face = face.at[:, -1, :].set(True)
            face = face.at[:, :, 0].set(True)
            face = face.at[:, :, -1].set(True)
            mask = face.ravel()
        else:
            mask = jnp.ones(err.shape, dtype=bool).ravel()

        # gather boundary points
        x = grid.coordinates[..., 0].ravel()[mask]
        y = grid.coordinates[..., 1].ravel()[mask]
        z = grid.coordinates[..., 2].ravel()[mask]
        c = err.ravel()[mask]

        surf = ax.scatter3D(
            x,
            y,
            z,
            c=c,
            norm=norm,
            cmap=cmap,
            rasterized=True,
            depthshade=False,
        )

        fig.colorbar(
            surf,
            label=cbar_label,
            orientation="horizontal",
            pad=0.05,
            ticks=ticks,
            shrink=0.9,
        )

        fig, ax = grid.domain.draw(show=False, fig=fig, ax=ax)

        if title is not None:
            ax.set_title(title)

    if save:
        out_dir = Path(save_location) if save_location else Path.cwd()
        out_dir.mkdir(parents=True, exist_ok=True)
        out_path = out_dir / (save_name or "comparison.pdf")
        ext = out_path.suffix.lower()
        fmt = (
            ext.lstrip(".")
            if ext
            else (save_format.lstrip(".") if isinstance(save_format, str) else "pdf")
        )
        fig.savefig(out_path, format=fmt or "pdf", dpi=dpi)

    if show:
        plt.show()
    else:
        plt.close()

    return fig, ax

    # TODO(@mjryande): Parts of this only work in 3D (x, y, z)
    if model is None and truth is None:
        msg = "Neither a model or truth has be provided."
        config.LOGGER.error(msg)
        raise ValueError(msg)

    if domain is None and point_type.casefold() == "percentage":
        msg = "When using point_type = 'percentage, a domain must be provided."
        config.LOGGER.error(msg)
        raise ValueError(msg)

    if time is not None:
        if isinstance(time, pinnjax.Time):
            time = jnp.linspace(time.min, time.max, int(time.range() / time.step))
        elif isinstance(time, Iterable):
            time = jnp.asarray(time)
        else:
            msg = f"Invalid time type: {type(time)}."
            config.LOGGER.error(msg)
            raise ValueError(msg)
    elif domain is not None:
        time = jnp.linspace(
            domain.times.min,
            domain.times.max,
            int(domain.times.range() / domain.times.step),
        )

    spatial_points = jnp.asarray(spatial_points)
    match point_type.casefold():
        case "percentage":
            percentages = [
                _percentage_to_point(domain, spatial_points[row])
                for row in range(spatial_points.shape[0])
            ]
            spatial_points = jnp.stack(percentages)
        case "coordinate":
            if domain is not None:
                for row in range(spatial_points.shape[0]):
                    if not _verify_domain(domain, spatial_points[row]):
                        msg = (
                            f"Selected Coordinate: {spatial_points[row]} "
                            f"is not in the Domain."
                        )
                        config.LOGGER.error(msg)
                        raise ValueError(msg)
            else:
                config.LOGGER.warning(
                    "Coordinates not guaranteed to be in the Domain.",
                )
            spatial_points = jnp.stack(spatial_points)
        case _:
            msg = (
                f"Invalid point_type: {point_type}. "
                f"Use either 'percentage' or 'coordinate'"
            )
            config.LOGGER.error(msg)
            raise ValueError(msg)

    truth_times = None
    truth_values_at_points = None
    if truth is not None:
        if isinstance(truth, (str, Path)):
            truth = Path(truth)
            if truth.suffix.lower() != ".csv":
                msg = f"Expected a CSV file, got {truth.suffix}."
                config.LOGGER.error(msg)
                raise ValueError(msg)
            truth = pl.read_csv(truth).to_numpy().astype(float)

                jnp.delete(truth, csv_truth_column, axis=1),

                [
                    jnp.tile(
                        pt[None, :],
                        (truth_times.size, 1),

                    truth_times[:, None],

            for pt in spatial_points
        ]

        tree = cKDTree(coordinates)
        truth_values_at_points = [
            truth_values[
                jnp.array([_nearest_neighbor_index(tree, pt_row) for pt_row in pts])
            ]
            for pts in spatial_temporal_points
        ]

    predictions = None
    if model is not None:
        if time is None and truth is not None:
            time = truth_times
        spatial_temporal_points = [
            jnp.hstack(
                [
                    jnp.tile(
                        pt[None, :],
                        (time.size, 1),
                    ),  # repeat spatial point for each time
                    time[:, None],  # append times
                ],
            )
            for pt in spatial_points
        ]

        predictions = []
        for points in spatial_temporal_points:
            _prediction = jax.vmap(model)(points)
            predictions.append(_prediction.reshape(points.shape[:-1]))

    # ----Plotting-----
    if show_points:
        # check if the points share an axis.
        axes = list(zip(*spatial_points, strict=False))
        shared_axes = [
            i
            for i, axis_values in enumerate(axes)
            if len({float(v) for v in axis_values}) == 1
        ]

        if not shared_axes:
            fig1 = plt.figure(constrained_layout=True)
            ax1 = fig1.add_subplot(111, projection="3d")
            _style(ax1)
            for i in range(len(spatial_points)):
                x, y, z = spatial_points[i]

                ax1.scatter(x, y, z)
                ax1.text(x, y, z, f"P{i + 1}", fontsize=14)

            if domain is not None:
                fig1, ax1 = domain.draw(show=False, fig=fig1, ax=ax1)

            if save:
                out_dir = Path(save_location) if save_location else Path.cwd()
                out_dir.mkdir(parents=True, exist_ok=True)
                out_path = out_dir / "representative_point_locations.pdf"
                ext = out_path.suffix.lower()
                fmt = (
                    ext.lstrip(".")
                    if ext
                    else (
                        save_format.lstrip(".")
                        if isinstance(save_format, str)
                        else "pdf"
                    )
                )
                fig1.savefig(out_path, format=fmt or "pdf", dpi=300)

        else:
            units = None
            all_axis = [0, 1, 2]
            for idx in shared_axes:  # plots each shared axis as a separate plot.
                other_axis = [ax for ax in all_axis if ax != idx]
                fig1, ax1 = plt.subplots(
                    figsize=(8, 2),
                    constrained_layout=True,
                )  # TODO(@mjryande): not hardcoded fig-size
                _style(ax1)
                reduced_points = [
                    list(p[:idx]) + list(p[idx + 1 :]) for p in spatial_points
                ]
                for i, (x, y) in enumerate(reduced_points):
                    ax1.scatter(x, y)
                    ax1.annotate(f"P{i + 1}", (x, y), fontsize=14)

                if domain is not None:
                    _mins = tuple(domain.min[:idx] + domain.min[idx + 1 :])
                    _maxs = tuple(domain.max[:idx] + domain.max[idx + 1 :])
                    units = getattr(domain, "UNITS", {}).get("min", None)

                    xs = jnp.linspace(_mins[0], _maxs[0], num=5)
                    ys = jnp.linspace(_mins[1], _maxs[1], num=5)
                    ax1.set_xticks(xs)
                    ax1.set_yticks(ys)

                ax1.set_xlabel(
                    str(IDX_TO_AXIS[other_axis[0]])
                    + (f" [{units}]" if units is not None else ""),
                )
                ax1.set_ylabel(
                    str(IDX_TO_AXIS[other_axis[1]])
                    + (f" [{units}]" if units is not None else ""),
                )

                ax1.set_aspect("equal", adjustable="box")

                if save:
                    out_dir = Path(save_location) if save_location else Path.cwd()
                    out_dir.mkdir(parents=True, exist_ok=True)
                    out_path = (
                        out_dir / f"representative_point_locations_"
                        f"{IDX_TO_AXIS[other_axis[0]]}"
                        f"-{IDX_TO_AXIS[other_axis[1]]}.pdf"
                    )
                    ext = out_path.suffix.lower()
                    fmt = (
                        ext.lstrip(".")
                        if ext
                        else (
                            save_format.lstrip(".")
                            if isinstance(save_format, str)
                            else "pdf"
                        )
                    )
                    fig1.savefig(out_path, format=fmt or "pdf", dpi=300)

    # Main Plot.
    fig, ax = plt.subplots(figsize=fig_size)
    _style(ax)
    for idx, _ in enumerate(spatial_points):
        if model is not None:
            ax.plot(time, predictions[idx], label=f"P{idx}-PINN")
        if truth is not None:
            ax.scatter(
                truth_times,
                truth_values_at_points[idx],
                edgecolors="black",
                marker="o",
                linewidths=0.25,
                s=12,
                label=f"P{idx}-GT",
            )

    if x_axis_label:
        ax.set_xlabel(x_axis_label, fontsize=15)
    if y_axis_label:
        ax.set_ylabel(y_axis_label, fontsize=15)

    if title:
        plt.title(title)

    if legend:
        ax.legend(fontsize=14, markerscale=1.0)

    ax.grid(
        color="grey",
        axis="both",
        linestyle="-",
        alpha=0.2,
        zorder=0,
        linewidth=0.75,
    )
    ax.xaxis.set_tick_params(labelsize=14)
    ax.yaxis.set_tick_params(labelsize=14)

    if save:
        out_dir = Path(save_location) if save_location else Path.cwd()
        out_dir.mkdir(parents=True, exist_ok=True)
        out_path = out_dir / (save_name or "value_over_time.pdf")
        ext = out_path.suffix.lower()
        fmt = (
            ext.lstrip(".")
            if ext
            else (save_format.lstrip(".") if isinstance(save_format, str) else "pdf")
        )
        fig.savefig(out_path, format=fmt or "pdf", dpi=300)

    if show:
        plt.show()
    else:
        plt.close()

    if show_points:
        return fig, ax, fig1, ax1

    return fig, ax

    def value_along_path_plot(
    time_instances: Iterable | float,
    path: pinnjax.Path,
    *,
    model: pinnjax.NeuralNetwork | None = None,
    truth: Path | str | None = None,
    domain: pinnjax.SpatialDomain | None = None,
    times: pinnjax.Time | None = None,
    csv_coordinate_column_order: Iterable | None = None,
    csv_truth_column: int = -1,
    fig_size: tuple | None = (10, 5),
    title: str | None = None,
    x_axis_label: str | None = None,
    y_axis_label: str | None = None,
    legend: bool = True,
    show: bool = True,
    save: bool | None = False,
    save_location: str | Path | None = None,
    save_name: str | None = "value_along_path.pdf",
    save_format: str = "pdf",
) -> tuple[Figure, Axes]:
    time_instances = jnp.asarray(time_instances)

    if model is None and truth is None:
        msg = "Neither a model or truth has be provided."
        config.LOGGER.error(msg)
        raise ValueError(msg)

    if domain is not None:
        if isinstance(domain, pinnjax.Domain):
            travel_times = jnp.linspace(
                domain.times.min,
                domain.times.max,
                int(domain.times.range() / domain.times.step),
            )
        else:
            msg = "'domain' should be type: pinnjax.Domain."
            config.LOGGER.error(msg)
            raise TypeError(msg)
    elif times is not None:
        if isinstance(times, pinnjax.Time):
            travel_times = jnp.linspace(
                times.min,
                times.max,
                int(times.range() / times.step),
            )
        else:
            msg = "'domain' should be type: pinnjax.Domain."
            config.LOGGER.error(msg)
            raise TypeError(msg)
    else:
        msg = "Neither a Domain or Time object has been provided."
        config.LOGGER.error(msg)
        raise ValueError(msg)

    path_locations = path.eval(travel_times)

    if truth is not None:
        if isinstance(truth, (str, Path)):
            truth = Path(truth)
            if truth.suffix.lower() != ".csv":
                msg = f"Expected a CSV file, got {truth.suffix}"
                config.LOGGER.error(msg)
                raise ValueError(msg)
            truth = pl.read_csv(truth).to_numpy().astype(float)
            coordinates = jnp.array(
                jnp.delete(truth, csv_truth_column, axis=1),
                copy=False,
            )
            if csv_coordinate_column_order is not None:
                coordinates = jnp.array(
                    coordinates[:, csv_coordinate_column_order],
                    copy=False,
                )
            else:
                coordinates = jnp.array(coordinates, copy=False)
            truth_values = jnp.array(truth[:, csv_truth_column], copy=False)
        else:
            msg = f"Invalid truth type: {type(truth)}. Use Path or str (CSV)."
            config.LOGGER.error(msg)
            raise TypeError(msg)

        if model is not None and coordinates.shape[1] != model.topology[0]:
            msg = (
                f"Truth coordinates [Columns: {coordinates.shape[1]}] is not "
                f"consistent with prediction topology [Columns: "
                f"{model.topology[0]}] "
            )
            config.LOGGER.error(msg)
            raise ValueError(msg)

        truth_times = jnp.unique(jnp.array(coordinates[:, -1], copy=False))

        tree = cKDTree(truth_times[:, None])
        for idx, _time in enumerate(time_instances):
            time_instances = time_instances.at[idx].set(
                truth_times.at[_nearest_neighbor_index(tree, _time)].get(),
            )
        time_instances = jnp.unique(time_instances)

        spatial_coordinates_at_t0 = coordinates[
            jnp.isclose(coordinates[:, -1], truth_times[0], rtol=0.0, atol=1e-12),
            :-1,
        ]

        tree = cKDTree(spatial_coordinates_at_t0)
        for idx, point in enumerate(path_locations):
            path_locations = path_locations.at[idx].set(
                spatial_coordinates_at_t0.at[
                    _nearest_neighbor_index(tree, point)
                ].get(),
            )

        path_locations = jnp.unique(path_locations, axis=0)

        spatial_temporal_points = [
            jnp.hstack([path_locations, jnp.full((path_locations.shape[0], 1), ti)])
            for ti in time_instances
        ]

        tree = cKDTree(coordinates)
        truth_values_at_points = [
            truth_values[
                jnp.array([_nearest_neighbor_index(tree, pt_row) for pt_row in pts])
            ]
            for pts in spatial_temporal_points
        ]

    predictions = None
    if model is not None:
        spatial_temporal_points = [
            jnp.hstack([path_locations, jnp.full((path_locations.shape[0], 1), ti)])
            for ti in time_instances
        ]

        predictions = []
        for points in spatial_temporal_points:
            _prediction = jax.vmap(model)(points)
            predictions.append(_prediction.reshape(points.shape[:-1]))

    # --Plotting--
    fig, ax = plt.subplots(figsize=fig_size)
    _style(ax)

    prop_colors = plt.rcParams.get("axes.prop_cycle").by_key().get("color", [])
    colors = cycle(prop_colors)

    legend_labels = []
    legend_handles = []
    for idx, _time in enumerate(time_instances):
        # TODO(@mjryande): Reconsider x-axis (distance somehow)
        #    euclidean distance between points (?)
        color = next(colors)
        if model is not None:
            ax.plot(range(len(predictions[idx])), predictions[idx], color=color)
        if truth is not None:
            scatter = ax.scatter(
                range(len(truth_values_at_points[idx])),
                truth_values_at_points[idx],
                edgecolors="black",
                marker="o",
                linewidths=0.25,
                s=12,
                color=color,
            )
        else:
            scatter = ax.scatter(
                [],
                [],  # empty data -> used only as a legend proxy
                facecolors=color,
                edgecolors="black",
                marker="o",
                linewidths=0.25,
                s=36,
                zorder=3,
            )
        legend_labels.append(f"t={_time:.2f}s")
        legend_handles.append(scatter)

    if x_axis_label:
        ax.set_xlabel(x_axis_label, fontsize=15)
    if y_axis_label:
        ax.set_ylabel(y_axis_label, fontsize=15)

    if title:
        plt.title(title)

    if legend and legend_handles:
        ax.legend(
            handles=legend_handles,
            labels=legend_labels,
            fontsize=14,
            markerscale=1.0,
        )

    ax.grid(
        color="grey",
        axis="both",
        linestyle="-",
        alpha=0.2,
        zorder=0,
        linewidth=0.75,
    )
    ax.xaxis.set_tick_params(labelsize=14)
    ax.yaxis.set_tick_params(labelsize=14)

    if save:
        out_dir = Path(save_location) if save_location else Path.cwd()
        out_dir.mkdir(parents=True, exist_ok=True)
        out_path = out_dir / (save_name or "value_over_time.pdf")
        ext = out_path.suffix.lower()
        fmt = (
            ext.lstrip(".")
            if ext
            else (save_format.lstrip(".") if isinstance(save_format, str) else "pdf")
        )
        fig.savefig(out_path, format=fmt or "pdf", dpi=300)

    if show:
        plt.show()
    else:
        plt.close()

    return fig, ax
